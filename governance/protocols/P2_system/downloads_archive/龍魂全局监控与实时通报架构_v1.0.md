```python
class LocalServerMonitor:
    """本地服务器监控器"""

    MONITOR_ITEMS = {
        # 系统级
        "cpu_usage": {
            "cmd": "top -l 1 | grep 'CPU usage'",
            "threshold": 80,  # CPU超过80%告警
            "interval": 60,   # 每60秒检查
        },
        "memory_usage": {
            "cmd": "vm_stat | awk '{print $3}'",
            "threshold": 85,  # 内存超过85%告警
            "interval": 60,
        },
        "disk_usage": {
            "cmd": "df -h / | tail -1 | awk '{print $5}' | sed 's/%//'",
            "threshold": 90,  # 磁盘超过90%告警
            "interval": 300,  # 每5分钟
        },
        "network_io": {
            "cmd": "netstat -ib | awk '{print $7, $10}'",
            "threshold": 100000000,  # 100MB/s
            "interval": 60,
        },

        # 龍魂服务级
        "longhun_process": {
            "cmd": "ps aux | grep 'longhun' | grep -v grep | wc -l",
            "threshold": 0,  # 进程数为0告警
            "interval": 30,
        },
        "log_growth": {
            "cmd": "du -sh ~/.longhun/logs/ | awk '{print $1}'",
            "threshold": 1000000000,  # 1GB
            "interval": 3600,
        },

        # 文件变更级
        "file_changes": {
            "watch_paths": [
                "~/longhun/bin/",
                "~/.longhun/",
                "~/longhun-system/",
            ],
            "interval": 10,  # 每10秒检查文件变更
        },

        # 安全级
        "login_attempts": {
            "cmd": "last | grep -v 'wtmp' | wc -l",
            "threshold": 10,  # 10分钟内登录超过10次告警
            "interval": 600,
        },
        "sudo_usage": {
            "cmd": "grep 'sudo' /var/log/system.log | tail -5",
            "threshold": 1,  # 任何sudo使用都通知
            "interval": 60,
        },
    }

    def check_all(self):
        """检查所有监控项"""
        alerts = []
        for item_name, config in self.MONITOR_ITEMS.items():
            result = self.check_item(item_name, config)
            if result["triggered"]:
                alerts.append(result)
        return alerts

    def check_item(self, name, config):
        """检查单个监控项"""
        import subprocess
        try:
            if "cmd" in config:
                result = subprocess.run(
                    config["cmd"], shell=True, capture_output=True, text=True, timeout=10
                )
                value = self.parse_value(result.stdout)
                triggered = value > config["threshold"]
                return {
                    "source": "local_server",
                    "item": name,
                    "value": value,
                    "threshold": config["threshold"],
                    "triggered": triggered,
                    "timestamp": datetime.now().isoformat(),
                }
        except Exception as e:
            return {
                "source": "local_server",
                "item": name,
                "value": None,
                "error": str(e),
                "triggered": True,  # 检查失败也算异常
                "timestamp": datetime.now().isoformat(),
            }
```

```python
            "threshold": 10,  # 慢查询超过10条告警
            "interval": 300,
        },
    }

    def check_via_ssh(self, server_ip, private_key):
        """通过SSH远程检查服务器"""
        import paramiko
        ssh = paramiko.SSHClient()
        ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
        ssh.connect(server_ip, username="root", key_filename=private_key)

        alerts = []
        for item_name, config in self.MONITOR_ITEMS.items():
            if "cmd" in config:
                stdin, stdout, stderr = ssh.exec_command(config["cmd"])
                output = stdout.read().decode().strip()
                value = self.parse_value(output)
                if value > config.get("threshold", 0):
                    alerts.append({
                        "source": f"cloud_server:{server_ip}",
                        "item": item_name,
                        "value": value,
                        "threshold": config["threshold"],
                        "timestamp": datetime.now().isoformat(),
                    })

        ssh.close()
        return alerts
```

```python
class BarkNotifier:
    """Bark推送统一接口"""

    BARK_KEY = "YOUR_BARK_KEY"  # 从Bark App获取
    BARK_SERVER = "https://api.day.app"

    # 告警级别与图标映射
    ALERT_LEVELS = {
        "info": {
            "icon": "🔵",
            "sound": "minimal",
            "group": "龍魂-信息",
        },
        "warning": {
            "icon": "🟡",
            "sound": "alarm",
            "group": "龍魂-警告",
        },
        "error": {
            "icon": "🔴",
            "sound": "emergency",
            "group": "龍魂-严重",
        },
        "critical": {
            "icon": "🚨",
            "sound": "critical",
            "group": "龍魂-紧急",
        },
        "success": {
            "icon": "✅",
            "sound": "success",
            "group": "龍魂-成功",
        },
    }

    def send(self, title: str, body: str, level: str = "info", 
             url: str = None, badge: int = 1):
        """
        发送Bark推送

        Args:
            title: 推送标题
            body: 推送内容
            level: 告警级别 (info/warning/error/critical/success)
            url: 点击跳转URL
            badge: 角标数字
        """
        import requests

        config = self.ALERT_LEVELS.get(level, self.ALERT_LEVELS["info"])

        payload = {
            "title": f"{config['icon']} {title}",
            "body": body,
            "sound": config["sound"],
            "group": config["group"],
            "badge": badge,
            "icon": "https://longhun-system.com/logo.png",
        }

        if url:
            payload["url"] = url

        try:
            response = requests.post(
                f"{self.BARK_SERVER}/{self.BARK_KEY}",
                json=payload,
                timeout=10
            )
            return response.status_code == 200
        except Exception as e:
            print(f"Bark推送失败: {e}")
            return False

    def send_alert(self, alert: dict):
        """根据告警字典自动发送推送"""
        # 判定级别
        level = self.determine_level(alert)

        # 格式化内容
        title, body = self.format_alert(alert)

        # 发送
        return self.send(title, body, level)

    def determine_level(self, alert: dict) -> str:
        """根据告警内容判定级别"""
        item = alert.get("item", "")
        value = alert.get("value", 0)
        threshold = alert.get("threshold", 0)

        # 服务异常/重启失败 = 严重
        if "restart" in item.lower() or "fail" in item.lower():
            return "critical"

        # 安全相关 = 紧急
        if "login" in item.lower() or "sudo" in item.lower() or "security" in item.lower():
            return "critical"

        # 资源超过阈值 = 警告
        if value and threshold and value > threshold:
            ratio = value / threshold
            if ratio > 2:
                return "critical"
            elif ratio > 1.5:
                return "error"
            else:
                return "warning"

        # 开源平台变化 = 信息
        if alert.get("source") in ["github", "gitee"]:
            return "info"

        return "warning"

    def format_alert(self, alert: dict) -> Tuple[str, str]:
        """格式化告警内容"""
        source = alert.get("source", "unknown")
        item = alert.get("item", "unknown")

        # 本地服务器
        if source == "local_server":
            title = f"本地服务器 · {item}"
            body = f"监控项: {item}
"
            if "value" in alert:
                body += f"当前值: {alert['value']}
"
            if "threshold" in alert:
                body += f"阈值: {alert['threshold']}
"
            body += f"时间: {alert.get('timestamp', 'now')}"

        # 云端服务器
        elif source.startswith("cloud_server"):
            server = source.split(":")[1]
            title = f"云端服务器({server}) · {item}"
            body = f"服务器: {server}
监控项: {item}
"
            if "value" in alert:
                body += f"当前值: {alert['value']}
"
            body += f"时间: {alert.get('timestamp', 'now')}"

        # GitHub
        elif source == "github":
            if "stars" in item:
                title = f"GitHub · ⭐ Star变化"
                body = f"当前Stars: {alert.get('current', 0)}
"
                body += f"变化: {alert.get('change', 0):+d}
"
            elif "forks" in item:
                title = f"GitHub · 🍴 Fork变化"
                body = f"当前Forks: {alert.get('current', 0)}
"
                body += f"变化: {alert.get('change', 0):+d}
"
            elif "issues" in item:
                title = f"GitHub · 🐛 新Issue"
                body = f"新Issue已创建
"
            elif "pull" in item:
                title = f"GitHub · 🔀 新PR"
                body = f"新Pull Request已提交
"
            elif "clone" in item:
                title = f"GitHub · 📥 代码被复刻"
                body = f"发现疑似复刻仓库:
"
                for clone in alert.get("clones", []):
                    body += f"  - {clone['repo']}
"
            else:
                title = f"GitHub · {item}"
                body = str(alert)
            body += f"时间: {alert.get('timestamp', 'now')}"

        # Gitee
        elif source == "gitee":
            title = f"Gitee · {item}"
            body = str(alert)

        else:
            title = f"龍魂系统 · {item}"
            body = str(alert)

        return title, body
```

```python
#!/usr/bin/env python3
# lh_global_monitor.py
# 龍魂全局监控编排器 · 每分钟检查所有源

class GlobalMonitor:
    """全局监控编排器"""

    def __init__(self):
        self.local_monitor = LocalServerMonitor()
        self.cloud_monitor = CloudServerMonitor()
        self.os_monitor = OpenSourceMonitor()
        self.notifier = BarkNotifier()

        # 服务器配置
        self.cloud_servers = [
            {"ip": "your.huawei.ip", "key": "~/.ssh/huawei_key"},
        ]

        # GitHub Token
        self.github_token = os.getenv("GITHUB_TOKEN", "")

    def run_cycle(self):
        """执行一轮监控检查"""
        all_alerts = []

        # 1. 检查本地服务器
        print("[监控] 检查本地服务器...")
        local_alerts = self.local_monitor.check_all()
        all_alerts.extend(local_alerts)

        # 2. 检查云端服务器
        for server in self.cloud_servers:
            print(f"[监控] 检查云端服务器 {server['ip']}...")
            try:
                cloud_alerts = self.cloud_monitor.check_via_ssh(
                    server["ip"], server["key"]
                )
                all_alerts.extend(cloud_alerts)
            except Exception as e:
                all_alerts.append({
                    "source": f"cloud_server:{server['ip']}",
                    "item": "connection_failed",
                    "error": str(e),
                    "timestamp": datetime.now().isoformat(),
                })

        # 3. 检查GitHub
        if self.github_token:
            print("[监控] 检查GitHub...")
            github_alerts = self.os_monitor.check_github(self.github_token)
            all_alerts.extend(github_alerts)

            # 检查代码复刻
            print("[监控] 检查代码复刻...")
            clones = self.os_monitor.check_code_clones(self.github_token)
            if clones:
                all_alerts.append({
                    "source": "github",
                    "item": "code_clone_detection",
                    "clones": clones,
                    "timestamp": datetime.now().isoformat(),
                })

        # 4. 发送推送
        print(f"[监控] 发现 {len(all_alerts)} 条告警，发送推送...")
        for alert in all_alerts:
            self.notifier.send_alert(alert)

        return len(all_alerts)

    def run_forever(self, interval=60):
        """持续运行"""
        print(f"[龍魂监控] 启动全局监控，检查间隔: {interval}秒")
        while True:
            try:
                count = self.run_cycle()
                print(f"[龍魂监控] 本轮完成，{count}条告警已推送")
            except Exception as e:
                print(f"[龍魂监控] 本轮异常: {e}")
                self.notifier.send(
                    "龍魂监控器异常",
                    f"监控编排器发生异常:
{e}",
                    level="critical"
                )

            time.sleep(interval)


# 主入口
if __name__ == "__main__":
    monitor = GlobalMonitor()
    monitor.run_forever(interval=60)  # 每60秒检查一轮
```

### 6.1 本地部署（Mac M4）

```bash
# 1. 安装依赖
pip install requests paramiko

# 2. 创建监控目录
mkdir -p ~/.longhun/monitor
mkdir -p ~/.longhun/monitor/logs

# 3. 放置脚本
cp lh_global_monitor.py ~/.longhun/monitor/
cp lh_local_monitor.py ~/.longhun/monitor/
cp lh_cloud_monitor.py ~/.longhun/monitor/
cp lh_os_monitor.py ~/.longhun/monitor/
cp lh_bark_notifier.py ~/.longhun/monitor/

# 4. 配置Bark Key
export BARK_KEY="your_bark_key_here"
export GITHUB_TOKEN="your_github_token_here"

# 5. 测试运行
python3 ~/.longhun/monitor/lh_global_monitor.py --test

# 6. 设置后台运行
nohup python3 ~/.longhun/monitor/lh_global_monitor.py > ~/.longhun/monitor/logs/monitor.log 2>&1 &

# 7. 设置开机自启
# 创建LaunchAgent plist
cat > ~/Library/LaunchAgents/com.longhun.monitor.plist << 'EOF'
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
    <key>Label</key>
    <string>com.longhun.monitor</string>
    <key>ProgramArguments</key>
    <array>
        <string>/usr/bin/python3</string>
        <string>/Users/YOUR_USERNAME/.longhun/monitor/lh_global_monitor.py</string>
    </array>
    <key>RunAtLoad</key>
    <true/>
    <key>KeepAlive</key>
    <true/>
    <key>StandardOutPath</key>
    <string>/Users/YOUR_USERNAME/.longhun/monitor/logs/monitor.log</string>
    <key>StandardErrorPath</key>
    <string>/Users/YOUR_USERNAME/.longhun/monitor/logs/monitor.error.log</string>
</dict>
</plist>
EOF

launchctl load ~/Library/LaunchAgents/com.longhun.monitor.plist
```

### 6.2 云端部署（华为鲲鹏）

```bash
# 1. SSH到华为云服务器
ssh -i ~/.ssh/huawei_key root@your.huawei.ip

# 2. 安装Python和依赖
apt update && apt install -y python3 python3-pip
pip3 install requests paramiko

# 3. 创建监控目录
mkdir -p /opt/longhun/monitor
mkdir -p /opt/longhun/monitor/logs

# 4. 放置脚本（从本地scp）
scp -i ~/.ssh/huawei_key ~/.longhun/monitor/*.py root@your.huawei.ip:/opt/longhun/monitor/

# 5. 配置环境变量
echo 'export BARK_KEY="your_bark_key"' >> /etc/environment
echo 'export GITHUB_TOKEN="your_github_token"' >> /etc/environment

# 6. 创建systemd服务
cat > /etc/systemd/system/longhun-monitor.service << 'EOF'
[Unit]
Description=LongHun Global Monitor
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/longhun/monitor
ExecStart=/usr/bin/python3 /opt/longhun/monitor/lh_global_monitor.py
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable longhun-monitor
systemctl start longhun-monitor

# 7. 查看状态
systemctl status longhun-monitor
journalctl -u longhun-monitor -f
```
