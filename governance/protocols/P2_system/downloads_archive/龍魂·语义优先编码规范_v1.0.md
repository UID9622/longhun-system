```python
def calculate_weighted_average(values: list, weights: list, divisor: float) -> float:
    """
    计算加权平均值
    假设：values和weights长度相等，divisor > 0
    DNA: #龍芯⚡️丙午·甲午·戊寅·戊午·䷕贲-WEIGHTED-AVG-v1.0
    """
    weighted_sum = sum(v * w for v, w in zip(values, weights))
    return weighted_sum / divisor
```

```python
char_code_sum = sum(ord(c) for c in content)  # 第一步：内容→数字和
```

```bash
#!/bin/bash
# longhun_scan_orphan.sh
# DNA: #龍芯⚡️丙午·甲午·戊寅·戊午·䷕贲-SCAN-ORPHAN-v1.0

echo "🐉 龍魂孤立文件扫描器"

# 扫描 ~/longhun-system 下所有文件
# 检查每个文件是否被其他文件引用（import/include/提及）

cd ~/longhun-system

for file in $(find . -type f \( -name "*.py" -o -name "*.md" -o -name "*.sh" -o -name "*.json" \)); do
    filename=$(basename "$file")
    # 检查文件名是否被其他文件引用
    refs=$(grep -rl "$filename" --include="*.py" --include="*.md" --include="*.sh" . | grep -v "$file" | wc -l)
    
    if [ "$refs" -eq 0 ]; then
        echo "🔴 孤立: $file (未被任何文件引用)"
    elif [ "$refs" -eq 1 ]; then
        echo "🟡 弱关联: $file (仅被1个文件引用)"
    else
        echo "🟢 正常: $file (被 $refs 个文件引用)"
    fi
done

echo "========================"
echo "扫描完成 · 请检查 🔴 和 🟡 标记的文件"
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
龍魂 · 语义规范检查器
DNA: #龍芯⚡️丙午·甲午·戊寅·戊午·䷕贲-SEMANTIC-CHECKER-v1.0
功能：自动检查代码/文档是否符合语义优先规范
"""

import os
import re
import json
from pathlib import Path
from datetime import datetime

# 配置
LONGHUN_ROOT = Path.home() / "longhun-system"
LOG_FILE = LONGHUN_ROOT / "logs" / "semantic_audit.jsonl"

# 检查规则
RULES = {
    "R1": {
        "name": "模块归属声明",
        "pattern": r"所属模块:|所属模块：",
        "severity": "🔴",
        "fix": "在文件头添加模块归属声明（见铁律5模板）"
    },
    "R2": {
        "name": "DNA追溯码",
        "pattern": r"#龍芯⚡️\d{4}-\d{2}-\d{2}-",
        "severity": "🔴",
        "fix": "添加DNA追溯码：#龍芯⚡️YYYY-MM-DD-主题-vX.X"
    },
    "R3": {
        "name": "显式假设声明",
        "pattern": r"假设：|假设:|Assumptions:|假设",
        "severity": "🟡",
        "fix": "在函数注释中添加假设和边界条件说明"
    },
    "R4": {
        "name": "函数命名语义化",
        "pattern": None,  # 特殊检查
        "severity": "🟡",
        "fix": "函数名应使用动词+名词结构，避免缩写"
    },
    "R5": {
        "name": "关联页面链接",
        "pattern": r"关联页面:|关联页面：|notion\.site",
        "severity": "🟡",
        "fix": "添加关联Notion页面链接"
    },
    "R6": {
        "name": "创建者信息",
        "pattern": r"创建者:|创建者：|UID9622",
        "severity": "🟢",
        "fix": "添加创建者信息"
    }
}


def check_file(filepath: Path) -> list:
    """检查单个文件是否符合规范"""
    issues = []
    content = filepath.read_text(encoding='utf-8', errors='ignore')
    
    # R1: 模块归属声明
    if not re.search(RULES["R1"]["pattern"], content):
        issues.append({"rule": "R1", **{k: v for k, v in RULES["R1"].items() if k != "pattern"}})
    
    # R2: DNA追溯码
    if not re.search(RULES["R2"]["pattern"], content):
        issues.append({"rule": "R2", **{k: v for k, v in RULES["R2"].items() if k != "pattern"}})
    
    # R3: 显式假设
    if not re.search(RULES["R3"]["pattern"], content):
        issues.append({"rule": "R3", **{k: v for k, v in RULES["R3"].items() if k != "pattern"}})
    
    # R4: 函数命名（简单检查：查找单字母函数名或calc/tmp等模糊命名）
    vague_names = re.findall(r'def\s+(calc|tmp|temp|x|y|z|fn|func)\s*\(', content)
    if vague_names:
        issues.append({
            "rule": "R4",
            "name": "函数命名语义化",
            "severity": "🟡",
            "fix": f"发现模糊函数名: {vague_names}，请使用动词+名词结构",
            "detail": vague_names
        })
    
    # R5: 关联页面
    if not re.search(RULES["R5"]["pattern"], content):
        issues.append({"rule": "R5", **{k: v for k, v in RULES["R5"].items() if k != "pattern"}})
    
    return issues


def scan_all():
    """扫描整个龍魂系统"""
    results = []
    total_files = 0
    total_issues = 0
    
    for ext in ["*.py", "*.md", "*.sh", "*.js"]:
        for filepath in LONGHUN_ROOT.rglob(ext):
            total_files += 1
            issues = check_file(filepath)
            total_issues += len(issues)
            
            if issues:
                results.append({
                    "file": str(filepath.relative_to(LONGHUN_ROOT)),
                    "issues": issues
                })
                
                # 打印问题
                print(f"\n🟡 {filepath.relative_to(LONGHUN_ROOT)}")
                for issue in issues:
                    print(f"   {issue['severity']} [{issue['rule']}] {issue['name']}")
                    print(f"      → {issue['fix']}")
    
    # 写入日志
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    audit_entry = {
        "time": datetime.now().isoformat(),
        "dna": "#龍芯⚡️丙午·甲午·戊寅·戊午·䷕贲-SEMANTIC-CHECK-v1.0",
        "total_files": total_files,
        "files_with_issues": len(results),
        "total_issues": total_issues,
        "compliance_rate": f"{((total_files - len(results)) / total_files * 100):.1f}%" if total_files > 0 else "N/A",
        "results": results
    }
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(json.dumps(audit_entry, ensure_ascii=False) + "\n")
    
    # 汇总
    print("\n" + "="*60)
    print("🐉 龍魂语义规范检查结果")
    print("="*60)
    print(f"扫描文件总数: {total_files}")
    print(f"合规文件数: {total_files - len(results)}")
    print(f"问题文件数: {len(results)}")
    print(f"总问题数: {total_issues}")
    print(f"合规率: {audit_entry['compliance_rate']}")
    print(f"日志位置: {LOG_FILE}")
    print("="*60)


if __name__ == "__main__":
    print("🐉 龍魂 · 语义规范检查器 v1.0")
    print("DNA: #龍芯⚡️丙午·甲午·戊寅·戊午·䷕贲-SEMANTIC-CHECKER-v1.0")
    print("="*60)
    scan_all()
```
