   * TypeScript 核心實現                                                   
   * DNA: #龍芯⚡️2026-06-07-MOBILE-MONITORING-SDK

  export interface MonitoringConfig {                                      
    appId: string;                                                         
    appName: string;                                                       
    version: string;                                                       
    environment: 'development' | 'staging' | 'production';                 
    logEndpoint: string;                                                   
    batchSize: number;                                                     
    flushInterval: number;                                                 
    enableEncryption: boolean;                                             
    encryptionKey?: string;                                                
  }                                                                        
                                                                           
  export interface PerformanceMetrics {                                    
    navigationTiming: PerformanceNavigationTiming;                         
    resourceTiming: PerformanceResourceTiming[];                           
    memoryUsage: MemoryMetrics;                                            
    fpsFps: number;                                                        
    networkLatency: number;                                                
    batteryLevel: number;                                                  
  }                                                                        
                                                                           
  export interface BehavioralSignature {                                   
    userId: string;                                                        
    timestamp: number;                                                     
    deviceId: string;                                                      
    sessionId: string;                                                     
    fingerprint: string;                                                   
    dnaHash: string;                                                       
  }                                                                        
                                                                           
  export interface ErrorReport {                                           
    type: 'js_error' | 'network_error' | 'promise_rejection' |             
  'custom_error';                                                          
    message: string;                                                       
    stack: string;                                                         
    context: Record<string, any>;                                          
    timestamp: number;                                                     
  }                                                                        
                                                                           
  export class LonghunMonitor {                                            
    private config: MonitoringConfig;                                      
    private queue: any[] = [];                                             
    private sessionId: string;                                             
    private deviceId: string;                                              
    private behavioralSignature: BehavioralSignature;                      
                                                                           
    constructor(config: MonitoringConfig) {                                
      this.config = config;                                                
      this.sessionId = this.generateSessionId();                           
      this.deviceId = this.getOrCreateDeviceId();                          
      this.init();                                                         
    }                                                                      
                                                                           
    private init(): void {                                                 
      console.log('🐉 龍魂監控 SDK 初始化中...');                          
                                                                           
      // 自動捕捉全局錯誤                                                  
      this.setupErrorHandlers();                                           
                                                                           
      // 自動採集性能數據                                                  
      this.setupPerformanceTracking();                                     
                                                                           
      // 自動採集用戶行為                                                  
      this.setupBehaviorTracking();                                        
                                                                           
      // 定時批量上報                                                      
      this.startBatchReporter();                                           
                                                                           
      console.log('✅ 龍魂監控 SDK 初始化完成');                           
    }                                                                      
                                                                           
    private setupErrorHandlers(): void {                                   
      // 捕捉 JS 錯誤                                                      
      window.addEventListener('error', (event) => {                        
        this.captureError({                                                
          type: 'js_error',                                                
          message: event.message,                                          
          stack: event.error?.stack || '',                                 
          context: { filename: event.filename, lineno: event.lineno },     
          timestamp: Date.now()                                            
        });                                                                
      });                                                                  
                                                                           
      // 捕捉未處理的 Promise 拒絕                                         
      window.addEventListener('unhandledrejection', (event) => {           
        this.captureError({                                                
          type: 'promise_rejection',                                       
          message: String(event.reason),                                   
          stack: event.reason?.stack || '',                                
          context: {},                                                     
          timestamp: Date.now()                                            
        });                                                                
      });                                                                  
    }                                                                      
                                                                           
    private setupPerformanceTracking(): void {                             
      if ('PerformanceObserver' in window) {                               
        const observer = new PerformanceObserver((list) => {               
          for (const entry of list.getEntries()) {                         
            this.trackMetric(entry);                                       
          }                                                                
        });                                                                
                                                                           
        observer.observe({ entryTypes: ['navigation', 'resource', 'paint'] 
   });                                                                     
      }                                                                    
    }                                                                      
                                                                           
    private setupBehaviorTracking(): void {                                
      // 追踪點擊事件                                                      
      document.addEventListener('click', (e) => {                          
        this.trackBehavior({                                               
          type: 'click',                                                   
          target: (e.target as HTMLElement).tagName,                       
          timestamp: Date.now()                                            
        });                                                                
      });                                                                  
                                                                           
      // 追踪滑動事件                                                      
      document.addEventListener('scroll', () => {                          
        this.trackBehavior({                                               
          type: 'scroll',                                                  
          position: window.scrollY,                                        
          timestamp: Date.now()                                            
        });                                                                
      }, { passive: true });                                               
    }                                                                      
                                                                           
    private startBatchReporter(): void {                                   
      setInterval(() => {                                                  
        this.flush();                                                      
      }, this.config.flushInterval);                                       
    }                                                                      
                                                                           
    public captureError(error: ErrorReport): void {                        
      this.queue.push({                                                    
        type: 'error',                                                     
        data: error,                                                       
        signature: this.behavioralSignature                                
      });                                                                  
    }                                                                      
                                                                           
    public trackMetric(metric: any): void {                                
      this.queue.push({                                                    
        type: 'metric',                                                    
        data: metric,                                                      
        signature: this.behavioralSignature                                
      });                                                                  
    }                                                                      
                                                                           
    public trackBehavior(behavior: any): void {                            
      this.queue.push({                                                    
        type: 'behavior',                                                  
        data: behavior,                                                    
        signature: this.behavioralSignature                                
      });                                                                  
    }                                                                      
                                                                           
    public async flush(): Promise<void> {                                  
      if (this.queue.length === 0) return;                                 
                                                                           
      const batch = this.queue.splice(0, this.config.batchSize);           
                                                                           
      try {                                                                
        const payload = {                                                  
          appId: this.config.appId,                                        
          sessionId: this.sessionId,                                       
          deviceId: this.deviceId,                                         
          timestamp: Date.now(),                                           
          events: batch                                                    
        };                                                                 
                                                                           
        // 加密（如果啟用）                                                
        const data = this.config.enableEncryption                          
          ? this.encrypt(JSON.stringify(payload))                          
          : JSON.stringify(payload);                                       
                                                                           
        await fetch(this.config.logEndpoint, {                             
          method: 'POST',                                                  
          headers: {                                                       
            'Content-Type': 'application/json',                            
            'X-App-Id': this.config.appId,                                 
            'X-Session-Id': this.sessionId                                 
          },                                                               
          body: data                                                       
        });                                                                
                                                                           
        console.log(`✅ 已上報 ${batch.length} 個事件`);                   
      } catch (error) {                                                    
        console.error('❌ 上報失敗:', error);                              
        // 重新放入隊列                                                    
        this.queue.unshift(...batch);                                      
      }                                                                    
    }                                                                      
                                                                           
    private encrypt(data: string): string {                                
      // AES-256-GCM 加密實現                                              
      // 本示例為簡化版本                                                  
      return btoa(data);                                                   
    }                                                                      
                                                                           
    private generateSessionId(): string {                                  
      return `session_${Date.now()}_${Math.random().toString(36).substr(2, 
   9)}`;                                                                   
    }                                                                      
                                                                           
    private getOrCreateDeviceId(): string {                                
      let deviceId = localStorage.getItem('longhun_device_id');            
      if (!deviceId) {                                                     
        deviceId =                                                         
  `device_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;       
        localStorage.setItem('longhun_device_id', deviceId);               
      }                                                                    
      return deviceId;                                                     
    }                                                                      
  }

    }                                                                      
  }                                                                        
                                                                           
  window.LonghunMonitor = LonghunMonitor;                                  
                                                                           
  export default LonghunMonitor;

  from pydantic import BaseModel

  @app.post("/api/v1/monitor/events")                                      
  async def receive_events(

      x_session_id: str = Header(None)

          # 檢查告警規則                                                   
          await check_alert_rules(payload.appId, event)                    
                                                                           
      logger.info(f"✅ 已存儲 {len(payload.events)} 個事件")               
                                                                           
      return {                                                             
          "status": "success",                                             
          "message": f"Received {len(payload.events)} events",             
          "batchId": hashlib.sha256(                                       
              f"{payload.sessionId}{payload.timestamp}".encode()           
          ).hexdigest()[:16]                                               
      }                                                                    
                                                                           
  @app.get("/api/v1/monitor/events/{app_id}")                              
  async def get_events(                                                    
      app_id: str,                                                         
      event_type: str = None,                                              
      limit: int = 100                                                     
  ) -> Dict[str, Any]:                                                     
      """查詢監控事件"""                                                   
                                                                           
      events = events_store.get(app_id, [])                                
                                                                           
      # 過濾                                                               
      if event_type:                                                       
          events = [e for e in events if e['type'] == event_type]          
                                                                           
      # 限制數量                                                           
      events = events[-limit:]                                             
                                                                           
      return {                                                             
          "appId": app_id,                                                 
          "total": len(events),                                            
          "events": events                                                 
      }

  async def create_alert_rule(rule: AlertRule) -> Dict[str, str]:          
      """創建告警規則"""                                                   
                                                                           
      rules_store[rule.id] = rule.dict()                                   
      logger.info(f"🚨 告警規則已創建: {rule.name}")                       
                                                                           
      return {                                                             
          "status": "success",                                             
          "ruleId": rule.id,                                               
          "message": f"Alert rule '{rule.name}' created"                   
      }                                                                    
                                                                           
  @app.get("/api/v1/monitor/alerts/{app_id}")                              
  async def get_alerts(                                                    
      app_id: str,                                                         
      severity: str = None,                                                
      limit: int = 50                                                      
  ) -> Dict[str, Any]:                                                     
      """查詢告警"""                                                       
                                                                           
      app_alerts = alerts_store.get(app_id, [])                            
                                                                           
      if severity:                                                         
          app_alerts = [a for a in app_alerts if a['severity'] ==          
  severity]                                                                
                                                                           
      app_alerts = app_alerts[-limit:]                                     
                                                                           
      return {                                                             
          "appId": app_id,                                                 
          "total": len(app_alerts),                                        
          "alerts": app_alerts                                             
      }                                                                    
                                                                           
  @app.get("/api/v1/monitor/health")                                       
  async def health_check() -> Dict[str, str]:                              
      """健康檢查"""                                                       
      return {                                                             
          "status": "healthy",                                             
          "timestamp": datetime.now().isoformat(),                         
          "version": "4.1"                                                 
      }                                                                    
                                                                           
  async def check_alert_rules(app_id: str, event: Event) -> None:          
      """檢查告警規則"""                                                   
                                                                           
      for rule_id, rule in rules_store.items():                            
          if not rule.get('enabled'):                                      
              continue                                                     
                                                                           
          # 簡化的規則檢查邏輯                                             
          if event.type == rule.get('metric'):                             
              alert = {                                                    
                  'ruleId': rule_id,                                       
                  'ruleName': rule.get('name'),                            
                  'appId': app_id,                                         
                  'severity': rule.get('severity'),                        
                  'triggeredAt': datetime.now().isoformat(),               
                  'event': event.dict()                                    
              }                                                            
                                                                           
              if app_id not in alerts_store:                               
                  alerts_store[app_id] = []                                
                                                                           
              alerts_store[app_id].append(alert)                           
              logger.warning(f"🚨 告警觸發: {rule.get('name')}             
  (severity={rule.get('severity')})")                                      
                                                                           
  if __name__ == '__main__':                                               
      import uvicorn
