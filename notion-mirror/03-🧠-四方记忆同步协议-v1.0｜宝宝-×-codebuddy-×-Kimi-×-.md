## §8 一键复制指令（给 codebuddy）

```bash
# 任务：生成本地资产全量清单，供 Notion 台账对齐
# 铁律：只读不写；不进 ~/Desktop/整理后/ （含私人文件）；不上传任何内容

cd ~/longhun-system

{
  echo "# LOCAL_ASSET_INVENTORY  $(date '+%Y-%m-%d %H:%M:%S')"
  echo ""
  echo "## 1. 目录级统计（文件数 / 大小）"
  for d in 01_protocols 08_BIN 12_DOCS 15_LABS articles papers brain config src scripts web_apps; do
    [ -d "$d" ] && printf "%-32s %6s files  %8s\n" "$d" "$(find "$d" -type f | wc -l | tr -d ' ')" "$(du -sh "$d" | cut -f1)"
  done
  echo ""
  echo "## 2. 最近 30 天改动过的文件"
  find . -type f -mtime -30 -not -path "./.git/*" | head -200
  echo ""
  echo "## 3. 所有 md 文档标题行（用于台账去重）"
  find . -name "*.md" -not -path "./.git/*" -exec sh -c 'echo "[$1] $(head -1 "$1")"' _ {} \; | head -400
  echo ""
  echo "## 4. SQLite 清点（表名 + 行数）"
  for db in $(find . -name "*.db" -not -path "./.git/*"); do
    echo "### $db"
    sqlite3 "$db" ".tables" 2>/dev/null
    for t in $(sqlite3 "$db" ".tables" 2>/dev/null); do
      echo "  $t: $(sqlite3 "$db" "SELECT COUNT(*) FROM \"$t\";" 2>/dev/null) rows"
    done
  done
} > LOCAL_ASSET_INVENTORY.md

echo "✅ 已生成 LOCAL_ASSET_INVENTORY.md，请把它粘给宝宝"
wc -l LOCAL_ASSET_INVENTORY.md
```

拿到这份清单后，我能把 🔴 仅本地·未上台账 那三行大包拆成逐件可追的真行。
