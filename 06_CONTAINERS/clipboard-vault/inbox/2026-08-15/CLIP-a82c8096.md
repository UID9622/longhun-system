@app.post("/api/browser/command")
async def execute_command(request: Request):
    """执行浏览器指令 (小艺/Kimi/CodeBuddy调用)"""
    data = await request.json()
    text = data.get("text", "")
    source = data.get("source", "unknown")
    dna = generate_dna("CMD")

    # 1. 通心译转译
    command = translate_command(text)

    if command.get("action") == "unknown":
        record_historian("unknown_command", dna, {"text": text, "source": source})
        return {
            "status": "error",
            "message": f"无法识别指令: {text}",
            "dna": dna,
            "suggestions": list(TONXINYI_BROWSER_MAP.keys())[:5]
        }

    # 2. 提取参数 (从自然语言中)
    params = command.get("params", {}).copy()
    # 尝试从文本中提取数值参数
    import re
    # 提取数字
    numbers = re.findall(r'\d+', text)
    if "set_viewport" in command.get("action", ""):
        if len(numbers) >= 2:
            params["width"] = int(numbers[0])
            params["height"] = int(numbers[1])
    elif "set_geolocation" in command.get("action", ""):
        if len(numbers) >= 2:
            params["latitude"] = float(numbers[0])
            params["longitude"] = float(numbers[1])

    # 3. 转发到Mac
    result = forward_to_mac(command["action"], params)

    # 4. 记录史官
    record_historian(command["action"], dna, {
        "source": source,
        "text": text,
        "params": params,
        "mac_result": result
    })

    # 5. 三色审计
    tricolor = "🟢"
    if result.get("status") == "error":
        tricolor = "🔴"
        write_shame_wall("browser_command_failed", {
            "source": source,
            "text": text,
            "error": result.get("message")
        })
    elif "warning" in result.get("status", ""):
        tricolor = "🟡"

    return {
        "status": "success" if result.get("status") != "error" else "error",
        "dna": dna,
        "command": command["action"],
        "text": text,
        "source": source,
        "result": result,
        "tricolor": tricolor,
        "timestamp": datetime.now().isoformat()
    }
