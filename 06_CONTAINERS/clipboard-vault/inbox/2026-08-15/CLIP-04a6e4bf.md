class LocalEncryption:
    """本地加密存储 - 数据永远不出境"""
    
    def __init__(self):
        self.key_path = ENCRYPT_DIR / "master.key"
        self._ensure_key()
    
    def _ensure_key(self):
        """确保加密密钥存在"""
        if not self.key_path.exists():
            # 生成32字节密钥
            import secrets
            key = secrets.token_bytes(32)
            with open(self.key_path, 'wb') as f:
                f.write(key)
            logger.info("🔑 主密钥已生成，保存于本地")
    
    def encrypt(self, data: str) -> Dict:
        """加密数据"""
        from cryptography.fernet import Fernet
        with open(self.key_path, 'rb') as f:
            key = f.read()
        fernet = Fernet(key)
        encrypted = fernet.encrypt(data.encode('utf-8'))
        return {
            "encrypted": encrypted.hex(),
            "algorithm": "Fernet(AES-128)",
            "stored_locally": True
        }
    
    def decrypt(self, encrypted_data: str) -> str:
        """解密数据"""
        from cryptography.fernet import Fernet
        with open(self.key_path, 'rb') as f:
            key = f.read()
        fernet = Fernet(key)
        decrypted = fernet.decrypt(bytes.fromhex(encrypted_data))
        return decrypted.decode('utf-8')

        """检查ADB是否可用"""

            result = subprocess.run(["adb", "version"], capture_output=True, text=True)

    def _get_device(self) -> str:
        """获取设备ID"""
        try:
            result = subprocess.run(["adb", "devices"], capture_output=True, text=True)
            lines = result.stdout.strip().split('\n')
            for line in lines:
                if "device" in line and "List" not in line:
                    return line.split()[0]
        except:
            pass
        return ""
    
    def execute(self, action: str, params: Dict = None) -> Dict:
        """执行手机操作"""

        """智能去重"""

        unique_items = []

            content_hash = hashlib.sha256(content.encode()).hexdigest()[:16]
