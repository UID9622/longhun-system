### 3.2 平台调度器实现 (`platform_dispatcher.py`)

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
🐉 龍魂浏览器托管 · 平台调度器
自动识别指令对应的平台，分发给各个平台执行器

DNA: #龍芯⚡️丙午·丙申·辛酉·丑时-PLATFORM-DISPATCH-UID9622
"""

import yaml
import re
from pathlib import Path
from typing import Dict, List, Optional, Tuple
from dataclasses import dataclass

@dataclass
class PlatformAction:
    platform: str
    action: str
    target: str
    params: Dict

class PlatformDispatcher:
    """平台指令调度器"""

    def __init__(self, config_path: str = None):
        self.config_path = config_path or Path(__file__).parent / "platforms.yaml"
        self.platforms = self._load_config()
        self._build_routing_table()

    def _load_config(self) -> Dict:
        """加载平台配置"""
        with open(self.config_path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)

    def _build_routing_table(self):
        """构建平台关键词路由表"""
        self.routing_table = {}
        for key, platform in self.platforms.get("platforms", {}).items():
            name = platform.get("name", "")
            description = platform.get("description", "")
            # 生成关键词
            keywords = [name.lower()]
            # 添加描述中的关键词
            for word in description.split():
                if len(word) > 2:
                    keywords.append(word.lower())
            # 添加别名
            aliases = {
                "csdn": ["博客", "技术博客", "csdn"],
                "zhihu": ["知乎", "问答"],
                "wechat_mp": ["公众号", "微信公众"],
                "weibo": ["微博", "热搜"],
                "douyin": ["抖音", "短视频"],
                "bilibili": ["b站", "哔哩哔哩"],
                "github": ["github", "代码托管"],
                "gitee": ["码云", "gitee"],
                "huawei_cloud": ["华为云", "华为"],
                "aliyun": ["阿里云", "阿里"],
                "tencent_cloud": ["腾讯云", "腾讯"],
                "notion": ["notion", "知识库"],
                "feishu": ["飞书"],
                "dingtalk": ["钉钉"],
                "wecom": ["企业微信"],
            }
            for alias in aliases.get(key, []):
                keywords.append(alias.lower())
            self.routing_table[key] = {
                "keywords": keywords,
                "platform": key,
                "name": name
            }

    def dispatch(self, command: str) -> List[PlatformAction]:
        """
        解析指令，返回平台动作列表
        示例: "在CSDN发布文章，同步到知乎和公众号"
        """
        command_lower = command.lower()
        actions = []

        # 解析平台
        matched_platforms = []
        for key, info in self.routing_table.items():
            for keyword in info["keywords"]:
                if keyword in command_lower:
                    matched_platforms.append(key)
                    break

        # 如果没有匹配到，尝试提取所有平台名
        if not matched_platforms:
            for key, info in self.routing_table.items():
                if info["name"].lower() in command_lower:
                    matched_platforms.append(key)

        # 如果还是没有，默认CSDN
        if not matched_platforms:
            matched_platforms = ["csdn"]

        # 解析动作
        action_map = {
            "发布": "publish",
            "同步": "sync",
            "推送": "push",
            "提交": "submit",
            "管理": "manage",
            "查看": "view",
            "查询": "query",
            "发送": "send",
            "群发": "mass_send",
            "创建": "create",
            "编辑": "edit",
            "删除": "delete",
            "调用": "call",
        }

        action_type = "publish"  # 默认
        for key, value in action_map.items():
            if key in command_lower:
                action_type = value
                break

        # 提取目标内容（简化版）
        import re
        content_match = re.search(r'["\']([^"\']+)["\']', command)
        target = content_match.group(1) if content_match else command

        for platform in matched_platforms:
            actions.append(PlatformAction(
                platform=platform,
                action=action_type,
                target=target[:100],
                params={"full_command": command}
            ))

        return actions
```
