class TongxinyiConfig:
    """通心译配置管理器 - 支持热加载"""

    _instance = None
    _config = None
    _last_modified = None
    _lock = threading.Lock()

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self):
        self.config_path = Path(__file__).parent / "tongxinyi_config.yaml"
        self._load()

    def _load(self):
        """加载配置文件"""
        if not self.config_path.exists():
            # 创建默认配置
            self._create_default_config()
            return

        with self._lock:
            try:
                with open(self.config_path, 'r', encoding='utf-8') as f:
                    self._config = yaml.safe_load(f)
                self._last_modified = self.config_path.stat().st_mtime
                logger.info(f"✅ 通心译配置已加载: {len(self._config.get('mappings', []))} 条映射")
            except Exception as e:
                logger.error(f"配置加载失败: {e}")
                self._config = {"mappings": []}

    def _create_default_config(self):
        """创建默认配置文件"""
        default_config = {
            "version": "1.0",
            "last_updated": datetime.now().isoformat(),
            "mappings": []
        }
        with open(self.config_path, 'w', encoding='utf-8') as f:
            yaml.dump(default_config, f, allow_unicode=True, default_flow_style=False)
        self._config = default_config

    def reload_if_changed(self):
        """检测并热加载配置"""
        if not self.config_path.exists():
            return
        current_mtime = self.config_path.stat().st_mtime
        if current_mtime != self._last_modified:
            with self._lock:
                self._load()
                logger.info("🔄 通心译配置已热加载")

    def get_mappings(self) -> List[Dict]:
        """获取所有映射"""
        self.reload_if_changed()
        return self._config.get("mappings", [])

    def get_by_key(self, key: str) -> Optional[Dict]:
        """根据关键字获取映射"""
        self.reload_if_changed()
        for mapping in self._config.get("mappings", []):
            if mapping.get("key") == key:
                return mapping
        return None

    def search(self, text: str) -> Optional[Dict]:
        """在文本中搜索匹配的映射"""
        self.reload_if_changed()
        for mapping in self._config.get("mappings", []):
            if mapping.get("key") in text:
                return mapping
        return None

    def get_all_keys(self) -> List[str]:
        """获取所有关键字"""
        self.reload_if_changed()
        return [m.get("key") for m in self._config.get("mappings", []) if m.get("key")]

class SemanticEngineV2:
    """语义理解引擎 v2.0"""

    def __init__(self):
        self.model = None
        self.tongxinyi = TongxinyiConfig()
        try:
            from sentence_transformers import SentenceTransformer
            self.model = SentenceTransformer('all-MiniLM-L6-v2')
            self.available = True
        except:
            self.available = False
            logger.warning("⚠️ 语义引擎未就绪，降级为关键词匹配")

    def tongxinyi_translate(self, text: str) -> str:
        """通心译转译"""
        for key in self.tongxinyi.get_all_keys():
            if key in text:
                mapping = self.tongxinyi.get_by_key(key)
                if mapping:
                    text = text.replace(key, mapping.get("term", key))
        return text

    def tongxinyi_extract(self, text: str) -> Optional[Dict]:
        """提取通心译映射"""
        for mapping in self.tongxinyi.get_mappings():
            if mapping.get("key") in text:
                return mapping
        return None

    def similarity(self, text1: str, text2: str) -> float:
        """计算语义相似度"""
        text1 = self.tongxinyi_translate(text1)
        text2 = self.tongxinyi_translate(text2)

        if not self.available:
            return self._keyword_similarity(text1, text2)

        try:
            emb1 = self.model.encode(text1)
            emb2 = self.model.encode(text2)
            from numpy import dot
            from numpy.linalg import norm
            return float(dot(emb1, emb2) / (norm(emb1) * norm(emb2)))
        except:
            return self._keyword_similarity(text1, text2)

    def _keyword_similarity(self, text1: str, text2: str) -> float:
        words1 = set(text1.split())
        words2 = set(text2.split())
        if not words1 or not words2:
            return 0.0
        overlap = len(words1 & words2)
        return overlap / max(len(words1), len(words2))

class MergeEngineV2:
    """智能合并引擎 v2.0"""

    def __init__(self):
        self.semantic = SemanticEngineV2()
        self.similarity_threshold = 0.7
        self.stats = {
            "total_merges": 0,
            "successful_merges": 0,
            "failed_merges": 0,
            "total_savings": 0
        }

    def group_similar(self, entries: List[Dict]) -> Tuple[List[List[Dict]], List[Dict]]:
        """将相似条目分组，同时检测冲突"""
        groups = []
        conflicts = []

        for entry in entries:
            matched = False
            for group in groups:
                # 检查是否有冲突 (相同key但不同内容)
                for g in group:
                    if (entry.get("key") and g.get("key") and
                        entry["key"] == g["key"] and
                        entry.get("content") != g.get("content")):
                        conflicts.append({
                            "key": entry["key"],
                            "existing": g.get("content"),
                            "new": entry.get("content"),
                            "conflict": "content_mismatch"
                        })
                        matched = True
                        break
                if matched:
                    break

                if self.semantic.similarity(
                    entry.get("content", ""),
                    group[0].get("content", ""),
                ) >= self.similarity_threshold:
                    group.append(entry)
                    matched = True
                    break

            if not matched:
                groups.append([entry])

        return groups, conflicts

    def merge_group(self, group: List[Dict], conflict_resolution: str = "latest") -> Dict:
        """合并一组相似条目"""
        if len(group) == 1:
            return group[0]

        # 按时间排序
        sorted_group = sorted(group, key=lambda x: x.get("timestamp", ""))

        # 冲突解决策略
        if conflict_resolution == "latest":
            base = sorted_group[-1].copy()
        elif conflict_resolution == "oldest":
            base = sorted_group[0].copy()
        elif conflict_resolution == "merge_all":
            # 合并所有内容
            base = sorted_group[-1].copy()
            all_contents = [e.get("content", "") for e in sorted_group]
            base["content"] = "\n---\n".join(all_contents)
        else:
            base = sorted_group[-1].copy()

        # 补充合并信息
        base["merged_from"] = [e.get("id") for e in sorted_group[:-1]]
        base["merge_count"] = len(sorted_group)
        base["dna"] = generate_dna("MERGED")
        base["merged_at"] = datetime.now().isoformat()
        base["merge_strategy"] = conflict_resolution

        # 通心译映射
        mapping = self.semantic.tongxinyi_extract(base.get("content", ""))
        if mapping:
            base["term"] = mapping.get("key")
            base["target_file"] = mapping.get("file")
            base["target_protocol"] = mapping.get("protocol")
            base["confidence"] = mapping.get("confidence", 0.8)

        return base

    def merge_all(self, entries: List[Dict],
                  conflict_resolution: str = "latest") -> Dict:
        """合并所有条目"""
        groups, conflicts = self.group_similar(entries)
        merged = [self.merge_group(g, conflict_resolution) for g in groups]

        # 更新统计
        self.stats["total_merges"] += 1
        self.stats["successful_merges"] += 1
        self.stats["total_savings"] += len(entries) - len(merged)

        # 三色审计
        audit_result = TricolorAudit.audit_merge_result(
            len(entries), len(merged), conflicts
        )

        return {
            "original_count": len(entries),
            "group_count": len(groups),
            "merged_count": len(merged),
            "groups": groups,
            "merged": merged,
            "conflicts": conflicts,
            "savings": len(entries) - len(merged),
            "savings_rate": (len(entries) - len(merged)) / len(entries) if len(entries) > 0 else 0,
            "dna": generate_dna("MERGE-ALL"),
            "audit": {
                "color": audit_result["color"].value,
                "issues": audit_result["issues"],
                "score": audit_result["score"]
            },
            "conflict_resolution": conflict_resolution
        }

    def auto_archive(self, merged: List[Dict]) -> List[Dict]:
        """自动归档到对应位置"""
        archived = []
        for item in merged:
            target_file = item.get("target_file")
            if target_file:
                filepath = Path.home() / "longhun-system" / target_file

                        "item": item,

                        "action": "updated"
                    })
                else:
                    filepath.parent.mkdir(parents=True, exist_ok=True)
                    with open(filepath, 'w', encoding='utf-8') as f:
                        f.write(f"# 🐉 龍魂 · 自动合并产物\n")
                        f.write(f"# DNA: {item.get('dna')}\n")
                        f.write(f"# 合并自: {item.get('merged_from', [])}\n\n")
                        f.write(item.get("content", ""))
                    archived.append({
                        "item": item,
                        "file": str(filepath),
                        "action": "created"
                    })
            else:
                archive_path = Path.home() / ".longhun" / "03_MEMORY" / "merged_archive"
                archive_path.mkdir(parents=True, exist_ok=True)
                filepath = archive_path / f"merged_{int(time.time())}.json"
                with open(filepath, 'w', encoding='utf-8') as f:
                    json.dump(item, f, indent=2, ensure_ascii=False)
                archived.append({
                    "item": item,
                    "file": str(filepath),
                    "action": "archived"
                })

        return archived

    def rollback(self, dna: str) -> Dict:
        """回滚到指定DNA版本"""
        # 查找历史记录
        audit_path = Path.home() / ".longhun" / "04_AUDIT" / "semantic_merge.jsonl"
        if not audit_path.exists():
            return {"status": "failed", "error": "无历史记录"}

        target_record = None
        with open(audit_path, 'r', encoding='utf-8') as f:
            for line in f:
                try:
                    record = json.loads(line)
                    if record.get("dna") == dna:
                        target_record = record
                        break
                except:
                    pass

        if not target_record:
            return {"status": "failed", "error": f"未找到DNA: {dna}"}

        # 回滚操作
        return {
            "status": "success",
            "rolled_back_to": dna,
            "timestamp": datetime.now().isoformat(),
            "dna": generate_dna("ROLLBACK")
        }

class UnifiedCaptureMergeV2:
    """统一对话采集与合并 v2.0"""

    def __init__(self):
        self.merge_engine = MergeEngineV2()
        self.audit = TricolorAudit()
        self.historian = Historian()

    def get_pending_entries(self, source: str = None) -> List[Dict]:
        """获取待合并条目"""
        storage_dir = Path.home() / ".longhun" / "03_MEMORY" / "ai_conversations"
        entries = []
        for jsonl_file in storage_dir.rglob("*.jsonl"):
            if source and source not in str(jsonl_file):
                continue
            with open(jsonl_file, 'r', encoding='utf-8') as f:
                for line in f:
                    try:
                        data = json.loads(line)
                        if not data.get("merged", False):
                            entries.append(data)
                    except:
                        pass
        return entries

    def run(self, source: str = None, conflict_resolution: str = "latest",
            auto_feed: bool = True, dry_run: bool = False) -> Dict:
        """执行合并"""
        logger.info(f"🚀 开始语义合并 (来源: {source or '全部'})")

        # 1. 获取待合并条目
        entries = self.get_pending_entries(source)
        logger.info(f"📋 待合并: {len(entries)} 条")

        if not entries:
            return {"status": "success", "message": "无待合并条目"}

        # 2. 执行合并
        result = self.merge_engine.merge_all(entries, conflict_resolution)

        # 3. 记录史官
        self.historian.record(
            action="semantic_merge",
            dna=result["dna"],
            details={
                "original_count": result["original_count"],
                "merged_count": result["merged_count"],
                "savings": result["savings"],
                "savings_rate": result["savings_rate"],
                "conflicts": len(result["conflicts"]),
                "audit_color": result["audit"]["color"]
            }
        )

        # 4. 自动归档 (非dry-run)
        if not dry_run:
            archived = self.merge_engine.auto_archive(result["merged"])
            result["archived"] = archived

        # 5. 知识图谱反哺
        if auto_feed and not dry_run:
            feed_result = KnowledgeGraphFeeder.feed(result["merged"])
            result["knowledge_feed"] = feed_result

        # 6. 标记已合并
        if not dry_run:
            self._mark_merged(result["merged"])

        logger.info(f"✅ 合并完成: {result['merged_count']} 条, 节省 {result['savings']} 条")
        return result

    def _mark_merged(self, merged: List[Dict]):
        """标记已合并"""
        storage_dir = Path.home() / ".longhun" / "03_MEMORY" / "ai_conversations"
        merged_ids = {m.get("id") for m in merged}
        for jsonl_file in storage_dir.rglob("*.jsonl"):
            temp_file = jsonl_file.with_suffix(".tmp")
            with open(jsonl_file, 'r', encoding='utf-8') as f, open(temp_file, 'w', encoding='utf-8') as out:
                for line in f:
                    try:
                        data = json.loads(line)
                        if data.get("id") in merged_ids:
                            data["merged"] = True
                            data["merged_dna"] = next((m.get("dna") for m in merged if m.get("id") == data.get("id")), "")
                        out.write(json.dumps(data, ensure_ascii=False) + "\n")
                    except:
                        out.write(line)
            temp_file.replace(jsonl_file)
