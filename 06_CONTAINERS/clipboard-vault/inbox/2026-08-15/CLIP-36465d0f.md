class BuildPipeline:
    """零件生产线 - 代码构建"""

    def __init__(self, workspace: Path):
        self.workspace = workspace
        self.artifacts_dir = workspace / "artifacts"
        self.artifacts_dir.mkdir(parents=True, exist_ok=True)
        self.build_queue = queue.Queue()
        self.history = []

    def build(self, source_path: Path, version: str = None) -> BuildArtifact:
        """构建单个零件"""
        dna = generate_dna("BUILD")
        if version is None:
            version = datetime.now().strftime("%Y%m%d.%H%M%S")

        # 1. 复制源码到构建目录
        build_dir = self.artifacts_dir / f"build_{int(time.time())}"
        build_dir.mkdir(exist_ok=True)

        if source_path.is_file():
            shutil.copy2(source_path, build_dir / source_path.name)
        elif source_path.is_dir():
            shutil.copytree(source_path, build_dir, dirs_exist_ok=True)

        # 2. 计算产物哈希
        artifact_hash = self._compute_hash(build_dir)

        # 3. 创建产物记录
        artifact = BuildArtifact(
            id=f"ART-{int(time.time())}",
            name=source_path.name,
            path=str(build_dir),
            version=version,
            size=self._compute_size(build_dir),
            hash=artifact_hash,
            created_at=datetime.now().isoformat(),
            dna=dna
        )

        self.history.append(artifact)
        return artifact

    def _compute_hash(self, path: Path) -> str:
        """计算目录哈希"""
        import hashlib
        sha256 = hashlib.sha256()
        for f in sorted(path.rglob("*")):
            if f.is_file():
                sha256.update(f.read_bytes())
        return sha256.hexdigest()[:16]

    def _compute_size(self, path: Path) -> int:
        """计算目录大小"""
        total = 0
        for f in path.rglob("*"):
            if f.is_file():
                total += f.stat().st_size
        return total

    def get_artifacts(self, limit: int = 20) -> List[Dict]:
        return [a.to_dict() for a in self.history[-limit:]]

class TestPipeline:
    """质检流水线 - 自动测试"""

    def __init__(self, workspace: Path):
        self.workspace = workspace
        self.test_dir = workspace / "tests"
        self.reports_dir = workspace / "reports"
        self.test_dir.mkdir(parents=True, exist_ok=True)
        self.reports_dir.mkdir(parents=True, exist_ok=True)

    def run_tests(self, artifact: BuildArtifact) -> TestReport:
        """运行质检"""
        dna = generate_dna("TEST")

        # 1. 复制测试文件到构建目录
        test_target = Path(artifact.path) / "tests"
        if self.test_dir.exists():
            shutil.copytree(self.test_dir, test_target, dirs_exist_ok=True)

        # 2. 运行pytest
        result = subprocess.run(
            [sys.executable, "-m", "pytest", test_target, "-v", "--json-report"],
            capture_output=True,
            text=True,
            cwd=artifact.path,
            timeout=300
        )

        # 3. 解析结果
        report_file = Path(artifact.path) / ".pytest_cache" / ".report.json"
        report_data = {}
        if report_file.exists():
            with open(report_file) as f:
                report_data = json.load(f)

        total = report_data.get("summary", {}).get("total", 0)
        passed = report_data.get("summary", {}).get("passed", 0)
        failed = report_data.get("summary", {}).get("failed", 0)
        skipped = report_data.get("summary", {}).get("skipped", 0)
        errors = report_data.get("summary", {}).get("errors", 0)

        # 4. 计算覆盖率
        coverage = self._run_coverage(artifact.path)

        # 5. 三色审计
        tricolor = self._tricolor_audit(passed, failed, total)

        report = TestReport(
            task_id=artifact.id,
            total=total,
            passed=passed,
            failed=failed,
            skipped=skipped,
            errors=errors,
            coverage=coverage,
            tricolor=tricolor,
            dna=dna,
            timestamp=datetime.now().isoformat(),
            details=report_data.get("tests", [])
        )

        # 6. 保存报告
        report_path = self.reports_dir / f"test_report_{int(time.time())}.json"
        with open(report_path, 'w', encoding='utf-8') as f:
            json.dump(report.to_dict(), f, indent=2, ensure_ascii=False)

        return report

    def _run_coverage(self, path: Path) -> float:
        """运行覆盖率检查"""
        try:
            result = subprocess.run(
                [sys.executable, "-m", "pytest", "--cov=.", "--cov-report=json"],
                capture_output=True,
                text=True,
                cwd=path,
                timeout=120
            )
            cov_file = path / "coverage.json"
            if cov_file.exists():
                with open(cov_file) as f:
                    data = json.load(f)
                    return data.get("totals", {}).get("percent_covered", 0.0)
        except:
            pass
        return 0.0

    def _tricolor_audit(self, passed: int, failed: int, total: int) -> str:
        """三色审计"""
        if total == 0:
            return "🟡"
        if failed == 0:
            return "🟢"
        if failed <= total * 0.1:
            return "🟡"
        return "🔴"

class RepairPipeline:
    """自动修复线 - AI修复"""

    def __init__(self, workspace: Path):
        self.workspace = workspace
        self.repair_log = []

    def repair(self, report: TestReport) -> Dict:
        """自动修复"""
        dna = generate_dna("REPAIR")
        repairs = []

        if report.tricolor == "🟢":
            return {"status": "ok", "message": "无需修复"}

        # 1. 分析失败测试
        for test in report.details:
            if test.get("outcome") == "failed":
                repair_result = self._repair_test(test)
                repairs.append(repair_result)

        # 2. 生成修复报告
        return {
            "dna": dna,
            "status": "repaired" if repairs else "no_repair_needed",
            "repairs": repairs,
            "timestamp": datetime.now().isoformat()
        }

    def _repair_test(self, test: Dict) -> Dict:
        """修复单个测试"""
        # 模拟修复逻辑 - 实际可接入AI修复
        test_name = test.get("name", "unknown")
        error = test.get("call", {}).get("crash", {}).get("message", "")

        fix_suggestions = []
        if "DNA" in error or "dna" in error.lower():
            fix_suggestions.append("添加或修复DNA追溯码")
        if "import" in error.lower():
            fix_suggestions.append("检查模块导入路径")
        if "assert" in error.lower():
            fix_suggestions.append("检查断言逻辑")

        return {
            "test": test_name,
            "error": error[:200],
            "fixes": fix_suggestions,
            "priority": "HIGH" if fix_suggestions else "LOW"
        }

class DeployPipeline:
    """部署上线线"""

    def __init__(self, workspace: Path):
        self.workspace = workspace
        self.deploy_dir = workspace / "deploy"
        self.deploy_dir.mkdir(parents=True, exist_ok=True)

    def deploy(self, artifact: BuildArtifact, target: str = "local") -> Dict:
        """部署"""
        dna = generate_dna("DEPLOY")

        # 1. 打包产物
        package_path = self.deploy_dir / f"{artifact.name}_{artifact.version}.tar.gz"
        shutil.make_archive(
            str(package_path).replace('.tar.gz', ''),
            'gztar',
            artifact.path
        )

        # 2. 部署到目标
        if target == "local":
            deploy_path = self.deploy_dir / "current"
            if deploy_path.exists():
                shutil.rmtree(deploy_path)
            shutil.copytree(artifact.path, deploy_path)

        return {
            "dna": dna,
            "status": "success",
            "target": target,
            "package": str(package_path),
            "timestamp": datetime.now().isoformat()
        }

class FeedbackLoop:
    """反馈闭环 - 学习进化"""

    def __init__(self, workspace: Path):
        self.workspace = workspace
        self.feedback_dir = workspace / "feedback"
        self.feedback_dir.mkdir(parents=True, exist_ok=True)

    def collect(self, report: TestReport, repair_result: Dict) -> Dict:
        """收集反馈"""
        dna = generate_dna("FEEDBACK")

        feedback = {
            "dna": dna,
            "test_report": report.to_dict(),
            "repair_result": repair_result,
            "timestamp": datetime.now().isoformat()
        }

        # 保存反馈
        feedback_path = self.feedback_dir / f"feedback_{int(time.time())}.json"
        with open(feedback_path, 'w', encoding='utf-8') as f:
            json.dump(feedback, f, indent=2, ensure_ascii=False)

        return feedback

    def learn(self) -> Dict:
        """从历史反馈中学习"""
        patterns = []
        for f in self.feedback_dir.glob("*.json"):
            with open(f) as fp:
                data = json.load(fp)
                if data.get("repair_result", {}).get("repairs"):
                    for repair in data["repair_result"]["repairs"]:
                        patterns.append({
                            "test": repair.get("test"),
                            "fixes": repair.get("fixes"),
                            "frequency": 1
                        })

        # 统计高频模式
        pattern_map = {}
        for p in patterns:
            key = p["test"]
            if key not in pattern_map:
                pattern_map[key] = {"test": key, "fixes": p["fixes"], "count": 0}
            pattern_map[key]["count"] += 1

        return {
            "total_patterns": len(pattern_map),
            "top_patterns": sorted(pattern_map.values(), key=lambda x: x["count"], reverse=True)[:10]
        }

class AutoFactory:
    """全自动工厂主控制器"""

    def __init__(self, workspace: Path = None):
        if workspace is None:
            workspace = Path.home() / ".longhun" / "factory"
        self.workspace = workspace
        self.workspace.mkdir(parents=True, exist_ok=True)

        self.build_pipeline = BuildPipeline(workspace)
        self.test_pipeline = TestPipeline(workspace)
        self.repair_pipeline = RepairPipeline(workspace)
        self.deploy_pipeline = DeployPipeline(workspace)
        self.feedback_loop = FeedbackLoop(workspace)

        self.status = "idle"
        self.current_task = None
        self.history = []
        self.dna = generate_dna("FACTORY-INIT")

    def run_full_cycle(self, source_path: Path, version: str = None) -> Dict:
        """执行完整工厂流程：造零件 → 质检 → 修复 → 部署 → 反馈"""
        task_dna = generate_dna("FULL-CYCLE")
        self.status = "running"
        results = {"dna": task_dna, "steps": {}}

        try:
            # Step 1: 造零件
            print("🏗️ 造零件...")
            artifact = self.build_pipeline.build(source_path, version)
            results["steps"]["build"] = {
                "artifact_id": artifact.id,
                "version": artifact.version,
                "size": artifact.size,
                "dna": artifact.dna
            }
            print(f"  ✅ 零件已造: {artifact.name} v{artifact.version}")

            # Step 2: 质检
            print("🔍 质检...")
            report = self.test_pipeline.run_tests(artifact)
            results["steps"]["test"] = {
                "total": report.total,
                "passed": report.passed,
                "failed": report.failed,
                "coverage": report.coverage,
                "tricolor": report.tricolor,
                "dna": report.dna
            }
            print(f"  ✅ 质检完成: {report.tricolor} ({report.passed}/{report.total})")

            # Step 3: 自动修复
            print("🔧 自动修复...")
            repair_result = self.repair_pipeline.repair(report)
            results["steps"]["repair"] = repair_result
            print(f"  ✅ 修复完成: {repair_result.get('status')}")

            # Step 4: 部署
            print("🚀 部署...")
            deploy_result = self.deploy_pipeline.deploy(artifact)
            results["steps"]["deploy"] = deploy_result
            print(f"  ✅ 部署完成: {deploy_result['target']}")

            # Step 5: 反馈
            print("📊 收集反馈...")
            feedback = self.feedback_loop.collect(report, repair_result)
            results["steps"]["feedback"] = {
                "dna": feedback["dna"],
                "timestamp": feedback["timestamp"]
            }
            print("  ✅ 反馈已收集")

            # 学习模式
            learn_result = self.feedback_loop.learn()
            results["steps"]["learn"] = learn_result

            self.status = "completed"
            results["status"] = "success"

        except Exception as e:
            self.status = "failed"
            results["status"] = "failed"
            results["error"] = str(e)
            print(f"❌ 工厂流程失败: {e}")

        self.history.append(results)
        return results

    def get_status(self) -> Dict:
        """获取工厂状态"""
        return {
            "dna": self.dna,
            "status": self.status,
            "workspace": str(self.workspace),
            "build_count": len(self.build_pipeline.history),
            "history_count": len(self.history),
            "timestamp": datetime.now().isoformat()
        }

    def get_artifacts(self) -> List[Dict]:
        return self.build_pipeline.get_artifacts()

    def get_feedback_patterns(self) -> Dict:
        return self.feedback_loop.learn()
