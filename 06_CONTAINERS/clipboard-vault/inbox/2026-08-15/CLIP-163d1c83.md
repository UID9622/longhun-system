```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
🐉 龍魂生态 · 数据模型
DNA: #龍芯⚡️丙午·丙申·庚申·亥时-DEV-MODELS-UID9622
"""

from sqlalchemy import create_engine, Column, String, Integer, Float, DateTime, Text, Boolean
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from datetime import datetime
from .config import DATABASE_URL

Base = declarative_base()
engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(bind=engine)


class Developer(Base):
    __tablename__ = "developers"

    id = Column(Integer, primary_key=True, autoincrement=True)
    dna = Column(String(128), unique=True, nullable=False)          # 开发者DNA
    name = Column(String(64), nullable=False)
    email = Column(String(128), unique=True, nullable=False)
    nickname = Column(String(64), nullable=False)
    gpg_public_key = Column(Text, nullable=True)
    registered_at = Column(DateTime, default=datetime.now)
    paid_at = Column(DateTime, nullable=True)                       # 支付时间
    payment_amount = Column(Float, default=0.0)
    status = Column(String(20), default="active")                  # active / suspended / revoked
    contribution_score = Column(Integer, default=0)                 # 贡献分
    dna_count = Column(Integer, default=0)                          # 生成的DNA数量

    def to_dict(self):
        return {
            "id": self.id,
            "dna": self.dna,
            "name": self.name,
            "email": self.email,
            "nickname": self.nickname,
            "registered_at": self.registered_at.isoformat(),
            "paid": self.paid_at is not None,
            "contribution_score": self.contribution_score,
            "dna_count": self.dna_count,
        }


class CodeDNA(Base):
    __tablename__ = "code_dna"

    id = Column(Integer, primary_key=True, autoincrement=True)
    dna = Column(String(128), unique=True, nullable=False)          # 代码DNA
    developer_dna = Column(String(128), nullable=False)             # 关联开发者
    file_path = Column(String(512), nullable=False)
    file_hash = Column(String(64), nullable=False)                  # 文件内容哈希
    created_at = Column(DateTime, default=datetime.now)
    updated_at = Column(DateTime, default=datetime.now, onupdate=datetime.now)
    line_count = Column(Integer, default=0)
    language = Column(String(32), nullable=True)

    def to_dict(self):
        return {
            "dna": self.dna,
            "developer_dna": self.developer_dna,
            "file_path": self.file_path,
            "file_hash": self.file_hash,
            "created_at": self.created_at.isoformat(),
            "line_count": self.line_count,
            "language": self.language,
        }


class Contribution(Base):
    __tablename__ = "contributions"

    id = Column(Integer, primary_key=True, autoincrement=True)
    developer_dna = Column(String(128), nullable=False)
    contribution_type = Column(String(32), nullable=False)          # code / protocol / doc / community
    content = Column(Text, nullable=True)
    score = Column(Integer, default=1)
    created_at = Column(DateTime, default=datetime.now)
    dna = Column(String(128), unique=True, nullable=False)


# 初始化数据库
def init_db():
    Base.metadata.create_all(engine)
```

```python
"""
🐉 龍魂生态 · DNA生成引擎
DNA: #龍芯⚡️丙午·丙申·庚申·亥时-DNA-GEN-UID9622
"""

import hashlib
import time
from datetime import datetime
from .config import DNA_PREFIX, DNA_UID


def get_ganzhi():
    """获取天干地支（简化版）"""
    tian = ["甲","乙","丙","丁","戊","己","庚","辛","壬","癸"]
    di = ["子","丑","寅","卯","辰","巳","午","未","申","酉","戌","亥"]
    now = datetime.now()
    gan = tian[(now.year - 4) % 10]
    zhi = di[(now.year - 4) % 12]
    return f"{gan}{zhi}"


def generate_dna(prefix: str = "", uid: str = DNA_UID) -> str:
    """生成DNA追溯码"""
    if not prefix:
        prefix = "DEV"
    timestamp = datetime.now().strftime("%Y-%m-%d")
    rand = hashlib.sha256(f"{prefix}{timestamp}{time.time()}".encode()).hexdigest()[:8].upper()
    ganzhi = get_ganzhi()
    return f"{DNA_PREFIX}{ganzhi}-{prefix}-{rand}-{uid}"


def generate_developer_dna(name: str, email: str) -> str:
    """生成开发者DNA"""
    seed = f"{name}{email}{time.time()}"
    rand = hashlib.sha256(seed.encode()).hexdigest()[:8].upper()
    timestamp = datetime.now().strftime("%Y-%m-%d")
    ganzhi = get_ganzhi()
    return f"{DNA_PREFIX}{ganzhi}-DEV-{rand}-{DNA_UID}"
```

```python
@app.post("/api/developer/register")
async def register_developer(req: DeveloperRegisterRequest):
    """开发者注册"""
    db = SessionLocal()

    # 检查邮箱是否已注册
    existing = db.query(Developer).filter(Developer.email == req.email).first()
    if existing:
        db.close()
        raise HTTPException(status_code=400, detail="邮箱已被注册")

    # 生成开发者DNA
    dna = generate_developer_dna(req.name, req.email)

    # 创建开发者
    dev = Developer(
        dna=dna,
        name=req.name,
        email=req.email,
        nickname=req.nickname,
        gpg_public_key=req.gpg_public_key,
        registered_at=datetime.now(),
        status="pending",  # 待支付

    db.add(dev)
    db.commit()

    # 生成支付订单
    order = create_payment_order(dna)

    db.close()

    return {
        "success": True,
        "dna": dna,
        "message": "注册成功，请完成1元支付激活",
        "order": order,
        "developer": dev.to_dict(),
        "confirm": CONFIRM,

    if not dev:
        raise HTTPException(status_code=404, detail="开发者不存在")

@app.post("/api/code/inject")
async def inject_code_dna(req: CodeInjectRequest):
    """注入代码DNA"""
    db = SessionLocal()

    # 验证开发者
    dev = db.query(Developer).filter(Developer.dna == req.developer_dna).first()
    if not dev:
        db.close()
        raise HTTPException(status_code=404, detail="开发者不存在")
    if dev.status != "active":
        db.close()
        raise HTTPException(status_code=403, detail="开发者未激活，请先完成1元支付")

    # 生成代码DNA
    code_dna = generate_dna("CODE")
    file_hash = hashlib.sha256(req.content.encode()).hexdigest()

    # 保存记录
    code_record = CodeDNA(
        dna=code_dna,
        developer_dna=req.developer_dna,
        file_path=req.file_path,
        file_hash=file_hash,
        line_count=len(req.content.split('\n')),
        language=req.language or "unknown",
        created_at=datetime.now(),
    )
    db.add(code_record)

    # 增加贡献分
    dev.contribution_score += 1
    dev.dna_count += 1
    db.commit()

    # 记录贡献
    contrib_dna = generate_dna("CONTRIB")
    contrib = Contribution(
        developer_dna=req.developer_dna,
        contribution_type="code",
        content=f"注入代码 {req.file_path}",
        score=1,
        dna=contrib_dna
    )
    db.add(contrib)
    db.commit()
    db.close()

    return {
        "success": True,
        "dna": code_dna,
        "message": f"✅ 代码DNA已注入: {code_dna}",
        "file_path": req.file_path,
        "developer_dna": req.developer_dna,
```

## ⚙️ CLI 注入工具

### 11. `cli/lh_dna_inject.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
🐉 龍魂生态 · DNA注入CLI工具
DNA: #龍芯⚡️丙午·丙申·庚申·亥时-DEV-INJECT-UID9622
用法:
    lh dna inject --path ./my-repo/ --developer-dna <DNA> [--recursive]
    lh dna inject --file ./main.py --developer-dna <DNA>
"""

import os
import sys
import argparse
import hashlib
import requests
from pathlib import Path
from datetime import datetime

API_URL = os.getenv("LONGHUN_API_URL", "http://localhost:8000")


def inject_dna_to_file(file_path, developer_dna, api_url=API_URL):
    """为单个文件注入DNA"""
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
    except:
        print(f"⚠️ 跳过无法读取的文件: {file_path}")
        return False

    # 检查是否已有DNA
    if "#龍芯⚡️" in content[:200]:
        print(f"⏭️ 已存在DNA: {file_path}")
        return False

    # 调用API注入
    resp = requests.post(
        f"{api_url}/api/code/inject",
        json={
            "file_path": str(file_path),
            "content": content,
            "developer_dna": developer_dna,
            "language": file_path.suffix.lstrip('.') or "txt"
        },
        timeout=30
    )

    if resp.status_code == 200:
        data = resp.json()
        dna = data.get("dna")
        # 在文件头插入DNA注释
        ext = file_path.suffix
        if ext in ['.py', '.js', '.ts', '.go', '.rs', '.c', '.cpp']:
            header = f'# 🐉 DNA: {dna}\n# 开发者: {developer_dna}\n# 生成时间: {datetime.now().isoformat()}\n\n'
        elif ext in ['.md', '.txt']:
            header = f'**DNA:** `{dna}`\n**开发者:** `{developer_dna}`\n\n'
        elif ext in ['.html', '.htm']:
            header = f'<!-- DNA: {dna} -->\n<!-- 开发者: {developer_dna} -->\n\n'
        else:
            header = f'# DNA: {dna}\n# 开发者: {developer_dna}\n\n'

        new_content = header + content
        with open(file_path, 'w', encoding='utf-8') as f:
            f.write(new_content)

        print(f"✅ 已注入: {file_path} → {dna}")
        return True
    else:
        print(f"❌ 注入失败: {file_path} - {resp.text}")
        return False


def main():
    parser = argparse.ArgumentParser(description="龍魂DNA注入CLI工具")
    parser.add_argument("--path", help="目录或文件路径")
    parser.add_argument("--file", help="单个文件路径")
    parser.add_argument("--developer-dna", required=True, help="开发者DNA")
    parser.add_argument("--recursive", "-r", action="store_true", help="递归扫描目录")
    parser.add_argument("--api", default=API_URL, help="API地址")
    args = parser.parse_args()

    if not args.path and not args.file:
        print("❌ 请指定 --path 或 --file")
        sys.exit(1)

    target_files = []
    if args.file:
        target_files = [Path(args.file)]
    elif args.path:
        p = Path(args.path)
        if p.is_file():
            target_files = [p]
        elif p.is_dir():
            if args.recursive:
                target_files = list(p.rglob("*"))
            else:
                target_files = list(p.glob("*"))

    # 过滤代码文件
    code_exts = {'.py', '.js', '.ts', '.html', '.css', '.md', '.txt', '.json', '.yaml', '.yml', '.sh', '.go', '.rs', '.c', '.cpp', '.h'}
    target_files = [f for f in target_files if f.is_file() and f.suffix in code_exts]

    print(f"🐉 龍魂DNA注入工具")
    print(f"开发者DNA: {args.developer_dna}")
    print(f"扫描文件: {len(target_files)} 个")

    success = 0
    for f in target_files:
        if inject_dna_to_file(f, args.developer_dna, args.api):
            success += 1

    print(f"\n✅ 完成: {success}/{len(target_files)} 个文件已注入DNA")
```

```bash
#!/bin/bash
# 🐉 龍魂 · Git pre-commit 钩子
# 每次提交前自动检查并注入DNA

set -e

echo "🐉 龍魂主权检查..."

# 获取开发者DNA（从环境变量或配置）
DEV_DNA="${LONGHUN_DEV_DNA:-}"
if [ -z "$DEV_DNA" ]; then
    echo "⚠️ 未设置 LONGHUN_DEV_DNA，请先注册开发者"
    echo "   export LONGHUN_DEV_DNA=#龍芯⚡️..."
    exit 1
fi

# 获取变更文件
FILES=$(git diff --cached --name-only --diff-filter=ACM | grep -E '\.(py|js|ts|html|css|md|txt|sh|go|rs|c|cpp|h|json|yaml|yml)$' || true)

if [ -z "$FILES" ]; then
    echo "✅ 无代码文件变更，跳过"
    exit 0
fi

# 检查是否已有DNA
NEED_INJECT=""
for f in $FILES; do
    if [ -f "$f" ] && ! head -20 "$f" | grep -q "#龍芯⚡️"; then
        NEED_INJECT="$NEED_INJECT $f"
    fi
done

if [ -n "$NEED_INJECT" ]; then
    echo "🔍 发现以下文件缺少DNA:"
    echo "$NEED_INJECT"
    echo ""
    echo "请运行: lh dna inject --file <file> --developer-dna $DEV_DNA"
    echo "或设置自动注入"
    exit 1
fi

echo "✅ 所有变更文件均含DNA，允许提交"
```
