```python
# backend/monthly_fee.py
# 新增模块：月费核对 + 支付 + 状态熔断

from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from .models import Developer, MonthlyFeeRecord, Contribution
from .dna_generator import generate_dna

MONTHLY_FEE_MIN = 1.0
GRACE_MONTHS = 3

def get_current_month() -> str:
    return datetime.now().strftime("%Y-%m")

def check_developer_status(dev_dna: str, db: Session) -> dict:
    dev = db.query(Developer).filter(Developer.dna == dev_dna).first()
    if not dev:
        return {"status": "invalid", "message": "开发者不存在"}
    
    current_month = get_current_month()
    last_paid = dev.last_paid_month or "1970-01"
    
    # 计算欠费月数
    last_year, last_month = map(int, last_paid.split("-"))
    cur_year, cur_month = map(int, current_month.split("-"))
    months_diff = (cur_year - last_year) * 12 + (cur_month - last_month)
    
    if dev.monthly_fee_status == "frozen":
        return {"status": "frozen", "message": "已冻结，需补缴欠费", "arrears": dev.fee_arrears}
    
    if months_diff == 0:
        return {"status": "active", "message": "当前月费已缴"}
    elif months_diff <= GRACE_MONTHS:
        return {"status": "grace", "message": f"宽限期内，欠缴 {months_diff} 个月", "arrears": months_diff}
    else:
        return {"status": "expired", "message": f"已欠缴 {months_diff} 个月，超过宽限期，已冻结", "arrears": months_diff}

def create_monthly_bill(dev_dna: str, db: Session) -> dict:
    dev = db.query(Developer).filter(Developer.dna == dev_dna).first()
    if not dev:
        return {"error": "开发者不存在"}
    
    current_month = get_current_month()
    existing = db.query(MonthlyFeeRecord).filter(
        MonthlyFeeRecord.developer_dna == dev_dna,
        MonthlyFeeRecord.year_month == current_month
    ).first()
    
    if existing and existing.status == "paid":
        return {"message": "本月已缴"}
    
    # 自动判断金额：老百姓 1 元，企业可上浮（通过请求体传入）
    amount = MONTHLY_FEE_MIN
    return {
        "year_month": current_month,
        "amount": amount,
        "min_amount": MONTHLY_FEE_MIN,
        "message": "请支付本月开发者月费"
    }

def confirm_monthly_payment(dev_dna: str, amount: float, db: Session) -> dict:
    dev = db.query(Developer).filter(Developer.dna == dev_dna).first()
    if not dev:
        return {"error": "开发者不存在"}
    
    current_month = get_current_month()
    record = MonthlyFeeRecord(
        developer_dna=dev_dna,
        year_month=current_month,
        amount=amount,
        paid_at=datetime.now(),
        status="paid"
    )
    db.add(record)
    
    dev.last_paid_month = current_month
    dev.monthly_fee_status = "active"
    dev.total_contributed += amount
    db.commit()
    
    # 记录贡献（10 分 / 月）
    contrib_dna = generate_dna("FEE")
    contrib = Contribution(
        developer_dna=dev_dna,
        contribution_type="monthly_fee",
        content=f"支付 {amount} 元月费",
        score=10,
        dna=contrib_dna
    )
    db.add(contrib)
    db.commit()
    
    return {"message": f"✅ {current_month} 月费支付成功", "amount": amount}

def freeze_expired_developers(db: Session):
    """定时任务：冻结超期开发者"""
    devs = db.query(Developer).filter(Developer.monthly_fee_status == "grace").all()
    for dev in devs:
        status = check_developer_status(dev.dna, db)
        if status["status"] == "expired":
            dev.monthly_fee_status = "frozen"
            dev.fee_arrears = status["arrears"]
    db.commit()
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
🐉 龍魂生态 · 月度开发者月费模块
DNA: #龍芯⚡️丙午·丙申·庚申·亥时-MONTHLY-FEE-UID9622
"""

import time
import random
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from .models import Developer, MonthlyFeeRecord, Contribution
from .dna_generator import generate_dna
from .config import DNA_UID

MONTHLY_FEE_MIN = 1.0
GRACE_MONTHS = 3
FREEZE_MONTHS = 3  # 超过3个月冻结


def get_current_month() -> str:
    """获取当前月份 YYYY-MM"""
    return datetime.now().strftime("%Y-%m")


def get_month_delta(month1: str, month2: str) -> int:
    """计算两个月份之间的月数差"""
    y1, m1 = map(int, month1.split("-"))
    y2, m2 = map(int, month2.split("-"))
    return (y2 - y1) * 12 + (m2 - m1)


def check_developer_fee_status(dev_dna: str, db: Session) -> dict:
    """
    检查开发者月费状态
    返回: {
        "status": "active" | "grace" | "frozen" | "invalid",
        "message": "...",
        "arrears": 0,
        "current_month": "YYYY-MM"
    }
    """
    dev = db.query(Developer).filter(Developer.dna == dev_dna).first()
    if not dev:
        return {"status": "invalid", "message": "开发者不存在"}

    current_month = get_current_month()
    last_paid = dev.last_paid_month or dev.fee_start_month or current_month
    months_diff = get_month_delta(last_paid, current_month)

    # 如果上月已缴，本月还没到月底，算活跃（月初有1天缓冲）
    if months_diff == 0:
        return {
            "status": "active",
            "message": "✅ 本月月费已缴",
            "arrears": 0,
            "current_month": current_month,
            "last_paid": last_paid
        }

    if months_diff <= GRACE_MONTHS:
        return {
            "status": "grace",
            "message": f"🟡 宽限期内，欠缴 {months_diff} 个月",
            "arrears": months_diff,
            "current_month": current_month,
            "last_paid": last_paid
        }

    # 超过宽限期 → 冻结
    return {
        "status": "frozen",
        "message": f"🔴 已冻结，欠缴 {months_diff} 个月，需补缴",
        "arrears": months_diff,
        "current_month": current_month,
        "last_paid": last_paid
    }


def create_monthly_bill(dev_dna: str, db: Session, amount: float = None) -> dict:
    """
    生成当月账单
    """
    dev = db.query(Developer).filter(Developer.dna == dev_dna).first()
    if not dev:
        return {"error": "开发者不存在"}

    current_month = get_current_month()

    # 检查是否已缴
    existing = db.query(MonthlyFeeRecord).filter(
        MonthlyFeeRecord.developer_dna == dev_dna,
        MonthlyFeeRecord.year_month == current_month,
        MonthlyFeeRecord.status == "paid"
    ).first()
    if existing:
        return {
            "message": "✅ 本月已缴",
            "year_month": current_month,
            "amount": existing.amount,
            "status": "paid"
        }

    # 金额：如果指定金额且≥1元则使用，否则1元
    fee_amount = max(MONTHLY_FEE_MIN, amount or MONTHLY_FEE_MIN)

    # 生成订单号
    order_id = f"FEE-{int(time.time())}-{random.randint(1000,9999)}"

    # 创建待支付记录
    record = MonthlyFeeRecord(
        developer_dna=dev_dna,
        year_month=current_month,
        amount=fee_amount,
        status="pending",
        order_id=order_id,
    )
    db.add(record)
    db.commit()

    return {
        "year_month": current_month,
        "amount": fee_amount,
        "min_amount": MONTHLY_FEE_MIN,
        "order_id": order_id,
        "message": f"📌 请支付 {current_month} 月费 {fee_amount} 元",
        "status": "pending"
    }


def confirm_monthly_payment(dev_dna: str, order_id: str, db: Session) -> dict:
    """
    确认月费支付
    """
    record = db.query(MonthlyFeeRecord).filter(
        MonthlyFeeRecord.developer_dna == dev_dna,
        MonthlyFeeRecord.order_id == order_id
    ).first()

    if not record:
        return {"error": "订单不存在"}

    if record.status == "paid":
        return {"message": "✅ 已支付"}

    # 更新记录
    record.status = "paid"
    record.paid_at = datetime.now()

    # 更新开发者
    dev = db.query(Developer).filter(Developer.dna == dev_dna).first()
    if dev:
        dev.last_paid_month = record.year_month
        dev.monthly_fee_status = "active"
        dev.total_contributed = (dev.total_contributed or 0) + record.amount
        dev.fee_arrears = 0

    db.commit()

    # 记录贡献
    contrib_dna = generate_dna("FEE")
    contrib = Contribution(
        developer_dna=dev_dna,
        contribution_type="monthly_fee",
        content=f"支付 {record.year_month} 月费 {record.amount} 元",
        score=10,
        dna=contrib_dna
    )
    db.add(contrib)
    db.commit()

    return {
        "message": f"✅ {record.year_month} 月费支付成功",
        "amount": record.amount,
        "year_month": record.year_month,
        "dna": contrib_dna
    }


def freeze_expired_developers(db: Session) -> dict:
    """
    定时任务：冻结超期开发者（每月1日执行）
    """
    current_month = get_current_month()
    devs = db.query(Developer).filter(
        Developer.monthly_fee_status != "frozen"
    ).all()

    frozen = 0
    for dev in devs:
        status = check_developer_fee_status(dev.dna, db)
        if status["status"] == "frozen":
            dev.monthly_fee_status = "frozen"
            dev.fee_arrears = status["arrears"]
            frozen += 1

    db.commit()
    return {"frozen": frozen, "checked": len(devs)}


def get_fee_history(dev_dna: str, db: Session, limit: int = 12) -> dict:
    """
    获取缴费历史
    """
    records = db.query(MonthlyFeeRecord).filter(
        MonthlyFeeRecord.developer_dna == dev_dna,
        MonthlyFeeRecord.status == "paid"
    ).order_by(MonthlyFeeRecord.year_month.desc()).limit(limit).all()

    return {
        "records": [
            {
                "year_month": r.year_month,
                "amount": r.amount,
                "paid_at": r.paid_at.isoformat() if r.paid_at else None,
            }
            for r in records
        ],
        "total": len(records)
    }


def get_public_fee_stats(db: Session) -> dict:
    """
    获取公开统计（可用于展示生态贡献榜）
    """
    total = db.query(MonthlyFeeRecord).filter(
        MonthlyFeeRecord.status == "paid"
    ).count()

    total_amount = db.query(MonthlyFeeRecord).filter(
        MonthlyFeeRecord.status == "paid"
    ).with_entities(
        db.func.sum(MonthlyFeeRecord.amount)
    ).scalar() or 0

    top_contributors = db.query(
        Developer.nickname,
        Developer.dna,
        db.func.sum(MonthlyFeeRecord.amount).label("total")
    ).join(
        MonthlyFeeRecord, Developer.dna == MonthlyFeeRecord.developer_dna
    ).filter(
        MonthlyFeeRecord.status == "paid"
    ).group_by(
        Developer.dna
    ).order_by(
        db.desc("total")
    ).limit(10).all()

    return {
        "total_payments": total,
        "total_amount": round(total_amount, 2),
        "top_contributors": [
            {"nickname": c.nickname, "total": round(c.total, 2)}
            for c in top_contributors
        ]
    }
```
