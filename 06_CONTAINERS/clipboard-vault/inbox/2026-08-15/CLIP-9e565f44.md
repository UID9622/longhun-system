(function() {
    console.log('🐉 龍魂采集器已注入 DeepSeek');

    let capturedMessages = new Set();

    function captureMessage(element) {
        if (capturedMessages.has(element)) return;

        const role = element.classList.contains('user') ? 'user' : 'assistant';
        const content = element.textContent?.trim() || '';
        if (!content) return;

        // 提取话题（从页面标题或上下文）
        const topic = document.title.replace('DeepSeek', '').trim() || 'DeepSeek对话';

        // 发送到本地采集服务
        fetch('http://localhost:8769/api/capture', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                source: 'deepseek',
                role: role,
                content: content,
                topic: topic,
                metadata: {
                    url: window.location.href,
                    timestamp: new Date().toISOString()
                }
            })
        }).catch(() => {
            // 本地服务未启动，存到IndexedDB
            saveToIndexedDB({role, content, topic});
        });

        capturedMessages.add(element);
        element.dataset.captured = 'true';
    }

    // MutationObserver 监听新消息
    const observer = new MutationObserver(() => {
        const messages = document.querySelectorAll('.message, .chat-message, [class*="message"]');
        messages.forEach(captureMessage);
    });

    observer.observe(document.body, {childList: true, subtree: true});

    // 也捕获现有消息
    document.querySelectorAll('.message, .chat-message, [class*="message"]')
        .forEach(captureMessage);
})();

# 08_BIN/lh_capture_cleanup.py
def auto_cleanup(days: int = 30):
    """自动清理过期对话（保留最近30天）"""
    cutoff = datetime.now() - timedelta(days=days)
    storage_dir = Path.home() / ".longhun" / "03_MEMORY" / "ai_conversations"

    for source_dir in storage_dir.iterdir():
        if not source_dir.is_dir():
            continue
        for jsonl_file in source_dir.glob("*.jsonl"):
            # 检查文件日期
            try:
                file_date = datetime.strptime(jsonl_file.stem, "%Y-%m-%d")
                if file_date < cutoff:
                    # 归档而非删除
                    archive_dir = storage_dir / "_archive" / source_dir.name
                    archive_dir.mkdir(parents=True, exist_ok=True)
                    shutil.move(str(jsonl_file), str(archive_dir / jsonl_file.name))
                    print(f"📦 已归档: {jsonl_file}")
            except:
                pass

# 08_BIN/lh_capture_vector.py
class VectorSearch:
    """对话向量检索"""

    def __init__(self):
        try:
            from sentence_transformers import SentenceTransformer
            self.model = SentenceTransformer('all-MiniLM-L6-v2')
            self.available = True
        except:
            self.available = False
            print("⚠️ sentence-transformers未安装，使用关键词检索")

    def search(self, query: str, conversations: List[Dict], top_k: int = 5) -> List[Dict]:
        if not self.available:
            return self._keyword_search(query, conversations, top_k)

        # 向量检索
        texts = [c.get("content", "") for c in conversations]
        query_vec = self.model.encode(query)
        text_vecs = self.model.encode(texts)
        from numpy import dot
        from numpy.linalg import norm
        scores = [dot(query_vec, v) / (norm(query_vec) * norm(v)) for v in text_vecs]
        # 按相似度排序
        indexed = list(enumerate(scores))
        indexed.sort(key=lambda x: x[1], reverse=True)
        return [conversations[i] for i, _ in indexed[:top_k]]

    def _keyword_search(self, query: str, conversations: List[Dict], top_k: int) -> List[Dict]:
        q = query.lower()
        results = []
        for c in conversations:
            if q in c.get("content", "").lower():
                results.append(c)
        return results[:top_k]
```

# 08_BIN/lh_capture_dedup.py
def deduplicate_entries(entries: List[Dict]) -> List[Dict]:
    """智能去重 - 基于内容相似度"""
    seen = {}
    unique = []

    for entry in entries:
        content = entry.get("content", "")
        # 内容哈希
        content_hash = hashlib.sha256(content.encode()).hexdigest()[:16]
        if content_hash not in seen:
            seen[content_hash] = entry
            unique.append(entry)
        else:
            # 记录重复到耻辱墙
            ShameWall.write({
                "reason": "重复对话",
                "dna": entry.get("dna"),
                "details": {"original": seen[content_hash].get("id")}
            })

    return unique
```
