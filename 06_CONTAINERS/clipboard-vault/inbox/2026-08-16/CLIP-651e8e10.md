                    params: Dict = None) -> MilitaryOrder:

            return None
        
        order = MilitaryOrder(
            order_id=f"ORD-{int(time.time())}-{hashlib.md5(command.encode()).hexdigest()[:6]}",
            commander=UID,
            target_unit=target_unit,
            command=command,
            priority=priority,
            timeout=timeout,
            callback_url=callback_url,
            params=params or {}
        )

        # 更新单位状态

            "target": target_unit,
            "command": command,
            "priority": priority
        })
        
        # 立即执行
        self._execute_order(order)
        
        return order

        order.executed_at = datetime.now().isoformat()
        
        unit = self.units.get(order.target_unit)
        if not unit:
            order.status = "违抗"
            order.result = "目标单位不存在"

            elif unit.branch == Branch.SIGNAL:
                result = self._signal_execute(order)
            elif unit.branch == Branch.GUARD:
                result = self._guard_execute(order)
            elif unit.branch == Branch.STAFF:
                result = self._staff_execute(order)
            else:
                result = {"status": "success", "message": f"执行完成: {order.command}"}

            order.result = json.dumps(result, ensure_ascii=False)
            
            # 更新单位状态

            unit.last_active = datetime.now().isoformat()
