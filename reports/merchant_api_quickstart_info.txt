📝 Python 调用示例 (可直接复制运行):
```python
import hashlib, hmac, time, json, requests

API_KEY = "lh_mch_0b51a799e39512aebcf9a89486cf74ad6346b659a23e1ba5"
BASE_URL = "http://localhost:9633"

def call_api(method, path, body=None):
    """签名 + 调用龍魂商户API"""
    body_str = json.dumps(body) if body else ''  # 默认ensure_ascii=True, 与requests一致
    ts = str(int(time.time()))
    body_hash = hashlib.sha256(body_str.encode()).hexdigest()
    # 签名串: METHOD + PATH + TIMESTAMP + SHA256(BODY)
    sign_str = f'{method}{path}{ts}{body_hash}'
    # HMAC-SHA256(API_KEY, 签名串)
    sig = hmac.new(API_KEY.encode(), sign_str.encode(), hashlib.sha256).hexdigest()
    
    headers = {
        'X-LH-API-Key': API_KEY,
        'X-LH-Timestamp': ts,
        'X-LH-Signature': sig,
        'Content-Type': 'application/json',
    }
    
    url = f'{BASE_URL}{path}'
    if method == 'POST':
        return requests.post(url, headers=headers, json=body)
    return requests.get(url, headers=headers)

# 示例: 计算数字根
resp = call_api("POST", "/v1/math/digital-root", {"n": 369})
print(resp.json())
# → {'code':0, 'data':{'digital_root':9, 'is_369_fixed_point':True}}

# 示例: 焦虑话术检测

# 示例: 五行判定
```
