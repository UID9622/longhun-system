### 1.1 DNA追溯引擎 `core/DNAEngine.ets`

每条操作都有唯一指纹：干支+卦名+时间戳。DNA格式: `#龍芯⚡️{干支·卦名}-{车辆ID}-{时间戳}-UID9622`

```typescript
import { GanzhiUtil } from '../common/utils/GanzhiUtil';
import { HashUtil } from '../common/utils/HashUtil';

export class DNAEngine {
  private static UID: string = 'UID9622';
  private static CONFIRM: string = '#CONFIRM🌌9622-ONLY-ONCE🧬LK9X-772Z';

  static generateDNA(vehicleId: string, action: string, detail: string): string {
    const ganzhi = GanzhiUtil.getCurrentGanzhi();
    const hexagram = GanzhiUtil.getCurrentHexagram();
    const timestamp = Date.now().toString(36);
    const hash = HashUtil.shortHash(`${vehicleId}${action}${detail}${timestamp}`);
    return `#龍芯⚡️${ganzhi}·${hexagram}-${vehicleId}-${timestamp}-${hash}-${this.UID}`;
  }

  static validateDNA(dna: string): boolean {
    return dna.startsWith('#龍芯⚡️') && dna.includes(this.UID);
  }

  static parseDNA(dna: string): {
    ganzhi: string, hexagram: string, vehicleId: string,
    timestamp: string, hash: string, uid: string
  } {
    const parts = dna.replace('#龍芯⚡️', '').split('-');
    const ganzhiHex = parts[0].split('·');
    return {
      ganzhi: ganzhiHex[0], hexagram: ganzhiHex[1] || '坤卦',
      vehicleId: parts[1] || 'unknown', timestamp: parts[2] || '',
      hash: parts[3] || '', uid: parts[4] || ''
    };
  }
}
```

> 道德经注释: 第1章 "道可道，非常道" → DNA不可篡改；第25章 "人法地，地法天，天法道，道法自然" → DNA遵循自然规律

### 1.2 三才算法引擎 `core/SancaiEngine.ets`

天时(0.34) + 地利(0.33) + 人和(0.33) = 决策。不是算路，是算"该不该走"。

```typescript
export interface SancaiInput {
  weather: 'sunny' | 'rainy' | 'snowy' | 'foggy' | 'windy';
  temperature: number;
  timeOfDay: 'dawn' | 'morning' | 'noon' | 'afternoon' | 'dusk' | 'night';
  trafficLevel: 0 | 1 | 2 | 3 | 4 | 5;
  roadCondition: 'excellent' | 'good' | 'fair' | 'poor' | 'hazardous';
  terrain: 'flat' | 'hilly' | 'mountainous' | 'urban' | 'rural';
  driverFatigue: 0 | 1 | 2 | 3 | 4 | 5;
  passengerComfort: 0 | 1 | 2 | 3 | 4 | 5;
  urgencyLevel: 0 | 1 | 2 | 3 | 4 | 5;
}

export interface SancaiOutput {
  score: number;
  decision: 'proceed' | 'caution' | 'reroute' | 'stop' | 'manual_override';
  hexagram: string;
  description: string;
  details: { heaven: number; earth: number; human: number; };
}

export class SancaiEngine {
  private static WEATHER_WEIGHTS = { sunny: 0.9, rainy: 0.6, snowy: 0.3, foggy: 0.2, windy: 0.5 };
  private static TRAFFIC_WEIGHTS = { 0:1.0, 1:0.9, 2:0.7, 3:0.5, 4:0.3, 5:0.1 };
  private static FATIGUE_WEIGHTS = { 0:1.0, 1:0.9, 2:0.8, 3:0.6, 4:0.4, 5:0.2 };

  static evaluate(input: SancaiInput): SancaiOutput {
    const heaven = this.calculateHeaven(input);
    const earth = this.calculateEarth(input);
    const human = this.calculateHuman(input);
    const score = heaven * 0.34 + earth * 0.33 + human * 0.33;
    const digitalRoot = this.digitalRoot(Math.round(score * 100));

    let decision: SancaiOutput['decision'];
    let hexagram: string;
    let description: string;

    if (digitalRoot === 3 || digitalRoot === 9) {
      decision = 'manual_override';
      hexagram = digitalRoot === 3 ? '震卦' : '艮卦';
      description = digitalRoot === 3 ? '天时异常，建议人工确认' : '高危场景，强制人工接管';
    } else if (score >= 0.8) {
      decision = 'proceed'; hexagram = '乾卦'; description = '天时地利人和，畅通无阻';
    } else if (score >= 0.6) {
      decision = 'caution'; hexagram = '坤卦'; description = '条件一般，谨慎驾驶';
    } else if (score >= 0.4) {
      decision = 'reroute'; hexagram = '坎卦'; description = '路况不佳，建议绕行';
    } else {
      decision = 'stop'; hexagram = '离卦'; description = '危险！立即停车';
    }

    return {
      score, decision, hexagram, description,
      details: {
        heaven: Math.round(heaven*100)/100,
        earth: Math.round(earth*100)/100,
        human: Math.round(human*100)/100
      }
    };
  }

}
```

### 1.3 数字根熔断引擎 `core/DigitalRootEngine.ets`

数字根为3或9时自动触发伦理审查。

```typescript
export class DigitalRootEngine {
  private static TRIGGER_ROOTS = [3, 9];

  static calculate(n: number): number {
    while (n >= 10) {
      let sum = 0;
      while (n > 0) { sum += n % 10; n = Math.floor(n / 10); }
      n = sum;
    }
    return n;
  }

  static shouldFuse(n: number): boolean {
    return this.TRIGGER_ROOTS.includes(this.calculate(n));
  }

  static getFuseLevel(n: number): 'normal' | 'caution' | 'fuse' {
    const root = this.calculate(n);
    if (root === 3) return 'caution';
    if (root === 9) return 'fuse';
    return 'normal';
  }

  static process(decision: any): { approved: boolean, reason: string, root: number } {
    const root = this.calculate(decision.score || 0);
    if (root === 9) return { approved: false, reason: '🔴 熔断触发：高危场景，强制人工接管', root };
    if (root === 3) return { approved: false, reason: '🟡 熔断触发：天时异常，建议人工确认', root };
    return { approved: true, reason: '✅ 通过', root };
  }
}
```
