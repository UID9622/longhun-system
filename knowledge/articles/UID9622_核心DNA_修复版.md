```python
class ZhongYongDecisionModule:

        """中庸决策：寻找最平衡的方案"""
        scores = {}
        
        for option in options:
            # 计算每个选项的平衡度
            balance_score = self._evaluate_balance(option)
            risk_score = self._evaluate_risk(option)
            opportunity_score = self._evaluate_opportunity(option)
            
            # 中庸评分：不求最高，但求最稳
            zhongyong_score = (
                balance_score * 0.4 +      # 平衡度40%
                risk_score * 0.3 +          # 风险控制30%
                opportunity_score * 0.3     # 机会把握30%
            )
            
            scores[option["name"]] = {
                "total_score": zhongyong_score,
                "balance": balance_score,
                "risk": risk_score,
                "opportunity": opportunity_score,
                "reasoning": self._generate_reasoning(option, zhongyong_score)
            }
        
        best_option = max(scores.items(), key=lambda x: x[1]["total_score"])
        return {
            "recommended": best_option[0],
            "score": best_option[1]["total_score"],
            "reason": best_option[1]["reasoning"],
            "all_scores": scores
        }
    
    def _evaluate_balance(self, option: Dict) -> float:
        """评估方案的平衡度"""
        factors = option.get("factors", {})
        if not factors:
            return 0.5
        
        values = list(factors.values())
        mean = sum(values) / len(values)
        variance = sum(abs(x - mean) for x in values) / len(values)
        return 1 - variance  # 方差越小越平衡
    
    def _evaluate_risk(self, option: Dict) -> float:
        """评估风险（风险越低分数越高）"""
        risk = option.get("risk", 0.5)
        return 1 - risk  # 反转：低风险高分
    
    def _evaluate_opportunity(self, option: Dict) -> float:
        """评估机会"""
        return option.get("opportunity", 0.5)
    
    def _generate_reasoning(self, option: Dict, score: float) -> str:
        """生成中庸决策理由"""
```

```python
# yijing_engine.py
import random
import datetime
from typing import Dict, List, Tuple
from yijing_data import GUA_DATABASE, SOLAR_TERMS

class YijingEngine:
    """UID9622易经推演引擎"""
    
    def __init__(self):
        self.gua_db = GUA_DATABASE
        self.solar_terms = SOLAR_TERMS
    
    def get_current_solar_term(self) -> str:
        """获取当前节气（简化版本）"""
        month = datetime.datetime.now().month
        day = datetime.datetime.now().day
        
        # 简化的节气映射（实际使用建议用sxtwl库精确计算）
        if month == 2 and day >= 3: return "立春"
        elif month == 3 and day >= 20: return "春分"
        elif month == 5 and day >= 5: return "立夏"
        elif month == 6 and day >= 21: return "夏至"
        elif month == 8 and day >= 7: return "立秋"
        elif month == 9 and day >= 23: return "秋分"
        elif month == 11 and day >= 7: return "立冬"
        elif month == 12 and day >= 22: return "冬至"
        else: return "春分"  # 默认
    
    def cast_gua(self, question: str = "") -> Dict:
        """起卦 - 生成一个卦象"""
        # 方法1：随机起卦（模拟蓍草）
        gua_number = random.randint(1, 64)
        
        # 方法2：基于问题的hash起卦（确定性）
        if question:
            hash_value = sum(ord(c) for c in question)
            gua_number = (hash_value % 64) + 1
        
        return self.gua_db.get(gua_number, self.gua_db[1])
    
    def analyze_with_solar_term(self, gua: Dict) -> Dict:
        """结合节气分析卦象"""
        current_term = self.get_current_solar_term()
        term_data = self.solar_terms.get(current_term, {"weight": 1.0, "keywords": []})
        
        # 节气加权
        adjusted_fortune = gua["fortune"] * term_data["weight"]
        adjusted_fortune = min(max(adjusted_fortune, 0.0), 1.0)  # 限制在0-1
        
        return {
            "gua": gua,
            "solar_term": current_term,
            "term_weight": term_data["weight"],
            "term_keywords": term_data["keywords"],
            "adjusted_fortune": adjusted_fortune,
            "final_advice": self._generate_advice(gua, term_data, adjusted_fortune)
        }
    
    def _generate_advice(self, gua: Dict, term_data: Dict, fortune: float) -> str:
        """生成综合建议"""
        if fortune >= 0.8:
            level = "大吉"
        elif fortune >= 0.6:
            level = "吉"
        elif fortune >= 0.4:
            level = "中平"
        else:
            level = "需谨慎"
        
        advice = f"【{level}】{gua['meaning']}\n\n"
        advice += f"当前节气：{term_data.get('keywords', [])}之时\n"
        advice += f"卦象建议：{gua['advice']}\n"
        
        return advice
    
    def predict_timeline(self, question: str, years: int = 1) -> List[Dict]:
        """时间线预测 - 未来N年的卦象变化"""
        results = []
        
        for month in range(12 * years):
            future_date = datetime.datetime.now() + datetime.timedelta(days=30 * month)
            
            # 基于时间的确定性起卦
            time_seed = future_date.year * 10000 + future_date.month * 100 + future_date.day
            question_seed = sum(ord(c) for c in question)
            gua_number = ((time_seed + question_seed) % 64) + 1
            
            gua = self.gua_db.get(gua_number, self.gua_db[1])
            
            results.append({
                "date": future_date.strftime("%Y-%m"),
                "gua": gua["name"],
                "symbol": gua["symbol"],
                "fortune": gua["fortune"],
                "keywords": gua["keywords"]
            })
        
        return results
    
    def full_divination(self, question: str) -> Dict:
        """完整占卜流程"""
        print(f"\n🔮 正在为您占卜：{question}\n")
        
        # 1. 起卦
        gua = self.cast_gua(question)
        print(f"📊 得卦：{gua['symbol']} {gua['name']}卦")
        
        # 2. 节气分析
        analysis = self.analyze_with_solar_term(gua)
        print(f"🌿 当前节气：{analysis['solar_term']}")
        print(f"⭐ 运势指数：{analysis['adjusted_fortune']:.2%}")
        
        # 3. 时间线预测
        timeline = self.predict_timeline(question, years=1)
        print(f"\n📅 未来12个月卦象趋势：")
        for t in timeline[:6]:  # 只显示前6个月
            print(f"  {t['date']}: {t['symbol']} {t['gua']}（{t['fortune']:.0%}）")
        
        # 4. 综合建议
        print(f"\n💡 综合建议：\n{analysis['final_advice']}")
        
        return {
            "question": question,
            "gua": gua,
            "analysis": analysis,
            "timeline": timeline
        }
```

```jsx
function 提取甲骨笔画(文本) {
  // 甲骨文基本笔画：圜、直、斜、点、折
  const 笔画映射 = {
    '𠂉': { 阴阳: '阳', 五行: '木' },

    '𠃑': { 阴阳: '阳', 五行: '火' },
    '𠃓': { 阴阳: '阴', 五行: '金' },
    '𠄌': { 阴阳: '阳', 五行: '土' }
  };
  
  // 转化为甲骨笔画能量
  let 能量矩阵 = [];
  for (const 字 of 文本) {
    const 古字 = 转为甲骨文(字);

}
```

```jsx
function 生成本卦(能量) {
  // 能量转化为阴阳爻
  const 爻值 = [];
  for (let i = 0; i < 6; i++) {
    // 计算阈值，决定阴阳
    const 阈值 = 计算能量阈值(能量, i);
    爻值.push(阈值 > 0.5 ? '阳' : '阴');
  }
  
  // 形成卦象：下卦（初爻、二爻、三爻），上卦（四爻、五爻、上爻）
  const 下卦 = 确定卦象(爻值.slice(0, 3));
  const 上卦 = 确定卦象(爻值.slice(3, 6));
  
  return { 名称: 确定卦名(下卦, 上卦), 爻值, 下卦, 上卦 };
}

function 演变互卦(本卦) {
  // 互卦：由本卦二、三、四爻构成下卦，三、四、五爻构成上卦
  const 互卦爻值 = [
    本卦.爻值[1], 本卦.爻值[2], 本卦.爻值[3],
    本卦.爻值[2], 本卦.爻值[3], 本卦.爻值[4]
  ];
  
  const 下卦 = 确定卦象(互卦爻值.slice(0, 3));
  const 上卦 = 确定卦象(互卦爻值.slice(3, 6));
  
  return { 名称: 确定卦名(下卦, 上卦), 爻值: 互卦爻值, 下卦, 上卦 };
}

function 推演变卦(本卦, 能量) {
  // 确定变爻位置
  const 变爻位置 = 确定变爻(本卦, 能量);
  
  // 变爻：阳变阴，阴变阳
  const 变卦爻值 = [...本卦.爻值];
  变爻位置.forEach(位置 => {
    变卦爻值[位置] = 变卦爻值[位置] === '阳' ? '阴' : '阳';
  });
  
  const 下卦 = 确定卦象(变卦爻值.slice(0, 3));
  const 上卦 = 确定卦象(变卦爻值.slice(3, 6));
  
  return { 名称: 确定卦名(下卦, 上卦), 爻值: 变卦爻值, 下卦, 上卦, 变爻: 变爻位置 };
}
```

```jsx
function 归纳天机(五行属性, 卦象解读) {
  // 天道：卦象总体趋势
  const 天道 = 分析卦象趋势(卦象解读.本卦卦辞, 卦象解读.变卦卦辞);
  
  // 地道：五行能量流转
  const 地道 = 分析五行走势(五行属性.趋势);
  
  // 人道：具体爻位吉凶
  const 人道 = 分析变爻吉凶(卦象解读.爻辞解读);
  
  // 三才归一：天地人合一之道
  return {
    卦象: {
      本卦: 卦象解读.本卦卦辞.卦名,
      互卦: 卦象解读.互卦卦辞.卦名,
      变卦: 卦象解读.变卦卦辞.卦名
    },
    太极判断: 综合三才(天道, 地道, 人道),
    具体解析: {
      天道趋势: 天道,
      地道流转: 地道,
      人事吉凶: 人道
    },
    谋事建议: 生成行动建议(天道, 地道, 人道),
    太极三色: 判断三色(天道, 地道, 人道)
  };
}

function 综合三才(天道, 地道, 人道) {
  // 根据三才判断总体吉凶
  const 天地人权重 = [0.4, 0.3, 0.3]; // 天、地、人权重
  
  // 计算综合得分
  const 综合分数 = 
    天道.吉凶值 * 天地人权重[0] + 
    地道.吉凶值 * 天地人权重[1] + 
    人道.吉凶值 * 天地人权重[2];
  
  // 判断吉凶等级
  if (综合分数 > 0.7) return { 判断: '大吉', 分数: 综合分数, 解释: '诸事顺遂，可行动' };
  if (综合分数 > 0.5) return { 判断: '小吉', 分数: 综合分数, 解释: '稍有阻碍，谨慎可行' };
  if (综合分数 > 0.3) return { 判断: '中平', 分数: 综合分数, 解释: '平常之象，守正待时' };
  if (综合分数 > 0.1) return { 判断: '小凶', 分数: 综合分数, 解释: '有险阻，宜谨慎' };
  return { 判断: '大凶', 分数: 综合分数, 解释: '诸事不宜，应当止步' };
}

function 判断三色(天道, 地道, 人道) {
  // 太极三色：🟢🟡🔴
  const 综合分数 = (天道.吉凶值 + 地道.吉凶值 + 人道.吉凶值) / 3;
  
  if (综合分数 > 0.6) return '🟢 可行';
  if (综合分数 > 0.3) return '🟡 谨慎';
  return '🔴 不宜';
}
```

### 2. 起卦算法

```python
import time
import hashlib

def generate_hexagram(input_text, timestamp=None):
    """
    生成本卦
    input_text: 用户问题或意念
    timestamp: 时间戳（默认当前时间）
    """
    if timestamp is None:
        timestamp = time.time()
    
    # 混合输入与时间生成种子
    seed = f"{input_text}{timestamp}".encode('utf-8')
    hash_value = hashlib.sha256(seed).hexdigest()
    
    # 生成6个爻（0=阴爻，1=阳爻）
    lines = []
    for i in range(6):
        # 取哈希值的不同位置
        byte_value = int(hash_value[i*2:(i+1)*2], 16)
        lines.append(1 if byte_value > 127 else 0)
    
    # 确定变爻（老阴、老阳）
    change_lines = []
    for i in range(6):
        byte_value = int(hash_value[i*2+12:(i+1)*2+12], 16)
        # 四分之一概率为变爻
        if byte_value < 64 or byte_value > 191:
            change_lines.append(i)
    
    return {
        "lines": lines,
        "change_lines": change_lines,
        "binary": "".join(map(str, lines)),
        "hexagram_id": binary_to_id(lines)
    }

def binary_to_id(lines):
    """将二进制爻象转换为卦序号（1-64）"""
    binary_str = "".join(map(str, lines))
    # 根据伏羲六十四卦序查表
    return int(binary_str, 2) % 64 + 1
```
