5.1 压缩率 ρ（继承公式总册F23）
```
ρ = 1 − S_out / S_in          S=字节量
参考基线（论文实测）：文本语义压缩≥10倍、图片≥10倍、视频≥100倍
ρ为负（越压越大）→ 压缩器故障，自动回滚并报🟡
```

5.2 迭代压缩不动点判据（GAP-02补全 · Banach落地）
```
压缩算子 C：语义空间→语义空间。迭代序列 x₀=原文, x_{n+1}=C(x_n)
Banach压缩映象原理：C在语义空间中为压缩映射（保距常数k<1）时，
存在唯一不动点 x* 使 C(x*) = x* —— "再压也不变的那一段"，即不动点核。

工程停止规则（三取一即停）:
  ① 收敛：sim(x_{n+1}, x_n) ≥ 0.995（嵌入余弦相似度）→ 到核
  ② 兜底：n ≥ N_max = 7 轮 → 强制停，取当前x₇为核并标🟡"未完全收敛"
  ③ 护魂：σ(x_{n+1}, x₀) 跌破本档阈值（3.2）→ 回退到x_n，标"到核但不可再压"
核的处置：x* + DNA + 迭代轨迹(n, sim曲线) → 写入ROM候选区（第七章）
```
含义：周总结→月总结→季总结就是 x₁→x₂→x₃；压到"再压还是这几句话"时，
那就是你这段生命的核——刻进ROM，永不再动。

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
龍魂系统 · 记忆外脑迭代压缩引擎 v1.0
作者：诸葛鑫（UID9622）
优先级：P0++（最高，不可绕过）
DNA: #龍芯⚡️2026-07-20-EXOBRAIN-ENGINE-V1.0
"""

import hashlib, math, re
from collections import Counter

# ---------- 第五章参数（上链公开，修改=修协议） ----------
N_MAX       = 7        # 迭代上限
SIM_CONV    = 0.995    # 不动点收敛阈值
SIGMA       = {"核心": 0.95, "重要": 0.92, "常规": 0.90, "临时": 0.85}
T_HALF      = {"核心": math.inf, "重要": 365.0, "常规": 90.0, "临时": 7.0}  # 天
PULSE       = 0.3      # 复习脉冲
锚词        = ("DNA", "龍芯", "确认码", "GPG", "#", "协议", "签章", "铁律", "焊死")


class CNSH_记忆外脑引擎:
    """创建者原话的代码形态：重复压缩，迭代，归档，总结，继续识别。"""

    DNA = "#龍芯⚡️2026-07-20-EXOBRAIN-ENGINE-V1.0"

    # ===== 5.1 压缩率 =====
    @staticmethod
    def 压缩率(原文: str, 压缩文: str) -> float:
        return 1 - len(压缩文.encode()) / max(len(原文.encode()), 1)

    # ===== 语义相似度（词频余弦，工程代理；生产换嵌入向量） =====
    @staticmethod
    def _相似(a: str, b: str) -> float:
        ta, tb = Counter(re.findall(r"[\u4e00-\u9fff]|\w+", a)), \
                 Counter(re.findall(r"[\u4e00-\u9fff]|\w+", b))
        keys = set(ta) | set(tb)
        dot = sum(ta[k] * tb[k] for k in keys)
        na = math.sqrt(sum(v * v for v in ta.values()))
        nb = math.sqrt(sum(v * v for v in tb.values()))
        return dot / max(na * nb, 1e-9)

    # ===== 抽取式压缩算子（幂等：对压缩结果再压=不变，天然收敛） =====
    def _压一遍(self, 文本: str) -> str:
        句 = [s.strip() for s in re.split(r"[。！？!?\n]", 文本) if s.strip()]
        核 = [s for s in 句 if any(a in s for a in 锚词) or re.search(r"\d", s)]
        return "。".join(核) if 核 else 文本[:80]

    # ===== 5.2 迭代压缩·不动点（核心档σ校验+回退护魂） =====
    def 迭代压缩(self, 原文: str, 档: str = "常规") -> dict:
        x, 轨迹 = 原文, []
        for n in range(1, N_MAX + 1):
            x_next = self._压一遍(x)
            sim = self._相似(x_next, x)
            σ = self._相似(x_next, 原文)
            轨迹.append((n, round(sim, 4), round(σ, 4)))
            if σ < SIGMA[档]:                      # 护魂：σ跌破档阈值回退
                return {"核": x, "代数": n - 1, "轨迹": 轨迹,
                        "状态": "🟡 到核但不可再压", "dna": self.DNA}
            x = x_next
            if sim >= SIM_CONV:                    # 收敛：C(x*)=x*
                return {"核": x, "代数": n, "轨迹": 轨迹,
                        "状态": "🟢 不动点核", "dna": self.DNA}
        return {"核": x, "代数": N_MAX, "轨迹": 轨迹,
                "状态": "🟡 未完全收敛(强制停)", "dna": self.DNA}

    # ===== 5.4 simhash 去重（64位指纹） =====
    @staticmethod
    def 指纹(文本: str) -> int:
        v = [0] * 64
        for tok in re.findall(r"[\u4e00-\u9fff]{2}|\w+", 文本):
            h = int(hashlib.sha256(tok.encode()).hexdigest(), 16)
            for i in range(64):
                v[i] += 1 if (h >> i) & 1 else -1
        return sum((1 << i) for i in range(64) if v[i] > 0)

    def 汉明(self, a: int, b: int) -> int:
        return bin(a ^ b).count("1")

    def 去重判定(self, 文a: str, 文b: str) -> str:
        sim = 64 - self.汉明(self.指纹(文a), self.指纹(文b))
        if sim >= 61: return f"重复(sim={sim})→合并留档"
        if sim >= 56: return f"近缘(sim={sim})→挂关联边"
        return f"不同(sim={sim})→各自独立"

    # ===== 5.5 衰减 + 复习脉冲 =====
    @staticmethod
    def 强度(M0: float, 天数: float, 档: str) -> float:
        h = T_HALF[档]
        return M0 if h == math.inf else M0 * math.exp(-math.log(2) / h * 天数)

    @staticmethod
    def 复习(M: float) -> float:
        return min(1.0, M + PULSE)

    # ===== 5.6 重要性评分 =====
    @staticmethod
    def 重要性(用户标记: int, 访问频率: float, 关联数: int, 情感: float) -> float:
        return round(0.35 * 用户标记 / 10 + 0.25 * min(访问频率, 1.0)
                     + 0.25 * min(关联数, 10) / 10 + 0.15 * min(情感, 1.0), 3)

    # ===== 5.7 分层决策 =====
    @staticmethod
    def 分层(I: float, 最近访问天数: float, 不动点核: bool = False,
             用户焊死: bool = False, 封存: bool = False) -> str:
        if 封存: return "封存(冻结不删)"
        if 不动点核 or 用户焊死: return "ROM永久(只读)"
        if I >= 0.50 or 最近访问天数 <= 30: return "热层"
        if I >= 0.20: return "温层"
        return "冷层"
