## 🧠 核心定义

### 什么是逻辑溯源（Logical Provenance）？

**逻辑溯源是龙魂系统的"诚实引擎"——让每一个思想、每一个算法、每一个设计决策都能追溯到它的根。**

```
普通系统 → 你问"这个想法从哪来？" → 沉默或编造
龙魂系统 → 你问"这个想法从哪来？" → 精确到章节、定理、论文、对话记录
```

### 为什么逻辑溯源是主权问题

| 无溯源的系统 | 有溯源的系统 |
|:---|:---|
| 假装思想是凭空产生的 | 承认所有思想都有根 |
| 无法区分原创与借用 | 原创与引用边界清晰 |
| 被质疑时无法自证 | 被质疑时一键出示来源 |
| 黑箱 | 白箱 |
| 文化盗窃的温床 | 文化主权的护城河 |

**逻辑溯源不是谦虚，是主权——龙魂系统的每一个逻辑都必须能说出"我从哪来"。**

def generate_dna(action_tag: str = "PROV", version: str = "v2.0") -> str:
    """生成rizhu v3.0标准DNA"""
    dt = datetime.now()
    year = dt.year
    if dt.month < 2 or (dt.month == 2 and dt.day < 4):
        year -= 1
    year_gz = GAN[(year - 4) % 10] + ZHI[(year - 4) % 12]
    year_gan_idx = (year - 4) % 10
    if year_gan_idx in [0, 5]: start_gan = 2
    elif year_gan_idx in [1, 6]: start_gan = 4
    elif year_gan_idx in [2, 7]: start_gan = 6
    elif year_gan_idx in [3, 8]: start_gan = 8
    else: start_gan = 0
    lunar_month = dt.month - 1 if dt.month > 1 else 12
    month_gz = GAN[(start_gan + lunar_month - 1) % 10] + ZHI[(lunar_month + 1) % 12]
    delta = (dt - datetime(1900, 1, 31)).days
    day_gz = GAN[delta % 10] + ZHI[delta % 12]
    zhi_idx = (dt.hour + 1) // 2 % 12
    day_gan_idx = GAN.index(day_gz[0])
    if day_gan_idx in [0, 5]: h_start = 0
    elif day_gan_idx in [1, 6]: h_start = 2

                    d = d.setdefault(p, {})
                d[parts[-1]] = value

        d = self.config
        for key in keys:
            if isinstance(d, dict) and key in d:
                d = d[key]
            else:
                return default
        return d

class SecurityLayer:
    """安全层 - P0: 确认码闸门"""

    CONFIRM_CODE = "#CONFIRM🌌9622-ONLY-ONCE🧬LK9X-772Z"

    def verify_confirm_code(self, code: str) -> bool:
        return hmac.compare_digest(code, self.CONFIRM_CODE)

    def compute_checksum(self, content: str, algorithm: str = "sha256") -> str:
        """计算内容校验和"""
        if algorithm == "sha256":
            return hashlib.sha256(content.encode()).hexdigest()
        raise ValueError(f"不支持的算法: {algorithm}")

    def verify_checksum(self, content: str, expected: str, algorithm: str = "sha256") -> bool:
        """验证内容校验和"""
        return hmac.compare_digest(self.compute_checksum(content, algorithm), expected)

    def audit_log(self, action: str, identity: str, result: str):
        import logging
        logging.getLogger("provenance.security").info(
            f"AUDIT | action={action} | identity={identity} | result={result}"
        )

class PersistenceLayer:
    """持久化层 - 溯源记录不丢、可回溯、可审计"""

    def __init__(self, data_dir: Path):
        self.data_dir = Path(data_dir).expanduser()
        self.data_dir.mkdir(parents=True, exist_ok=True)
        self.db_path = self.data_dir / "provenance.db"
        self.lock = threading.RLock()
        self._init_db()

    def _init_db(self):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS provenance_records (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    module TEXT NOT NULL,
                    dna TEXT NOT NULL,
                    timestamp TEXT NOT NULL,
                    checksum TEXT NOT NULL,
                    content TEXT NOT NULL,
                    status TEXT DEFAULT '🟢',
                    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
                )
            """)
            conn.execute("CREATE INDEX IF NOT EXISTS idx_module ON provenance_records(module)")
            conn.execute("CREATE INDEX IF NOT EXISTS idx_dna ON provenance_records(dna)")
            conn.commit()

    def save(self, record: Dict):
        with self.lock:
            with sqlite3.connect(self.db_path) as conn:
                conn.execute("""
                    INSERT INTO provenance_records (module, dna, timestamp, checksum, content, status)
                    VALUES (?, ?, ?, ?, ?, ?)
                """, (
                    record.get("module", ""),
                    record.get("dna", ""),
                    record.get("timestamp", datetime.now().isoformat()),
                    record.get("checksum", ""),
                    json.dumps(record, ensure_ascii=False),
                    record.get("status", "🟢")
                ))
                conn.commit()

    def load_by_module(self, module: str, limit: int = 100) -> List[Dict]:
        with self.lock:
            with sqlite3.connect(self.db_path) as conn:
                conn.row_factory = sqlite3.Row
                cursor = conn.execute(
                    "SELECT * FROM provenance_records WHERE module = ? ORDER BY timestamp DESC LIMIT ?",
                    (module, limit)
                )
                return [dict(row) for row in cursor.fetchall()]

    def load_all(self, limit: int = 1000) -> List[Dict]:
        with self.lock:
            with sqlite3.connect(self.db_path) as conn:
                conn.row_factory = sqlite3.Row
                cursor = conn.execute(
                    "SELECT * FROM provenance_records ORDER BY timestamp DESC LIMIT ?", (limit,)
                )
                return [dict(row) for row in cursor.fetchall()]

    def get_stats(self) -> Dict:
        with self.lock:
            with sqlite3.connect(self.db_path) as conn:
                total = conn.execute("SELECT COUNT(*) FROM provenance_records").fetchone()[0]
                modules = conn.execute("SELECT COUNT(DISTINCT module) FROM provenance_records").fetchone()[0]
                return {"total_records": total, "unique_modules": modules}

class LogicalProvenanceEngine:
    """逻辑溯源引擎 v2.0"""

    def __init__(self, config_path: Optional[str] = None):
        self.config = ConfigManager(config_path)
        self.security = SecurityLayer()
        data_dir = Path(self.config.get("provenance", "data_dir", default="~/.longhun/provenance/data"))
        self.persistence = PersistenceLayer(data_dir)
        self.db = PROVENANCE_DB
        self.lock = threading.RLock()
        self.dna = generate_dna("PROVENANCE-ENGINE", "v2.0")

    def inject(self, module_name: str, confirm_code: str = "") -> Dict:
        """为模块注入逻辑溯源 - P0: 确认码闸门"""
        if self.config.get("security", "confirm_code_required", default=True):
            if not self.security.verify_confirm_code(confirm_code):
                self.security.audit_log("inject", module_name, "REJECTED: invalid confirm code")
                return {"error": "确认码验证失败", "status": "🔴", "code": 403}

        with self.lock:
            sources = self.db.get(module_name, [])
            if not sources:
                # 尝试模糊匹配
                sources = self._fuzzy_match(module_name)

            record = {
                "module": module_name,
                "dna": generate_dna(f"PROV-{module_name}", "v2.0"),
                "timestamp": datetime.now().isoformat(),
                "sources": [s.to_dict() for s in sources],
                "status": "🟢"
            }

            # 计算内容校验和
            content_json = json.dumps(record, ensure_ascii=False, sort_keys=True)
            record["checksum"] = self.security.compute_checksum(content_json)

            # 持久化
            if self.config.get("provenance", "auto_persist", default=True):
                self.persistence.save(record)

            self.security.audit_log("inject", module_name, "SUCCESS")
            return record

    def _fuzzy_match(self, module_name: str) -> List[SourceRecord]:
        """模糊匹配模块名"""
        for key in self.db:
            if key in module_name or module_name in key:
                return self.db[key]
        return [SourceRecord("original", module_name, "UID9622·auto",
                             "自动标记为原创（无匹配来源）", 0.5)]

    def batch_inject(self, directory: str, pattern: str = "*.py", confirm_code: str = "") -> List[Dict]:
        """批量注入目录下所有模块"""
        results = []
        path = Path(directory)
        for file in path.rglob(pattern):
            module_name = file.stem
            result = self.inject(module_name, confirm_code)
            results.append(result)
        return results

    def verify(self, module_name: str, expected_checksum: str) -> Dict:
        """验证模块溯源记录的完整性"""
        records = self.persistence.load_by_module(module_name, limit=1)
        if not records:
            return {"verified": False, "reason": "无溯源记录", "status": "🔴"}

        record = json.loads(records[0]["content"])
        content = {k: v for k, v in record.items() if k != "checksum"}
        content_json = json.dumps(content, ensure_ascii=False, sort_keys=True)

        if self.security.verify_checksum(content_json, expected_checksum):
            return {"verified": True, "dna": record["dna"], "status": "🟢"}
        else:
            return {"verified": False, "reason": "校验和不匹配", "status": "🔴"}

    def generate_markdown(self, record: Dict) -> str:
        """生成Markdown格式溯源报告"""
        lines = []
        lines.append(f"## 🧬 逻辑溯源 · {record['module']}")
        lines.append("")
        lines.append(f"**DNA:** `{record['dna']}`")
        lines.append(f"**时间:** {record['timestamp']}")
        lines.append(f"**校验和:** `{record.get('checksum', 'N/A')}`")
        lines.append("")

        # 按分类分组
        categories = {}
        for src in record.get("sources", []):
            cat = src.get("category", "unknown")
            categories.setdefault(cat, []).append(src)

        cat_names = {
            "philosophy": "📜 哲学根基",
            "mathematics": "🔢 数学依据",
            "engineering": "⚙️ 工程参考",
            "original": "🐉 龙魂原创",
            "ai_collaboration": "🤖 AI协作"
        }

        for cat, sources in categories.items():
            lines.append(f"### {cat_names.get(cat, cat)}")
            lines.append("")
            for src in sources:
                lines.append(f"- **{src['source_name']}**")
                if src.get('reference'):
                    lines.append(f"  - 引用: `{src['reference']}`")
                lines.append(f"  - 影响: {src.get('influence', '')}")
                conf = src.get('confidence', 0)
                color = "🟢" if conf >= 0.9 else "🟡" if conf >= 0.7 else "🔴"
                lines.append(f"  - 置信度: {color} {conf}")
                lines.append("")

        return "\n".join(lines)

    def generate_json(self, record: Dict) -> str:
        """生成JSON格式溯源报告"""
        return json.dumps(record, ensure_ascii=False, indent=2)

    def list_modules(self) -> List[str]:
        """列出所有已注册模块"""
        return list(self.db.keys())

    def get_stats(self) -> Dict:
        """获取引擎统计"""
        db_stats = self.persistence.get_stats()
        return {
            "registered_modules": len(self.db),
            "persistence": db_stats,
            "engine_dna": self.dna
        }

def main():
    import argparse
    parser = argparse.ArgumentParser(description="🐉 逻辑溯源引擎 (LPE) v2.0")
    parser.add_argument("--inject", "-i", type=str, help="注入单个模块")
    parser.add_argument("--batch", "-b", type=str, help="批量注入目录")
    parser.add_argument("--pattern", type=str, default="*.py", help="批量匹配模式")
    parser.add_argument("--verify", "-v", type=str, help="验证模块校验和")
    parser.add_argument("--checksum", type=str, help="期望校验和")
    parser.add_argument("--list", "-l", action="store_true", help="列出所有模块")
    parser.add_argument("--stats", "-s", action="store_true", help="统计信息")
    parser.add_argument("--json", "-j", action="store_true", help="JSON输出")
    parser.add_argument("--confirm-code", type=str, default="", help="确认码（P0安全闸门）")

    args = parser.parse_args()
    engine = LogicalProvenanceEngine()

    if args.list:
        modules = engine.list_modules()
        print("\n🐉 已注册溯源模块:")
        for m in modules:
            print(f"  · {m}")
        return

    if args.stats:
        stats = engine.get_stats()
        if args.json:
            print(json.dumps(stats, indent=2, ensure_ascii=False))
        else:
            print(f"\n📊 统计信息:")
            print(f"  注册模块: {stats['registered_modules']}")
            print(f"  持久化记录: {stats['persistence']['total_records']}")
            print(f"  唯一模块: {stats['persistence']['unique_modules']}")
            print(f"  引擎DNA: {stats['engine_dna']}")
        return

    if args.inject:
        result = engine.inject(args.inject, confirm_code=args.confirm_code or CONFIRM_CODE)
        if "error" in result:
            print(f"\n🔴 错误: {result['error']}")
            return
        if args.json:
            print(engine.generate_json(result))
        else:
            print(engine.generate_markdown(result))
        return

    if args.batch:
        results = engine.batch_inject(args.batch, args.pattern, confirm_code=args.confirm_code or CONFIRM_CODE)
        print(f"\n📦 批量注入完成: {len(results)} 个模块")
        for r in results:
            status = r.get("status", "🟢")
            name = r.get("module", "unknown")
            print(f"  {status} {name}")
        return

    if args.verify and args.checksum:
        result = engine.verify(args.verify, args.checksum)
        print(f"\n{'🟢' if result['verified'] else '🔴'} 验证结果: {result}")
        return

    parser.print_help()

if __name__ == "__main__":
    main()
