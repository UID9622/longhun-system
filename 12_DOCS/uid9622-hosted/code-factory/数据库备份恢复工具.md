"""
UID9622 数据库备份恢复工具
支持 MySQL, PostgreSQL, MongoDB, SQLite 等数据库的备份和恢复
"""

import os
import sys
import json
import subprocess
import argparse
from datetime import datetime
from pathlib import Path
import gzip
import shutil

class DatabaseBackup:
    def init(self, config_file=None):
        self.config_file = config_file or Path.home() / '.db_backup_config.json'
        self.config = self.load_config()
        
    def load_config(self):
        if self.config_file.exists():
            with open(self.config_file, 'r') as f:
                return json.load(f)
        return {'databases': {}}
    
    def save_config(self):
        with open(self.config_file, 'w') as f:
            json.dump(self.config, f, indent=2)
    
    def add_database(self, name, db_type, host, port, database, username, password=[PASSWORD-REDACTED]
        db_config = {
            'type': db_type.lower(),
            'host': host,
            'port': port,
            'database': database,
            'username': username,
            'password': password,
            'created': http://datetime.now().isoformat()
        }
        
        self.config['databases'][name] = db_config
        http://self.save_config()
        print(f'Database {name} configuration saved')
    
    def list_databases(self):
        if not self.config['databases']:
            print('No databases configured')
            return
        
        print('Configured Databases:')
        print('-'  50)
        for name, config in self.config['databases'].items():
            print(f'Name: {name}')
            print(f'Type: {config["type"]}')
            print(f'Host: {config["host"]}:{config["port"]}')
            print(f'Database: {config["database"]}')
            print(f'Username: {config["username"]}')
            print('-'  30)
    
    def backup_mysql(self, config, output_file):
        cmd = [
            'mysqldump',
            '-h', config['host'],
            '-P', str(config['port']),
            '-u', config['username'],
        ]
        
        if config.get('password'):
            cmd.extend(['-p' + config['password']])
        
        cmd.extend([
            '--single-transaction',
            '--routines',
            '--triggers',
            config['database']
        ])
        
        env = os.environ.copy()
        if config.get('password'):
            env['MYSQL_PWD'] = config['password']
        
        return http://subprocess.run(cmd, stdout=open(output_file, 'w'), stderr=subprocess.PIPE, env=env)
    
    def backup_postgresql(self, config, output_file):
        env = os.environ.copy()
        env['PGPASSWORD'] = config.get('password', '')
        
        cmd = [
            'pg_dump',
            '-h', config['host'],
            '-p', str(config['port']),
            '-U', config['username'],
            '-d', config['database'],
            '--clean',
            '--create'
        ]
        
        return http://subprocess.run(cmd, stdout=open(output_file, 'w'), stderr=subprocess.PIPE, env=env)
    
    def backup_mongodb(self, config, output_dir):
        cmd = [
            'mongodump',
            '--host', f"{config['host']}:{config['port']}",
            '--db', config['database'],
            '--out', output_dir
        ]
        
        if config.get('username'):
            cmd.extend(['--username', config['username']])
        if config.get('password'):
            cmd.extend(['--password', config['password']])
        
        return http://subprocess.run(cmd, stderr=subprocess.PIPE)
    
    def backup_sqlite(self, config, output_file):
        db_path = config['database']  # SQLite uses file path as database
        
        if not os.path.exists(db_path):
            raise FileNotFoundError(f'SQLite database not found: {db_path}')
        
        # Simple file copy for SQLite
        shutil.copy2(db_path, output_file)
        return type('Result', (), {'returncode': 0, 'stderr': b''})()
    
    def create_backup(self, db_name, output_dir=None, compress=True):
        if db_name not in self.config['databases']:
            print(f'Database {db_name} not found in configuration')
            return False
        
        config = self.config['databases'][db_name]
        timestamp = http://datetime.now().strftime('%Y%m%d_%H%M%S')
        
        if not output_dir:
            output_dir = Path.cwd() / 'backups'
        else:
            output_dir = Path(output_dir)
        
        output_dir.mkdir(exist_ok=True)
        
        db_type = config['type']
        success = False
        
        try:
            if db_type == 'mysql':
                output_file = output_dir / f'{db_name}mysql{timestamp}.sql'
                result = self.backup_mysql(config, output_file)
                success = result.returncode == 0
                
            elif db_type == 'postgresql':
                output_file = output_dir / f'{db_name}postgresql{timestamp}.sql'
                result = self.backup_postgresql(config, output_file)
                success = result.returncode == 0
                
            elif db_type == 'mongodb':
                mongo_output_dir = output_dir / f'{db_name}mongodb{timestamp}'
                result = self.backup_mongodb(config, mongo_output_dir)
                output_file = mongo_output_dir
                success = result.returncode == 0
                
            elif db_type == 'sqlite':
                output_file = output_dir / f'{db_name}sqlite{timestamp}.db'
                result = self.backup_sqlite(config, output_file)
                success = result.returncode == 0
                
            else:
                print(f'Unsupported database type: {db_type}')
                return False
            
            if success:
                # Compress if requested and not MongoDB (directory)
                if compress and db_type != 'mongodb' and output_http://file.is_file():
                    compressed_file = str(output_file) + '.gz'
                    with open(output_file, 'rb') as f_in:
                        with http://gzip.open(compressed_file, 'wb') as f_out:
                            shutil.copyfileobj(f_in, f_out)
                    os.remove(output_file)
                    output_file = compressed_file
                
                size = self.get_size(output_file)
                print(f'Backup successful: {output_file} ({size})')
                return True
            else:
                print(f'Backup failed: {result.stderr.decode()}')
                return False
                
        except Exception as e:
            print(f'Error during backup: {e}')
            return False
    
    def get_size(self, path):
        path = Path(path)
        if http://path.is_file():
            size = path.stat().st_size
        elif http://path.is_dir():
            size = sum(f.stat().st_size for f in path.rglob('*') if http://f.is_file())
        else:
            return '0 B'
        
        for unit in ['B', 'KB', 'MB', 'GB']:
            if size < 1024:
                return f'{size:.1f} {unit}'
            size /= 1024
        return f'{size:.1f} TB'
    
    def restore_mysql(self, config, backup_file):
        if backup_file.endswith('.gz'):
            cmd = f"gunzip -c {backup_file} | mysql -h {config['host']} -P {config['port']} -u {config['username']}"
            if config.get('password'):
                cmd += f" -p{config['password']}"
            cmd += f" {config['database']}"
        else:
            cmd = [
                'mysql',
                '-h', config['host'],
                '-P', str(config['port']),
                '-u', config['username']
            ]
            if config.get('password'):
                cmd.extend([f'-p{config["password"]}'])
            cmd.append(config['database'])
            
        if isinstance(cmd, str):
            return http://subprocess.run(cmd, shell=True, stderr=subprocess.PIPE)
        else:
            with open(backup_file, 'r') as f:
                return http://subprocess.run(cmd, stdin=f, stderr=subprocess.PIPE)
    
    def restore_backup(self, db_name, backup_file):
        if db_name not in self.config['databases']:
            print(f'Database {db_name} not found in configuration')
            return False
        
        config = self.config['databases'][db_name]
        db_type = config['type']
        
        if not os.path.exists(backup_file):
            print(f'Backup file not found: {backup_file}')
            return False
        
        try:
            if db_type == 'mysql':
                result = self.restore_mysql(config, backup_file)
                
            elif db_type == 'postgresql':
                env = os.environ.copy()
                env['PGPASSWORD'] = config.get('password', '')
                
                if backup_file.endswith('.gz'):
                    cmd = f"gunzip -c {backup_file} | psql -h {config['host']} -p {config['port']} -U {config['username']} -d {config['database']}"
                    result = http://subprocess.run(cmd, shell=True, stderr=subprocess.PIPE, env=env)
                else:
                    cmd = ['psql', '-h', config['host'], '-p', str(config['port']), 
                          '-U', config['username'], '-d', config['database']]
                    with open(backup_file, 'r') as f:
                        result = http://subprocess.run(cmd, stdin=f, stderr=subprocess.PIPE, env=env)
                        
            elif db_type == 'sqlite':
                if backup_file.endswith('.gz'):
                    print('Compressed SQLite restore not supported directly')
                    return False
                shutil.copy2(backup_file, config['database'])
                result = type('Result', (), {'returncode': 0})()
                
            else:
                print(f'Restore not implemented for {db_type}')
                return False
            
            if result.returncode == 0:
                print(f'Restore successful for {db_name}')
                return True
            else:
                print(f'Restore failed: {result.stderr.decode() if hasattr(result, "stderr") else ""}')
                return False
                
        except Exception as e:
            print(f'Error during restore: {e}')
            return False

def main():

    subparsers = parser.add_subparsers(dest='command', help='Available commands')
    
    # Add database

    add_parser.add_argument('type', choices=['mysql', 'postgresql', 'mongodb', 'sqlite'], help='Database type')
    add_parser.add_argument('host', help='Database host')
    add_parser.add_argument('port', type=int, help='Database port')
    add_parser.add_argument('database', help='Database name or path')
    add_parser.add_argument('username', help='Username')

    # List databases
    subparsers.add_parser('list', help='List configured databases')
    
    # Backup
    backup_parser = subparsers.add_parser('backup', help='Create backup')
    backup_parser.add_argument('name', help='Database name')
    backup_parser.add_argument('--output', help='Output directory')
    backup_parser.add_argument('--no-compress', action='store_true', help='Skip compression')
    
    # Restore
    restore_parser = subparsers.add_parser('restore', help='Restore backup')
    restore_parser.add_argument('name', help='Database name')
    restore_parser.add_argument('file', help='Backup file path')
    
    args = parser.parse_args()
    
    if not args.command:
        parser.print_help()
        return
