import psutil
import socket
import subprocess
import hashlib
import os
import json
from datetime import datetime

class UID9622_DeviceProtection:
    def __init__(self):
        self.baseline_processes = self.get_current_processes()
        self.baseline_network = self.get_network_connections()
        self.log_file = 'uid9622_security.log'
        
    def get_current_processes(self):
        processes = []
        for proc in psutil.process_iter(['pid', 'name', 'exe']):
            try:
                processes.append(proc.info)
            except:
                continue
        return processes
        
    def detect_new_processes(self):
        current = self.get_current_processes()
        new_processes = []
        
        for proc in current:
            if proc not in self.baseline_processes:
                if self.is_suspicious_process(proc):
                    new_processes.append(proc)
                    self.log_security_event('SUSPICIOUS_PROCESS', proc)
        
        return new_processes
    
    def check_network_activity(self):
        connections = psutil.net_connections()
        suspicious_connections = []
        
        for conn in connections:
            if conn.status == 'ESTABLISHED' and conn.raddr:
                if self.is_suspicious_connection(conn):
                    suspicious_connections.append(conn)
                    self.log_security_event('SUSPICIOUS_NETWORK', conn)
        
        return suspicious_connections
    
    def run_protection_scan(self):
        print('UID9622 设备保护扫描开始...')
        
        new_procs = self.detect_new_processes()
        susp_network = self.check_network_activity()
        
        report = {
            'scan_time': datetime.now().isoformat(),
            'new_processes': len(new_procs),
            'suspicious_network': len(susp_network),
            'status': 'PROTECTED' if not any([new_procs, susp_network]) else 'THREATS_DETECTED'
        }
        
        print(f'扫描完成: {report["status"]}')
        return report
    
    # 需要实现的辅助方法
    def is_suspicious_process(self, proc):
        # 检测可疑进程的逻辑
        suspicious_names = ['keylogger', 'capture', 'monitor', 'spy']
        if proc['name'] and any(s in proc['name'].lower() for s in suspicious_names):
            return True
        return False
    
    def is_suspicious_connection(self, conn):
        # 检测可疑网络连接的逻辑
        suspicious_ports = [31337, 12345, 1080]
        if conn.raddr.port in suspicious_ports:
            return True
        return False

        return psutil.net_connections()

        with open(self.log_file, 'a') as f:
            event = {
                'timestamp': datetime.now().isoformat(),
                'type': event_type,
                'data': str(data)
            }

2. 执行以下命令删除桌面上的脚本文件：

```bash
rm ~/Desktop/device_protection.py
rm ~/Desktop/scan_results.log
rm ~/Desktop/uid9622_security.log

```

### 方法二：清空文件内容

1. 打开终端应用
2. 执行以下命令清空脚本文件但保留文件本身：

```bash
echo "" > ~/Desktop/device_protection.py
echo "" > ~/Desktop/scan_results.log
touch ~/Desktop/uid9622_security.log

```

### 方法三：移除定时任务

如果你已经设置了crontab任务，需要将其移除：

1. 打开终端应用
2. 查看当前的crontab任务：

```bash
crontab -l

```

1. 编辑crontab移除相关任务：

```bash
crontab -e

```

然后在编辑界面中删除包含"device_protection.py"的行，保存并退出。

### 确认清理完成

1. 验证文件已被删除或清空：

```bash
ls -la ~/Desktop/device_protection.py 2>/dev/null || echo "文件已成功删除"

```

1. 验证crontab任务已移除：

```bash
crontab -l | grep device_protection || echo "crontab任务已移除"

```

完成以上步骤后，您已经彻底清除了脚本及其相关组件。如果您之后想要重新安装，可以从头开始创建新的脚本文件。
