```html
        const userLang = navigator.language || navigator.userLanguage;
```

**原理：AI自动翻译成用户母语**

```python
#!/usr/bin/env python3
# ~/UID9622/🌍 翻译系统/auto_[translate.py](http://translate.py)

import ollama
import langdetect

class AdaptiveTranslator:
    """自适应人群翻译器"""
    
    def __init__(self):
        self.supported_languages = {
            'zh-cn': '中文（简体）',
            'zh-tw': '中文（繁体）',
            'en': 'English',
            'ja': '日本语',
            'ko': '한국어',
            'es': 'Español',
            'fr': 'Français',
            'de': 'Deutsch',
            'ru': 'Русский',
            'ar': 'العربية'
        }
    
    def detect_user_language(self, text):
        """检测用户使用的语言"""
        try:
            lang = langdetect.detect(text)
            return lang
        except:
            return 'en'  # 默认英文
    
    def translate_content(self, content, target_lang):
        """用本地AI翻译内容"""
        
        prompt = f"""
你是UID9622翻译系统。

任务：将以下内容翻译成{self.supported_languages.get(target_lang, '英文')}。

要求：
1. 保持原意
2. 符合目标语言习惯
3. 保留专有名词（如UID9622）
4. 保持语气和风格

原文：
{content}

翻译：
"""
        
        response = ollama.generate(
            model='qwen2.5:72b',
            prompt=prompt
        )
        
        return response['response']
    
    def auto_adapt(self, user_input):
        """自动适配用户语言"""
        # 检测用户语言
        user_lang = self.detect_user_language(user_input)
        print(f"检测到用户语言：{self.supported_languages.get(user_lang, '未知')}")
        
        # 系统回复（默认中文）
        system_reply = "UID9622为人民服务！这是一个属于普通人的AI系统。"
        
        # 如果用户不是中文，自动翻译
        if user_lang != 'zh-cn':
            print("正在翻译成用户母语...")
            translated = self.translate_content(system_reply, user_lang)
            return translated
        else:
            return system_reply

if __name__ == "__main__":

    # 测试：美国用户
```

```bash
            try {
                const response = await fetch('/translate', {

                    headers: {

                    },
                    body: JSON.stringify({text: text})

                resultDiv.innerHTML = `

                    &lt;p&gt;${data.translated}&lt;/p&gt;

if __name__ == "__main__":

    app.run(host='0.0.0.0', port=5000, debug=True)
```
