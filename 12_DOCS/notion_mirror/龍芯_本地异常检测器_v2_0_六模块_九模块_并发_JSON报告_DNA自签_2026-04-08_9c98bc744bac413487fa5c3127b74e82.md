def run_cmd(cmd, timeout=15):
    try:
        result = subprocess.run(
            cmd, shell=True, capture_output=True,
            text=True, timeout=timeout
        )
        return result.stdout + result.stderr
    except Exception:
        return ""


def is_suspicious(text):
    text_lower = text.lower()
    for safe in SAFE_KEYWORDS:
        if safe in text_lower:
            return False, None
    for kw in SUSPICIOUS_KEYWORDS:
        if kw in text_lower:
            return True, kw
    return False, None


def add_result(level, item):

    with results_lock:
        results[level].append(item)

            log(f"  🚨 发现含'{kw}'的证书！", color=C.RED)
            add_result("p0", f"危险证书·含'{kw}': {output[:200]}")

    self_signed = run_cmd(
        f'security find-certificate -a "{Path.home()}/Library/Keychains/login.keychain-db" '
        f'2>/dev/null | grep "labl" | grep -v "Apple\|DigiCert\|Comodo\|Let\'s Encrypt" | head -20'
    )
    if self_signed.strip():
        for line in self_signed.split("\n"):
            if line.strip():
                log(f"  🟡 非主流CA证书: {line.strip()[:80]}", color=C.ORANGE)
                add_result("p2", f"非主流CA证书: {line.strip()[:80]}")

        log(f"  🔴 发现 {len(suspicious_certs)} 个可疑证书！", color=C.RED)

def scan_network():
    header("🌐 M3·网络异常扫描")

    log("[3.1] 检查 DNS 服务器...", color=C.BLUE)
    dns_output = run_cmd("scutil --dns | grep 'nameserver' | sort -u")
    if dns_output.strip():
        safe_dns = [
            "8.8.8.8", "8.8.4.4", "1.1.1.1", "1.0.0.1",
            "114.114.114.114", "223.5.5.5", "119.29.29.29",
            "180.76.76.76", "223.6.6.6"
        ]
        for line in dns_output.split("\n"):
            if not line.strip():
                continue
            ip = line.split(":")[-1].strip() if ":" in line else line.strip()
            if any(s in ip for s in safe_dns) or ip.startswith("192.168") or ip.startswith("10."):
                log(f"  ✅ DNS: {ip} (正常)", color=C.GREEN)
            else:
                log(f"  ⚠️  DNS: {ip} (未知DNS，需关注)", color=C.ORANGE)
                add_result("p1", f"未知DNS服务器: {ip}")

    log("[3.2] 检查系统代理设置...", color=C.BLUE)
    proxy_output = run_cmd("scutil --proxy")
    for line in proxy_output.split("\n"):
        if "Enable : 1" in line:
            log(f"  ⚠️  代理已启用: {line.strip()}", color=C.ORANGE)
            add_result("p1", f"代理已启用: {line.strip()}")
    else:
        log("  ✅ 未发现异常代理", color=C.GREEN)

    log("[3.3] 检查 /etc/hosts ...", color=C.BLUE)
    hosts_output = run_cmd("cat /etc/hosts")
    for line in hosts_output.split("\n"):
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        suspicious, kw = is_suspicious(line)
        if suspicious:
            log(f"  🚨 可疑hosts·含'{kw}': {line}", color=C.RED)
            add_result("p0", f"可疑hosts: {line}")
        elif not any(x in line for x in ["127.0.0.1", "::1", "255.", "broadcast",
                                          "localhost", "0.0.0.0", "fe80", "ff0"]):
            log(f"  ⚠️  非标准hosts: {line}", color=C.ORANGE)
            add_result("p1", f"非标准hosts: {line}")

    log("[3.4] 检查活跃网络连接...", color=C.BLUE)
    conn_output = run_cmd("netstat -an | grep ESTABLISHED | head -40")
    for line in conn_output.split("\n"):
        suspicious, kw = is_suspicious(line)
        if suspicious:
            log(f"  🚨 可疑连接·含'{kw}': {line.strip()[:100]}", color=C.RED)
            add_result("p1", f"可疑连接: {line.strip()[:100]}")

    # v2.0新增：外连非标准端口检测
    log("[3.5] 检查可疑外连端口...", color=C.BLUE)
    lsof_out = run_cmd("lsof -i -n -P 2>/dev/null | grep ESTABLISHED | head -30")
    suspicious_ports = []
    for line in lsof_out.split("\n"):
        if not line.strip():
            continue
        # 检查是否有非标准端口
        parts = line.split()
        for part in parts:
            if "->" in part:
                dst = part.split("->")[-1]
                if ":" in dst:
                    port = dst.split(":")[-1]
                    try:
                        port_num = int(port)
                        if port_num not in [80, 443, 53, 22, 25, 587, 993, 995, 8080, 8443]:
                            suspicious, kw = is_suspicious(line)
                            if suspicious:
                                log(f"  🚨 可疑外连·端口{port_num}·含'{kw}': {line[:80]}", color=C.RED)
                                add_result("p0", f"可疑外连[端口{port_num}]: {line[:80]}")
                    except ValueError:
                        pass

    log("  ✅ 网络扫描完成", color=C.GREEN)

def scan_launch_items():
    header("⚙️  M4·启动项扫描（LaunchAgents/Daemons）")

    scan_dirs = [
        "/Library/LaunchAgents",
        "/Library/LaunchDaemons",
        "/System/Library/LaunchAgents",
        "/System/Library/LaunchDaemons",
        f"{Path.home()}/Library/LaunchAgents",
    ]

    all_plist_files = []
    suspicious_items = []

    for d in scan_dirs:
        if not os.path.exists(d):
            continue
        try:
            for f in os.listdir(d):
                if f.endswith(".plist"):
                    full_path = os.path.join(d, f)
                    all_plist_files.append(full_path)
                    suspicious, kw = is_suspicious(f)
                    if suspicious:
                        suspicious_items.append((full_path, kw))
        except PermissionError:
            pass

    log(f"  扫描到 {len(all_plist_files)} 个启动项 plist 文件", color=C.BLUE)

    for plist_path in all_plist_files:
        try:
            content = open(plist_path, "r", errors="ignore").read()
            suspicious, kw = is_suspicious(content)
            if suspicious:
                suspicious_items.append((plist_path, kw))
        except Exception:
            pass

    seen = set()
    unique_suspicious = []
    for item in suspicious_items:
        if item[0] not in seen:
            seen.add(item[0])
            unique_suspicious.append(item)

    if unique_suspicious:
        log(f"  🔴 发现 {len(unique_suspicious)} 个可疑启动项！", color=C.RED)
        for path, kw in unique_suspicious:
            log(f"  🚨 含'{kw}': {path}", color=C.RED)
            add_result("p1", f"可疑启动项: {path}")
    else:
        log("  ✅ 未发现可疑启动项", color=C.GREEN)
        add_result("info", "M4: 启动项正常")
