import sqlite3

class DNAIndexer:
    def __init__(self, db_path="~/UID9622_DNA_index.db"):
        self.conn = sqlite3.connect(db_path)
        self.create_index_table()
    
    def create_index_table(self):
        """
        创建DNA索引表（B-Tree索引）
        查询性能：O(log n)
        """
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS dna_index (
                dna_code TEXT PRIMARY KEY,
                page_url TEXT NOT NULL,
                page_title TEXT,
                created_time INTEGER,
                creator TEXT,
                INDEX idx_dna_code (dna_code),
                INDEX idx_created_time (created_time)
            )
        """)
    
    def search_dna(self, dna_code):
        """
        DNA码检索
        旧方案：遍历所有页面（O(n)）
        新方案：索引查询（O(log n)）
        """
        cursor = self.conn.execute(
            "SELECT * FROM dna_index WHERE dna_code = ?",
            (dna_code,)
        )
        return cursor.fetchone()
