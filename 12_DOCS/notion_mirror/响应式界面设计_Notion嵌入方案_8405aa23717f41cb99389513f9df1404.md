### 1. 基础响应式CSS
```css
/* responsive.css - UID9622响应式设计 */

/* 基础变量 */
:root {

  --secondary-color: #FFD700;
  --bg-light: #FFFFFF;
  --bg-dark: #1A1A1A;

  }

@media (prefers-color-scheme: dark) {
  :root {
    --bg-color: var(--bg-dark);
    --text-color: var(--text-dark);
  }
}
```

### 2. Notion嵌入优化
```javascript
// notion-embed.js
class NotionEmbed {
    constructor(notionUrl, containerId) {
        this.notionUrl = notionUrl;
        this.container = document.getElementById(containerId);
        this.init();
    }
    
    init() {
        // 创建响应式iframe
        const iframe = document.createElement('iframe');
        iframe.src = this.notionUrl;
        iframe.style.width = '100%';
        iframe.style.border = 'none';
        iframe.style.height = this.calculateHeight();
        
        // 监听窗口大小变化
        window.addEventListener('resize', () => {
            iframe.style.height = this.calculateHeight();
        });
        
        this.container.appendChild(iframe);
    }
    
    calculateHeight() {
        // 根据屏幕尺寸计算高度
        const vh = window.innerHeight;
        return `${vh * 0.8}px`;  // 80% viewport height
    }
}

// 使用
const embed = new NotionEmbed(
    'https://notion.so/UID9622',
    'notion-container'
);
```
