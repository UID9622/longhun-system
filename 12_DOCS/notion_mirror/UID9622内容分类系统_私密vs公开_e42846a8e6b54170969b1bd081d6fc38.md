### 内容分类脚本
```python
#!/usr/bin/env python3
# ~/UID9622/📚 内容分类/classify_content.py

import os
import re
from pathlib import Path
import shutil

class ContentClassifier:
    """内容自动分类器"""
    
    def __init__(self):
        self.base_path = Path.home() / "UID9622"
        self.private_path = self.base_path / "🔒 私密区"
        self.public_path = self.base_path / "🌐 公开区"
        
        # 创建目录
        self.private_path.mkdir(exist_ok=True)
        self.public_path.mkdir(exist_ok=True)
    
    def is_private_content(self, content):
        """判断是否为私密内容"""
        private_keywords = [
            '老大', '宝宝', '心情', '感受',
            'API密钥', '密码', 'token',
            '未完成', '测试中', 'WIP',
            '私密', '不公开', '内部'
        ]
        
        for keyword in private_keywords:
            if keyword in content:
                return True
        return False
    
    def classify_file(self, file_path):
        """分类单个文件"""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            
            if self.is_private_content(content):
                # 私密内容
                dest = self.private_path / file_path.name
                print(f"🔒 私密: {file_path.name}")
            else:
                # 公开内容
                dest = self.public_path / file_path.name
                print(f"🌐 公开: {file_path.name}")
            
            shutil.copy2(file_path, dest)
            
        except Exception as e:
            print(f"⚠️  错误: {file_path.name} - {e}")
    
    def scan_and_classify(self):
        """扫描并分类所有文件"""
        print("🔍 开始扫描 UID9622 目录...\n")
        
        # 扫描 Python 文件
        for py_file in self.base_path.rglob("*.py"):
            if '🔒' not in str(py_file) and '🌐' not in str(py_file):
                self.classify_file(py_file)
        
        # 扫描 Markdown 文件
        for md_file in self.base_path.rglob("*.md"):
            if '🔒' not in str(md_file) and '🌐' not in str(md_file):
                self.classify_file(md_file)
        
        print("\n✅ 分类完成！")
        print(f"🔒 私密区: {self.private_path}")
        print(f"🌐 公开区: {self.public_path}")

if __name__ == "__main__":
    classifier = ContentClassifier()
    classifier.scan_and_classify()
```

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
UID9622 Notion API 内容管理器
自动扫描、分类、导出Notion内容

老大：🚀 Lucky｜UID9622
创建：2025-11-10
"""

import os
import json
import requests
from pathlib import Path
from datetime import datetime

class NotionContentManager:
    """Notion内容自动管理器"""
    
    def __init__(self, token):
        self.token = token
        self.headers = {
            "Authorization": f"Bearer {token}",
            "Notion-Version": "2022-06-28",
            "Content-Type": "application/json"
        }
        self.base_url = "https://api.notion.com/v1"
        
        # 本地路径
        self.base_path = Path.home() / "UID9622"
        self.private_path = self.base_path / "🔒 私密区"
        self.public_path = self.base_path / "🌐 公开区"
        
        # 创建目录
        self.private_path.mkdir(parents=True, exist_ok=True)
        self.public_path.mkdir(parents=True, exist_ok=True)
    
    def search_all_pages(self):
        """搜索所有Notion页面"""
        print("🔍 正在扫描Notion workspace...")
        
        url = f"{self.base_url}/search"
        payload = {
            "filter": {
                "property": "object",
                "value": "page"
            },
            "sort": {
                "direction": "descending",
                "timestamp": "last_edited_time"
            }
        }
        
        response = requests.post(url, headers=self.headers, json=payload)
        
        if response.status_code == 200:
            results = response.json()
            pages = results.get("results", [])
            print(f"✅ 找到 {len(pages)} 个页面")
            return pages
        else:
            print(f"❌ 错误：{response.status_code}")
            print(response.text)
            return []
    
    def get_page_content(self, page_id):
        """获取页面内容"""
        url = f"{self.base_url}/blocks/{page_id}/children"
        response = requests.get(url, headers=self.headers)
        
        if response.status_code == 200:
            return response.json().get("results", [])
        return []
    
    def is_private_content(self, title, content_text):
        """判断是否为私密内容"""
        private_keywords = [
            '老大', '宝宝', '心情', '感受', '私密',
            'API密钥', '密码', 'token', 'secret',
            '未完成', '测试中', 'WIP', '草稿',
            '内部', '不公开'
        ]
        
        # 检查标题
        for keyword in private_keywords:
            if keyword in title.lower():
                return True
        
        # 检查内容
        for keyword in private_keywords:
            if keyword in content_text.lower():
                return True
        
        return False
    
    def extract_text_from_blocks(self, blocks):
        """从blocks中提取文本"""
        text = ""
        for block in blocks:
            block_type = block.get("type")
            if block_type == "paragraph":
                rich_texts = block.get("paragraph", {}).get("rich_text", [])
                for rt in rich_texts:
                    text += rt.get("plain_text", "")
            elif block_type == "heading_1":
                rich_texts = block.get("heading_1", {}).get("rich_text", [])
                for rt in rich_texts:
                    text += rt.get("plain_text", "")
            # 可以添加更多block类型
        return text
    
    def classify_and_export_page(self, page):
        """分类并导出页面"""
        page_id = page["id"]
        
        # 获取标题
        title_parts = page.get("properties", {}).get("title", {}).get("title", [])
        title = "".join([t.get("plain_text", "") for t in title_parts])
        
        if not title:
            title = "Untitled"
        
        # 获取内容
        blocks = self.get_page_content(page_id)
        content_text = self.extract_text_from_blocks(blocks)
        
        # 分类
        is_private = self.is_private_content(title, content_text)
        
        # 准备导出数据
        export_data = {
            "title": title,
            "page_id": page_id,
            "url": page.get("url"),
            "created_time": page.get("created_time"),
            "last_edited_time": page.get("last_edited_time"),
            "content": content_text
        }
        
        # 选择保存路径
        if is_private:
            save_path = self.private_path / f"{title}.json"
            category = "🔒 私密"
        else:
            save_path = self.public_path / f"{title}.json"
            category = "🌐 公开"
        
        # 保存到本地
        with open(save_path, 'w', encoding='utf-8') as f:
            json.dump(export_data, f, ensure_ascii=False, indent=2)
        
        print(f"{category}: {title}")
        
        return {
            "title": title,
            "category": "private" if is_private else "public",
            "path": str(save_path)
        }
    
    def run(self):
        """运行完整流程"""
        print("\n" + "="*60)
        print("🚀 UID9622 Notion内容自动管理器")
        print("="*60 + "\n")
        
        # 1. 搜索所有页面
        pages = self.search_all_pages()
        
        if not pages:
            print("❌ 没有找到页面")
            return
        
        # 2. 分类统计
        private_count = 0
        public_count = 0
        results = []
        
        print("\n📋 开始分类和导出...\n")
        
        for page in pages:
            result = self.classify_and_export_page(page)
            results.append(result)
            
            if result["category"] == "private":
                private_count += 1
            else:
                public_count += 1
        
        # 3. 生成报告
        print("\n" + "="*60)
        print("✅ 处理完成！")
        print("="*60)
        print(f"\n📊 统计：")
        print(f"  - 总页面数：{len(pages)}")
        print(f"  - 🔒 私密内容：{private_count}")
        print(f"  - 🌐 公开内容：{public_count}")
        print(f"\n📂 导出位置：")
        print(f"  - 私密区：{self.private_path}")
        print(f"  - 公开区：{self.public_path}")
        
        # 4. 生成分类报告
        report_path = self.base_path / "分类报告.json"
        report = {
            "generated_at": datetime.now().isoformat(),
            "total": len(pages),
            "private": private_count,
            "public": public_count,
            "results": results
        }
        
        with open(report_path, 'w', encoding='utf-8') as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        
        print(f"\n📄 详细报告：{report_path}\n")

def main():

def watch_notion_changes(self):
    """实时监控Notion变化"""
    import time
    
    print("👀 开始监控Notion变化...")
    
    last_sync = {}
    
    while True:
        pages = self.search_all_pages()
        
        for page in pages:
            page_id = page["id"]
            last_edited = page.get("last_edited_time")
            
            # 检查是否有更新
            if page_id not in last_sync or last_sync[page_id] != last_edited:
                print(f"🔄 检测到更新：{page.get('title')}")
                self.classify_and_export_page(page)
                last_sync[page_id] = last_edited
        
        # 每5分钟检查一次
        time.sleep(300)
