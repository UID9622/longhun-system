## 一、🚀 一键全查脚本（最省事·一行跑完·结果自动存到桌面）
```bash
cat > ~/Desktop/mac_info_$(date +%Y%m%d).md << 'EOF_LONGHUN'
# 🖥️ Mac 设备信息全查 · $(date '+%Y-%m-%d %H:%M:%S')

## 1. 基础型号
$(sw_vers)

## 2. 硬件概览
$(system_profiler SPHardwareDataType | grep -E "Model Name|Model Identifier|Chip|Total Number of Cores|Memory|Serial Number")

## 3. CPU 详情
- CPU 型号: $(sysctl -n machdep.cpu.brand_string)
- 物理核心: $(sysctl -n hw.physicalcpu)
- 逻辑核心: $(sysctl -n hw.logicalcpu)

## 4. 内存
- 总内存: $(echo "scale=2; $(sysctl -n hw.memsize)/1024/1024/1024" | bc) GB

## 5. 磁盘
$(df -h | grep -E "^/dev/disk|Filesystem")

## 6. 显卡
$(system_profiler SPDisplaysDataType | grep -E "Chipset Model|VRAM|Resolution|Vendor")

## 7. 电池(笔记本)
$(pmset -g batt 2>/dev/null | head -3)

## 8. 网络硬件
$(networksetup -listallhardwareports | grep -E "Hardware Port|Device")

## 9. 系统运行时间
$(uptime)

EOF_LONGHUN
echo "✅ 完成! 文件已存到桌面: ~/Desktop/mac_info_$(date +%Y%m%d).md"
open ~/Desktop/mac_info_$(date +%Y%m%d).md
```
