class HistoryWisdomExtractor:
    def __init__(self):
        self.decision_database = []  # 您的历史决策记录

    def find_similar_situations(self, current_situation):
        """找到您过去的相似经历"""
        similar_cases = []

        for past_decision in self.decision_database:
            similarity_score = self.calculate_similarity(
                current_situation,
                past_decision
            )

            if similarity_score > 0.6:  # 60%以上相似
                case_analysis = {
                    'situation': past_decision['situation'],
                    'your_choice': past_decision['chosen_option'],
                    'outcome': past_decision['satisfaction_score'],
                    'lessons_learned': past_decision['key_insights'],
                    'what_worked': past_decision['success_factors'],
                    'what_didnt': past_decision['pain_points'],
                    'similarity_score': similarity_score
                }
                similar_cases.append(case_analysis)

        return sorted(similar_cases, key=lambda x: x['similarity_score'], reverse=True)[:3]
