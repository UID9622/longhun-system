```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
UID9622身份验证插件
确认码：#ZHUGEXIN⚡️2025-PLUGIN-V1.0
"""

import json
import hashlib
from datetime import datetime

def load_config():
    """加载配置"""
    with open('config.json', 'r', encoding='utf-8') as f:
        return json.load(f)

def verify_identity(input_fingerprint, input_confirm_code):
    """验证身份"""
    config = load_config()
    
    # 验证GPG指纹
    if input_fingerprint != config['gpg_fingerprint']:
        return {
            "status": "FAIL",
            "reason": "GPG指纹不匹配",
            "timestamp": datetime.now().isoformat()
        }
    
    # 验证确认码
    if input_confirm_code != config['confirm_code']:
        return {
            "status": "FAIL",
            "reason": "确认码不匹配",
            "timestamp": datetime.now().isoformat()
        }
    
    # 验证通过
    return {
        "status": "SUCCESS",
        "identity": config['identity'],
        "uid": config['uid'],
        "system": config['system'],
        "timestamp": datetime.now().isoformat(),
        "message": f"✅ 身份验证成功：{config['identity']}"
    }

def main():
    """主函数"""
    print("="*70)
    print("⚡ UID9622身份验证插件")
    print("="*70)
    
    # 读取配置
    config = load_config()
    print(f"\n📋 系统信息：")
    print(f"   身份标识：{config['identity']}")
    print(f"   UID：{config['uid']}")
    print(f"   系统：{config['system']}")
    
    print("\n" + "="*70)
    print("🔐 请输入身份信息进行验证")
    print("="*70)
    
    # 输入验证信息
    input_fingerprint = input("\n请输入GPG数字指纹：").strip()
    input_confirm_code = input("请输入一次性确认码：").strip()
    
    # 执行验证
    result = verify_identity(input_fingerprint, input_confirm_code)
    
    # 显示结果
    print("\n" + "="*70)
    if result['status'] == 'SUCCESS':
        print("✅ 验证通过")
        print(f"   身份：{result['identity']}")
        print(f"   UID：{result['uid']}")
        print(f"   时间：{result['timestamp']}")
    else:
        print("❌ 验证失败")
        print(f"   原因：{result['reason']}")
        print(f"   时间：{result['timestamp']}")
    print("="*70)
    
    return result

if __name__ == "__main__":
    main()
```
