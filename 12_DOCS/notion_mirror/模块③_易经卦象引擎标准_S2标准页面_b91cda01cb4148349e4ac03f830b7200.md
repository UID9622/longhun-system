### 三维卦象叠加引擎
```python
def hexagram_engine(time_gua, space_gua, human_gua, 
                     alpha=0.3, beta=0.3, gamma=0.4):
    """
    三维卦象叠加引擎
    输入：时间卦、空间卦、人事卦
    输出：综合决策权重 + 建议方向
    """
    # 获取各维度数值
    t_val = BAGUA[time_gua]["value"]
    s_val = BAGUA[space_gua]["value"]
    h_val = BAGUA[human_gua]["value"]
    
    # 加权叠加
    W = alpha * t_val + beta * s_val + gamma * h_val
    
    # 判定方向
    if W >= 0.7:
        direction = "大吉·宜进取"
    elif W >= 0.5:
        direction = "中吉·宜稳进"
    elif W >= 0.3:
        direction = "小吉·宜守成"
    else:
        direction = "凶·宜静待"
    
    return {
        "weight": round(W, 3),
        "direction": direction,
        "time": f"{BAGUA[time_gua]['symbol']} {time_gua}·{BAGUA[time_gua]['attribute']}",
        "space": f"{BAGUA[space_gua]['symbol']} {space_gua}·{BAGUA[space_gua]['attribute']}",
        "human": f"{BAGUA[human_gua]['symbol']} {human_gua}·{BAGUA[human_gua]['attribute']}",
    }

# 使用示例
result = hexagram_engine("离", "乾", "兑")
# 输出：weight=0.7, direction="大吉·宜进取"
```
