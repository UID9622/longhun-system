## 🎯 系统概述
本地离线的 Knowledge Work Plugins 管理系统。
七大特点：
- ✅ 完全离线运行（无需连 GitHub）
- ✅ 可视化仪表盘（Web + CLI 双界面）
- ✅ 本地缓存所有插件库
- ✅ 支持完全自定义（改词汇、调工作流、自造插件）
- ✅ 纯文本配置（Markdown + JSON）
- ✅ 支持快速安装到 Claude Code / Claude Cowork
- ✅ 零代码依赖（纯 Python stdlib）

## 🔧 核心 Python 脚本
### plugin_manager.py（插件管理引擎）
```python
#!/usr/bin/env python3
"""龍魂知识工作插件库管理引擎"""

import json
import os
from pathlib import Path
import subprocess
import hashlib

class PluginManager:
    def __init__(self):
        self.plugins_dir = Path.home() / ".龍魂_plugins"
        self.config_dir = Path.home() / ".龍魂_config"
        self.claude_plugins = Path.home() / ".claude_code" / "plugins"

    def list_plugins(self):
        """列出所有可用插件"""
        if not self.plugins_dir.exists():
            return []
        plugins = []
        for plugin_dir in self.plugins_dir.iterdir():
            if plugin_dir.is_dir() and not plugin_dir.name.startswith('.'):
                config_file = plugin_dir / 'config.json'
                if config_file.exists():
                    with open(config_file) as f:
                        config = json.load(f)
                        plugins.append({
                            'name': plugin_dir.name,
                            'title': config.get('title'),
                            'description': config.get('description'),
                            'installed': self._is_installed(plugin_dir.name),
                        })
        return plugins

    def install_plugin(self, plugin_name):
        """安装插件到 Claude Code"""
        plugin_dir = self.plugins_dir / plugin_name
        if not plugin_dir.exists():
            raise FileNotFoundError(f"插件不存在: {plugin_name}")
        claude_plugin_dir = self.claude_plugins / plugin_name
        claude_plugin_dir.mkdir(parents=True, exist_ok=True)
        for file in plugin_dir.glob('*'):
            if file.is_file():
                subprocess.run(['cp', str(file), str(claude_plugin_dir)])
        self._record_installation(plugin_name)
        return f"✅ 已安装 {plugin_name} 到 Claude Code"

    def _is_installed(self, plugin_name):
        return (self.claude_plugins / plugin_name).exists()

    def _record_installation(self, plugin_name):
        installed_file = self.config_dir / 'installed.json'
        installed = {}
        if installed_file.exists():
            with open(installed_file) as f:
                installed = json.load(f)
        installed[plugin_name] = {
            'installed_at': os.popen('date -u +%Y-%m-%dT%H:%M:%SZ').read().strip(),
            'version': '1.0.0',
        }
        with open(installed_file, 'w') as f:
            json.dump(installed, f, indent=2)

if __name__ == '__main__':
    import sys
    pm = PluginManager()
    if len(sys.argv) < 2:
        print("用法: python plugin_manager.py [list|install|search] [args]")
        sys.exit(1)
    cmd = sys.argv[1]
    if cmd == 'list':
        for plugin in pm.list_plugins():
            status = "✅" if plugin['installed'] else "⭕"
            print(f"{status} {plugin['name']:20} {plugin['title']}")
    elif cmd == 'install' and len(sys.argv) > 2:
        print(pm.install_plugin(sys.argv[2]))
    elif cmd == 'search' and len(sys.argv) > 2:
        query = sys.argv[2].lower()
        for plugin in pm.list_plugins():
            if query in plugin['name'].lower() or query in plugin['description'].lower():
                print(f"📌 {plugin['name']:20} {plugin['description']}")
```
### web_dashboard.py（Web 仪表盘服务器）
```python
#!/usr/bin/env python3
"""Web 仪表盘服务器"""

from flask import Flask, jsonify, request
import json
from pathlib import Path
from plugin_manager import PluginManager

app = Flask(__name__)
pm = PluginManager()

@app.route('/')
def dashboard():
    plugins = pm.list_plugins()
    installed_count = sum(1 for p in plugins if p['installed'])
    return jsonify({
        'status': 'ok',
        'total_plugins': len(plugins),
        'installed': installed_count,
        'plugins': plugins,
    })

@app.route('/api/plugins')
def api_plugins():
    search = request.args.get('search', '')
    plugins = pm.list_plugins()
    if search:
        plugins = [p for p in plugins if search.lower() in p['name'].lower()]
    return jsonify(plugins)

@app.route('/api/plugins/<name>')
def api_plugin_detail(name):
    plugin_dir = Path.home() / ".龍魂_plugins" / name
    if not plugin_dir.exists():
        return jsonify({'error': '插件不存在'}), 404
    with open(plugin_dir / 'config.json') as f:
        config = json.load(f)
    with open(plugin_dir / 'commands.json') as f:
        commands = json.load(f)
    with open(plugin_dir / 'connectors.json') as f:
        connectors = json.load(f)
    return jsonify({'config': config, 'commands': commands, 'connectors': connectors})

@app.route('/api/plugins/<name>/install', methods=['POST'])
def api_install(name):
    result = pm.install_plugin(name)
    return jsonify({'message': result}), 200

@app.route('/api/plugins/<name>/uninstall', methods=['POST'])
def api_uninstall(name):
    import shutil
    claude_plugin_dir = Path.home() / ".claude_code" / "plugins" / name
    if claude_plugin_dir.exists():
        shutil.rmtree(claude_plugin_dir)
    return jsonify({'message': f'✅ 已删除 {name}'}), 200

if __name__ == '__main__':
    print("🚀 Web 仪表盘启动 → http://localhost:8888")
    app.run(host='localhost', port=8888, debug=True)
```
