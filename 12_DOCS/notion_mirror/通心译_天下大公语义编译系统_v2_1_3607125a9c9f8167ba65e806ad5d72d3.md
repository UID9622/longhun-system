```python
def tongxinyi_parse(raw_input: str) -> ParseResult:
    # L0: 原话保留
    archive_raw(raw_input)
    
    # L1: 情绪净化
    emotion = detect_emotion(raw_input)  # 返回 {类型, 强度, 方向}
    cleaned = remove_emotion_noise(raw_input, emotion)
    
    # L2: 意图骨架
    intent = extract_intent(cleaned)  # 返回 {主体, 状态, 动作, 优先级}
    
    # L3: SAST 生成
    sast = generate_sast(intent)
    
    # L4: 审计
    tricolor = watchdog_audit(sast)  # 返回 AUTO_OK / NEED_CONFIRM / BLOCKED
    
    # L5: 文化适配
    adapted = culture_adapt(sast, target_culture)
    
    return ParseResult(emotion, intent, sast, tricolor, adapted)
```
