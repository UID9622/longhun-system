## 🚀 快速开始
### 系统要求
### 5分钟快速安装
方式一：NPM 安装（推荐）
```bash
# 全局安装
npm install -g @uid9622/notion-mcp-server

# 或项目内安装
npm install @uid9622/notion-mcp-server

# 验证安装
uid9622-mcp --version
```
方式二：从源码构建
```bash
# 克隆仓库
git clone https://github.com/UID9622/notion-mcp-server.git
cd notion-mcp-server

# 安装依赖
npm install

# 构建项目
npm run build

# 运行
npm start
```
方式三：Docker 部署
```bash
# 拉取镜像
docker pull uid9622/notion-mcp-server:latest

# 运行容器
docker run -d -p 3000:3000 \
  -e NOTION_API_KEY=your_api_key \
  uid9622/notion-mcp-server:latest
```

创建 .uid9622.config.json 配置文件：


## 💻 代码示例
### Node.js 基础使用
```javascript
const { UID9622MCP } = require('@uid9622/notion-mcp-server');

// 初始化客户端
const client = new UID9622MCP({
  notionApiKey: process.env.NOTION_API_KEY,
  locale: 'zh-CN'
});

// 查询数据库
async function queryDatabase() {
  try {
    const response = await client.databases.query({
      database_id: 'your_database_id',
      filter: {
        property: '状态',
        select: {
          equals: '进行中'
        }
      },
      sorts: [
        {
          property: '优先级',
          direction: 'descending'
        }
      ]
    });
    
    console.log('查询结果：', response.results);
    return response.results;
  } catch (error) {
    console.error('查询失败：', error.message);
    throw error;
  }
}

// 创建页面（支持中文）
async function createPage() {
  const page = await client.pages.create({
    parent: { database_id: 'your_database_id' },
    properties: {
      '任务名称': {
        title: [{ text: { content: '完成技术文档' } }]
      },
      '状态': {
        select: { name: '进行中' }
      },
      '优先级': {
        select: { name: '高' }
      }
    },
    children: [
      {
        object: 'block',
        type: 'heading_2',
        heading_2: {
          rich_text: [{ text: { content: '任务详情' } }]
        }
      },
      {
        object: 'block',
        type: 'paragraph',
        paragraph: {
          rich_text: [
            { text: { content: '这是使用 UID9622-MCP 创建的页面。' } }
          ]
        }
      }
    ]
  });
  
  console.log('页面创建成功：', page.url);
  return page;
}

// 智能搜索（中文优化）
async function smartSearch(query) {
  const results = await client.search({
    query: query,
    filter: { property: 'object', value: 'page' },
    sort: { direction: 'descending', timestamp: 'last_edited_time' }
  });
  
  return results;
}

// 执行示例
(async () => {
  await queryDatabase();
  await createPage();
  const searchResults = await smartSearch('技术文档');
  console.log('搜索结果：', searchResults);
})();
```
### Python SDK 使用
```python
from uid9622_mcp import UID9622Client
import os

# 初始化客户端
client = UID9622Client(
    notion_api_key=os.getenv('NOTION_API_KEY'),
    locale='zh-CN'
)

# 查询数据库
def query_database():
    response = client.databases.query(
        database_id='your_database_id',
        filter={
            'property': '状态',
            'select': {'equals': '进行中'}
        },
        sorts=[{
            'property': '优先级',
            'direction': 'descending'
        }]
    )
    
    print(f'查询到 {len(response["results"])} 条记录')
    return response['results']

# 创建页面
def create_page():
    page = client.pages.create(
        parent={'database_id': 'your_database_id'},
        properties={
            '任务名称': {
                'title': [{'text': {'content': '完成技术文档'}}]
            },
            '状态': {'select': {'name': '进行中'}},
            '优先级': {'select': {'name': '高'}}
        },
        children=[
            {
                'object': 'block',
                'type': 'heading_2',
                'heading_2': {
                    'rich_text': [{'text': {'content': '任务详情'}}]
                }
            },
            {
                'object': 'block',
                'type': 'paragraph',
                'paragraph': {
                    'rich_text': [
                        {'text': {'content': '使用 Python SDK 创建的页面。'}}
                    ]
                }
            }
        ]
    )
    
    print(f'页面创建成功：{page["url"]}')
    return page

# 批量操作
def batch_update(page_ids, new_status):
    results = []
    for page_id in page_ids:
        result = client.pages.update(
            page_id=page_id,
            properties={
                '状态': {'select': {'name': new_status}}
            }
        )
        results.append(result)
    
    print(f'批量更新完成：{len(results)} 个页面')
    return results

if __name__ == '__main__':
    query_database()
    create_page()
    batch_update(['page_id_1', 'page_id_2'], '已完成')
```

## 📖 API 完整文档
### 数据库操作
### databases.query(options)
查询数据库中的页面。
参数：
- database_id (string, 必需) - 数据库ID
- filter (object, 可选) - 过滤条件
- sorts (array, 可选) - 排序规则
- start_cursor (string, 可选) - 分页游标
- page_size (number, 可选) - 每页数量，默认100
返回值：
```typescript
{
  results: Page[],
  has_more: boolean,
  next_cursor: string | null
}
```
示例：
```javascript
const response = await client.databases.query({
  database_id: 'xxx',
  filter: {
    and: [
      { property: '状态', select: { equals: '进行中' } },
      { property: '优先级', select: { equals: '高' } }
    ]
  },
  sorts: [
    { property: '创建时间', direction: 'descending' }
  ],
  page_size: 50
});
```
### databases.create(options)
创建新数据库。
参数：
- parent (object, 必需) - 父页面信息
- title (array, 必需) - 数据库标题
- properties (object, 必需) - 属性定义
- icon (object, 可选) - 数据库图标
示例：
```javascript
const database = await client.databases.create({
  parent: { page_id: 'parent_page_id' },
  title: [{ text: { content: '项目管理' } }],
  properties: {
    '任务名称': { title: {} },
    '状态': {
      select: {
        options: [
          { name: '待开始', color: 'gray' },
          { name: '进行中', color: 'blue' },
          { name: '已完成', color: 'green' }
        ]
      }
    },
    '优先级': {
      select: {
        options: [
          { name: '低', color: 'gray' },
          { name: '中', color: 'yellow' },
          { name: '高', color: 'red' }
        ]
      }
    },
    '截止日期': { date: {} },
    '负责人': { people: {} }
  },
  icon: { emoji: '📋' }
});
```

### 错误处理
```javascript
try {
  const page = await client.pages.create(pageData);
  console.log('创建成功：', page.id);
} catch (error) {
  if (error.code === 'validation_error') {
    console.error('数据验证失败：', error.message);
  } else if (error.code === 'unauthorized') {
    console.error('认证失败，请检查API密钥');
  } else if (error.code === 'rate_limited') {
    console.error('请求过于频繁，请稍后重试');
    // 实现重试逻辑
    await sleep(error.retryAfter * 1000);
  } else {
    console.error('未知错误：', error);
  }
}
```

```javascript
// 批量操作
const batchSize = 10;
const pageIds = [...]; // 大量页面ID

for (let i = 0; i < pageIds.length; i += batchSize) {

      page_id: id,
      properties: { /* ... */ }
    }))

  // 避免速率限制
```
