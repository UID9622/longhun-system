import pyperclip
from notion_client import Client
from datetime import datetime

class OneClickArchive:
    def __init__(self):
        self.notion = Client(auth="你的Notion_Token")
        self.db_id = "数据库ID"
        self.uid = "UID9622"
    
    def archive(self):
        # 1. 读取剪贴板
        content = pyperclip.paste()
        if not content:
            print("❌ 请先复制对话内容")
            return
        
        # 2. 询问平台
        platform = input("平台？(1=ChatGPT 2=通义 3=智谱 4=其他): ")
        platforms = {"1": "ChatGPT", "2": "通义千问", "3": "智谱清言", "4": "其他"}
        
        # 3. 生成DNA编号
        timestamp = datetime.now().strftime("%Y%m%d-%H%M%S")
        dna_code = f"#UID9622-DIALOGUE-{platforms.get(platform, '其他')}-{timestamp}"
        
        # 4. 智能分析（调用AI）
        analysis = self.analyze(content)
        
        # 5. 上传到Notion
        self.notion.pages.create(
            parent={"database_id": self.db_id},
            properties={
                "对话标题": {"title": [{"text": {"content": analysis["标题"]}}]},
                "DNA编号": {"rich_text": [{"text": {"content": dna_code}}]},
                "对话平台": {"select": {"name": platforms.get(platform, "其他")}},
                "重要程度": {"select": {"name": analysis["重要程度"]}},
                "存储策略": {"select": {"name": "永久保存"}},
                "对话时间": {"date": {"start": datetime.now().isoformat()}}
            },
            children=[{"object": "block", "type": "paragraph",
                      "paragraph": {"rich_text": [{"text": {"content": content[:2000]}}]}}]
        )
        
        print(f"✅ 归档成功！DNA编号：{dna_code}")
    
    def analyze(self, content):
        # 简化版：直接返回基础分析
        return {"标题": f"对话 {datetime.now().strftime('%Y-%m-%d')}", "重要程度": "P1重要记忆"}

if __name__ == "__main__":
    OneClickArchive().archive()

from notion_client import Client

@app.route('/api/archive', methods=['POST'])

    data = request.json
    notion.pages.create(
        parent={"database_id": "数据库ID"},
        properties={
            "对话标题": {"title": [{"text": {"content": f"{data['platform']}对话"}}]},

                  "paragraph": {"rich_text": [{"text": {"content": data['content'][:2000]}}]}}]
    )
