## 📦 完整安装脚本代码
```bash
#!/bin/bash
#═══════════════════════════════════════════════════════
# UID9622 龙魂终端配置安装脚本 v3.0
# 作者：诸葛鑫 (UID9622)
# 功能：自动化配置终端环境，支持断点续传和多服务器容错
#═══════════════════════════════════════════════════════

set -e

#═══════════════════════════════════════════════════════
# 全局配置
#═══════════════════════════════════════════════════════

VERSION="3.0"
INSTALL_DIR="$HOME/.uid9622"
CONFIG_DIR="$INSTALL_DIR/config"
LOG_FILE="$INSTALL_DIR/install.log"
STATE_FILE="$INSTALL_DIR/install.state"
BACKUP_DIR="$INSTALL_DIR/backups/$(date +%Y%m%d_%H%M%S)"

# 服务器列表（多服务器容错）
PRIMARY_SERVER="https://your-primary-server.com"
BACKUP_SERVERS=(
    "https://backup1.example.com"
    "https://backup2.example.com"
    "https://mirror.example.com"
)

# 完整性哈希值（可选）
EXPECTED_HASH=""

#═══════════════════════════════════════════════════════
# 工具函数
#═══════════════════════════════════════════════════════

# 日志函数
log() {
    local level=$1
    shift
    local message="$@"
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "[$timestamp] [$level] $message" | tee -a "$LOG_FILE"
}

# 错误处理
error_exit() {
    log "ERROR" "$1"
    echo ""
    echo "❌ 安装失败：$1"
    echo "📋 查看详细日志：cat $LOG_FILE"
    echo "🔧 恢复建议："
    echo "   1. 检查网络连接"
    echo "   2. 确认系统权限"
    echo "   3. 重新运行脚本（支持断点续传）"
    exit 1
}

# 保存安装状态
save_state() {
    echo "$1" > "$STATE_FILE"
    log "INFO" "保存状态：$1"
}

# 读取安装状态
load_state() {
    if [ -f "$STATE_FILE" ]; then
        cat "$STATE_FILE"
    else
        echo "init"
    fi
}

# 检测操作系统
detect_os() {
    if [[ "$OSTYPE" == "darwin"* ]]; then
        echo "macos"
    elif [[ "$OSTYPE" == "linux-gnu"* ]]; then
        if [ -f /etc/os-release ]; then
            . /etc/os-release
            echo "$ID"
        else
            echo "linux"
        fi
    else
        echo "unknown"
    fi
}

# 多服务器下载函数
download_with_fallback() {
    local file_path="$1"
    local output="$2"
    local servers=("$PRIMARY_SERVER" "${BACKUP_SERVERS[@]}")
    
    for server in "${servers[@]}"; do
        log "INFO" "尝试从 $server 下载..."
        if curl -fsSL -o "$output" "$server/$file_path"; then
            log "INFO" "下载成功：$server/$file_path"
            return 0
        else
            log "WARN" "下载失败，尝试下一个服务器..."
        fi
    done
    
    error_exit "所有服务器下载失败"
}

# 备份现有配置
backup_config() {
    log "INFO" "备份现有配置..."
    mkdir -p "$BACKUP_DIR"
    
    local files=(
        "$HOME/.zshrc"
        "$HOME/.bashrc"
        "$HOME/.config/starship.toml"
    )
    
    for file in "${files[@]}"; do
        if [ -f "$file" ]; then
            cp "$file" "$BACKUP_DIR/" || log "WARN" "备份 $file 失败"
            log "INFO" "已备份：$file"
        fi
    done
}

# 安装步骤
#═══════════════════════════════════════════════════════

# 步骤1：初始化
init_installation() {
    log "INFO" "开始安装 UID9622 龙魂终端 v$VERSION"
    
    mkdir -p "$INSTALL_DIR" "$CONFIG_DIR"
    
    echo "╔════════════════════════════════════════════════╗"
    echo "║     🐉 UID9622 龙魂终端安装程序 v$VERSION 🐉     ║"
    echo "╚════════════════════════════════════════════════╝"
    echo ""
    
    save_state "initialized"
}

# 步骤2：检查系统环境
check_system() {
    log "INFO" "检查系统环境..."
    
    local os=$(detect_os)
    log "INFO" "检测到操作系统：$os"
    
    # 检查必要工具
    local required_tools=("curl" "git")
    for tool in "${required_tools[@]}"; do
        if ! command -v $tool &> /dev/null; then
            error_exit "缺少必要工具：$tool，请先安装"
        fi
    done
    
    save_state "system_checked"
}

# 步骤3：安装依赖
install_dependencies() {
    log "INFO" "安装依赖..."
    
    local os=$(detect_os)
    
    # 安装 Oh My Zsh
    if [ ! -d "$HOME/.oh-my-zsh" ]; then
        log "INFO" "安装 Oh My Zsh..."
        sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" "" --unattended || error_exit "Oh My Zsh 安装失败"
    fi
    
    # 安装 Starship
    if ! command -v starship &> /dev/null; then
        log "INFO" "安装 Starship..."
        curl -sS https://starship.rs/install.sh | sh -s -- -y || error_exit "Starship 安装失败"
    fi
    
    save_state "dependencies_installed"
}

# 步骤4：配置终端
configure_terminal() {
    log "INFO" "配置终端..."
    
    # 备份现有配置
    backup_config
    
    # 创建 Starship 配置
    mkdir -p "$HOME/.config"
    cat > "$HOME/.config/starship.toml" << 'EOF'
format = """
[╭─](bold cyan)$username[@](bold cyan)$hostname \
$directory\
$git_branch\
$git_status
[╰─](bold cyan)$character"""

[username]
format = "[$user]($style)"
style_user = "bold red"
show_always = true

[hostname]
format = "[$hostname]($style)"
style = "bold yellow"
ssh_only = false

[directory]
format = "[$path]($style)[$read_only]($read_only_style) "
style = "bold cyan"
truncation_length = 3
truncate_to_repo = true

[git_branch]
format = "[$symbol$branch]($style) "
symbol = "🌱 "
style = "bold green"

[git_status]
format = '([\[$all_status$ahead_behind\]]($style) )'
style = "bold yellow"

[character]
success_symbol = "[⚡](bold green)"
error_symbol = "[⚡](bold red)"
EOF
    
    # 配置 zshrc
    if [ ! -f "$HOME/.zshrc.uid9622.bak" ]; then
        cp "$HOME/.zshrc" "$HOME/.zshrc.uid9622.bak" 2>/dev/null || true
    fi
    
    cat >> "$HOME/.zshrc" << 'EOF'

# 卸载功能
#═══════════════════════════════════════════════════════

uninstall() {
    log "INFO" "开始卸载..."
    
    # 恢复备份
    if [ -f "$HOME/.zshrc.uid9622.bak" ]; then
        mv "$HOME/.zshrc.uid9622.bak" "$HOME/.zshrc"
        log "INFO" "已恢复 .zshrc"
    fi
    
    # 删除配置文件
    rm -f "$HOME/.config/starship.toml"
    
    # 删除安装目录
    rm -rf "$INSTALL_DIR"
    
    echo "✅ 卸载完成"
    log "INFO" "卸载完成"
}

#═══════════════════════════════════════════════════════
# 主流程
#═══════════════════════════════════════════════════════

main() {
    local current_state=$(load_state)
    
    case "$1" in
        install)
            case "$current_state" in
                "init")
                    init_installation
                    ;&
                "initialized")
                    check_system
                    ;&
                "system_checked")
                    install_dependencies
                    ;&
                "dependencies_installed")
                    configure_terminal
                    ;&
                "terminal_configured")
                    finish_installation
                    ;;
                "completed")
                    echo "已经安装完成！"
                    echo "如需重新安装，请先运行：$0 uninstall"
                    ;;
            esac
            ;;
        uninstall)
            uninstall
            ;;
        *)
            echo "用法：$0 {install|uninstall}"
            exit 1
            ;;
    esac
}

main "$@"
```

```python
# UID9622龙魂系统激活码生成逻辑
# 开源于 2025年10月11日

import hashlib
import time

def generate_activation_code(user_id, purpose, timestamp):
    """
    生成激活码
    user_id: 用户唯一标识
    purpose: 使用目的（必须是正向价值）
    timestamp: 时间戳
    """
    # 核心种子
    seed = f"UID9622-LONGHUN-{user_id}-{purpose}-{timestamp}"
    
    # 价值观验证
    positive_purposes = [
        "教育", "励志", "法律", "人性", "责任",
        "成长", "学习", "帮助", "创造", "贡献"
    ]
    
    # 检查目的是否正向
    if not any(p in purpose for p in positive_purposes):
        return None  # 拒绝非正向用途
    
    # 生成激活码
    hash_object = hashlib.sha256(seed.encode())
    activation_code = hash_object.hexdigest()[:16].upper()
    
    return f"LONGHUN-{activation_code}"

# 使用示例
```

错误在第324行附近，问题出在 ;& 这个符号上。在脚本的 main() 函数里用了 ;& 来实现"贯穿执行"（fallthrough），但这个语法只在某些 shell 版本的 case 语句中有效。
快速修复方法：
把脚本中 main() 函数里的这部分：
```bash
case "$current_state" in
    "init")
        init_installation
        ;&
    "initialized")
        check_system
        ;&
    "dependencies_installed")
        configure_terminal
        ;&
    "terminal_configured")
        finish_installation
        ;;

```
改成这样（移除 ;&，改用函数调用链）：
```bash
case "$current_state" in
    "init")
        init_installation
        check_system
        install_dependencies
        configure_terminal
        finish_installation
        ;;
    "initialized")
        check_system
        install_dependencies
        configure_terminal
        finish_installation
        ;;
    "system_checked")
        install_dependencies
        configure_terminal
        finish_installation
        ;;
    "dependencies_installed")
        configure_terminal
        finish_installation
        ;;
    "terminal_configured")
        finish_installation
        ;;

```
