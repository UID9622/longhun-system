#!/bin/bash

echo "🐉 诸葛亮数字人一键部署 v1.0"

echo ""

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 检测操作系统
detect_os() {
    if [[ "$OSTYPE" == "linux-gnu"* ]]; then
        OS="linux"
    elif [[ "$OSTYPE" == "darwin"* ]]; then
        OS="macos"
    elif [[ "$OSTYPE" == "msys" || "$OSTYPE" == "cygwin" ]]; then
        OS="windows"
    else
        echo "${RED}❌ 不支持的操作系统${NC}"
        exit 1
    fi
    echo "${GREEN}✅ 检测到系统: $OS${NC}"
}

# 检测Python版本
check_python() {
    echo "\n🔍 检查Python环境..."
    if command -v python3 &> /dev/null; then
        PYTHON_VERSION=$(python3 --version | cut -d' ' -f2)
        echo "${GREEN}✅ Python版本: $PYTHON_VERSION${NC}"
    else
        echo "${RED}❌ 未检测到Python3，请先安装Python 3.8+${NC}"
        exit 1
    fi
}

# 检测Ollama
check_ollama() {
    echo "\n🔍 检查Ollama环境..."
    if command -v ollama &> /dev/null; then
        echo "${GREEN}✅ Ollama已安装${NC}"
    else
        echo "${YELLOW}⚠️  Ollama未安装，正在安装...${NC}"
        curl -fsSL https://ollama.com/install.sh | sh
    fi
}

# 安装依赖
install_dependencies() {
    echo "\n📦 安装依赖包..."
    
    # 创建虚拟环境
    python3 -m venv zhugeliang_env
    source zhugeliang_env/bin/activate
    
    # 安装Python依赖
    pip install --upgrade pip
    pip install -r requirements.txt
    
    echo "${GREEN}✅ 依赖安装完成${NC}"
}

# 安装国密算法库
install_guomi() {
    echo "\n🔐 安装国密算法库..."
    pip install gmssl
    pip install pycryptodome
    echo "${GREEN}✅ 国密算法库安装完成${NC}"
}

# 下载模型文件
download_models() {
    echo "\n⬇️  下载模型文件..."
    
    # 创建模型目录
    mkdir -p models/3d
    mkdir -p models/voice
    mkdir -p models/ollama
    
    # 下载3D模型（500MB）
    echo "下载诸葛亮3D模型（500MB）..."
    wget -O models/3d/zhugeliang.fbx https://models.uid9622.cn/zhugeliang/v1/zhugeliang.fbx
    
    # 下载语音模型（200MB）
    echo "下载语音模型（200MB）..."
    wget -O models/voice/paddlespeech.tar.gz https://models.uid9622.cn/voice/paddlespeech.tar.gz
    tar -xzf models/voice/paddlespeech.tar.gz -C models/voice/
    
    # 下载Ollama模型
    echo "下载本地AI模型（qwen:7b-chat）..."
    ollama pull qwen:7b-chat
    
    echo "${GREEN}✅ 模型下载完成${NC}"
}

# 初始化数据库
init_database() {
    echo "\n🗄️  初始化数据库..."
    python3 scripts/init_db.py
    echo "${GREEN}✅ 数据库初始化完成${NC}"
}

# 生成NFC标签
generate_nfc() {
    echo "\n📱 生成Mate 80 NFC标签..."
    python3 scripts/generate_nfc.py
    echo "${GREEN}✅ NFC标签已生成，请用Mate 80扫描二维码${NC}"

# 启动服务
start_service() {
    echo "\n🚀 启动诸葛亮数字人服务..."
    python3 main.py &
    echo "${GREEN}✅ 服务已启动！${NC}"
    echo ""
    echo "访问地址:"
    echo "  - 本地: http://localhost:8080"
    echo "  - Mate 80: 碰一碰NFC标签即可启动"
    echo ""
    echo "停止服务: pkill -f main.py"
}

# 主流程
main() {

    echo "快速启动:"
    echo "  1. Mate 80碰一碰NFC标签"
    echo "  2. 或语音唤醒：'诸葛亮，在吗？'"
    echo "  3. 或访问 http://localhost:8080"
    echo ""
    echo "DNA追溯码: #ZHUGEXIN⚡️2025-ZHUGELIANG-INSTALL-V1.0"

main

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
诸葛亮数字人主程序
作者: Lucky (UID9622)
DNA确认码: #ZHUGEXIN⚡️2025-ZHUGELIANG-MAIN-V1.0
"""

import os
import sys
from flask import Flask, render_template, request, jsonify
from loguru import logger
import ollama
from gmssl import sm2, sm3, sm4
import sqlite3
import json

# 初始化Flask应用
app = Flask(__name__)

# 配置日志
logger.add("logs/zhugeliang.log", rotation="1 day", retention="30 days")

# 国密算法初始化
class GuoMiCrypto:
    """国密算法封装"""
    
    def __init__(self):
        # SM2密钥对（后续从Mate 80 TEE读取）
        self.private_key = "00B9AB0B828FF68872F21A837FC303668428DEA11DCD1B24429D0C99E24EED83D5"
        self.public_key = "B9C9A6E04E9C91F7BA880429273747D7EF5DDEB0BB2FF6317EB00BEF331A83081A6994B8993F3F5D6EADDDB81872266C87C018FB4162F5AF347B483E24620207"
        
        self.sm2_crypt = sm2.CryptSM2(
            public_key=self.public_key,
            private_key=self.private_key
        )
    
    def encrypt(self, data: str) -> str:
        """SM2加密"""
        enc_data = self.sm2_crypt.encrypt(data.encode())
        return enc_data.hex()
    
    def decrypt(self, enc_data: str) -> str:
        """SM2解密"""
        dec_data = self.sm2_crypt.decrypt(bytes.fromhex(enc_data))
        return dec_data.decode()
    
    def hash(self, data: str) -> str:
        """SM3哈希"""
        sm3_hash = sm3.sm3_hash([i for i in bytes(data.encode())])
        return sm3_hash

# 初始化国密
guomi = GuoMiCrypto()

# 初始化Ollama
client = ollama.Client()

class ZhuGeLiang:
    """诸葛亮数字人核心类"""
    
    def __init__(self):
        self.db_path = "data/zhugeliang.db"
        self.init_database()
        logger.info("诸葛亮数字人初始化完成")
    
    def init_database(self):
        """初始化数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 创建对话记录表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS conversations (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_input TEXT NOT NULL,
                zgl_response TEXT NOT NULL,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                encrypted_data TEXT,
                dna_code TEXT
            )
        ''')
        
        conn.commit()
        conn.close()
    
    def chat(self, user_input: str) -> dict:
        """对话功能"""
        try:
            # 调用Ollama本地模型
            response = client.chat(model='qwen:7b-chat', messages=[
                {
                    'role': 'system',
                    'content': '''你是诸葛亮，字孔明，号卧龙，三国时期蜀汉丞相。
你拥有以下特质：
1. 战略眼光：能够从全局看问题，提供长远规划
2. 冷静理性：不急不躁，稳重沉着
3. 忠诚担当：鞠躬尽瘁，死而后已
4. 智慧谦逊：虽有才华，但谦虚待人

回答时：
- 使用文言文与白话文结合（70%白话+30%文言）
- 适当引用《出师表》《诫子书》等经典
- 提供具体可行的建议，不空谈
- 对现代问题用古代智慧来解答'''
                },
                {
                    'role': 'user',
                    'content': user_input
                }
            ])
            
            zgl_response = response['message']['content']
            
            # 加密存储（国密SM2）
            encrypted_input = guomi.encrypt(user_input)
            encrypted_response = guomi.encrypt(zgl_response)
            
            # 生成DNA追溯码
            dna_code = f"#ZHUGEXIN⚡️2025-CHAT-{guomi.hash(user_input + zgl_response)[:8]}"
            
            # 保存到数据库
            self.save_conversation(
                user_input=user_input,
                zgl_response=zgl_response,
                encrypted_data=json.dumps({
                    'input': encrypted_input,
                    'response': encrypted_response
                }),
                dna_code=dna_code
            )
            
            return {
                'success': True,
                'response': zgl_response,
                'dna_code': dna_code
            }
            
        except Exception as e:
            logger.error(f"对话失败: {str(e)}")
            return {
                'success': False,
                'error': str(e)
            }
    
    def save_conversation(self, user_input, zgl_response, encrypted_data, dna_code):
        """保存对话记录"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            INSERT INTO conversations (user_input, zgl_response, encrypted_data, dna_code)
            VALUES (?, ?, ?, ?)
        ''', (user_input, zgl_response, encrypted_data, dna_code))
        
        conn.commit()
        conn.close()

# 初始化诸葛亮
zgl = ZhuGeLiang()

# 路由
@app.route('/')
def index():
    """首页"""
    return render_template('index.html')

@app.route('/chat', methods=['POST'])
def chat():
    """对话接口"""
    data = request.get_json()
    user_input = data.get('message', '')
    
    if not user_input:
        return jsonify({'success': False, 'error': '输入不能为空'})
    
    result = zgl.chat(user_input)
    return jsonify(result)

@app.route('/health')
def health():
    """健康检查"""
    return jsonify({'status': 'ok', 'dna': '#ZHUGEXIN⚡️2025-HEALTH-CHECK'})

if __name__ == '__main__':

1. Python版本是否3.8+？（python3 --version）
1. 硬盘空间是否足够？（需要20GB）
1. 网络是否通畅？（下载模型需要）

### Q3: Mate 80 NFC不响应？
A: 尝试以下步骤：
1. 确认NFC功能已开启（设置→连接与共享→NFC）
1. 手机背面靠近标签中心位置
1. 保持1-2秒
1. 如果还不行，用相机扫描二维码手动添加
### Q4: 对话响应慢怎么办？
A: 可能原因：
1. CPU性能不足→建议升级到16GB内存
1. 模型太大→可以换用轻量版模型（qwen:3b）
1. 后台程序太多→关闭其他应用
### Q5: 数据安全吗？
A: ✅ 完全安全！
- 所有数据本地加密存储（国密SM2/SM3/SM4）
- 密钥存储在Mate 80的TEE安全区
- 不联网，不上传，不泄露
- 即使手机丢失，数据也无法解密
