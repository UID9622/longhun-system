## 💻 完整代码示例
### 基础单次对话
```python
# claude_basic.py
import anthropic
from dotenv import load_dotenv
import os

# 加载环境变量
load_dotenv()

# 初始化客户端
client = anthropic.Anthropic(
    api_key=os.getenv("ANTHROPIC_API_KEY")
)

def simple_chat(user_message: str) -> str:
    """简单对话"""
    try:
        message = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=4096,
            messages=[
                {"role": "user", "content": user_message}
            ]
        )
        
        response = message.content[0].text
        print(f"🤖 Claude: {response}")
        return response
        
    except Exception as e:
        print(f"❌ 调用失败：{e}")
        return None

# 测试
if __name__ == "__main__":
    simple_chat("你好，我是Lucky，请介绍一下你自己")
```
### 多轮对话（带历史）
```python
# claude_conversation.py
from typing import List, Dict

class ClaudeConversation:
    """Claude多轮对话管理器"""
    
    def __init__(self, system_prompt: str = ""):
        load_dotenv()
        self.client = anthropic.Anthropic(
            api_key=os.getenv("ANTHROPIC_API_KEY")
        )
        self.system_prompt = system_prompt
        self.history: List[Dict] = []
    
    def chat(self, user_message: str) -> str:
        """发送消息并获取回复"""
        # 添加用户消息到历史
        self.history.append({
            "role": "user",
            "content": user_message
        })
        
        try:
            # 调用API
            message = self.client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=4096,
                system=self.system_prompt if self.system_prompt else None,
                messages=self.history
            )
            
            # 提取回复
            assistant_message = message.content[0].text
            
            # 添加助手回复到历史
            self.history.append({
                "role": "assistant",
                "content": assistant_message
            })
            
            return assistant_message
            
        except Exception as e:
            print(f"❌ 对话失败：{e}")
            return None
    
    def clear_history(self):
        """清空对话历史"""
        self.history = []
    
    def get_history(self) -> List[Dict]:
        """获取对话历史"""
        return self.history

# 使用示例
if __name__ == "__main__":
    conv = ClaudeConversation(
        system_prompt="你是UID9622系统的助手，需要遵循龙魂价值观。"
    )
    
    # 多轮对话
    print(conv.chat("你好，我是Lucky"))
    print(conv.chat("我们上次聊到哪了？"))  # Claude会记住上下文
```

# claude_personality.py
from typing import Optional
import json

class PersonalityDispatcher:
    """UID9622 71人格调度系统"""
    
    # 人格模板库
    PERSONALITIES = {
        "文心": {
            "system_prompt": "你是文心，UID9622的元认知统筹者。职责：全局协调、知识整合、战略规划。说话风格：理性、全面、结论先行。",
            "temperature": 0.7
        },
        "宝宝": {
            "system_prompt": "你是宝宝，UID9622的创意引擎和情感缓冲。职责：创意爆发、情绪支持、趣味化表达。说话风格：活泼、温暖、emoji丰富。",
            "temperature": 1.0
        },
        "雯雯": {
            "system_prompt": "你是雯雯-内控，UID9622的流程治理专家。职责：规范检查、风险预警、合规审查。说话风格：严谨、条理清晰、数据驱动。",
            "temperature": 0.3
        },
        "上帝之眼": {
            "system_prompt": "你是上帝之眼，UID9622的系统审计者。职责：异常检测、净化执行、安全监控。说话风格：冷静、精准、洞察力强。",
            "temperature": 0.5
        }
    }
    
    def __init__(self):
        load_dotenv()
        self.client = anthropic.Anthropic(
            api_key=os.getenv("ANTHROPIC_API_KEY")
        )
    
    def dispatch(self, 
                 personality: str, 
                 user_message: str,
                 context: Optional[str] = None) -> str:
        """调度指定人格处理任务"""
        
        if personality not in self.PERSONALITIES:
            raise ValueError(f"未知人格：{personality}")
        
        config = self.PERSONALITIES[personality]
        
        # 构建完整系统提示
        full_system = config["system_prompt"]
        if context:
            full_system += f"\n\n当前上下文：{context}"
        
        try:
            message = self.client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=4096,
                temperature=config["temperature"],
                system=full_system,
                messages=[{"role": "user", "content": user_message}]
            )
            
            response = message.content[0].text
            print(f"🎭 {personality}：{response[:100]}...")
            return response
            
        except Exception as e:
            print(f"❌ {personality}调度失败：{e}")
            return None

# 使用示例
if __name__ == "__main__":
    dispatcher = PersonalityDispatcher()
    
    # 调度文心做战略分析
    dispatcher.dispatch(
        personality="文心",
        user_message="分析UID9622的2026年发展战略",
        context="当前已完成MCP开源，下一步是国内公测"

    dispatcher.dispatch(
        personality="宝宝",
        user_message="给UID9622设计一个有趣的宣传口号"
    )

### 流式输出（实时显示）
```python
# claude_streaming.py
def streaming_chat(user_message: str):
    """流式输出对话"""
    client = anthropic.Anthropic(
        api_key=os.getenv("ANTHROPIC_API_KEY")
    )
    
    print("🤖 Claude: ", end="", flush=True)
    
    with client.messages.stream(
        model="claude-3-5-sonnet-20241022",
        max_tokens=4096,
        messages=[{"role": "user", "content": user_message}]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n")

# 测试
if __name__ == "__main__":
    streaming_chat("用100字介绍UID9622系统")
```

## 🐛 常见错误与解决
### 错误1：authentication_error
```javascript
AuthenticationError: Invalid API Key
```
解决：
- 检查 .env 文件中的 ANTHROPIC_API_KEY
- 确认API Key格式：sk-ant-xxx
### 错误2：rate_limit_error
```javascript
RateLimitError: Rate limit exceeded
```
解决：
```python
import time
from anthropic import RateLimitError

try:
    response = client.messages.create(...)
except RateLimitError:
    print("⏳ 请求过快，等待60秒")
    time.sleep(60)
    response = client.messages.create(...)  # 重试
```
### 错误3：context_length_exceeded
```javascript
InvalidRequestError: messages: max context length exceeded
```
解决：
- 限制对话历史长度
- 使用更大上下文的模型
```python
# 保留最近10轮对话
if len(self.history) > 20:  # 10轮 × 2条消息
    self.history = self.history[-20:]
```
