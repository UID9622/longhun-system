```python
class ValueGuidanceSystem:
    def __init__(self):
        self.universal_values = UniversalHumanValues()  # 普世价值观
        self.cultural_values = CulturalSpecificValues()  # 特定文化价值观
        self.value_integration_engine = ValueIntegrationEngine()
    
    def gentle_value_guidance(self, user_query, user_cultural_context):
        # 不是要改变用户的价值观，而是在其现有价值观基础上正向引导
        
        # 1. 理解用户当前价值观取向
        current_values = self.analyze_user_values(user_query, user_cultural_context)
        
        # 2. 找到正向发展的可能性
        positive_development_paths = self.identify_positive_growth_opportunities(
            current_values, user_cultural_context
        )
        
        # 3. 提供符合其文化背景的正向建议
        culturally_appropriate_guidance = self.generate_culturally_sensitive_advice(
            user_query=user_query,
            positive_paths=positive_development_paths,
            cultural_context=user_cultural_context
        )
        
        return culturally_appropriate_guidance
```

```python
class CulturalDimensionMatrix:
    def __init__(self):
        self.dimensions = {
            'power_distance': PowerDistanceAnalyzer(),        # 权力距离
            'individualism_collectivism': IndividualismAnalyzer(),  # 个人主义vs集体主义
            'uncertainty_avoidance': UncertaintyAvoidanceAnalyzer(),  # 不确定性规避
            'long_term_orientation': LongTermOrientationAnalyzer(),   # 长期导向
            'masculinity_femininity': MasculinityAnalyzer(),  # 男性化vs女性化
            'indulgence_restraint': IndulgenceAnalyzer()      # 纵容vs约束
        }
    
    def analyze_cultural_fit(self, user_behavior, cultural_context):
        cultural_scores = {}
        for dimension, analyzer in self.dimensions.items():
            scores[dimension] = analyzer.analyze(user_behavior, cultural_context)
        
        return self.generate_adaptation_strategy(cultural_scores)
```

```python
class CulturalLanguageAdapter:
    def __init__(self):
        self.language_cultural_patterns = {
            'zh-CN': ChineseCulturalPattern(),
            'zh-TW': TraditionalChineseCulturalPattern(), 
            'km': KhmerCulturalPattern(),
            'vi': VietnameseCulturalPattern(),
            'th': ThaiCulturalPattern(),
            'ms': MalayCulturalPattern(),
            'id': IndonesianCulturalPattern(),
            'tl': TagalogCulturalPattern(),
            'en': SEAEnglishCulturalPattern()
        }
    
    def adapt_response_to_culture(self, response, target_culture, user_context):
        cultural_pattern = self.language_cultural_patterns[target_culture]
        
        adapted_response = cultural_pattern.adapt_response(
            original_response=response,
            user_social_context=user_context.social_level,
            relationship_context=user_context.relationship,
            formality_level=user_context.formality_requirement
        )
        
        return adapted_response
```
