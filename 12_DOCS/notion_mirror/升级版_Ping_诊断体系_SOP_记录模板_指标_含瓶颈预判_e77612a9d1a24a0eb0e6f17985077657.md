## 📋 一键自检脚本
### Windows 自检脚本
复制以下内容到 PowerShell（管理员权限）：
```powershell
$ts=(Get-Date -Format "yyyyMMdd_HHmmss"); $out="$env:USERPROFILE\Desktop\net_diag_$ts.txt"
"=== NET DIAG $ts ===" | Out-File -FilePath $out -Encoding UTF8
"# 基本信息" | Tee-Object -FilePath $out -Append
systeminfo | Select-String 'OS Name|OS Version' | Tee-Object -FilePath $out -Append
Get-NetAdapter | Format-Table -Auto | Tee-Object -FilePath $out -Append
"# IP/路由" | Tee-Object -FilePath $out -Append
ipconfig /all | Tee-Object -FilePath $out -Append
route print | Tee-Object -FilePath $out -Append
"# ARP/邻居" | Tee-Object -FilePath $out -Append
arp -a | Tee-Object -FilePath $out -Append
"# 连通性测试" | Tee-Object -FilePath $out -Append
ping 127.0.0.1 -n 3 | Tee-Object -FilePath $out -Append
$gw=(Get-NetIPConfiguration | Where-Object {$_.IPv4DefaultGateway} | Select-Object -First 1).IPv4DefaultGateway.NextHop
if ($gw) { ping $gw -n 5 | Tee-Object -FilePath $out -Append }
ping 8.8.8.8 -n 10 | Tee-Object -FilePath $out -Append
"# MTU测试" | Tee-Object -FilePath $out -Append
ping 8.8.8.8 -f -l 1472 | Tee-Object -FilePath $out -Append
ping 8.8.8.8 -f -l 1400 | Tee-Object -FilePath $out -Append
"# DNS测试" | Tee-Object -FilePath $out -Append
nslookup baidu.com | Tee-Object -FilePath $out -Append
Write-Host "✅ 诊断完成，报告保存在桌面：$out"
```
### Linux/macOS 自检脚本
复制以下内容到终端：
```bash
#!/usr/bin/env bash
ts=$(date +%Y%m%d_%H%M%S); out="$HOME/net_diag_$ts.txt"
{
  echo "=== NET DIAG $ts ==="
  echo "# 基本信息"; uname -a
  echo "# 网卡状态"; ip -brief a 2>/dev/null || ifconfig
  echo "# 路由表"; ip r 2>/dev/null || netstat -rn
  echo "# DNS配置"; cat /etc/resolv.conf
  echo "# ARP表"; ip neigh show 2>/dev/null || arp -a
  echo "# 连通性测试"
  ping -c 3 127.0.0.1
  gw=$(ip route 2>/dev/null | awk '/default/ {print $3; exit}')
  if [ -n "$gw" ]; then ping -c 5 "$gw"; fi
  ping -c 10 8.8.8.8
  echo "# MTU测试"
  ping -M do -s 1472 -c 2 8.8.8.8 2>/dev/null || echo "MTU测试需要root权限"
  echo "# 路由跟踪"
  traceroute -I -m 15 8.8.8.8 2>/dev/null || echo "traceroute需要安装"
} | tee "$out"
echo "✅ 诊断完成：$out"
```

## 🎯 标准排查流程（5步定位）
### 第1步：本地自检（30秒）
检查项目：
- 网卡是否启用？IP地址是否正确？
- 子网掩码、网关、DNS是否配置？
- 是否有VPN或代理软件干扰？
快速命令：
- Windows：ipconfig /all
- Linux：ip addr show
- 测试本机：ping 127.0.0.1
### 第2步：网关连通（1分钟）
检查项目：
- 能否ping通网关？
- ARP表是否有网关记录？
- 网线/WiFi是否正常？
快速命令：
- Windows：ping 网关IP + arp -a
- Linux：ping 网关IP + ip neigh
### 第3步：外网连通（2分钟）
检查项目：
- 能否ping通外网IP（如8.8.8.8）？
- 路由表是否有默认网关？
- 防火墙是否拦截？
快速命令：
- ping 8.8.8.8
- traceroute 8.8.8.8（Linux）
- tracert 8.8.8.8（Windows）
### 第4步：域名解析（1分钟）
检查项目：
- 能否ping通域名（如baidu.com）？
- DNS服务器是否可达？
- hosts文件是否有错误配置？
快速命令：
- nslookup baidu.com
- ping baidu.com
### 第5步：深度诊断（5分钟）
检查项目：
- MTU是否过大？（测试：ping -f -l 1472）
- 是否有路由黑洞？（查看traceroute）
- 防火墙/ACL策略是否拦截？

## 🔍 常见故障场景
### 场景1：小包通、大包不通
现象：
- ping通，但下载/上传失败
- 网页打不开，但QQ能用
原因： MTU设置过大
解决：
```bash
# Windows测试MTU
ping 8.8.8.8 -f -l 1472  # 通过
ping 8.8.8.8 -f -l 1500  # 不通
# 说明MTU在1472-1500之间

# 调整MTU（Windows）
netsh interface ipv4 set subinterface "以太网" mtu=1400 store=persistent

# 调整MTU（Linux）
sudo ip link set eth0 mtu 1400
```
### 场景2：ping域名不通，ping IP通
现象：
- ping 8.8.8.8 通
- ping baidu.com 不通
原因： DNS解析问题
解决：
```bash
# 检查DNS
nslookup baidu.com

# 更换DNS为公共DNS
# Windows：网络适配器设置中修改
# Linux：编辑 /etc/resolv.conf
nameserver 8.8.8.8
nameserver 114.114.114.114
```
### 场景3：局域网互ping失败
现象：
- 同网段机器ping不通
- 能ping通网关
原因： ARP冲突或防火墙
解决：
```bash
# 清理ARP缓存
arp -d  # Windows
sudo ip neigh flush all  # Linux

# 检查防火墙
# Windows：关闭Windows防火墙测试
# Linux：sudo iptables -L
```

### 🐍 Python统一诊断脚本（跨平台）

完整脚本（保存为 net_diag_pro.py）：
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
网络诊断工业级脚本 v2.0
支持: Windows / Linux / macOS
特性: 自动诊断 + HTML报告 + 安全确认
"""
import os
import sys
import platform
import subprocess
import json
from datetime import datetime
from pathlib import Path

# 安全确认机制
def confirm_execution():
    print("\n⚠️  即将执行网络诊断，将会:")
    print("  1. 收集网络配置信息")
    print("  2. 执行连通性测试")
    print("  3. 生成诊断报告")
    print("\n✅ 本脚本为只读诊断，不会修改任何配置")
    response = input("\n是否继续? (yes/no): ").strip().lower()
    if response not in ['yes', 'y']:
        print("❌ 已取消执行")
        sys.exit(0)

# 跨平台命令执行
def run_cmd(cmd):
    try:
        result = subprocess.run(
            cmd, shell=True, capture_output=True, 
            text=True, timeout=30
        )
        return result.stdout + result.stderr
    except Exception as e:
        return f"Error: {str(e)}"

# 主诊断逻辑
def diagnose():
    confirm_execution()
    
    system = platform.system()
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    output_dir = Path.home() / "Desktop"
    report_file = output_dir / f"net_diag_pro_{timestamp}.html"
    
    print(f"\n🔍 开始诊断 ({system})...\n")
    
    diag_data = {
        'timestamp': timestamp,
        'system': system,
        'tests': []
    }
    
    # 收集信息（跨平台适配）
    if system == "Windows":
        tests = [
            ("网络配置", "ipconfig /all"),
            ("路由表", "route print"),
            ("ARP表", "arp -a"),
            ("网关测试", "ping -n 5 %gateway%"),
            ("外网测试", "ping -n 10 8.8.8.8"),
            ("DNS测试", "nslookup baidu.com")
        ]
    else:  # Linux/macOS
        tests = [
            ("网络配置", "ip addr show || ifconfig"),
            ("路由表", "ip route || netstat -rn"),
            ("ARP表", "ip neigh || arp -a"),
            ("网关测试", "ping -c 5 $(ip route | awk '/default/{print $3}')"),
            ("外网测试", "ping -c 10 8.8.8.8"),
            ("DNS测试", "nslookup baidu.com")
        ]
    
    for name, cmd in tests:
        print(f"  📌 {name}...", end=" ")
        output = run_cmd(cmd)
        diag_data['tests'].append({'name': name, 'cmd': cmd, 'output': output})
        print("✅")
    
    # 生成HTML报告
    html_content = generate_html_report(diag_data)
    report_file.write_text(html_content, encoding='utf-8')
    
    print(f"\n✅ 诊断完成！报告已保存:")
    print(f"   {report_file}\n")

def generate_html_report(data):
    html = f"""
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>网络诊断报告 - {data['timestamp']}</title>
    <style>
        body  font-family: monospace; margin: 20px; background: #1e1e1e; color: #d4d4d4; 
        .header  background: #0e639c; padding: 20px; border-radius: 8px; 
        .test-section  background: #252526; margin: 15px 0; padding: 15px; border-radius: 5px; 
        .test-title  color: #4ec9b0; font-weight: bold; 
        pre  background: #1e1e1e; padding: 10px; overflow-x: auto; border-left: 3px solid #007acc; 
    </style>
</head>
<body>
    <div class="header">
        <h1>🛰️ 网络诊断报告</h1>
        <p>时间: {data['timestamp']} | 系统: {data['system']}</p>
    </div>
"""
    for test in data['tests']:
        html += f"""
    <div class="test-section">
        <div class="test-title">📌 {test['name']}</div>
        <code>$ {test['cmd']}</code>
        <pre>{test['output']}</pre>
    </div>
"""
    html += """
</body>
</html>
"""
    return html

if __name__ == "__main__":
    diagnose()
```

```python
#!/usr/bin/env python3
"""
根据诊断结果自动生成修复建议
"""
import re

def analyze_and_suggest(diag_output):
    suggestions = []
    
    # 检测1: MTU问题
    if "packet needs to be fragmented" in diag_output or "Message too long" in diag_output:
        suggestions.append({
            'issue': 'MTU过大导致大包丢失',
            'risk': '🟢 低风险',
            'fix_windows': 'netsh interface ipv4 set subinterface "以太网" mtu=1400 store=persistent',
            'fix_linux': 'sudo ip link set eth0 mtu 1400',
            'rollback': 'MTU改为1500恢复默认值'
        })
    
    # 检测2: DNS解析失败
    if "can't find" in diag_output or "NXDOMAIN" in diag_output:
        suggestions.append({
            'issue': 'DNS解析失败',
            'risk': '🟢 低风险',
            'fix_windows': 'netsh interface ip set dns "以太网" static 8.8.8.8',
            'fix_linux': 'echo "nameserver 8.8.8.8" | sudo tee /etc/resolv.conf',
            'rollback': '改回原DNS服务器'
        })
    
    # 检测3: 网关不通
    if re.search(r'ping.*gateway.*100% packet loss', diag_output, re.IGNORECASE):
        suggestions.append({
            'issue': '网关不可达（可能网线/WiFi断开）',
            'risk': '🟡 中风险',
            'fix_windows': '检查网线连接 或 netsh wlan disconnect && netsh wlan connect',
            'fix_linux': 'sudo systemctl restart NetworkManager',
            'rollback': '无需回滚（重启网络服务）'
        })
    
    # 生成修复脚本
    if not suggestions:
        print("✅ 未发现明显问题")
        return
    
    print("\n🔍 发现以下问题:\n")
    for i, sug in enumerate(suggestions, 1):
        print(f"问题 {i}: {sug['issue']}")
        print(f"  风险等级: {sug['risk']}")
        print(f"  Windows修复: {sug['fix_windows']}")
        print(f"  Linux修复: {sug['fix_linux']}")
        print(f"  回滚方案: {sug['rollback']}\n")

# 使用示例
if __name__ == "__main__":
    with open("net_diag_output.txt", "r") as f:
        output = f.read()
    analyze_and_suggest(output)
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Lucky网络诊断修复终极脚本 v3.0 - 老大专用"""
import os, sys, platform, subprocess, socket, shutil, time
from datetime import datetime
from pathlib import Path

class LuckyNetFixer:
    def __init__(self):
        self.system = platform.system()
        self.issues = []
        self.fixes = []
        self.backup_dir = Path.home() / ".netfix_backups"
        self.backup_dir.mkdir(exist_ok=True)
        self.report_data = []
        
    def run_cmd(self, cmd, silent=True):
        """执行命令"""
        try:

                                   text=True, timeout=30, encoding='utf-8', errors='ignore')
            output = result.stdout + result.stderr
            return output, result.returncode

    def check_and_fix_dns(self):
        """DNS检测和修复"""
        print("🔍 检测DNS...", end=" ")
        try:
            socket.gethostbyname("baidu.com")
            print("✅ 正常")
            self.report_data.append(("DNS解析", "✅ 正常", "无需修复"))
            return True
        except:
            print("⚠️ 失败，修复中...")
            self.issues.append("DNS解析失败")
            
            if self.system == "Darwin":  # macOS
                self.run_cmd("networksetup -setdnsservers Wi-Fi 8.8.8.8 1.1.1.1")
                self.run_cmd("networksetup -setdnsservers Ethernet 8.8.8.8 1.1.1.1")
            elif self.system == "Linux":
                self.backup_config("/etc/resolv.conf", "DNS配置")
                self.run_cmd('echo -e "nameserver 8.8.8.8\\nnameserver 1.1.1.1" | sudo tee /etc/resolv.conf')
            else:  # Windows
                self.run_cmd('netsh interface ip set dns "以太网" static 8.8.8.8')
                self.run_cmd('netsh interface ip set dns "WLAN" static 8.8.8.8')
            
            time.sleep(1)
            try:
                socket.gethostbyname("baidu.com")
                print("  ✅ 修复成功")
                self.fixes.append("DNS → 8.8.8.8 / 1.1.1.1")
                self.report_data.append(("DNS解析", "⚠️ 失败", "✅ 已修复为8.8.8.8"))
                return True
            except:
                print("  ❌ 修复失败")
                self.report_data.append(("DNS解析", "❌ 失败", "❌ 修复失败"))
                return False
    
    def check_and_fix_mtu(self):
        """MTU检测和修复"""
        print("🔍 检测MTU...", end=" ")
        if self.system == "Windows":
            output, code = self.run_cmd("ping 8.8.8.8 -f -l 1472 -n 2")
            if "需要拆分" in output or code != 0:
                print("⚠️ 过大，优化中...")
                self.issues.append("MTU过大")
                self.run_cmd('netsh interface ipv4 set subinterface "以太网" mtu=1400 store=persistent')
                self.run_cmd('netsh interface ipv4 set subinterface "WLAN" mtu=1400 store=persistent')
                self.fixes.append("MTU → 1400")
                self.report_data.append(("MTU配置", "⚠️ 过大", "✅ 已优化为1400"))
            else:
                print("✅ 正常")
                self.report_data.append(("MTU配置", "✅ 正常", "无需修复"))
        else:
            output, code = self.run_cmd("ping -M do -s 1472 -c 2 8.8.8.8")
            if code != 0:
                print("⚠️ 过大，优化中...")
                self.issues.append("MTU过大")
                iface_output, _ = self.run_cmd("ip route | awk '/default/{print $5}'")
                iface = iface_output.strip()
                if iface:
                    self.run_cmd(f"sudo ip link set {iface} mtu 1400")
                    self.fixes.append(f"MTU ({iface}) → 1400")
                    self.report_data.append(("MTU配置", "⚠️ 过大", f"✅ 已优化{iface}为1400"))
            else:
                print("✅ 正常")
                self.report_data.append(("MTU配置", "✅ 正常", "无需修复"))
    
    def check_gateway(self):
        """网关连通性"""
        print("🔍 检测网关...", end=" ")
        if self.system == "Windows":
            gw_cmd = 'powershell "(Get-NetIPConfiguration).IPv4DefaultGateway.NextHop"'
        else:
            gw_cmd = "ip route | awk '/default/{print $3}'"
        
        gateway, _ = self.run_cmd(gw_cmd)
        gateway = gateway.strip()
        
        if gateway:
            output, code = self.run_cmd(f"ping -{'n' if self.system=='Windows' else 'c'} 3 {gateway}")
            if code == 0:
                print(f"✅ 可达 ({gateway})")
                self.report_data.append(("网关连通", f"✅ {gateway}", "无需修复"))
            else:
                print(f"❌ 不可达 ({gateway})")
                self.issues.append(f"网关{gateway}不可达")
                self.report_data.append(("网关连通", f"❌ {gateway}不可达", "检查网线/WiFi"))
        else:
            print("❌ 未配置")
            self.issues.append("未配置网关")
            self.report_data.append(("网关连通", "❌ 未配置", "检查网络设置"))
    
    def optimize_caches(self):
        """清理缓存优化"""
        print("⚡ 优化网络缓存...", end=" ")
        
        if self.system == "Darwin":
            self.run_cmd("sudo dscacheutil -flushcache")
            self.run_cmd("sudo killall -HUP mDNSResponder")
            self.fixes.append("已刷新DNS缓存")
        elif self.system == "Linux":
            self.run_cmd("sudo ip neigh flush all")
            self.fixes.append("已清理ARP缓存")
        else:  # Windows
            self.run_cmd("ipconfig /flushdns")
            self.run_cmd("arp -d *")
            self.fixes.append("已刷新DNS和ARP")
        
        print("✅")
        self.report_data.append(("缓存清理", "✅ 完成", "DNS/ARP已刷新"))
    
    def speed_test(self):
        """简单速度测试"""
        print("🚀 测速（ping延迟）...", end=" ")
        targets = [("国内", "baidu.com"), ("国际", "google.com"), ("DNS", "8.8.8.8")]
        results = []
        
        for name, target in targets:
            output, code = self.run_cmd(f"ping -{'n' if self.system=='Windows' else 'c'} 4 {target}")
            if code == 0:
                # 提取平均延迟
                if self.system == "Windows":
                    import re
                    match = re.search(r'平均 = (\d+)ms', output)
                    if match:
                        results.append(f"{name}: {match.group(1)}ms")
                else:
                    import re
                    match = re.search(r'avg[=/]\s*(\d+\.\d+)', output)
                    if match:
                        results.append(f"{name}: {match.group(1)}ms")
        
        if results:
            print(f"✅ {' | '.join(results)}")
            self.report_data.append(("网络延迟", " | ".join(results), "✅"))
        else:
            print("⚠️")
            self.report_data.append(("网络延迟", "⚠️ 测试失败", "-"))
```
