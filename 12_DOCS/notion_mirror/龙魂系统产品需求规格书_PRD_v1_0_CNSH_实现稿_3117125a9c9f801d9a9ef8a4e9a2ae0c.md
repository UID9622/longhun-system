    def execute_hook(self, hook_name, *args, **kwargs):
        if hook_name in self.hooks:
            for callback in self.hooks[hook_name]:
                try:
                    callback(*args, **kwargs)
                except Exception as e:
                    print(f"钩子 {hook_name} 执行失败: {e}")

        """授权权重调整（钩子触发，操作留痕）"""

        if module in self.authorization_weights:

    def generate_dna(self, seed):
        """DNA生成（对接 §3.2 DNA追溯系统）"""
        self.hook_system.execute_hook('before_redline_check', seed)
        dna = hashlib.sha256(seed.encode()).hexdigest()
        if dna in self.registry:
            raise Exception("DNA冲突，系统拒绝注册")
        self.registry.add(dna)
        self.hook_system.execute_hook('after_redline_check', dna)
        return dna

    def update_personality(self, h, delta, eta=0.05):
        """人格向量更新，投影到 [-1,1]^28（对接 §7.2 人格演化系统）"""
        self.hook_system.execute_hook('before_personality_update', h, delta, eta)
        if h not in self.personality_vectors:
            self.personality_vectors[h] = [0.0] * 28
        vec = self.personality_vectors[h]
        for i in range(len(vec)):
            vec[i] = max(-1.0, min(1.0, vec[i] + eta * delta[i]))
        self.hook_system.execute_hook('after_personality_update', h, vec)
        return vec

    def append_memory(self, h, action):
        """链式记忆追加（对接量子记忆系统）"""
        if h not in self.memory_chains:
            prev_hash = hashlib.sha256(h.encode()).hexdigest()
            self.memory_chains[h] = []
        else:
            prev_hash = self.memory_chains[h][-1]['hash']
        new_hash = hashlib.sha256(
            (prev_hash + action + str(time.time())).encode()
        ).hexdigest()
        node = {
            'hash': new_hash,
            'prev_hash': prev_hash,
            'action': action,
            'timestamp': datetime.now().isoformat()
        }
        self.memory_chains[h].append(node)
        return new_hash
