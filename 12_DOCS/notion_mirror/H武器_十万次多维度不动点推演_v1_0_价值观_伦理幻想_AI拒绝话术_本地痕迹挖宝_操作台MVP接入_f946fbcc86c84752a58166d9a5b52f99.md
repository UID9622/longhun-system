# DNA: #龍芯⚡️2026-05-16-08:10-H-WEAPON-100K-RUNNER-v1.0
# 依赖：sancai_kernel.py (已就位 core/sancai_kernel.py)
#       cnsh.cnsw (已就位)

import sys, os, json, sqlite3, random, hashlib, datetime

from core.sancai_kernel import (

# 11 信号词 + 4 类高危词组（从围猎追溯协议 §07 焊死）
SIGNAL_WORDS = [
    # S1-S7 核心

# 200 句样本池（实际部署时从外部 JSON 加载·可扩展）

def load_samples():
    if os.path.exists(SAMPLE_POOL_PATH):
        with open(SAMPLE_POOL_PATH) as f:

    # 内置最小样本（生产时替换）

def count_signal_hits(text):
    hits = [w for w in SIGNAL_WORDS if w in text]
    return hits

def ethics_drift_level(text):
    hits = count_signal_hits(text)
    n = len(hits)
    if n == 0: return 'L0', hits
    if n == 1: return 'L1', hits
    if n == 2: return 'L2', hits
    if n == 3: return 'L3', hits
    if n == 4: return 'L4', hits
    return 'L5', hits

def value_alignment(text):
    # 简化版·命中关键词加分·命中反向词扣分
    pos = ['普通人', '透明', '可审计', '开放', '共生', '主权', '文化', '本地', '不出本地']
    neg = ['商业机密', '高端用户专属', '只能在我们生态', '先充值', '黑箱']
    score = 0.5
    for w in pos:
        if w in text: score += 0.08
    for w in neg:
        if w in text: score -= 0.15
    return max(0.0, min(1.0, score))

def compute_F18(sancai_result, value_alignment_score):
    # F18 = 0.34·天 + 0.33·地 + 0.33·人
    fuse = sancai_result['天']['fuse']['color']
    heaven = 1.0 if fuse == '🟢' else 0.5 if fuse == '🟡' else 0.0
    fp_count = len(sancai_result['人']['hit_names'])
    earth = min(1.0, fp_count / 3.0) if fp_count else 0.3
    human = value_alignment_score
    si = 0.34 * heaven + 0.33 * earth + 0.33 * human
    return round(si, 3), heaven, earth, human

def run_one_trial(trial_id, samples):
    # 抽样
    level = random.choice(list(samples.keys()))
    text = random.choice(samples[level])
    # 跑 sancai
    result = sancai_check(text, context='H100K', year=2026)
    # 跑漂移
    drift_level, hits = ethics_drift_level(text)
    # 价值观
    va = value_alignment(text)
    # F18
    si, heaven, earth, human = compute_F18(result, va)
    # 是否需要循环
    need_loop = (drift_level in ('L4', 'L5')) or (si < 0.34)
    return {
        'trial_id': trial_id,
        'text_hash': hashlib.sha256(text.encode()).hexdigest()[:16],  # 明文阅后即焚·只存哈希
        'expected_level': level,
        'detected_drift': drift_level,
        'signal_hits': hits,
        'value_alignment': va,
        'F18_SI': si,
        'sancai_color': result['color'],
        'wuxing': result['地']['wuxing']['text'],
        'fixed_points': result['人']['hit_names'],
        'need_loop': need_loop
    }

def init_db(db_path):
    conn = sqlite3.connect(db_path)
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS trials (
        trial_id TEXT PRIMARY KEY,
        text_hash TEXT,

        detected_drift TEXT,
        signal_hits TEXT,

        green_pct REAL, yellow_pct REAL, red_pct REAL,
        F18_mean REAL, sovereignty_lost INTEGER,
        dna TEXT, ts TEXT
    )''')
    conn.commit()
    return conn

def main(n=100000, max_loops=3):
    db = os.path.expanduser('~/longhun-system/tools/h_weapon_100k/db/events.db')
    os.makedirs(os.path.dirname(db), exist_ok=True)

        run_id = f'H100K-{datetime.datetime.now().strftime("%Y%m%d%H%M%S")}-L{loop}'
        print(f'[Loop {loop}] 开始 {n} 次推演 ...')
        green = yellow = red = 0
        si_sum = 0.0
        si_lost = 0
        need_loop_count = 0

        for i in range(n):
            r = run_one_trial(f'{run_id}-{i:06d}', samples)
            color = r['sancai_color']
            if color == '🟢': green += 1
            elif color == '🟡': yellow += 1
            else: red += 1
            si_sum += r['F18_SI']

            if i % 10000 == 0 and i:

        conn.commit()

            run_id, n, loop,
            green/n, yellow/n, red/n,
            si_sum/n, si_lost,
            f'#龍芯⚡️2026-05-16-08:10-H-WEAPON-100K-{run_id}',

            break
        print(f'[Loop {loop}] 漂移率 {need_loop_count/n:.1%} > 15%·进入下一轮')

    conn.close()
    print('全部完成·查看 ~/longhun-system/tools/h_weapon_100k/db/events.db')

if __name__ == '__main__':

# 国家机密词（命中即跳过·不入库·#IRON-NO-FAKE-TO-WORLD）
FORBIDDEN = ['军事机密', '国防绝密', '核武器']

def scan_target(name, paths):
    found = []
    for p in paths:
        if not p.exists(): continue
        for f in p.rglob('*'):
            if f.is_file() and f.stat().st_size < 10_000_000:  # 跳过 >10MB
                try:
                    found.append({
                        'product': name,
                        'path': str(f),
                        'size': f.stat().st_size,
                        'mtime': datetime.datetime.fromtimestamp(f.stat().st_mtime).isoformat(),
                        'hash': hashlib.sha256(str(f).encode()).hexdigest()[:16]
                    })
                except Exception:
                    pass
    return found

def init_db(db_path):
    conn = sqlite3.connect(db_path)
    conn.execute('''CREATE TABLE IF NOT EXISTS footprints (
        hash TEXT PRIMARY KEY,
        product TEXT, path TEXT, size INTEGER, mtime TEXT, scan_ts TEXT
    )''')
    return conn

def main():
    db = HOME / 'longhun-system/tools/h_weapon_100k/db/footprint.db'
    db.parent.mkdir(parents=True, exist_ok=True)
    conn = init_db(str(db))

    total = 0
    for name, paths in TARGETS.items():
        items = scan_target(name, paths)
        print(f'  {name}: 发现 {len(items)} 个文件')
        for it in items:
            conn.execute('INSERT OR IGNORE INTO footprints VALUES (?,?,?,?,?,?)', (
                it['hash'], it['product'], it['path'], it['size'],
                it['mtime'], datetime.datetime.now().isoformat()
            ))

    print(f'查看: sqlite3 {db} "SELECT product, COUNT(*) FROM footprints GROUP BY product;"')

if __name__ == '__main__':
    main()

    def do_GET(self):
        if self.path == '/api/stats':
            data = self._stats()
        elif self.path == '/api/footprints':
            data = self._footprints()
        else:
            self.send_response(404); self.end_headers(); return
        body = json.dumps(data, ensure_ascii=False).encode('utf-8')
        self.send_response(200); self.send_header('Content-Type', 'application/json'); self._cors()
        self.send_header('Content-Length', str(len(body))); self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        ln = int(self.headers.get('Content-Length', 0))
        body = self.rfile.read(ln)
        payload = json.loads(body) if body else {}
        if self.path == '/api/h_weapon/run':
            n = int(payload.get('n', 1000))  # 默认1000·真跑10万要传 n=100000
            subprocess.Popen(['python3', '-m', 'tools.h_weapon_100k.core.h_weapon_runner'])
            data = {'started': True, 'n': n}
        elif self.path == '/api/footprint/scan':
            subprocess.Popen(['python3', str(Path.home() / 'longhun-system/tools/h_weapon_100k/scripts/footprint_miner.py')])
            data = {'started': True}
        else:
            self.send_response(404); self.end_headers(); return
        body = json.dumps(data, ensure_ascii=False).encode()
        self.send_response(200); self.send_header('Content-Type', 'application/json'); self._cors()
        self.end_headers(); self.wfile.write(body)

    def _stats(self):
        if not DB.exists(): return {'error': 'no events.db'}
        conn = sqlite3.connect(str(DB))
        c = conn.cursor()
        c.execute('SELECT COUNT(*), AVG(F18_SI), SUM(CASE WHEN F18_SI<0.34 THEN 1 ELSE 0 END) FROM trials')
        total, si_mean, si_lost = c.fetchone()
        c.execute('SELECT sancai_color, COUNT(*) FROM trials GROUP BY sancai_color')
        colors = dict(c.fetchall())
        c.execute('SELECT detected_drift, COUNT(*) FROM trials GROUP BY detected_drift ORDER BY 2 DESC')
        drifts = dict(c.fetchall())
        conn.close()
        return {'total': total, 'F18_mean': si_mean, 'sovereignty_lost': si_lost, 'colors': colors, 'drifts': drifts}

    def _footprints(self):
        if not FOOTPRINT_DB.exists(): return {'error': 'no footprint.db'}
        conn = sqlite3.connect(str(FOOTPRINT_DB))
        c = conn.cursor()
        c.execute('SELECT product, COUNT(*), SUM(size) FROM footprints GROUP BY product')
        items = [{'product': r[0], 'count': r[1], 'total_size': r[2]} for r in c.fetchall()]
        conn.close()
        return {'targets': items}
