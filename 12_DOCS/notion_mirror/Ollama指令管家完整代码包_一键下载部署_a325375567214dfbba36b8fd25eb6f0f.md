#!/usr/bin/env python3
"""指令投喂脚本 - Lucky专用

DNA: #UID9622-CMD-MANAGER-FEEDER-V1.0
作者: 宝宝 + Lucky
创建: 2025-12-12
"""

import sys
import argparse
from pathlib import Path
from datetime import datetime

# 添加src到路径
sys.path.insert(0, str(Path(__file__).parent.parent / 'src'))

from extractor import CommandExtractor
from optimizer import CommandOptimizer
from notion_sync import NotionSync
from utils import load_env

def main():
    parser = argparse.ArgumentParser(
        description='投喂指令到Ollama管家',
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog='''
使用示例：
  # 交互式输入
  python feed_command.py --interactive
  
  # 从文件读取
  python feed_command.py --file commands.txt
  
  # 直接输入文本
  python feed_command.py --text "/LU-TEST - 测试指令"
        '''
    )
    parser.add_argument('--text', help='直接输入指令文本')
    parser.add_argument('--file', help='从文件读取指令')
    parser.add_argument('--interactive', action='store_true', help='交互式输入')
    parser.add_argument('--dry-run', action='store_true', help='只提取不同步')
    
    args = parser.parse_args()
    
    try:
        # 加载环境变量
        config = load_env()
        
        # 获取输入文本
        if args.text:
            raw_text = args.text
        elif args.file:
            raw_text = Path(args.file).read_text(encoding='utf-8')
        elif args.interactive:
            print("请输入指令内容（输入EOF或按Ctrl+D结束）：")
            print("-" * 50)
            lines = []
            try:
                while True:
                    line = input()
                    lines.append(line)
            except EOFError:
                pass
            raw_text = '\n'.join(lines)
        else:
            parser.print_help()
            return 1
        
        if not raw_text.strip():
            print("❌ 错误：输入内容为空")
            return 1
        
        print("\n" + "=" * 50)
        print("🔍 正在提取指令...")
        print("=" * 50)
        
        # 1. 提取指令
        extractor = CommandExtractor()
        result = extractor.extract(raw_text)
        commands = result.get('commands', [])
        
        if not commands:
            print("❌ 未找到任何指令")
            return 1
        
        print(f"✅ 找到 {len(commands)} 个指令\n")
        
        # 2. 优化分析
        optimizer = CommandOptimizer()
        
        # 3. 同步到Notion
        if not args.dry_run:
            notion = NotionSync(
                token=config['NOTION_TOKEN'],
                database_id=config['NOTION_DATABASE_ID']
            )
        
        for i, cmd in enumerate(commands, 1):
            print(f"\n{'=' * 50}")
            print(f"📝 处理第 {i}/{len(commands)} 个指令")
            print(f"{'=' * 50}")
            print(f"名称: {cmd.get('name', 'N/A')}")
            print(f"语法: {cmd.get('syntax', 'N/A')}")
            print(f"分类: {cmd.get('category', 'N/A')}")
            print(f"优先级: {cmd.get('priority', 'N/A')}")
            
            # 优化分析
            print("\n⚡ 正在分析优化建议...")
            analysis = optimizer.analyze(cmd)
            print(f"评分: {analysis.get('current_score', 0)}/100")
            
            # 同步到Notion
            if not args.dry_run:
                cmd['optimization'] = analysis
                try:
                    page = notion.create_command_page(cmd)
                    page_url = page.get('url', 'N/A')
                    print(f"✅ 已同步到Notion: {page_url}")
                except Exception as e:
                    print(f"❌ Notion同步失败: {e}")
            else:
                print("⏭️  跳过Notion同步（dry-run模式）")
            
            # 显示优化建议
            suggestions = analysis.get('suggestions', [])
            if suggestions:
                print("\n💡 优化建议:")
                for sug in suggestions:
                    print(f"  • {sug}")
        
        print("\n" + "=" * 50)
        print("🎉 全部完成！")
        print("=" * 50)
        return 0
        
    except KeyboardInterrupt:
        print("\n\n⚠️  用户中断")
        return 130
    except Exception as e:
        print(f"\n❌ 错误: {e}")
        import traceback
        traceback.print_exc()
        return 1

if __name__ == '__main__':
    sys.exit(main())

"""指令提取器

DNA: #UID9622-CMD-MANAGER-EXTRACTOR-V1.0
功能: 使用Ollama从文本中提取指令信息
"""

import ollama
import json
from datetime import datetime

class CommandExtractor:
    def __init__(self, model="qwen2.5:14b"):
        self.model = model
    
    def extract(self, raw_text):
        """从文本中提取指令信息
        
        Args:
            raw_text: 原始文本内容
            
        Returns:
            dict: {
                'commands': [
                    {
                        'name': '/LU-XXX',
                        'syntax': '完整语法',
                        'description': '功能描述',
                        'category': '分类',
                        'priority': 'P0/P1/P2',
                        'use_cases': ['场景1', '场景2'],
                        'parameters': {'param1': 'desc1'}
                    }
                ]
            }
        """
        
        prompt = f"""分析以下内容，提取所有指令信息。

输入文本：
{raw_text}

请以JSON格式输出，格式如下：
{{
  "commands": [
    {{
      "name": "指令名称（如/LU-XXX）",
      "syntax": "完整语法（如/LU-XXX [--param]）",
      "description": "功能描述",
      "category": "分类（系统类/记忆类/人格类/数据类）",
      "priority": "优先级（P0/P1/P2/P3）",
      "use_cases": ["使用场景1", "使用场景2"],
      "parameters": 
        "param_name": "参数说明"
      
    }}
  ]
}}

要求：
1. 识别所有类似 /LU-XXX 或 /XXX 的指令
2. 提取参数（以 -- 开头的）
3. 理解使用场景
4. 判断优先级（P0=永恒级，P1=核心，P2=常用，P3=辅助）
5. 只返回JSON，不要其他说明文字
"""
        
        try:
            response = ollama.generate(
                model=self.model,
                prompt=prompt,
                options={
                    'temperature': 0.3,
                    'top_p': 0.9
                }
            )
            
            # 提取JSON部分
            response_text = response['response'].strip()
            
            # 尝试提取JSON（可能被包裹在```json```中）
            if '```json' in response_text:
                start = response_text.find('```json') + 7
                end = response_text.find('```', start)
                response_text = response_text[start:end].strip()
            elif '```' in response_text:
                start = response_text.find('```') + 3
                end = response_text.find('```', start)
                response_text = response_text[start:end].strip()
            
            result = json.loads(response_text)
            return result
            
        except json.JSONDecodeError as e:
            print(f"JSON解析错误: {e}")
            print(f"原始响应: {response.get('response', 'N/A')[:200]}...")
            # 返回空结果
            return {'commands': []}
        except Exception as e:
            print(f"提取错误: {e}")
            return {'commands': []}
```

"""Notion同步器

DNA: #UID9622-CMD-MANAGER-NOTION-SYNC-V1.0
功能: 将指令信息同步到Notion数据库
"""

import requests
from datetime import datetime
import hashlib

class NotionSync:
    def __init__(self, token, database_id):
        self.token = token
        self.database_id = database_id
        self.headers = {
            "Authorization": f"Bearer {token}",
            "Content-Type": "application/json",
            "Notion-Version": "2022-06-28"
        }
        self.base_url = "https://api.notion.com/v1"
    
    def generate_dna(self, command_name):
        """生成DNA编号
        
        格式: #UID9622-CMD-{指令名}-{时间戳}
        """
        timestamp = datetime.now().strftime('%Y%m%d%H%M%S')
        clean_name = command_name.replace('/', '').replace(' ', '-').replace('_', '-')
        return f"#UID9622-CMD-{clean_name}-{timestamp}"
    
    def create_command_page(self, command_data):
        """创建指令页面
        
        Args:
            command_data: 指令信息字典，包含optimization字段
            
        Returns:
            dict: Notion API响应
        """
        
        # 生成DNA编号
        dna = self.generate_dna(command_data['name'])
        
        # 获取优化信息
        opt = command_data.get('optimization', {})
        
        # 构建页面数据
        page_data = {
            "parent": {"database_id": self.database_id},
            "properties": {
                "指令名称": {
                    "title": [{"text": {"content": command_data.get('name', 'Unnamed')}}]
                },
                "DNA编号": {
                    "rich_text": [{"text": {"content": dna}}]
                },
                "指令语法": {
                    "rich_text": [{"text": {"content": command_data.get('syntax', '')}}]
                },
                "功能描述": {
                    "rich_text": [{"text": {"content": command_data.get('description', '')}}]
                },
                "分类": {
                    "select": {"name": command_data.get('category', '未分类')}
                },
                "优先级": {
                    "select": {"name": command_data.get('priority', 'P2')}
                },
                "状态": {
                    "select": {"name": "活跃"}
                },
                "当前版本": {
                    "rich_text": [{"text": {"content": "v1.0"}}]
                },
                "优化评分": {
                    "number": opt.get('current_score', 70)
                },
                "创建时间": {
                    "date": {"start": datetime.now().isoformat()}
                }
            }
        }
        
        # 调用Notion API
        response = requests.post(
            f"{self.base_url}/pages",
            headers=self.headers,
            json=page_data
        )
        
        if response.status_code == 200:
            return response.json()
        else:
            error_msg = f"Notion同步失败: {response.status_code}"
            try:
                error_detail = response.json()
                error_msg += f" - {error_detail.get('message', '')}"
            except:
                error_msg += f" - {response.text}"
            raise Exception(error_msg)

"""工具函数

DNA: #UID9622-CMD-MANAGER-UTILS-V1.0
"""

import os
from pathlib import Path
from dotenv import load_dotenv

def load_env():
    """加载环境变量
    
    Returns:
        dict: 配置字典
        
    Raises:
        FileNotFoundError: .env文件不存在
        ValueError: 缺少必需的环境变量
    """
    
    # 查找.env文件
    env_path = Path('.env')
    if not env_path.exists():
        raise FileNotFoundError(
            "找不到.env文件！请创建.env文件并配置：\n"
            "NOTION_TOKEN=your_token\n"
            "NOTION_DATABASE_ID=your_database_id"
        )
    
    load_dotenv(env_path)
    
    config = {
        'NOTION_TOKEN': os.getenv('NOTION_TOKEN'),
        'NOTION_DATABASE_ID': os.getenv('NOTION_DATABASE_ID')
    }
    
    # 验证必需变量
    missing = [k for k, v in config.items() if not v]
    if missing:
        raise ValueError(f"缺少环境变量: {', '.join(missing)}")
    
    # 验证Token格式
    if config['NOTION_TOKEN'] == 'your_notion_integration_token_here':
        raise ValueError("请在.env文件中配置真实的NOTION_TOKEN")
    
    return config

## 🎯 Notion数据库结构

创建数据库时需要包含以下属性：

- **指令名称** (Title) - 主标题
- **DNA编号** (Text) - 唯一追溯码
- **指令语法** (Text) - 完整语法
- **功能描述** (Text) - 功能说明
- **分类** (Select) - 系统类/记忆类/人格类/数据类
- **优先级** (Select) - P0/P1/P2/P3
- **状态** (Select) - 活跃/待优化/已废弃
- **当前版本** (Text) - v1.0/v2.0/v3.0
- **优化评分** (Number) - 0-100分
- **创建时间** (Date) - 自动记录
