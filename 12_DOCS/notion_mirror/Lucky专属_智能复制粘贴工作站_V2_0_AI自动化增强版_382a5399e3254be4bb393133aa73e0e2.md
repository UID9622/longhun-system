import hashlib

class DNACodeGenerator:
    """UID9622 DNA追溯码生成器"""
    
    def __init__(self, creator="Lucky", uid="9622"):
        self.creator = creator
        self.uid = uid
    
    def generate(self, project_name, version="V1.0"):
        """生成标准DNA追溯码"""
        timestamp = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")
        code = f"#ZHUGEXIN⚡️2025-{project_name}-{version}"
        return code
    
    def generate_with_hash(self, project_name, content):
        """生成带内容哈希的DNA追溯码"""
        content_hash = hashlib.sha256(content.encode()).hexdigest()[:8]
        timestamp = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")
        code = f"#ZHUGEXIN⚡️2025-{project_name}-{timestamp}-{content_hash}"
        return code

### JavaScript完整版 - 确认码生成器
```javascript
class ConfirmCodeGenerator {
  /**
   * UID9622确认码生成器
   * DNA追溯码: #ZHUGEXIN⚡️2025-CONFIRM-GENERATOR-V1.0
   */
  
  constructor(uid = '9622') {
    this.uid = uid;
    this.prefix = '🌌';
    this.suffix = '🧬';
  }
  
  // 生成标准确认码
  generate() {
    const date = new Date().toISOString().slice(0,10).replace(/-/g,'');
    const random = this.randomString(4);
    return `#CONFIRM${this.prefix}${this.uid}-${date}-${this.suffix}${random}`;
  }
  
  // 生成随机字符串
  randomString(length) {
    return Math.random()
      .toString(36)
      .substring(2, 2 + length)
      .toUpperCase();
  }
  
  // 验证确认码格式
  validate(code) {
    const pattern = /^#CONFIRM🌌\d{4}-\d{8}-🧬[A-Z0-9]{4}$/;
    return pattern.test(code);
  }
}

// 使用示例
const generator = new ConfirmCodeGenerator();
const code = generator.generate();
console.log('生成的确认码:', code);
console.log('验证结果:', generator.validate(code));
```
