```python
# spider_net_crawler.py
import scrapy
from scrapy.crawler import CrawlerProcess
from fake_useragent import UserAgent
import random
import time
from typing import List, Dict
import hashlib

class SpiderNetCrawler:
    """
    UID9622蜘蛛网爬虫系统
    
    策略：
    - 60% 主流源（高权威性）
    - 30% 边缘源（独特视角）
    - 10% 疯狂源（前沿创新）
    """
    
    def __init__(self):
        self.ua = UserAgent()
        self.sources = self._initialize_sources()
        self.visited = set()
        self.data_lake = []
    
    def _initialize_sources(self) -> Dict[str, List[str]]:
        """初始化信息源分类"""
        return {
            'mainstream': [  # 60%主流
                'https://news.ycombinator.com',  # HackerNews
                'https://www.reddit.com/r/MachineLearning',
                'https://arxiv.org/list/cs.AI/recent',
                'https://techcrunch.com/category/artificial-intelligence/',
                # ... 更多主流源
            ],
            'edge': [  # 30%边缘
                'https://www.lesswrong.com',
                'https://marginalrevolution.com',
                # 小众技术博客
                # 独立研究者网站
            ],
            'crazy': [  # 10%疯狂
                # 实验性项目
                # GitHub trending
                # 前沿研究组
            ]
        }
    
    def get_random_headers(self) -> Dict:
        """生成随机请求头（反爬）"""
        return {
            'User-Agent': self.ua.random,
            'Accept': 'text/html,application/xhtml+xml',
            'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.8',
            'Accept-Encoding': 'gzip, deflate, br',
            'Connection': 'keep-alive',
            'Upgrade-Insecure-Requests': '1',
            'Cache-Control': 'max-age=0',
            'DNT': '1',
            'Referer': random.choice([
                'https://www.google.com',
                'https://www.bing.com',
                'https://www.baidu.com'
            ])
        }
    
    def calculate_sleep_time(self, source_type: str) -> float:
        """智能延迟（避免触发限流）"""
        base_delays = {
            'mainstream': (2, 5),  # 主流站点更谨慎
            'edge': (1, 3),
            'crazy': (0.5, 2)
        }
        
        min_delay, max_delay = base_delays.get(source_type, (1, 3))
        return random.uniform(min_delay, max_delay)
    
    def crawl_with_strategy(self, topic: str, max_pages: int = 100):
        """
        按蜘蛛网策略爬取
        """
        results = {
            'mainstream': [],
            'edge': [],
            'crazy': []
        }
        
        # 分配爬取配额
        quotas = {
            'mainstream': int(max_pages * 0.6),
            'edge': int(max_pages * 0.3),
            'crazy': int(max_pages * 0.1)
        }
        
        for source_type, urls in self.sources.items():
            quota = quotas[source_type]
            pages_per_source = quota // len(urls)
            
            for url in urls:
                try:
                    # 爬取
                    data = self._fetch_page(url, topic, pages_per_source)
                    results[source_type].extend(data)
                    
                    # 智能延迟
                    time.sleep(self.calculate_sleep_time(source_type))
                    
                except Exception as e:
                    print(f"❌ 爬取失败 {url}: {e}")
                    continue
        
        return results
    
    def _fetch_page(self, url: str, topic: str, max_items: int) -> List[Dict]:
        """实际爬取逻辑"""
        import requests
        from bs4 import BeautifulSoup
        
        headers = self.get_random_headers()
        
        try:
            response = requests.get(
                url, 
                headers=headers, 
                timeout=10
            )
            response.raise_for_status()
            
            soup = BeautifulSoup(response.content, 'lxml')
            
            # 提取标题和链接（需根据实际网站调整）
            items = []
            for item in soup.find_all('a', limit=max_items):
                title = item.get_text(strip=True)
                link = item.get('href')
                
                if title and topic.lower() in title.lower():
                    items.append({
                        'title': title,
                        'url': link,
                        'source': url,
                        'timestamp': time.time()
                    })
            
            return items
            
        except Exception as e:
            print(f"⚠️ 请求失败: {e}")
            return []
```

```python
# data_cleaner.py
import jieba
import hashlib
from typing import List, Dict

class DataCleaner:
    """数据清洗器"""
    
    def __init__(self):
        self.seen_hashes = set()
    
    def deduplicate(self, items: List[Dict]) -> List[Dict]:
        """去重"""
        unique_items = []
        
        for item in items:
            # 基于标题生成哈希
            content_hash = hashlib.md5(
                item['title'].encode()
            ).hexdigest()
            
            if content_hash not in self.seen_hashes:
                self.seen_hashes.add(content_hash)
                unique_items.append(item)
        
        return unique_items
    
    def extract_keywords(self, text: str, top_k: int = 10) -> List[str]:
        """提取关键词（中文）"""
        import jieba.analyse
        
        keywords = jieba.analyse.extract_tags(
            text,
            topK=top_k,
            withWeight=False
        )
        
        return keywords
    
    def filter_by_relevance(self, items: List[Dict], keywords: List[str]) -> List[Dict]:
        """按相关性过滤"""
        filtered = []
        
        for item in items:
            title_lower = item['title'].lower()
            
            # 计算匹配度
            matches = sum(1 for kw in keywords if kw.lower() in title_lower)
            relevance = matches / len(keywords)
            
            if relevance >= 0.3:  # 至少30%关键词匹配
                item['relevance'] = relevance
                filtered.append(item)
        
        # 按相关性排序
        filtered.sort(key=lambda x: x['relevance'], reverse=True)
        
        return filtered
```
