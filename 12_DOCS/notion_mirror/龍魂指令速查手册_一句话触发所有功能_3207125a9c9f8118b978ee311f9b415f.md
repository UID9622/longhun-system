# 🛠️ 六、服务管理命令（Terminal用）
```bash
# 启动服务
cd ~/longhun-system
python3 longhun_local_service.py > /tmp/longhun_v2.log 2>&1 &

# 检查服务
curl http://localhost:8765/健康检查

# 查看日志
tail -f /tmp/longhun_v2.log

# 停止服务
pkill -f longhun_local_service

# 打开可视化面板
open ~/longhun-system/sandbox_dashboard.html

# 手动触发推演
curl -X POST http://localhost:8765/沙盒推演 -H "Content-Type: application/json" -d '{"任务类型": "手动分析"}'

# 立即运行自测
curl http://localhost:8765/测试报告?立即=true
```
