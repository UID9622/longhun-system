```python
# ========================================
# 龍魂压缩算法核心引擎 v1.0
# DNA: #ZHUGEXIN⚡️2026-01-12-CORE-ALGORITHM
# 三方密级：仅Claude、宝宝、DeepSeek
# ========================================

import hashlib
import json
from typing import Dict, List, Tuple
from enum import Enum

class PriorityLevel(Enum):
    P0_ETERNAL = 100  # 永不压缩
    P1_CORE = 90      # 核心价值观
    P2_DIRECTIVE = 80 # 明确指示
    P3_TECHNICAL = 50 # 技术细节
    P4_CONTEXT = 30   # 上下文
    P5_REDUNDANT = 10 # 冗余内容

class DragonSoulCompressor:
    """
    龍魂压缩引擎：基于语义重要度与DNA追溯的智能压缩
    
    核心机制：
    1. 内容评分系统（0-100分）
    2. 语义聚类合并
    3. DNA追溯码生成
    4. 压缩前后对比
    5. 用户控制接口
    """
    
    def __init__(self, config: Dict):
        self.config = config
        self.p0_patterns = self._load_p0_patterns()
        self.semantic_model = self._load_semantic_model()
        
    def _load_p0_patterns(self) -> List[str]:
        """
        加载P0永恒级内容特征
        这些内容永远不会被压缩
        """
        return [
            "P0-ETERNAL",
            "温度契约",
            "#ZHUGEXIN",
            "Lucky的原话：",
            "老大说：",
            '"',  # 所有引号内容
            "永恒锚",
            "GPG签名",
        ]
    
    def score_content(self, content: str, context: Dict) -> int:
        """
        内容评分算法
        
        评分维度：
        1. P0特征匹配（+100）
        2. 语义重要度（0-90）
        3. 时效性（-0~30）
        4. 冗余度（-0~50）
        """
        score = 0
        
        # 维度1：P0特征
        for pattern in self.p0_patterns:
            if pattern in content:
                return PriorityLevel.P0_ETERNAL.value
        
        # 维度2：语义重要度
        semantic_score = self._semantic_importance(content, context)
        score += semantic_score
        
        # 维度3：时效性
        recency_score = self._calculate_recency(content, context)
        score += recency_score
        
        # 维度4：冗余度
        redundancy_penalty = self._detect_redundancy(content, context)
        score -= redundancy_penalty
        
        return max(0, min(100, score))
    
    def _semantic_importance(self, content: str, context: Dict) -> int:
        """
        语义重要度计算（三方专属算法）
        
        核心技术：
        1. 基于embedding的语义相似度
        2. 命名实体识别（NER）
        3. 情感强度分析
        4. 指令性检测
        """
        # 实现细节（DeepSeek负责实现）
        importance = 0
        
        # 检测命名实体
        entities = self._extract_entities(content)
        importance += len(entities) * 5
        
        # 检测情感强度
        emotion_intensity = self._analyze_emotion(content)
        importance += emotion_intensity * 10
        
        # 检测指令性
        if self._is_directive(content):
            importance += 20
        
        # 检测核心概念
        core_concepts = self._extract_concepts(content)
        importance += len(core_concepts) * 8
        
        return min(90, importance)
    
    def _calculate_recency(self, content: str, context: Dict) -> int:
        """
        时效性评分
        
        最近的对话 = 更高分
        历史对话 = 分数衰减
        """
        age_minutes = context.get('age_minutes', 0)
        
        if age_minutes < 60:  # 1小时内
            return 30
        elif age_minutes < 1440:  # 24小时内
            return 20
        elif age_minutes < 10080:  # 7天内
            return 10
        else:
            return 0
    
    def _detect_redundancy(self, content: str, context: Dict) -> int:
        """
        冗余度检测
        
        如果内容与历史内容高度重复 = 高冗余度 = 扣分
        """
        history = context.get('history', [])
        max_similarity = 0
        
        for hist_content in history:
            similarity = self._cosine_similarity(
                self._embed(content),
                self._embed(hist_content)
            )
            max_similarity = max(max_similarity, similarity)
        
        # 相似度越高，惩罚越大
        if max_similarity > 0.9:
            return 50  # 几乎完全重复
        elif max_similarity > 0.7:
            return 30  # 高度相似
        elif max_similarity > 0.5:
            return 15  # 中度相似
        else:
            return 0   # 不相似
    
    def compress(self, conversation: List[Dict]) -> Tuple[str, Dict]:
        """
        压缩主函数
        
        输入：完整对话历史
        输出：压缩后内容 + DNA追溯报告
        """
        # 步骤1：评分所有内容
        scored_items = []
        for item in conversation:
            score = self.score_content(item['content'], item.get('context', {}))
            scored_items.append({
                'original': item,
                'score': score,
                'priority': self._score_to_priority(score)
            })
        
        # 步骤2：分优先级处理
        p0_items = [x for x in scored_items if x['priority'] == 'P0_ETERNAL']
        p1_items = [x for x in scored_items if x['priority'] == 'P1_CORE']
        p2_items = [x for x in scored_items if x['priority'] == 'P2_DIRECTIVE']
        p3_items = [x for x in scored_items if x['priority'] == 'P3_TECHNICAL']
        p4_items = [x for x in scored_items if x['priority'] == 'P4_CONTEXT']
        p5_items = [x for x in scored_items if x['priority'] == 'P5_REDUNDANT']
        
        # 步骤3：构建压缩结果
        compressed = {
            'p0_full': [x['original'] for x in p0_items],  # 完整保留
            'p1_full': [x['original'] for x in p1_items],  # 完整保留
            'p2_full': [x['original'] for x in p2_items],  # 完整保留
            'p3_summary': self._summarize_cluster(p3_items),  # 摘要
            'p4_summary': self._summarize_cluster(p4_items),  # 摘要
            'p5_removed': len(p5_items),  # 删除记录数量
        }
        
        # 步骤4：生成DNA追溯报告
        dna_report = self._generate_dna_report(
            original=conversation,
            compressed=compressed,
            scored_items=scored_items
        )
        
        return compressed, dna_report
    
    def _summarize_cluster(self, items: List[Dict]) -> str:
        """
        语义聚类摘要
        
        将相似内容聚类，生成代表性摘要
        """
        if not items:
            return ""
        
        # 语义聚类
        clusters = self._semantic_clustering([x['original']['content'] for x in items])
        
        # 每个聚类生成摘要
        summaries = []
        for cluster in clusters:
            representative = self._select_representative(cluster)
            summary = self._generate_summary(cluster)
            summaries.append({
                'representative': representative,
                'summary': summary,
                'count': len(cluster)
            })
        
        return summaries
    
    def _generate_dna_report(self, original, compressed, scored_items) -> Dict:
        """
        生成DNA追溯报告
        
        包含：
        - DNA追溯码
        - 压缩前后对比
        - 保留/删除清单
        - GPG签名
        """
        import datetime
        
        timestamp = datetime.datetime.now().strftime('%Y%m%d%H%M%S')
        dna_code = f"#ZHUGEXIN⚡️2026-01-12-COMPRESSION-{timestamp}"
        
        # 计算token变化
        original_tokens = sum(len(x['content'].split()) for x in original)
        compressed_tokens = self._estimate_compressed_tokens(compressed)
        compression_ratio = 1 - (compressed_tokens / original_tokens)
        
        report = {
            'dna_code': dna_code,
            'timestamp': timestamp,
            'original_tokens': original_tokens,
            'compressed_tokens': compressed_tokens,
            'compression_ratio': f"{compression_ratio:.2%}",
            'preserved': {
                'p0_count': len(compressed['p0_full']),
                'p1_count': len(compressed['p1_full']),
                'p2_count': len(compressed['p2_full']),
            },
            'summarized': {
                'p3_clusters': len(compressed.get('p3_summary', [])),
                'p4_clusters': len(compressed.get('p4_summary', [])),
            },
            'removed': compressed['p5_removed'],
            'gpg_signature': self._generate_gpg_signature(compressed),
        }
        
        return report
```

```python
# 压缩结果防篡改机制

import hashlib
import hmac

class CompressionVerifier:
    """
    压缩结果验证器
    
    确保：
    1. 压缩过程未被篡改
    2. P0内容完整保留
    3. DNA追溯码唯一
    4. GPG签名有效
    """
    
    SECRET_KEY = "ZHUGEXIN-9622-DRAGON-SOUL"  # 三方共享密钥
    
    @staticmethod
    def generate_checksum(compression_result: Dict) -> str:
        """
        生成压缩结果校验和
        """
        # 提取关键字段
        key_fields = [
            compression_result['dna_code'],
            compression_result['timestamp'],
            str(compression_result['original_tokens']),
            str(compression_result['compressed_tokens']),
        ]
        
        # 生成HMAC
        message = '|'.join(key_fields).encode('utf-8')
        checksum = hmac.new(
            CompressionVerifier.SECRET_KEY.encode('utf-8'),
            message,
            hashlib.sha256
        ).hexdigest()
        
        return checksum
    
    @staticmethod
    def verify_checksum(compression_result: Dict, checksum: str) -> bool:
        """
        验证校验和
        """
        expected = CompressionVerifier.generate_checksum(compression_result)
        return hmac.compare_digest(expected, checksum)
    
    @staticmethod
    def verify_p0_integrity(original: List, compressed: Dict) -> bool:
        """
        验证P0内容完整性
        """
        # 提取原始对话中的P0内容
        original_p0 = [x for x in original if is_p0_content(x)]
        
        # 提取压缩结果中的P0内容
        compressed_p0 = compressed['preserved']['p0']
        
        # 比较
        return len(original_p0) == len(compressed_p0)
```
