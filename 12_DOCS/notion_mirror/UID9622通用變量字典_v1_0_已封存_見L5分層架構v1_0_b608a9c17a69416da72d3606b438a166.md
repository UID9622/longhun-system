## 🔧 工具函数
### 变量类型检测
```python
def detect_variable_type(var_name: str) -> str:
    """
    检测变量类型
    """
    # L0真实：标准编程变量命名
    if var_name.islower() and '_' in var_name:
        return "L0真实"
    
    # L1示意：全大写或中文
    if var_name.isupper() or any('\u4e00' <= c <= '\u9fff' for c in var_name):
        return "L1示意"
    
    # L2比喻：包含特殊符号或emoji
    if any(c in var_name for c in ['🎯', '🐱', '⚙️', '👁️', '💙']):
        return "L2比喻"
    
    return "未知"
```
### 变量净化
```python
def purify_variable(var_name: str, var_type: str) -> str:
    """
    净化变量，转换为真实可执行的形式
    """
    if var_type in ["L1示意", "L2比喻"]:
        # 查映射表
        if var_name in CONCEPT_TO_REAL:
            return CONCEPT_TO_REAL[var_name]
        else:
            return f"# [需要翻译] {var_name}"
    return var_name
```
