def semantic_route(user_text: str) -> dict:
    # L1 主键字硬匹配（瞬响）
    if any(k in user_text for k in ["贡献分", "功劳", "元老"]):
        return {"hit": "contribution_matrix", "confidence": 1.0}
    # L2 同义词/口语变体（毫秒级）
    for concept, synonyms in SYNONYMS.items():
        if any(s in user_text for s in synonyms):
            return {"hit": concept, "confidence": 0.8}
    # L3 语义向量（语义总结·约200ms）
    emb = embed(user_text)  # sentence-transformers/multilingual-MiniLM
    top = cosine_topk(emb, CONCEPT_VECTORS, k=1)
    if top.score > 0.65:
        return {"hit": top.concept, "confidence": round(top.score, 2)}
    # 兵关驿：不端不硬问老大
    return {"hit": "UNKNOWN", "ask": "您说的是指容错·报酬·还是举报？"}
