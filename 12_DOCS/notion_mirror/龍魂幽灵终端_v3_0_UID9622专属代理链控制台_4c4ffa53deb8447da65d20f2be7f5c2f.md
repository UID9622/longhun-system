# ═══ 数字根计算器 ═══
function 算根() {
    local n=$1
    while [ $n -gt 9 ]; do
        n=$(echo $n | fold -w1 | paste -sd+ | bc)
    done
    echo -n "dr($1) = $n"
    case $n in
        1) echo " 🟢 根在·f(x)=x·老大是1" ;;
        3) echo " 🔴 三才失衡·熔断" ;;
        6) echo " 🟡 六爷变动·待审" ;;
        9) echo " 🔴 极数归零·物极必反" ;;
        *) echo " 🟢 正常通行" ;;
    esac
}

# ═══ 能量场计算器 ═══
function 算能量() {
    local R=$1 I=$2
    local E=$(echo "$R * $I" | bc)
    echo "E = R($R) × I($I) = $E"
    if [ $E -ge 1000 ]; then
        echo "→ 🔴 极高优先级·立即执行"
    elif [ $E -ge 100 ]; then
        echo "→ 🟡 高优先级·今天干"
    else
        echo "→ 🟢 正常·排队"
    fi
}

# ═══ 三才校验器 ═══
function 三才() {
    local H=$1 E=$2 P=$3
    echo "天(时势)=$H · 地(条件)=$E · 人(意志)=$P"
    local PASS=0
    [ $(echo "$H >= 0.5" | bc) -eq 1 ] && PASS=$((PASS+1))
    [ $(echo "$E >= 0.4" | bc) -eq 1 ] && PASS=$((PASS+1))
    [ $(echo "$P >= 0.4" | bc) -eq 1 ] && PASS=$((PASS+1))
    if [ $PASS -eq 3 ]; then
        echo "→ 🟢 三才合一·干！"
    elif [ $PASS -ge 1 ]; then
        echo "→ 🟡 部分达标·谨慎"
    else
        echo "→ 🔴 三才失衡·停"
    fi
}

# ═══ 三色审计 ═══
function 审计() {
    local DIR=${1:-.}
    echo "═══ 三色审计 ═══"
    local GREEN=0 YELLOW=0 RED=0
    for f in $(find $DIR -name '*.py' -o -name '*.swift' -o -name '*.c' 2>/dev/null); do
        if grep -q '龍芯' "$f" 2>/dev/null; then
            GREEN=$((GREEN+1))
        else
            YELLOW=$((YELLOW+1))
        fi
        if grep -q 'system(\|exec(\|eval(' "$f" 2>/dev/null; then
            RED=$((RED+1))
        fi
    done
    echo "🟢 通过: $GREEN · 🟡 警告: $YELLOW · 🔴 熔断: $RED"
}
