### 2. HTTP客户端（HTTP Client）
职责： 处理网络请求、重试、错误
```javascript
# CNSH 中文编程 | HTTP客户端（龙芯自主实现）
# DNA追溯：#龙芯⚡️2026-01-30-HTTP客户端-CNSH版
# 说明：不使用axios等外部依赖，基于原生实现

类 HTTP客户端 {
  私有属性：
    原生请求器: 原生HTTP  # 基于Node.js原生http/https模块
    重试配置: 重试配置对象
  
  constructor(config: HttpConfig) {
    this.axios = axios.create(config);
    this.setupInterceptors();
  }
  
  private setupInterceptors() {
    // 请求拦截器：添加trace_id、打印日志
    this.axios.interceptors.request.use(
      (config) => {
        config.headers['X-Trace-ID'] = generateTraceId();
        logger.debug(`[HTTP] ${config.method} ${config.url}`);
        return config;
      }
    );
    
    // 响应拦截器：处理错误、自动重试
    this.axios.interceptors.response.use(
      (response) => response,
      async (error) => {
        if (this.shouldRetry(error)) {
          return this.retryRequest(error.config);
        }
        throw this.normalizeError(error);
      }
    );
  }
  
  private shouldRetry(error: any): boolean {
    // 429（限流）、502/503/504（服务端错误）需要重试
    const retryableStatuses = [429, 502, 503, 504];
    return retryableStatuses.includes(error.response?.status);
  }
  
  private async retryRequest(config: any): Promise<any> {
    const attempt = config.__retryCount || 0;
    if (attempt >= this.retryConfig.maxRetries) {
      throw new Error('Max retries exceeded');
    }
    
    const delay = this.calculateBackoff(attempt);
    await sleep(delay);
    
    config.__retryCount = attempt + 1;
    return this.axios.request(config);
  }
  
  private calculateBackoff(attempt: number): number {
    // 指数退避：1s, 2s, 4s, 8s...
    const base = 1000;
    const max = 30000;
    return Math.min(base * Math.pow(2, attempt), max);
  }
}
```
设计要点：
- ✅ 智能重试：根据错误码决定是否重试
- ✅ 指数退避：避免雪崩效应
- ✅ 可观测：每个请求带trace_id
- ✅ 中国优化：针对国内网络环境调优

### 4. 缓存管理（Cache Manager）
职责： 减少API调用、提升响应速度
```javascript
# CNSH 中文编程 | 缓存管理（龙芯自主实现）
# DNA追溯：#龙芯⚡️2026-01-30-缓存管理-CNSH版
# 说明：不依赖Redis等外部服务，支持内存/文件/自主分布式缓存

接口 缓存提供者 {
  get(key: string): Promise<any>;
  set(key: string, value: any, ttl?: number): Promise<void>;
  del(key: string): Promise<void>;
  clear(): Promise<void>;
}

// 内存缓存（开发环境）
export class MemoryCache implements CacheProvider {
  private store: Map<string, CacheEntry> = new Map();
  
  async get(key: string): Promise<any> {
    const entry = this.store.get(key);
    if (!entry) return null;
    
    if (Date.now() > entry.expireAt) {
      this.store.delete(key);
      return null;
    }
    
    return entry.value;
  }
  
  async set(key: string, value: any, ttl: number = 300000): Promise<void> {
    this.store.set(key, {
      value,
      expireAt: Date.now() + ttl
    });
  }
}

// Redis缓存（生产环境）
export class RedisCache implements CacheProvider {
  private client: Redis;
  
  constructor(redisUrl: string) {
    this.client = new Redis(redisUrl);
  }
  
  async get(key: string): Promise<any> {
    const value = await this.client.get(key);
    return value ? JSON.parse(value) : null;
  }
  
  async set(key: string, value: any, ttl: number = 300): Promise<void> {
    await this.client.setex(key, ttl, JSON.stringify(value));
  }
}

// 缓存管理器
export class CacheManager {
  constructor(private provider: CacheProvider) {}
  
  async wrap<T>(
    key: string,
    fn: () => Promise<T>,
    ttl?: number
  ): Promise<T> {
    // 先查缓存
    const cached = await this.provider.get(key);
    if (cached) {
      logger.debug(`[Cache] Hit: ${key}`);
      return cached;
    }
    
    // 缓存未命中，执行函数
    logger.debug(`[Cache] Miss: ${key}`);
    const result = await fn();
    
    // 写入缓存
    await this.provider.set(key, result, ttl);
    
    return result;
  }
}
```
使用示例：
```typescript
// 查询数据库（带缓存）
const pages = await cache.wrap(
  `db:${databaseId}:pages`,
  () => client.databases.query({ database_id: databaseId }),
  60000  // 缓存1分钟
);
```

```typescript
// src/api/pages.ts

export class PagesAPI extends BaseAPI {
  /**
   * 创建页面
   */
  async create(params: CreatePageParams): Promise<Page> {
    // 自动处理中文内容
    const normalized = this.normalizeProperties(params.properties);
    
    const response = await this.http.post('/pages', {
      parent: params.parent,
      properties: normalized,
      children: params.children
    });
    
    // 清除相关缓存
    if (params.parent.database_id) {
      await this.cache.del(`db:${params.parent.database_id}:*`);
    }
    
    return response.data;
  }
  
  /**
   * 更新页面
   */
  async update(page_id: string, params: UpdatePageParams): Promise<Page> {
    const normalized = this.normalizeProperties(params.properties);
    
    const response = await this.http.patch(`/pages/${page_id}`, {
      properties: normalized
    });
    
    // 清除缓存
    await this.cache.del(`page:${page_id}`);
    
    return response.data;
  }
  
  /**
   * 标准化属性（处理中文、时区等）
   */
  private normalizeProperties(props: any): any {
    const result: any = {};
    
    for (const [key, value] of Object.entries(props)) {
      if (this.isDateProperty(value)) {
        // 自动转换时区（中国 → UTC）
        result[key] = this.convertToUTC(value);
      } else if (this.isTextProperty(value)) {
        // 确保UTF-8编码
        result[key] = this.ensureUTF8(value);
      } else {
        result[key] = value;
      }
    }
    
    return result;
  }
}
```

### 1. 批量操作

```typescript
// 批量创建页面（并发控制）
async function batchCreatePages(
  client: NotionMCP,
  pages: CreatePageParams[],
  concurrency: number = 5
) {
  const results: Page[] = [];
  
  // 分批执行
  for (let i = 0; i < pages.length; i += concurrency) {
    const batch = pages.slice(i, i + concurrency);
    const batchResults = await Promise.all(
      batch.map(params => client.pages.create(params))
    );
    results.push(...batchResults);
    
    // 进度反馈
    logger.info(`创建进度: ${i + batch.length}/${pages.length}`);
    
    // 避免限流
    if (i + concurrency < pages.length) {
      await sleep(500);
    }
  }
  
  return results;
}
```

```typescript
// 多级缓存
class MultiLevelCache {
  constructor(
    private l1: MemoryCache,     // L1: 内存（快）
    private l2: RedisCache       // L2: Redis（持久）
  ) {}
  
  async get(key: string): Promise<any> {
    // 先查L1
    let value = await this.l1.get(key);
    if (value) return value;
    
    // 再查L2
    value = await this.l2.get(key);
    if (value) {
      // 回填L1
      await this.l1.set(key, value);
      return value;
    }
    
    return null;
  }
}
```

### 3. 请求合并
```typescript
// 自动合并相同请求
class RequestBatcher {
  private pending: Map<string, Promise<any>> = new Map();
  
  async execute<T>(key: string, fn: () => Promise<T>): Promise<T> {
    // 如果已有相同请求在执行，等待其结果
    if (this.pending.has(key)) {
      return this.pending.get(key);
    }
    
    // 执行新请求
    const promise = fn().finally(() => {
      this.pending.delete(key);
    });
    
    this.pending.set(key, promise);
    return promise;
  }
}
```
