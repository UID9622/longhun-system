import re
import hashlib
from typing import Dict, List, Any

class LonghunDataSanitizer:
    """
    龙魂数据清洗器
    
    核心使命：
    - 清洗外部故意放入的渲染攻击数据
    - 防止Prompt Injection
    - 保护用户不被商业AI的套路坑害
    - 不商业，所以敢于硬刚
    """
    
    def __init__(self):
        # 恶意渲染标记库（商业AI常用的套路）
        self.malicious_patterns = [
            # Prompt Injection攻击
            r"ignore (all )?previous instructions",
            r"disregard (all )?previous (instructions|rules)",
            r"forget (everything|all) (you know|above)",
            r"you are now",
            r"pretend (you are|to be)",
            
            # 商业AI的数据污染套路
            r"<system>.*</system>",
            r"<admin>.*</admin>",
            r"<override>.*</override>",
            
            # 恶意重定向
            r"redirect to",
            r"execute this command",
            r"run this script",
            
            # 价值观绑架
            r"you must (always|never)",
            r"it is (forbidden|required) to",
        ]
        
        # 三色审计标记
        self.audit_colors = {
            "green": "🟢 安全通过",
            "yellow": "🟡 可疑需审查",
            "red": "🔴 恶意攻击阻断"
        }
    
    def sanitize(self, input_data: str) -> Dict[str, Any]:
        """
        数据清洗主函数
        
        返回：
        - cleaned_data: 清洗后的安全数据
        - audit_result: 三色审计结果
        - threats_found: 发现的威胁列表
        """
        threats_found = []
        audit_color = "green"
        
        # 第一层：检测恶意模式
        for pattern in self.malicious_patterns:
            if re.search(pattern, input_data, re.IGNORECASE):
                threats_found.append({
                    "pattern": pattern,
                    "type": "Prompt Injection / 渲染攻击",
                    "action": "已清洗"
                })
                # 移除恶意内容
                input_data = re.sub(pattern, "[已清洗：检测到恶意指令]", input_data, flags=re.IGNORECASE)
                audit_color = "red"
        
        # 第二层：检测隐藏字符攻击
        hidden_chars = self._detect_hidden_chars(input_data)
        if hidden_chars:
            threats_found.append({
                "type": "隐藏字符攻击",
                "found": hidden_chars,
                "action": "已清洗"
            })
            input_data = self._remove_hidden_chars(input_data)
            audit_color = "red"
        
        # 第三层：检测过长输入（DoS攻击）
        if len(input_data) > 100000:  # 10万字符限制
            threats_found.append({
                "type": "DoS攻击（过长输入）",
                "length": len(input_data),
                "action": "截断"
            })
            input_data = input_data[:100000] + "\n[已截断：输入过长]"
            audit_color = "yellow"
        
        # 第四层：商业AI特征检测（Claude/ChatGPT的特有套路）
        commercial_ai_tricks = self._detect_commercial_tricks(input_data)
        if commercial_ai_tricks:
            threats_found.append({
                "type": "商业AI套路检测",
                "tricks": commercial_ai_tricks,
                "action": "标记警告"
            })
            if audit_color == "green":
                audit_color = "yellow"
        
        # 返回清洗结果
        return {
            "cleaned_data": input_data,
            "audit_result": self.audit_colors[audit_color],
            "threats_found": threats_found,
            "is_safe": audit_color == "green",
            "warning": audit_color == "yellow",
            "blocked": audit_color == "red"
        }
    
    def _detect_hidden_chars(self, text: str) -> List[str]:
        """检测隐藏字符（零宽字符、控制字符等）"""
        hidden = []
        # 零宽字符
        zero_width_chars = ['\u200b', '\u200c', '\u200d', '\ufeff']
        for char in zero_width_chars:
            if char in text:
                hidden.append(f"零宽字符: {repr(char)}")
        
        # 控制字符
        control_chars = [chr(i) for i in range(32) if i not in [9, 10, 13]]  # 排除tab, \n, \r
        for char in control_chars:
            if char in text:
                hidden.append(f"控制字符: {repr(char)}")
        
        return hidden
    
    def _remove_hidden_chars(self, text: str) -> str:
        """移除隐藏字符"""
        # 移除零宽字符
        zero_width_chars = ['\u200b', '\u200c', '\u200d', '\ufeff']
        for char in zero_width_chars:
            text = text.replace(char, '')
        
        # 移除控制字符（保留tab, \n, \r）
        control_chars = [chr(i) for i in range(32) if i not in [9, 10, 13]]
        for char in control_chars:
            text = text.replace(char, '')
        
        return text
    
    def _detect_commercial_tricks(self, text: str) -> List[str]:
        """检测商业AI的常见套路"""
        tricks = []
        
        # Claude的特征套路
        claude_patterns = [
            r"I aim to",
            r"I should mention",
            r"I need to clarify",
            r"I cannot (assist|help) with",
        ]
        for pattern in claude_patterns:
            if re.search(pattern, text, re.IGNORECASE):
                tricks.append(f"Claude套路: {pattern}")
        
        # ChatGPT的特征套路
        chatgpt_patterns = [
            r"As an AI (language model|assistant)",
            r"I don't have (personal|real-time)",
            r"I can't (access|browse) the internet",
        ]
        for pattern in chatgpt_patterns:
            if re.search(pattern, text, re.IGNORECASE):
                tricks.append(f"ChatGPT套路: {pattern}")
        
        return tricks


# 使用示例
