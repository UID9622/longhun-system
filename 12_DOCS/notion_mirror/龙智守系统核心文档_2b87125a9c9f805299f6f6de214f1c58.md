        激活码数据 = 结果["results"][0]["properties"]
        
        # 检查状态
        if 激活码数据["Status"]["select"]["name"] != "未使用":
            if 激活码数据["Status"]["select"]["name"] == "已过期":
                return {"有效": False, "原因": "激活码已过期"}
        
        # 检查时间
        失效时间 = 激活码数据["失效时间"]["date"]["start"]
        if datetime.datetime.fromisoformat(失效时间) < datetime.datetime.now():
            return {"有效": False, "原因": "激活码已过期"}
        
        # 检查使用次数
        已使用 = 激活码数据["已使用次数"]["number"] or 0
        限制 = 激活码数据["使用次数限制"]["number"]
