```python
import hashlib

def generate_dna_hash(content, timestamp, creator_id):
    # 将所有元数据拼接
    raw_data = f"{content}|{timestamp}|{creator_id}"
    # SHA-256哈希
    hash_obj = hashlib.sha256(raw_data.encode('utf-8'))
    # 返回前8位
    return hash_obj.hexdigest()[:8].upper()
```
