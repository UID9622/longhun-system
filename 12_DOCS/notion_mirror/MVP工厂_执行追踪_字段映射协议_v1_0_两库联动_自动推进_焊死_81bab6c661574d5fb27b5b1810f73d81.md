## §5 自动化执行流（伪代码·给 Cursor / Agent）
```javascript
// MVP工厂→追踪·自动推进伪代码
// 触发：工厂「状态」字段变为「已完成」或「已熔断」

function autoPushToTracker(factoryRow) {
  // ① 只有两种状态触发
  if (factoryRow.status !== '已完成' && factoryRow.status !== '已熔断') {
    return { skipped: true, reason: '状态非已完成/已熔断' }
  }

  // ② 构建追踪行数据（按§2 12字段映射）
  const trackerRow = {
    '名称': factoryRow['协议名称'],
    '快照ID': factoryRow['协议ID'],
    'DNA 签名': factoryRow['DNA标识'],
    '协议输入': factoryRow['输入数据']?.substring(0, 2000),
    '协议输出': factoryRow['协议内容'],
    '执行状态': factoryRow['状态'] === '已熔断' ? '熔断' : '未开始',
    '风险等级': mapRiskScore(factoryRow['风险评分']),
    'Tier等级': mapTier(factoryRow['Tier等级']),
    '审计链ID': factoryRow['审计链ID'],
    '回滚点': factoryRow['回滚点'],
    '执行者': mapExecutor(factoryRow['路由节点']),
    '执行时间': new Date().toISOString()
  }

  // ③ 三色判定补充覆盖执行状态
  if (factoryRow['三色判定'] === '🟡 待确认') {
    trackerRow['执行状态'] = '待确认'
  }
  if (factoryRow['三色判定'] === '🔴 熔断') {
    trackerRow['执行状态'] = '熔断'
  }

  // ④ 在追踪库创建新行
  return createPage(trackerDataSource, trackerRow)
}

// 风险评分映射
function mapRiskScore(score) {
  if (score >= 10) return '3-永久熔断'
  if (score >= 7) return '2-高风险'
  if (score >= 4) return '1-中风险'
  return '0-低风险'
}

// Tier映射
function mapTier(tier) {
  return tier?.replace('Tier1-DNA认证', 'TIER_1')
              ?.replace('Tier2-实名认证', 'TIER_2')
              ?.replace('Tier3-拒入', 'TIER_3') || tier
}

// 执行者映射
function mapExecutor(routes) {
  if (!routes || routes.length === 0) return 'Claude'
  const first = routes[0]
  if (first === 'ChatGPT') return 'ChatGPT'
  if (first === 'Claude') return 'Claude'
  return 'Local Runtime'
}
```
