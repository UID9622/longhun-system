#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
龍魂DNA库管理系统
DNA追溯码: #ZHUGEXIN⚡️2026-01-04-DNA-DATABASE-SYSTEM-v1.0
作者: UID9622 | Lucky
数据主权: 中华人民共和国公民
"""

import json
import hashlib
import os
from datetime import datetime
from typing import Dict, List, Optional

class DNADatabase:
    """龍魂DNA库管理系统"""
    
    def __init__(self, db_path="dna_database.json"):
        self.db_path = db_path
        self.database = self._load_database()
    
    def _load_database(self) -> Dict:
        """加载DNA库"""
        if os.path.exists(self.db_path):
            with open(self.db_path, 'r', encoding='utf-8') as f:
                return json.load(f)
        return {
            "metadata": {
                "creator": "UID9622",
                "creator_name": "诸葛鑫 (Lucky)",
                "created_at": datetime.now().isoformat(),
                "version": "1.0",
                "total_dna": 0
            },
            "dna_entries": []
        }
    
    def _save_database(self):
        """保存DNA库"""
        self.database["metadata"]["total_dna"] = len(self.database["dna_entries"])
        self.database["metadata"]["last_updated"] = datetime.now().isoformat()
        
        with open(self.db_path, 'w', encoding='utf-8') as f:
            json.dump(self.database, f, ensure_ascii=False, indent=2)
        
        print(f"✅ DNA库已保存: {self.db_path}")
    
    def add_dna(self,
                dna_code: str,
                title: str,
                doc_type: str,
                file_path: str = "",
                description: str = "",
                category: str = "技术类",
                status: str = "活跃"):
        """
        添加DNA记录
        
        参数:
        - dna_code: DNA追溯码
        - title: 文档标题
        - doc_type: 文档类型（文章/代码/系统/方案等）
        - file_path: 文件路径
        - description: 描述
        - category: 分类（技术类/文档类/理念类/资产类）
        - status: 状态（活跃/归档/冻结）
        """
        
        # 生成文件哈希（如果文件存在）
        file_hash = ""
        if file_path and os.path.exists(file_path):
            with open(file_path, 'rb') as f:
                file_hash = hashlib.sha256(f.read()).hexdigest()
        
        entry = {
            "dna_code": dna_code,
            "title": title,
            "doc_type": doc_type,
            "file_path": file_path,
            "file_hash": file_hash,
            "description": description,
            "category": category,
            "status": status,
            "created_at": datetime.now().isoformat(),
            "creator": "UID9622",
            "creator_name": "诸葛鑫 (Lucky)"
        }
        
        self.database["dna_entries"].append(entry)
        self._save_database()
        
        print(f"✅ DNA记录已添加: {dna_code}")
        return entry
    
    def search_dna(self, keyword: str) -> List[Dict]:
        """搜索DNA记录"""
        results = []
        keyword_lower = keyword.lower()
        
        for entry in self.database["dna_entries"]:
            if (keyword_lower in entry["dna_code"].lower() or
                keyword_lower in entry["title"].lower() or
                keyword_lower in entry["description"].lower()):
                results.append(entry)
        
        return results
    
    def verify_dna(self, dna_code: str) -> Optional[Dict]:
        """验证DNA码"""
        for entry in self.database["dna_entries"]:
            if entry["dna_code"] == dna_code:
                return entry
        return None
    
    def get_by_category(self, category: str) -> List[Dict]:
        """按分类获取DNA记录"""
        return [e for e in self.database["dna_entries"] if e["category"] == category]
    
    def get_statistics(self) -> Dict:
        """获取统计信息"""
        total = len(self.database["dna_entries"])
        
        by_category = {}
        by_status = {}
        by_type = {}
        
        for entry in self.database["dna_entries"]:
            # 按分类统计
            cat = entry["category"]
            by_category[cat] = by_category.get(cat, 0) + 1
            
            # 按状态统计
            status = entry["status"]
            by_status[status] = by_status.get(status, 0) + 1
            
            # 按类型统计
            doc_type = entry["doc_type"]
            by_type[doc_type] = by_type.get(doc_type, 0) + 1
        
        return {
            "total": total,
            "by_category": by_category,
            "by_status": by_status,
            "by_type": by_type
        }
    
    def generate_certificate(self, dna_code: str) -> str:
        """生成DNA证明文件"""
        entry = self.verify_dna(dna_code)
        if not entry:
            return "❌ DNA码不存在"
        
        certificate = f"""
{'='*60}
龍魂系统 - DNA原创证明
{'='*60}

DNA追溯码: {entry['dna_code']}
文档标题: {entry['title']}
文档类型: {entry['doc_type']}
分类: {entry['category']}
状态: {entry['status']}

创建时间: {entry['created_at']}
创建者: {entry['creator_name']} (UID{entry['creator']})

文件路径: {entry['file_path']}
文件哈希: {entry['file_hash']}

描述:
{entry['description']}

数据主权: 中华人民共和国公民
认证机构: 龍魂数字身份系统 v1.0

{'='*60}
此证明由龍魂DNA库系统自动生成
生成时间: {datetime.now().isoformat()}
查询验证: https://github.com/uid9622/cnsh
{'='*60}
"""
        return certificate
    
    def export_to_markdown(self, output_path="DNA库总表.md"):
        """导出为Markdown表格"""
        stats = self.get_statistics()
        
        md_content = f"""# 龍魂系统 - DNA库总表

## 🔐 创作者数字身份认证

**身份指纹**: `b83c74d108660082581f9ebbb9506f65849d9d48d21d328daf13f7c4d66cf6c1`

**主权声明**: 所有DNA记录均为UID9622原创作品，数据主权归中华人民共和国公民所有。

**DNA库系统**: #ZHUGEXIN⚡️2026-01-04-DNA-DATABASE-SYSTEM-v1.0

---

## 📊 统计信息

- **DNA总数**: {stats['total']}
- **创建者**: UID9622 | 诸葛鑫 (Lucky)
- **最后更新**: {self.database['metadata']['last_updated']}

### 分类统计

"""
        for cat, count in stats['by_category'].items():
            md_content += f"- {cat}: {count}\n"
        
        md_content += "\n### 状态统计\n\n"
        for status, count in stats['by_status'].items():
            md_content += f"- {status}: {count}\n"
        
        md_content += "\n---\n\n## 📋 DNA记录总表\n\n"
        md_content += "| DNA追溯码 | 标题 | 类型 | 分类 | 状态 | 创建时间 |\n"
        md_content += "|----------|------|------|------|------|----------|\n"
        
        for entry in self.database["dna_entries"]:
            created_date = entry['created_at'].split('T')[0]
            md_content += f"| `{entry['dna_code']}` | {entry['title']} | {entry['doc_type']} | {entry['category']} | {entry['status']} | {created_date} |\n"
        
        md_content += f"""
---

## 🔍 DNA验证

访问 https://github.com/uid9622/cnsh 验证任何DNA码的真实性。

## 📧 联系方式

- 邮箱: uid9622@petalmail.com
- 项目: UID9622龍魂系统

**生成时间**: {datetime.now().isoformat()}

**不可篡改**: ✅ 已锁定
"""
        
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write(md_content)
        
        print(f"✅ DNA库已导出: {output_path}")
        return md_content
