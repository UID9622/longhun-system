## 📦 核心工具：netfix.py（完整版）
### 完整脚本
保存为 ~/netfix.py，然后 sudo python3 ~/netfix.py 即可：
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Lucky网络诊断修复终极脚本 v3.0"""
import os, sys, platform, subprocess, socket, shutil, time
from datetime import datetime
from pathlib import Path

class LuckyNetFixer:
    def __init__(self):
        self.system = platform.system()
        self.issues = []
        self.fixes = []
        self.backup_dir = Path.home() / ".netfix_backups"
        self.backup_dir.mkdir(exist_ok=True)
        self.report_data = []
        
    def run_cmd(self, cmd, silent=True):
        try:
            result = subprocess.run(cmd, shell=True, capture_output=True, 
                                   text=True, timeout=30, encoding='utf-8', errors='ignore')
            return result.stdout + result.stderr, result.returncode
        except Exception as e:
            return f"Error: {e}", 1
    
    def backup_config(self, config_file, description):
        if os.path.exists(config_file):
            ts = datetime.now().strftime("%Y%m%d_%H%M%S")
            backup_file = self.backup_dir / f"{Path(config_file).name}.{ts}.bak"
            shutil.copy2(config_file, backup_file)
            print(f"  💾 已备份: {description}")
            return backup_file
        return None
    
    def check_and_fix_dns(self):
        print("🔍 检测DNS...", end=" ")
        try:
            socket.gethostbyname("baidu.com")
            print("✅ 正常")
            self.report_data.append(("DNS解析", "✅ 正常", "无需修复"))
            return True
        except:
            print("⚠️ 失败，修复中...")
            self.issues.append("DNS解析失败")
            
            if self.system == "Darwin":
                self.run_cmd("networksetup -setdnsservers Wi-Fi 8.8.8.8 1.1.1.1")
                self.run_cmd("networksetup -setdnsservers Ethernet 8.8.8.8 1.1.1.1")
            elif self.system == "Linux":
                self.backup_config("/etc/resolv.conf", "DNS配置")
                self.run_cmd('echo -e "nameserver 8.8.8.8\\nnameserver 1.1.1.1" | sudo tee /etc/resolv.conf')
            else:
                self.run_cmd('netsh interface ip set dns "以太网" static 8.8.8.8')
                self.run_cmd('netsh interface ip add dns "以太网" 1.1.1.1 index=2')
            
            time.sleep(1)
            try:
                socket.gethostbyname("baidu.com")
                print("  ✅ 修复成功")
                self.fixes.append("DNS → 8.8.8.8 / 1.1.1.1")
                self.report_data.append(("DNS解析", "⚠️ 失败", "✅ 已修复为8.8.8.8"))
                return True
            except:
                print("  ❌ 修复失败")
                self.report_data.append(("DNS解析", "❌ 失败", "❌ 修复失败"))
                return False
    
    def check_and_fix_mtu(self):
        print("🔍 检测MTU...", end=" ")
        if self.system == "Windows":
            output, code = self.run_cmd("ping 8.8.8.8 -f -l 1472 -n 2")
            if "需要拆分" in output or code != 0:
                print("⚠️ 过大，优化中...")
                self.issues.append("MTU过大")
                self.run_cmd('netsh interface ipv4 set subinterface "以太网" mtu=1400 store=persistent')
                self.run_cmd('netsh interface ipv4 set subinterface "WLAN" mtu=1400 store=persistent')
                self.fixes.append("MTU → 1400")
                self.report_data.append(("MTU配置", "⚠️ 过大", "✅ 已优化为1400"))
            else:
                print("✅ 正常")
                self.report_data.append(("MTU配置", "✅ 正常", "无需修复"))
        else:
            output, code = self.run_cmd("ping -M do -s 1472 -c 2 8.8.8.8")
            if code != 0:
                print("⚠️ 过大，优化中...")
                self.issues.append("MTU过大")
                iface_output, _ = self.run_cmd("ip route | awk '/default/{print $5}'")
                iface = iface_output.strip()
                if iface:
                    self.run_cmd(f"sudo ip link set {iface} mtu 1400")
                    self.fixes.append(f"MTU ({iface}) → 1400")
                    self.report_data.append(("MTU配置", "⚠️ 过大", f"✅ 已优化{iface}为1400"))
            else:
                print("✅ 正常")
                self.report_data.append(("MTU配置", "✅ 正常", "无需修复"))
    
    def check_gateway(self):
        print("🔍 检测网关...", end=" ")
        if self.system == "Windows":
            gw_cmd = 'powershell "(Get-NetIPConfiguration).IPv4DefaultGateway.NextHop"'
        else:
            gw_cmd = "ip route | awk '/default/{print $3}'"
        
        gateway, _ = self.run_cmd(gw_cmd)
        gateway = gateway.strip().split('\n')[0]
        
        if gateway and len(gateway) < 20:
            output, code = self.run_cmd(f"ping -{'n' if self.system=='Windows' else 'c'} 3 {gateway}")
            if code == 0:
                print(f"✅ 可达 ({gateway})")
                self.report_data.append(("网关连通", f"✅ {gateway}", "无需修复"))
            else:
                print(f"❌ 不可达 ({gateway})")
                self.issues.append(f"网关{gateway}不可达")
                self.report_data.append(("网关连通", f"❌ {gateway}不可达", "检查网线/WiFi"))
        else:
            print("❌ 未配置")
            self.issues.append("未配置网关")
            self.report_data.append(("网关连通", "❌ 未配置", "检查网络设置"))
    
    def optimize_caches(self):
        print("⚡ 优化网络缓存...", end=" ")
        
        if self.system == "Darwin":
            self.run_cmd("sudo dscacheutil -flushcache")
            self.run_cmd("sudo killall -HUP mDNSResponder")
            self.fixes.append("已刷新DNS缓存")
        elif self.system == "Linux":
            self.run_cmd("sudo ip neigh flush all")
            self.fixes.append("已清理ARP缓存")
        else:
            self.run_cmd("ipconfig /flushdns")
            self.run_cmd("arp -d *")
            self.fixes.append("已刷新DNS和ARP")
        
        print("✅")
        self.report_data.append(("缓存清理", "✅ 完成", "DNS/ARP已刷新"))
    
    def speed_test(self):
        print("🚀 测速（ping延迟）...", end=" ")
        targets = [("国内", "baidu.com"), ("国际", "google.com"), ("DNS", "8.8.8.8")]
        results = []
        
        for name, target in targets:
            output, code = self.run_cmd(f"ping -{'n' if self.system=='Windows' else 'c'} 4 {target}")
            if code == 0:
                import re
                if self.system == "Windows":
                    match = re.search(r'平均 = (\d+)ms', output)
                    if match:
                        results.append(f"{name}: {match.group(1)}ms")
                else:
                    match = re.search(r'avg[=/]\s*(\d+\.\d+)', output)
                    if match:
                        results.append(f"{name}: {match.group(1)}ms")
        
        if results:
            print(f"✅ {' | '.join(results)}")
            self.report_data.append(("网络延迟", " | ".join(results), "✅"))
        else:
            print("⚠️")
            self.report_data.append(("网络延迟", "⚠️ 测试失败", "-"))
    
    def generate_report(self):
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        report_file = Path.home() / "Desktop" / f"netfix_report_{ts}.html"
        
        status = "🟢 网络健康" if not self.issues else f"🟡 已修复 {len(self.issues)} 个问题"
        
        html = f"""<!DOCTYPE html>
<html><head><meta charset="UTF-8">
<title>网络诊断报告 - {ts}</title>
<style>
*  margin:0; padding:0; box-sizing:border-box; 
body  font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif; background: linear-gradient(135deg, #667eea 0%, #764ba2 100%); padding: 20px; min-height: 100vh; 
.container  max-width: 900px; margin: 0 auto; 
.header  background: white; border-radius: 12px; padding: 30px; margin-bottom: 20px; box-shadow: 0 4px 6px rgba(0,0,0,0.1); 
.header h1  font-size: 28px; margin-bottom: 10px; 
.header p  color: #666; 
.status  display: inline-block; padding: 8px 16px; border-radius: 20px; background: #10b981; color: white; font-weight: bold; margin-top: 10px; 
.grid  display: grid; gap: 20px; 
.card  background: white; border-radius: 12px; padding: 25px; box-shadow: 0 4px 6px rgba(0,0,0,0.1); 
.card h3  margin-bottom: 15px; font-size: 18px; color: #333; 
.item  padding: 12px 0; border-bottom: 1px solid #eee; 
.item:last-child  border: none; 
.ok  color: #10b981;  .warn  color: #f59e0b;  .error  color: #ef4444; 
.footer  text-align: center; color: white; margin-top: 30px; opacity: 0.9; font-size: 14px; 
</style></head><body>
<div class="container">
<div class="header">
<h1>🛰️ Lucky网络诊断报告</h1>
<p>📅 {datetime.now().strftime("%Y-%m-%d %H:%M:%S")} | 💻 {self.system}</p>
<div class="status">{status}</div>
</div>
"""
        
        html += '<div class="grid">'
        html += '<div class="card"><h3>📊 诊断结果</h3>'
        for name, status_text, action in self.report_data:
            icon = "✅" if "✅" in status_text else "⚠️" if "⚠️" in status_text else "❌"
            html += f'<div class="item">{icon} <strong>{name}:</strong> {status_text}<br><small style="color:#9ca3af">{action}</small></div>'
        html += '</div>'
        
        if self.fixes:
            html += '<div class="card"><h3>🔧 已执行修复</h3>'
            for fix in self.fixes:
                html += f'<div class="item">✅ {fix}</div>'
            html += '</div>'
        
        html += f'<div class="card"><h3>💾 配置备份</h3><div class="item">备份目录: <code>{self.backup_dir}</code><br><small style="color:#9ca3af">如需回滚，手动恢复备份文件</small></div></div>'
        html += '</div>'
        html += f'<div class="footer">Lucky网络诊断工具 v3.0 | UID9622系统<br>备份位置: {self.backup_dir}</div>'
        html += '</div></body></html>'
        
        report_file.write_text(html, encoding='utf-8')
        print(f"\n📊 报告已生成: {report_file}")
        
        try:
            if self.system == "Darwin":
                subprocess.run(["open", report_file])
            elif self.system == "Windows":
                os.startfile(report_file)
            else:
                subprocess.run(["xdg-open", report_file])
        except:
            pass
    
    def run(self):
        print("\n" + "="*60)
        print("🚀 Lucky网络诊断修复工具 v3.0 - 老大专用版")
        print("="*60 + "\n")
        
        self.check_and_fix_dns()
        self.check_and_fix_mtu()
        self.check_gateway()
        self.optimize_caches()
        self.speed_test()
        
        print("\n" + "="*60)
        if self.issues:
            print(f"✅ 完成！发现 {len(self.issues)} 个问题，已自动修复 {len(self.fixes)} 项")
        else:
            print("✅ 完成！网络健康，无需修复")
        print("="*60 + "\n")
        
        self.generate_report()

if __name__ == "__main__":
    try:
        fixer = LuckyNetFixer()
        fixer.run()
    except KeyboardInterrupt:
        print("\n\n❌ 用户取消")
    except Exception as e:
        print(f"\n❌ 出错: {e}")
```

## 🎁 额外工具：快速诊断别名
### macOS/Linux 版本
添加到 ~/.zshrc 或 ~/.bashrc：
```bash
# Lucky的快速网络工具集
alias ping8='ping 8.8.8.8'
alias pingcn='ping baidu.com'
alias pinggoogle='ping google.com'
alias myip='curl -s ipinfo.io/ip'
alias myiploc='curl -s ipinfo.io'
alias dnsflush='sudo dscacheutil -flushcache && sudo killall -HUP mDNSResponder && echo "✅ DNS缓存已清理"'
alias netinfo='ifconfig | grep "inet " | grep -v 127.0.0.1'
alias ports='lsof -iTCP -sTCP:LISTEN -n -P'
alias netspeed='curl -s https://raw.githubusercontent.com/sivel/speedtest-cli/master/speedtest.py | python3 -'

# 快速网络报告（无需Python）
netreport() {
    ts=$(date +%Y%m%d_%H%M%S)
    out=~/Desktop/net_report_$ts.txt
    {
        echo "=== Lucky网络诊断报告 $ts ==="
        echo ""; echo "【系统信息】"; uname -a
        echo ""; echo "【网络配置】"; ifconfig
        echo ""; echo "【路由表】"; netstat -rn
        echo ""; echo "【DNS配置】"; cat /etc/resolv.conf
        echo ""; echo "【网关测试】"; ping -c 3 $(ip route | awk '/default/{print $3}')
        echo ""; echo "【外网测试】"; ping -c 5 8.8.8.8
        echo ""; echo "【DNS测试】"; nslookup baidu.com
        echo ""; echo "【公网IP】"; curl -s ipinfo.io
    } > $out
    echo "✅ 报告已保存: $out"
    open $out 2>/dev/null || cat $out
}

# 快速测速
speedcheck() {
    echo "🚀 Lucky快速测速工具"
    echo "━━━━━━━━━━━━━━━━━━━━━"
    echo -n "本地网关: "; ping -c 4 $(ip route | awk '/default/{print $3}') 2>/dev/null | tail -1 | awk '{print $4}' | cut -d'/' -f2
    echo -n "国内节点: "; ping -c 4 baidu.com 2>/dev/null | tail -1 | awk '{print $4}' | cut -d'/' -f2
    echo -n "国际节点: "; ping -c 4 google.com 2>/dev/null | tail -1 | awk '{print $4}' | cut -d'/' -f2
    echo -n "公网IP: "; curl -s ipinfo.io/ip
}
```
### Windows PowerShell 版本
添加到 $PROFILE（用 notepad $PROFILE 打开编辑）：
```powershell
# Lucky的快速网络工具集
function ping8 { ping 8.8.8.8 }
function pingcn { ping baidu.com }
function pinggoogle { ping google.com }
function myip { (Invoke-WebRequest -Uri "http://ipinfo.io/ip").Content }
function myiploc { (Invoke-WebRequest -Uri "http://ipinfo.io").Content | ConvertFrom-Json }
function dnsflush { ipconfig /flushdns; Write-Host "✅ DNS缓存已清理" -ForegroundColor Green }
function netinfo { ipconfig | Select-String "IPv4" }
function ports { netstat -ano | Select-String "LISTENING" }

# 快速网络报告
function netreport {
    $ts = Get-Date -Format "yyyyMMdd_HHmmss"
    $out = "$env:USERPROFILE\Desktop\net_report_$ts.txt"
    @"
=== Lucky网络诊断报告 $ts ===

【系统信息】
$(systeminfo | Select-String 'OS Name|OS Version')

【网络配置】
$(ipconfig /all)

【路由表】
$(route print)

【网关测试】
$(ping -n 3 (Get-NetIPConfiguration).IPv4DefaultGateway.NextHop)

【外网测试】
$(ping -n 5 8.8.8.8)

【DNS测试】
$(nslookup baidu.com)

【公网IP】
$((Invoke-WebRequest ipinfo.io).Content)
"@ | Out-File -FilePath $out -Encoding UTF8
    Write-Host "✅ 报告已保存: $out" -ForegroundColor Green
    notepad $out
}

# 快速测速
function speedcheck {
    Write-Host "🚀 Lucky快速测速工具" -ForegroundColor Cyan
    Write-Host "━━━━━━━━━━━━━━━━━━━━━"
    Write-Host -NoNewline "本地网关: "
    ping -n 4 (Get-NetIPConfiguration).IPv4DefaultGateway.NextHop | Select-String "平均" 
    Write-Host -NoNewline "国内节点: "
    ping -n 4 baidu.com | Select-String "平均"
    Write-Host -NoNewline "公网IP: "
    (Invoke-WebRequest ipinfo.io/ip).Content
}
```
