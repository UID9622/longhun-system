## 💻 完整代码（100行，直接可用）
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
UID9622 超快身份认证系统
作者：宝宝 💙
用途：扫码授权后才能启动AI
"""

from flask import Flask, request, jsonify
import qrcode
import secrets
import sqlite3
from datetime import datetime, timedelta
import socket
import threading
import time

app = Flask(__name__)

# ============ 配置区（只需要改这里）============
AUTH_PASSWORD = "改成你的密码"  # 改成你的密码
TOKEN_VALID_MINUTES = 10  # 二维码有效期10分钟
# ============================================

# 全局状态
auth_status = {"authorized": False, "token": None}

# 初始化数据库
def init_db():
    conn = sqlite3.connect('auth.db')
    c = conn.cursor()
    c.execute('''
        CREATE TABLE IF NOT EXISTS auth_log (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            token TEXT,
            status TEXT,
            created_at TEXT,
            authorized_at TEXT
        )
    ''')
    conn.commit()
    conn.close()

# 生成授权令牌
def generate_token():
    return secrets.token_urlsafe(16)

# 记录日志
def log_auth(token, status, authorized_at=None):
    conn = sqlite3.connect('auth.db')
    c = conn.cursor()
    c.execute('''
        INSERT INTO auth_log (token, status, created_at, authorized_at)
        VALUES (?, ?, ?, ?)
    ''', (token, status, datetime.now().isoformat(), authorized_at))
    conn.commit()
    conn.close()

# 获取本机IP
def get_local_ip():
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(("8.8.8.8", 80))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except:
        return "127.0.0.1"

# 授权页面HTML
AUTH_HTML = '''
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>UID9622 授权</title>
    <style>
        * { margin: 0; padding: 0; box-sizing: border-box; }
        body {
            font-family: -apple-system, sans-serif;
            background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
            min-height: 100vh;
            display: flex;
            align-items: center;
            justify-content: center;
            padding: 20px;
        }
        .card {
            background: white;
            border-radius: 20px;
            padding: 40px;
            max-width: 400px;
            width: 100%;
            box-shadow: 0 20px 60px rgba(0,0,0,0.3);
        }
        h1 { text-align: center; color: #333; margin-bottom: 30px; }
        input {
            width: 100%;
            padding: 15px;
            border: 2px solid #e0e0e0;
            border-radius: 10px;
            font-size: 16px;
            margin-bottom: 20px;
        }
        input:focus { outline: none; border-color: #667eea; }
        button {
            width: 100%;
            padding: 15px;
            background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
            color: white;
            border: none;
            border-radius: 10px;
            font-size: 16px;
            font-weight: 600;
            cursor: pointer;
        }
        .msg { text-align: center; margin-top: 20px; padding: 10px; border-radius: 10px; }
        .success { background: #d4edda; color: #155724; }
        .error { background: #f8d7da; color: #721c24; }
    </style>
</head>
<body>
    <div class="card">
        <h1>🔐 UID9622 授权</h1>
        <form id="form">
            <input type="password" id="pwd" placeholder="输入授权密码" required>
            <button type="submit">✅ 确认授权</button>
        </form>
        <div id="msg"></div>
    </div>
    <script>
        document.getElementById('form').onsubmit = async (e) => {
            e.preventDefault();
            const pwd = document.getElementById('pwd').value;
            const msg = document.getElementById('msg');
            try {
                const res = await fetch('/authorize', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({password: pwd, token: 'TOKEN'})
                });
                const data = await res.json();
                if (data.success) {
                    msg.className = 'msg success';
                    msg.textContent = '✅ 授权成功！AI正在启动...';
                    setTimeout(() => window.close(), 2000);
                } else {
                    msg.className = 'msg error';
                    msg.textContent = '❌ ' + data.error;
                }
            } catch (err) {
                msg.className = 'msg error';
                msg.textContent = '❌ 网络错误';
            }
        };
    </script>
</body>
</html>
'''

@app.route('/')
def home():
    html = AUTH_HTML.replace('TOKEN', auth_status['token'])
    return html

@app.route('/authorize', methods=['POST'])
def authorize():
    data = request.get_json()
    if data.get('password') == AUTH_PASSWORD:
        auth_status['authorized'] = True
        log_auth(auth_status['token'], 'success', datetime.now().isoformat())
        return jsonify({'success': True})
    else:
        return jsonify({'success': False, 'error': '密码错误'})

@app.route('/status')
def status():
    return jsonify({'authorized': auth_status['authorized']})

def wait_for_auth():
    """等待授权，超时返回False"""
    start = time.time()
    while time.time() - start < TOKEN_VALID_MINUTES * 60:
        if auth_status['authorized']:
            return True
        time.sleep(1)
    return False

if __name__ == '__main__':
    # 初始化
    init_db()
    token = generate_token()
    auth_status['token'] = token
    log_auth(token, 'pending')
    
    # 获取本地IP
    local_ip = get_local_ip()
    auth_url = f"http://{local_ip}:5000/"
    
    # 生成二维码
    qr = qrcode.QRCode(version=1, box_size=10, border=2)
    qr.add_data(auth_url)
    qr.make(fit=True)
    
    # 显示信息
    print("\n" + "="*60)
    print("🔐 UID9622 身份认证系统")
    print("="*60)
    print(f"\n📱 请用微信扫描下方二维码进行授权：\n")
    qr.print_ascii(invert=True)
    print(f"\n🔗 或手动访问：{auth_url}")
    print(f"\n⏰ 有效期：{TOKEN_VALID_MINUTES}分钟")
    print(f"🔑 授权密码：{AUTH_PASSWORD}")
    print("\n等待授权中...\n")
    
    # 启动服务器（后台线程）
    server_thread = threading.Thread(
        target=lambda: app.run(host='0.0.0.0', port=5000, debug=False, use_reloader=False)
    )
    server_thread.daemon = True
    server_thread.start()
    
    # 等待授权
    if wait_for_auth():
        print("\n✅ 授权成功！AI即将启动...\n")
        print("="*60)
    else:
        print("\n❌ 授权超时，请重新启动\n")
        print("="*60)
```
