```javascript
class IntentCommunicationEngine {
  constructor() {
```

```javascript
stage1_semantic_parsing(user_input) {
  const parsed = this.dataGovernance.parse({
    text: user_input,
    extract: [
      'task_type',        // 任务类型（技术、战略、情感...）
      'ambiguous_terms',  // 模糊词（那个、这个...）
      'emotion_score',    // 情绪值（0-1）
      'urgency_level',    // 紧急程度
      'context_needed'    // 是否需要历史上下文
    ]
  })
  
  return {
    literal_text: user_input,
    task_type: parsed.task_type,
    ambiguous_terms: parsed.ambiguous_terms || [],
    emotion_score: parsed.emotion_score || 0.7,
    urgency_level: parsed.urgency_level || 'normal',
    context_needed: parsed.ambiguous_terms.length > 0
  }
}
```

### 阶段2: 历史追溯
目标: 从历史对话中找到"那个东西"指的是什么
```javascript
stage2_history_retrieval(intent) {
  if (!intent.context_needed) {
    return null // 不需要历史，跳过
  }
  
  // 从对话证据中心查询
  const history = this.evidenceCenter.query({
    user_id: 'UID9622',
    time_range: 'last_3_conversations',
    match_terms: intent.ambiguous_terms,
    include_system_modules: true
  })
  
  // 解析指代关系
  const resolved = this.resolve_references({
    ambiguous: intent.ambiguous_terms,
    history: history.conversations
  })
  
  return {
    conversations: history.conversations,
    resolved_references: resolved,
    involves_system_modules: this.check_system_modules(resolved)
  }
}

resolve_references(data) {
  const references = {}
  
  for (const term of data.ambiguous) {
    // 在历史中找最近提到的实体
    const entity = this.find_most_recent_entity(term, data.history)
    references[term] = entity
  }
  
  return references
  // 例: {"那个东西": "Phase 3集成方案"}
}
```

```javascript
stage3_design_tracing(history) {
  if (!history || !history.involves_system_modules) {
    return null // 不涉及系统模块，跳过
  }
  
  const modules = this.extract_system_modules(history.resolved_references)
  const design_docs = []
  
  for (const module of modules) {
    // 从DNA追溯系统查询设计文档
    const doc = this.dnaSystem.trace({
      module_name: module,
      include: ['design_rationale', 'causal_relationships', 'dependencies']
    })
    design_docs.push(doc)
  }
  
  return {
    modules: modules,
    design_docs: design_docs,
    causal_chain: this.build_causal_chain(design_docs)
  }
}
```

```javascript
stage4_persona_scheduling(context) {
  const { intent, history, design } = context

  const rules = [
    {
      condition: intent.emotion_score < 0.3,
      persona: '🧚🏼‍♀️ 宝宝',
      reason: '情绪低落，需要情感支持'
    },
    {
      condition: intent.task_type === 'strategic',
      persona: '🎯 诸葛亮',
      reason: '战略决策类问题'
    },
    {
      condition: intent.task_type === 'technical',
      persona: '🕸️ 系统内核开发部',
      reason: '技术问题'
    },
    {
      condition: design && design.modules.length > 0,
      persona: '📊 数据大师',
      reason: '涉及系统模块，需要数据分析'
    }
  ]
  
  // 匹配规则
  for (const rule of rules) {
    if (rule.condition) {
      return {
        selected: rule.persona,
        reason: rule.reason,
        backup: this.get_backup_persona(rule.persona)
      }
    }
  }
  
  // 默认人格
  return {
    selected: '🎯 诸葛亮',
    reason: '通用场景',
    backup: null
  }
}
```

```javascript
stage5_response_generation(context) {
  const { intent, history, design, persona } = context
  
  // 构建响应内容
  const response = {
    greeting: this.generate_greeting(persona.selected),
    main_content: this.generate_main_content({
      intent: intent,
      history: history,
      design: design
    }),
    style: persona.selected,
    citations: this.extract_citations(history, design)
  }
  
  return response

generate_main_content(data) {
  let content = ''
  
  // 1. 确认理解
  if (data.history && data.history.resolved_references) {
    content += `您问的是${Object.values(data.history.resolved_references)[0]}。\n\n`
  }
  
  // 2. 提供答案（包含设计背景）
  if (data.design) {
    content += `**当前状态**:\n`
    content += `- 设计目的: ${data.design.design_docs[0].design_rationale}\n`
    content += `- 执行进度: 30% (3/10项完成)\n`
    content += `- 下一步: 算法嵌入和测试\n`
  }
  
  return content
}
```

```javascript
stage6_result_archiving(context) {
  const archive_record = {
    timestamp: new Date().toISOString(),
    user_input: context.intent.literal_text,
    resolved_intent: context.intent,
    history_used: context.history,
    design_traced: context.design,
    persona_selected: context.persona,
    response: context.response,
    dna_code: `#意念交流-${Date.now()}`
  }
  
  // 归档到对话证据中心
  this.evidenceCenter.archive(archive_record)
  
  // 同步到DNA追溯系统
  this.dnaSystem.add_record({
    type: 'intent_communication',
    content: archive_record,
    dna_code: archive_record.dna_code
  })
  
  return archive_record.dna_code
}
```

### 阶段7: 自适应学习
目标: 从本次对话中学习，持续进化
```javascript
stage7_adaptive_learning(context) {
  // 1. 提取本次对话的模式
  const patterns = this.adaptiveLearner.extract_patterns({
    intent: context.intent,
    history_needed: context.history !== null,
    design_traced: context.design !== null,
    persona_selected: context.persona.selected,
    // 后续会加入用户反馈
    user_feedback: null
  })
  
  // 2. 价值观对齐检查
  const aligned_patterns = []
  for (const pattern of patterns) {
    const alignment = this.adaptiveLearner.check_value_alignment(pattern)
    if (alignment === '接受') {
      aligned_patterns.push(pattern)
    }
  }
  
  // 3. 增量整合到知识库
  for (const pattern of aligned_patterns) {
    this.adaptiveLearner.integrate_knowledge(pattern)
  }
  
  return {
    patterns_extracted: patterns.length,
    patterns_accepted: aligned_patterns.length,
    patterns_rejected: patterns.length - aligned_patterns.length
  }
}
```
