## 🛠️ Day 3-5：开发工具链配置（24小时）
### Task 2.1：CI/CD 流程（GitHub Actions）
```bash
mkdir -p .github/workflows

# 创建 lint.yml
cat > .github/workflows/lint.yml << 'EOF'
name: Lint

on: [push, pull_request]

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v4
        with:
          python-version: '3.11'
      - run: pip install black flake8
      - run: black --check backend/
      - run: flake8 backend/
EOF

# 创建 test.yml
cat > .github/workflows/test.yml << 'EOF'
name: Test

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:15-alpine
        env:
          POSTGRES_PASSWORD: test
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432

    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v4
        with:
          python-version: '3.11'
      - run: pip install -r backend/requirements.txt
      - run: cd backend && pytest --cov=. --cov-report=xml
      - uses: codecov/codecov-action@v3
EOF
```
### Task 2.2：本地开发环境配置
```bash
# 创建 Makefile
cat > Makefile << 'EOF'
.PHONY: help setup dev test lint format clean

help:
    @echo "龍魂终端开发命令"
    @echo "make setup      - 初始化开发环境"
    @echo "make dev        - 启动开发服务"
    @echo "make test       - 运行测试"
    @echo "make lint       - 代码检查"
    @echo "make format     - 代码格式化"
    @echo "make clean      - 清理临时文件"

setup:
    cd backend && python3.11 -m venv venv
    cd backend && source venv/bin/activate && pip install -r requirements.txt
    cd frontend-tauri && npm install
    docker-compose up -d

dev:
    docker-compose up -d
    cd backend && source venv/bin/activate && uvicorn main:app --reload
    # 另一个终端
    cd frontend-tauri && npm run dev

test:
    cd backend && source venv/bin/activate && pytest -v

lint:
    cd backend && source venv/bin/activate && flake8 . && black --check .
    cd frontend-tauri && npm run lint

format:
    cd backend && source venv/bin/activate && black .
    cd frontend-tauri && npx prettier --write src

clean:
    find . -type d -name __pycache__ -exec rm -rf {} +
    find . -type f -name "*.pyc" -delete
    rm -rf backend/venv
    rm -rf frontend-tauri/node_modules
    docker-compose down
EOF
```
### Task 2.3：数据库初始化脚本
```bash
mkdir -p scripts

cat > scripts/init-db.sh << 'EOF'
#!/bin/bash

# 等待 PostgreSQL 就绪
sleep 10

# 创建必要的表
psql -h localhost -U longhun -d longhun << SQL
-- DNA码注册表
CREATE TABLE IF NOT EXISTS dna_codes (
    id SERIAL PRIMARY KEY,
    dna_code VARCHAR(255) UNIQUE NOT NULL,
    code_type VARCHAR(50),
    content_hash VARCHAR(64),
    timestamp BIGINT,
    device_id VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    verified BOOLEAN DEFAULT FALSE
);

-- 记忆表
CREATE TABLE IF NOT EXISTS memories (
    id SERIAL PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL,
    content TEXT,
    dna_code VARCHAR(255) REFERENCES dna_codes(dna_code),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 任务执行记录表
CREATE TABLE IF NOT EXISTS task_executions (
    id SERIAL PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL,
    personality_id VARCHAR(10),
    task_description TEXT,
    dna_code VARCHAR(255) REFERENCES dna_codes(dna_code),
    status VARCHAR(50),
    started_at TIMESTAMP,
    completed_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 同步状态表
CREATE TABLE IF NOT EXISTS sync_status (
    id SERIAL PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL,
    resource_type VARCHAR(50),
    last_sync_at TIMESTAMP,
    local_version VARCHAR(255),
    notion_version VARCHAR(255),
    conflict BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建索引
CREATE INDEX IF NOT EXISTS idx_dna_codes_type ON dna_codes(code_type);
CREATE INDEX IF NOT EXISTS idx_memories_user ON memories(user_id);
CREATE INDEX IF NOT EXISTS idx_tasks_user ON task_executions(user_id);
CREATE INDEX IF NOT EXISTS idx_sync_user ON sync_status(user_id);

ECHO "✅ 数据库初始化完成"
SQL
EOF

chmod +x scripts/init-db.sh
```
