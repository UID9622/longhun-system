def sanitize_filename(filename):
    """清理文件名中的非法字符"""
    invalid_chars = '<>:"/\\|?*'
    for char in invalid_chars:
        filename = filename.replace(char, '_')
    return filename

def backup_page(notion, page_url, backup_dir):
    """备份单个页面"""
    try:
        # 提取page ID
        page_id = page_url.split('/')[-1].split('?')[0]
        
        # 获取页面信息
        page = notion.pages.retrieve(page_id)
        
        # 获取页面标题
        title_prop = page['properties'].get('title', {})
        if title_prop:
            title = title_prop.get('title', [{}])[0].get('plain_text', 'Untitled')
        else:
            title = page.get('title', [{}])[0].get('plain_text', 'Untitled')
        
        # 获取页面内容块
        blocks = notion.blocks.children.list(page_id)
        
        # 创建备份文件
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"{sanitize_filename(title)}_{timestamp}.json"
        filepath = os.path.join(backup_dir, filename)
        
        # 保存完整数据
        backup_data = {
            "page_id": page_id,
            "page_url": page_url,
            "title": title,
            "properties": page['properties'],
            "content": blocks,
            "backup_time": timestamp
        }
        
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(backup_data, f, ensure_ascii=False, indent=2)
        
        print(f"✅ 已备份: {title}")
        return True
        
    except Exception as e:
        print(f"❌ 备份失败 {page_url}: {str(e)}")
        return False

def main():
    """主函数"""
    # 创建备份目录
    os.makedirs(BACKUP_DIR, exist_ok=True)
    
    # 初始化Notion客户端

    print(f"🚀 开始备份 {len(PAGES_TO_BACKUP)} 个页面...")

    print("-" * 50)
    
    success_count = 0

    print(f"📁 备份位置: {BACKUP_DIR}")

if __name__ == "__main__":
