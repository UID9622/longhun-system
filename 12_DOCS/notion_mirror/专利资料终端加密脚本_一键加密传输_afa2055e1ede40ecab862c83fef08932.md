import tkinter as tk
from tkinter import filedialog, messagebox, scrolledtext, ttk
import os, zipfile, tarfile, hashlib, datetime, threading
from pathlib import Path

try:
    from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
    from cryptography.hazmat.backends import default_backend
    from cryptography.hazmat.primitives import padding
except:
    messagebox.showerror("错误", "请先运行：pip3 install cryptography")
    exit(1)

class NotionEncrypt:
    def __init__(self, root):
        self.root = root
        self.root.title("🔐 Notion导出加密工具")
        self.root.geometry("750x600")
        
        self.config_dir = Path.home() / ".uid9622"
        self.config_dir.mkdir(exist_ok=True)
        self.password_file = self.config_dir / "password.txt"
        self.selected_folder = None
        self.password = self.load_password()
        self.create_ui()
    
    def load_password(self):
        if self.password_file.exists():
            return self.password_file.read_text().strip()
        pwd = hashlib.sha256(os.urandom(32)).hexdigest()[:32]
        self.password_file.write_text(pwd)
        self.password_file.chmod(0o600)
        return pwd
    
    def create_ui(self):
        # 顶部
        header = tk.Frame(self.root, bg="#DC143C", height=100)
        header.pack(fill="x")
        header.pack_propagate(False)
        
        tk.Label(header, text="🔐 Notion导出加密工具", 
                font=("Arial", 22, "bold"), fg="white", bg="#DC143C").pack(pady=20)
        tk.Label(header, text="一键完成：智能压缩 + 自动加密", 
                font=("Arial", 12), fg="#FFD700", bg="#DC143C").pack()
        
        # 主区域
        main = tk.Frame(self.root, padx=25, pady=20)
        main.pack(fill="both", expand=True)
        
        tk.Label(main, text="📋 选择Notion导出的文件夹，工具会自动完成压缩和加密",
                font=("Arial", 11)).pack(pady=(0, 20))
        
        # 选择按钮
        self.select_btn = tk.Button(main, text="📂 选择Notion导出的文件夹",
                                    command=self.select_folder,
                                    font=("Arial", 14, "bold"),
                                    bg="#4CAF50", fg="white",
                                    cursor="hand2", height=2)
        self.select_btn.pack(fill="x", pady=(0, 15))
        
        # 信息显示
        self.info_frame = tk.Frame(main, bg="#e8f5e9", relief="solid", bd=1)
        self.info_frame.pack(fill="x", pady=(0, 20))
        self.folder_label = tk.Label(self.info_frame, text="📄 未选择文件夹",
                                     font=("Arial", 10), bg="#e8f5e9", padx=10, pady=8)
        self.folder_label.pack(fill="x")
        
        # 开始按钮
        self.start_btn = tk.Button(main, text="🚀 开始加密（自动压缩+加密）",
                                   command=self.start_process,
                                   font=("Arial", 16, "bold"),
                                   bg="#DC143C", fg="white",
                                   cursor="hand2", height=2, state="disabled")
        self.start_btn.pack(fill="x", pady=(0, 15))
        
        # 进度条
        self.progress = ttk.Progressbar(main, mode='indeterminate')
        self.progress.pack(fill="x", pady=(0, 10))
        
        # 日志
        tk.Label(tk.Frame(main).pack(fill="x"), text="📋 操作日志",
                font=("Arial", 11, "bold")).pack(side="left")
        
        self.log = scrolledtext.ScrolledText(main, height=12, font=("Monaco", 9),
                                            bg="#f5f5f5", fg="#333")
        self.log.pack(fill="both", expand=True)
        
        # 查看密码按钮
        tk.Button(tk.Frame(main).pack(fill="x", pady=(10,0)), text="🔑 查看密码",
                 command=self.show_password, font=("Arial", 10),
                 bg="#FF9800", fg="white", cursor="hand2").pack(side="left")
        
        self.log_msg("✅ 系统就绪")
        self.log_msg(f"🔑 密码位置：{self.password_file}")
    
    def log_msg(self, msg):
        t = datetime.datetime.now().strftime("%H:%M:%S")
        self.log.insert("end", f"[{t}] {msg}\n")
        self.log.see("end")
        self.root.update()
    
    def show_password(self):
        messagebox.showinfo("密码", f"🔑 您的密码：\n\n{self.password}\n\n位置：{self.password_file}")
    
    def select_folder(self):
        folder = filedialog.askdirectory(title="选择Notion导出的文件夹",
                                        initialdir=Path.home()/"Downloads")
        if folder:
            self.selected_folder = folder
            size = sum(f.stat().st_size for f in Path(folder).rglob('*') if f.is_file())/1024/1024
            self.folder_label.config(text=f"✅ {Path(folder).name} ({size:.1f} MB)", fg="#2e7d32")
            self.start_btn.config(state="normal")
            self.log_msg(f"✅ 已选择：{Path(folder).name} ({size:.1f} MB)")
    
    def start_process(self):
        if not self.selected_folder:
            return
        self.start_btn.config(state="disabled")
        self.select_btn.config(state="disabled")
        self.progress.start()
        threading.Thread(target=self.process, daemon=True).start()
    
    def process(self):
        try:
            folder = Path(self.selected_folder)
            ts = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
            size_mb = sum(f.stat().st_size for f in folder.rglob('*') if f.is_file())/1024/1024
            
            self.log_msg(f"📊 文件夹大小：{size_mb:.1f} MB")
            
            # 智能选择格式
            fmt = "ZIP" if size_mb<100 else "ZIP高压缩" if size_mb<500 else "TAR.GZ"
            ext = "zip" if size_mb<500 else "tar.gz"
            self.log_msg(f"🎯 智能选择：{fmt}")
            
            # 压缩
            self.log_msg("📦 压缩中...")
            temp = Path.home()/"Desktop"/f"temp_{ts}.{ext}"
            
            if ext=="zip":
                with zipfile.ZipFile(temp,'w',zipfile.ZIP_DEFLATED,compresslevel=9 if size_mb>=100 else 6) as z:
                    for f in folder.rglob('*'):
                        if f.is_file():
                            z.write(f, f.relative_to(folder))
            else:
                with tarfile.open(temp,'w:gz') as t:
                    t.add(folder, arcname=folder.name)
            
            comp_size = temp.stat().st_size/1024/1024
            self.log_msg(f"✅ 压缩完成：{comp_size:.1f}MB (压缩{100-comp_size*100/size_mb:.0f}%)")
            
            # 加密
            self.log_msg("🔐 加密中...")
            enc_file = Path.home()/"Desktop"/f"Notion导出_{ts}.encrypted"
            
            key = self.password.encode()[:32].ljust(32,b'0')
            iv = os.urandom(16)
            cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
            
            with open(temp,'rb') as f:
                data = f.read()
            
            padder = padding.PKCS7(128).padder()
            padded = padder.update(data) + padder.finalize()
            encrypted = cipher.encryptor().update(padded) + cipher.encryptor().finalize()
            
            with open(enc_file,'wb') as f:
                f.write(iv + encrypted)
            
            self.log_msg("✅ 加密完成")
            
            # 校验
            self.log_msg("🔍 生成校验...")
            sha_file = Path.home()/"Desktop"/f"Notion导出_{ts}.sha256"
            h = hashlib.sha256()
            with open(enc_file,'rb') as f:
                h.update(f.read())
            sha_file.write_text(f"{h.hexdigest()}  {enc_file.name}\n")
            
            temp.unlink()
            
            self.log_msg("="*50)
            self.log_msg("🎉 全部完成！")
            self.log_msg(f"📁 加密文件：{enc_file.name}")
            self.log_msg(f"📁 校验文件：{sha_file.name}")
            self.log_msg(f"🔑 密码：{self.password}")
            self.log_msg("="*50)
            
            messagebox.showinfo("完成",f"🎉 加密完成！\n\n文件：{enc_file.name}\n\n密码：{self.password}")
            
        except Exception as e:
            self.log_msg(f"❌ 错误：{e}")
            messagebox.showerror("错误",f"失败：{e}")
        finally:
            self.progress.stop()
            self.start_btn.config(state="normal")
            self.select_btn.config(state="normal")

root = tk.Tk()
NotionEncrypt(root)
root.mainloop()
PYTHON_END

import tkinter as tk
from tkinter import filedialog, messagebox, scrolledtext, simpledialog
import os, zipfile, tarfile, hashlib, datetime, threading
from pathlib import Path

try:
    from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
    from cryptography.hazmat.backends import default_backend
    from cryptography.hazmat.primitives import padding
except:
    messagebox.showerror("错误","请先运行：pip3 install cryptography")
    exit(1)

class NotionDecrypt:
    def __init__(self, root):
        self.root = root
        self.root.title("🔓 Notion解密工具")
        self.root.geometry("700x500")
        
        self.enc_file = None
        self.password = None
        self.create_ui()
    
    def create_ui(self):
        # 顶部
        header = tk.Frame(self.root, bg="#2196F3", height=80)
        header.pack(fill="x")
        header.pack_propagate(False)
        tk.Label(header, text="🔓 Notion解密工具", 
                font=("Arial", 20, "bold"), fg="white", bg="#2196F3").pack(pady=10)
        tk.Label(header, text="选择文件 → 输入密码 → 解密", 
                font=("Arial", 12), fg="#FFD700", bg="#2196F3").pack()
        
        main = tk.Frame(self.root, padx=20, pady=20)
        main.pack(fill="both", expand=True)
        
        # 步骤1
        tk.Label(main, text="步骤1：", font=("Arial",11,"bold")).pack(anchor="w")
        self.file_btn = tk.Button(main, text="📂 选择加密文件(.encrypted)",
                                  command=self.select_file,
                                  font=("Arial",12,"bold"), bg="#4CAF50", fg="white",
                                  cursor="hand2", height=2)
        self.file_btn.pack(fill="x", pady=(5,10))
        self.file_label = tk.Label(main, text="未选择", font=("Arial",10), fg="#999")
        self.file_label.pack(anchor="w", pady=(0,20))
        
        # 步骤2
        tk.Label(main, text="步骤2：", font=("Arial",11,"bold")).pack(anchor="w")
        self.pwd_btn = tk.Button(main, text="🔑 输入密码",
                                command=self.input_password,
                                font=("Arial",12,"bold"), bg="#FF9800", fg="white",
                                cursor="hand2", height=2, state="disabled")
        self.pwd_btn.pack(fill="x", pady=(5,10))
        self.pwd_label = tk.Label(main, text="未输入", font=("Arial",10), fg="#999")
        self.pwd_label.pack(anchor="w", pady=(0,20))
        
        # 步骤3
        tk.Label(main, text="步骤3：", font=("Arial",11,"bold")).pack(anchor="w")
        self.dec_btn = tk.Button(main, text="🔓 开始解密",
                                command=self.start_decrypt,
                                font=("Arial",14,"bold"), bg="#2196F3", fg="white",
                                cursor="hand2", height=2, state="disabled")
        self.dec_btn.pack(fill="x", pady=(5,15))
        
        self.progress = tk.ttk.Progressbar(main, mode='indeterminate')
        self.progress.pack(fill="x", pady=(0,10))
        
        self.log = scrolledtext.ScrolledText(main, height=10, font=("Monaco",9),
                                            bg="#f5f5f5", fg="#333")
        self.log.pack(fill="both", expand=True)
        
        self.log_msg("✅ 系统就绪")
    
    def log_msg(self, msg):
        t = datetime.datetime.now().strftime("%H:%M:%S")
        self.log.insert("end", f"[{t}] {msg}\n")
        self.log.see("end")
        self.root.update()
    
    def select_file(self):
        f = filedialog.askopenfilename(title="选择加密文件",
                                      initialdir=Path.home()/"Downloads",
                                      filetypes=[("加密文件","*.encrypted")])
        if f:
            self.enc_file = f
            self.file_label.config(text=f"✅ {Path(f).name}", fg="#2e7d32")
            self.pwd_btn.config(state="normal")
            self.log_msg(f"✅ 已选择：{Path(f).name}")
    
    def input_password(self):
        pwd = simpledialog.askstring("密码","请输入32位密码：",show='*')
        if pwd and len(pwd)==32:
            self.password = pwd
            self.pwd_label.config(text="✅ 已输入", fg="#2e7d32")
            self.dec_btn.config(state="normal")
            self.log_msg("✅ 密码已输入")
        elif pwd:
            messagebox.showerror("错误","密码必须32位！")
    
    def start_decrypt(self):
        if not self.enc_file or not self.password:
            return
        self.dec_btn.config(state="disabled")
        self.progress.start()
        threading.Thread(target=self.decrypt, daemon=True).start()
    
    def decrypt(self):
        try:
            self.log_msg("🔓 解密中...")
            
            key = self.password.encode()[:32].ljust(32,b'0')
            
            with open(self.enc_file,'rb') as f:
                iv = f.read(16)
                enc_data = f.read()
            
            cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
            padded = cipher.decryptor().update(enc_data) + cipher.decryptor().finalize()
            
            unpadder = padding.PKCS7(128).unpadder()
            data = unpadder.update(padded) + unpadder.finalize()
            
            # 保存
            temp_zip = Path.home()/"Desktop"/"temp_decrypt.zip"
            with open(temp_zip,'wb') as f:
                f.write(data)
            
            self.log_msg("✅ 解密成功")
            self.log_msg("📦 解压中...")
            
            out_folder = Path.home()/"Desktop"/"解密后的文件"
            out_folder.mkdir(exist_ok=True)
            
            # 尝试ZIP
            try:
                with zipfile.ZipFile(temp_zip,'r') as z:
                    z.extractall(out_folder)
            except:
                # 尝试TAR.GZ
                temp_tar = temp_zip.with_suffix('.tar.gz')
                temp_zip.rename(temp_tar)
                with tarfile.open(temp_tar,'r:gz') as t:
                    t.extractall(out_folder)
                temp_tar.unlink()
            else:
                temp_zip.unlink()
            
            self.log_msg("="*50)
            self.log_msg("🎉 解密完成！")
            self.log_msg(f"📁 位置：{out_folder}")
            self.log_msg("="*50)
            
            messagebox.showinfo("完成",f"🎉 解密成功！\n\n文件位置：\n{out_folder}")
            
        except Exception as e:
            self.log_msg(f"❌ 失败：{e}")
            messagebox.showerror("错误",f"解密失败\n可能密码错误或文件损坏\n\n{e}")
        finally:
            self.progress.stop()
            self.dec_btn.config(state="normal")

root = tk.Tk()
NotionDecrypt(root)
root.mainloop()
