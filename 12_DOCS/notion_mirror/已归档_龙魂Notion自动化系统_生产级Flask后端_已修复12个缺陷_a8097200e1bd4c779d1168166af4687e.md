"""
P2修复：业务逻辑解耦到 Service 层
发布服务 - 包含事务回滚机制
DNA追溯码：#ZHUGEXIN⚡️2025-🐉-PUBLISH-SERVICE-V2.0
"""

from sqlalchemy.orm import Session
from sqlalchemy.exc import SQLAlchemyError
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

class PublishService:
    """发布服务层 - 处理发布业务逻辑"""
    
    def __init__(self, notion_sync, publisher, db):
        self.notion_sync = notion_sync
        self.publisher = publisher
        self.db = db
    
    def publish_with_transaction(self, content_id, platform, wechat_version=None, knowledge_version=None):
        """
        P0修复：带事务回滚的发布流程
        如果任何步骤失败，自动回滚
        
        参数：
            content_id: 内容ID
            platform: 发布平台 ('wechat', 'knowledge_base', 'both')
            wechat_version: 微信版本内容
            knowledge_version: 知识库版本内容
        
        返回：
            发布结果字典
        """
        session = self.db.session
        
        try:
            # 开始嵌套事务（支持回滚）
            session.begin_nested()
            
            logger.info(f"开始发布内容 {content_id} 到 {platform}")
            
            # 步骤1：发布到平台
            publish_result = self.publisher.publish_to_platform(
                content_id=content_id,
                platform=platform,
                wechat_version=wechat_version,
                knowledge_version=knowledge_version,
                timeout=60  # P0修复：添加超时
            )
            
            if not publish_result.get('success'):
                raise Exception(f"发布失败: {publish_result.get('error')}")
            
            # 步骤2：更新 Notion 状态
            self.notion_sync.update_content_status(
                content_id=content_id,
                status="已发布",
                platform=platform,
                timeout=10  # P0修复：超时控制
            )
            
            # 步骤3：记录发布历史到数据库
            record = self._save_publish_record(
                content_id=content_id,
                platform=platform,
                result=publish_result
            )
            
            # 提交事务
            session.commit()
            
            logger.info(f"内容 {content_id} 发布成功")
            
            return {
                'success': True,
                'content_id': content_id,
                'platform': platform,
                'publish_url': publish_result.get('url'),
                'published_at': datetime.now().isoformat(),
                'record_id': record.id
            }
            
        except Exception as e:
            # 回滚事务
            session.rollback()
            logger.error(f"发布失败，已回滚: {str(e)}", exc_info=True)
            
            # 记录失败日志
            self._save_error_log(content_id, platform, str(e))
            
            raise Exception(f"发布失败: {str(e)}")
    
    def _save_publish_record(self, content_id, platform, result):
        """
        保存发布记录到数据库
        
        返回：
            PublishRecord 对象
        """
        from models.publish_record import PublishRecord
        
        record = PublishRecord(
            content_id=content_id,
            platform=platform,
            status='success',
            publish_url=result.get('url'),
            published_at=datetime.now(),
            metadata=result  # JSON字段，存储完整结果
        )
        
        self.db.session.add(record)
        self.db.session.flush()  # 刷新以获取ID，但不提交
        
        logger.info(f"已记录发布历史 ID: {record.id}")
        return record
    
    def _save_error_log(self, content_id, platform, error_msg):
        """
        保存错误日志（独立事务，不受主事务影响）
        """
        try:
            from models.error_log import ErrorLog
            
            error_log = ErrorLog(
                content_id=content_id,
                platform=platform,
                error_message=error_msg,
                created_at=datetime.now()
            )
            
            # 使用独立会话保存错误日志
            from sqlalchemy.orm import sessionmaker
            Session = sessionmaker(bind=self.db.engine)
            independent_session = Session()
            
            independent_session.add(error_log)
            independent_session.commit()
            independent_session.close()
            
            logger.info(f"已记录错误日志 ID: {error_log.id}")
            
        except Exception as e:
            # 错误日志保存失败不影响主流程
            logger.warning(f"保存错误日志失败: {str(e)}")

"""
P1修复：速率限制器
防止触发 Notion API 的频率限制（3次/秒）
DNA追溯码：#ZHUGEXIN⚡️2025-🐉-RATE-LIMITER-V2.0
"""

import time
import threading

class RateLimiter:
    """线程安全的速率限制器"""
    
    def __init__(self, max_calls=3, period=1):
        self.max_calls = max_calls
        self.period = period
        self.calls = []
        self.lock = threading.Lock()
    
    def __call__(self):
        with self.lock:
            now = time.time()
            # 清理过期的调用记录
            self.calls = [c for c in self.calls if now - c < self.period]
            
            # 如果达到限制，等待
            if len(self.calls) >= self.max_calls:
                sleep_time = self.period - (now - self.calls[0])
                if sleep_time > 0:
                    time.sleep(sleep_time)
                self.calls = self.calls[1:]  # 移除最旧的记录
            
            self.calls.append(time.time())

def process_content_task(self, content_id, content):
    """
    异步处理内容
    
    任务流程：
    1. AI润色优化
    2. 加密敏感信息
    3. 生成双版本（公开版+内部版）
    4. 返回处理结果
    
    自动重试：最多3次，每次间隔60秒
    """
    try:
        logger.info(f"开始处理内容 {content_id}")
        
        # 步骤1：AI润色
        self.update_state(
            state='STARTED',
            meta={'step': 'AI润色中', 'progress': 25}
        )
        
        optimized_content = ai_engine.optimize_content(
            content=content,
            timeout=30  # P0修复：超时控制
        )
        
        # 步骤2：加密处理
        self.update_state(
            state='STARTED',
            meta={'step': '加密处理中', 'progress': 50}
        )
        
        encrypted_data = cnsh_encryption.encrypt_sensitive_data(
            content=optimized_content
        )
        
        # 步骤3：生成双版本
        self.update_state(
            state='STARTED',
            meta={'step': '生成版本中', 'progress': 75}
        )
        
        versions = content_processor.generate_versions(
            content=encrypted_data,
            content_id=content_id
        )
        
        # 步骤4：完成
        logger.info(f"内容 {content_id} 处理完成")
        
        return {
            'success': True,
            'content_id': content_id,
            'versions': versions,
            'processed_at': time.time()
        }
        
    except Exception as e:
        logger.error(f"处理内容失败: {str(e)}", exc_info=True)
        
        # 自动重试
        try:
            raise self.retry(exc=e)
        except self.MaxRetriesExceededError:
            # 达到最大重试次数，返回失败
            return {
                'success': False,
                'content_id': content_id,
                'error': str(e),
                'retries': self.request.retries
            }

"""

from celery import Celery
from celery.schedules import crontab
import os

# 使用 Redis 作为消息队列和结果后端
celery = Celery(
    'dragon_soul',
    broker=os.environ.get('REDIS_URL', 'redis://localhost:6379/0'),
    backend=os.environ.get('REDIS_URL', 'redis://localhost:6379/0')
)

# Celery 配置
celery.conf.update(
    # 任务配置
    task_serializer='json',
    accept_content=['json'],
    result_serializer='json',
    timezone='Asia/Shanghai',
    enable_utc=True,
    
    # 任务结果过期时间（1天）
    result_expires=86400,
    
    # 任务超时时间（10分钟）
    task_soft_time_limit=600,
    task_time_limit=660,
    
    # 任务重试配置

    task_reject_on_worker_lost=True,  # Worker丢失时拒绝任务

    worker_prefetch_multiplier=1,  # 每次只取1个任务
    worker_max_tasks_per_child=1000,  # Worker执行1000个任务后重启
    
    # 定时任务配置
    beat_schedule={
        # 每天凌晨2点清理旧数据
        'cleanup-old-data': {
            'task': 'tasks.cleanup_old_data',
            'schedule': crontab(hour=2, minute=0),

        # 每小时同步一次到Notion

            'task': 'tasks.sync_to_notion',

        },
    },

# 自动发现任务
celery.autodiscover_tasks(['tasks'])

if __name__ == '__main__':
    celery.start()

## 🚀 部署指南
### 开发环境部署
```bash
# 1. 安装依赖
pip install -r requirements.txt

# 2. 配置环境变量
cp .env.example .env
# 编辑 .env 文件，填写真实配置

# 3. 启动 Redis
redis-server

# 4. 启动 Celery Worker
celery -A celery_app worker --loglevel=info

# 5. 启动 Flask（开发模式）
FLASK_ENV=development python app.py
```
### 生产环境部署（Docker）
```bash
# 1. 构建并启动所有服务
docker-compose up -d

# 2. 查看服务日志
docker-compose logs -f web

# 3. 查看Worker日志
docker-compose logs -f worker

# 4. 停止所有服务
docker-compose down

# 5. 完全清理（包括数据卷）
docker-compose down -v
```
