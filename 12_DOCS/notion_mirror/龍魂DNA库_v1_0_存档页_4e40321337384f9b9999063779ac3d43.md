## 二、Step 3 · push_dna.sh 源码全文
```bash
#!/bin/bash
# push_dna.sh v1.0 · 本地→Notion DNA库单向回写
# DNA格式：DNA::CN-{SHA16}-{SIZE}-{TIME8}
# DNA自身：#龍芯⚡️20260422-PUSH-DNA-01

set -u
DNA_FILE="$1"
[ ! -f "$DNA_FILE" ] && { echo "[push_dna] file not found: $DNA_FILE" >&2; exit 0; }

# 读取 DNA 字符串
DNA_STR=$(cat "$DNA_FILE" | tr -d '\n')
FILENAME=$(basename "$DNA_FILE" .dna)

# 拆解 DNA::CN-SHA-SIZE-TIME  → 字段从 -F'-' 后依次 3/4/5
SHA=$(echo "$DNA_STR"  | awk -F'-' '{print $3}')
SIZE=$(echo "$DNA_STR" | awk -F'-' '{print $4}')
TIME=$(echo "$DNA_STR" | awk -F'-' '{print $5}')

# SHA 首字符 → 十进制 → 数字根 (0-9)
FIRST_CHAR=$(echo "$SHA" | cut -c1)
ORD=$(printf '%d' "'$FIRST_CHAR")
DR=$(awk -v n="$ORD" 'BEGIN{while(n>9){s=0;while(n>0){s+=n%10;n=int(n/10)};n=s};print n}')

# 五行映射：1/6水 2/7火 3/8木 4/9金 5/0土
case "$DR" in
	1|6) WUXING="水" ;;
	2|7) WUXING="火" ;;
	3|8) WUXING="木" ;;
	4|9) WUXING="金" ;;
	*)   WUXING="土" ;;
esac

# 三色：默认🟢；dr=6→🟡；dr=3或9→🔴
COLOR="🟢通行"
[ "$DR" = "6" ] && COLOR="🟡待審"
[ "$DR" = "3" ] && COLOR="🔴熔斷"
[ "$DR" = "9" ] && COLOR="🔴熔斷"

# 生成时间（文件 mtime → ISO8601）
GEN_TIME=$(date -r "$DNA_FILE" +"%Y-%m-%dT%H:%M:%S%z" 2>/dev/null || date +"%Y-%m-%dT%H:%M:%S%z")

# 载入 token / DB
source ~/.cnsh/notion.env
HOSTNAME_SHORT=$(hostname -s)

# 推到 Notion
RESP=$(curl -sS -X POST "https://api.notion.com/v1/pages" \
	-H "Authorization: Bearer $NOTION_TOKEN" \
	-H "Notion-Version: 2022-06-28" \
	-H "Content-Type: application/json" \
	-d @- <<EOF
{
	"parent": {"database_id": "$DNA_DB_ID"},
	"properties": {
		"DNA碼":    {"title":[{"text":{"content":"$DNA_STR"}}]},
		"文件名":   {"rich_text":[{"text":{"content":"$FILENAME"}}]},
		"SHA16":    {"rich_text":[{"text":{"content":"$SHA"}}]},
		"文件大小": {"number": $SIZE},
		"數字根":   {"number": $DR},
		"三色審計": {"select":{"name":"$COLOR"}},
		"五行":     {"select":{"name":"$WUXING"}},
		"狀態":     {"select":{"name":"✅已歸檔"}},
		"來源設備": {"rich_text":[{"text":{"content":"$HOSTNAME_SHORT"}}]},
		"生成時間": {"date":{"start":"$GEN_TIME"}}
	}
}
EOF
)

echo "[push_dna] $(date +%H:%M:%S) $DNA_STR → $COLOR · $WUXING" >> ~/cnsh/入口/push.log

# ====== Step 5 · 草日志自动留痕 ======
COUNTER_FILE=~/cnsh/入口/counter.txt
[ ! -f "$COUNTER_FILE" ] && echo "0" > "$COUNTER_FILE"
N=$(cat "$COUNTER_FILE"); N=$((N+1)); echo "$N" > "$COUNTER_FILE"
SEQ=$(printf "%03d" "$N")
TODAY=$(date +"%Y%m%d")
NOW=$(date +"%H:%M")
SUMMARY="S-${TODAY}-${SEQ} · ${NOW} · DNA回寫 · ${DNA_STR} · ${COLOR} ${WUXING} · 本地→Notion"

# append 到草日志页（block_children API）
curl -sS -X PATCH "https://api.notion.com/v1/blocks/${CAO_PAGE_ID}/children" \
	-H "Authorization: Bearer $NOTION_TOKEN" \
	-H "Notion-Version: 2022-06-28" \
	-H "Content-Type: application/json" \
	-d @- > /dev/null <<EOF
{
	"children":[{
		"object":"block",
		"type":"bulleted_list_item",
		"bulleted_list_item":{
			"rich_text":[{"type":"text","text":{"content":"$SUMMARY"}}]
		}
	}]
}
EOF

exit 0
```
