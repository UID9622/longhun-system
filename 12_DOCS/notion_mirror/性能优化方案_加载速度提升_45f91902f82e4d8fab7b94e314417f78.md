## 💻 性能优化实现
### 1. 代码分割
```javascript
// lazy-loading.js
import { lazy, Suspense } from 'react';

const HWeapon = lazy(() => import('./components/HWeapon'));

function App() {
  return (
    <Suspense fallback={<div>Loading...</div>}>
      <HWeapon />
    </Suspense>
  );
}
```
