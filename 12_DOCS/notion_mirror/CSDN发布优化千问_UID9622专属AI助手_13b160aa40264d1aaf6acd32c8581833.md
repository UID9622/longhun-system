import hashlib
import base64
from datetime import datetime

def generate_dna_codes(article_title, article_content):
    """
    自动生成DNA追溯码
    """
    # 提取主题关键词
    theme = extract_theme(article_title)
    theme_upper = theme.upper().replace(' ', '-')
    
    # 生成完整DNA码
    dna_full = f"#ZHUGEXIN⚡️2025-{theme_upper}-V1.0"
    
    # 生成确认码
    date_str = datetime.now().strftime('%Y-%m-%d')
    confirm_code = f"#CONFIRM🌌9622-{theme}-{date_str}"
    
    # 生成短追溯码（8位哈希）
    content_hash = hashlib.sha256(article_content.encode()).digest()
    short_code = base64.b62encode(content_hash)[:8].decode()
    
    return {
        'dna_full': dna_full,
        'confirm_code': confirm_code,
        'short_code': short_code
    }
