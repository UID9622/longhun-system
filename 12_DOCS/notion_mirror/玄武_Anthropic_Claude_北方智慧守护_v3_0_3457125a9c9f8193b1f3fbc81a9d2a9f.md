## 🐍 第三章 · C++代码落地（从 ARC-05 保留）
```c++
// 三才决策评分 v1.0 · Mac原生·零依赖
// 编译: clang++ -std=c++17 -o sancai sancai.cpp
#include <algorithm>
struct SancaiIn { double H, E, P; };
struct SancaiOut { double score, w_human; bool pass; };

SancaiOut score(SancaiIn s) {
    double w = std::max(0.5, 1.0 - (s.H*0.3 + s.E*0.3));
    double sc = s.H*0.25 + s.E*0.25 + s.P*w;
    return {sc, w, sc >= 0.5};
}
```
