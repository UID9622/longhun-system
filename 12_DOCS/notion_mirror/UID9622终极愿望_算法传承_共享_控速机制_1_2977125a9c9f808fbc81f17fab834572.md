# 不同功能分阶段开放
class FeatureControl:
    def __init__(self):
        self.features = {
            "basic_ai": {
                "stage": 1,  # 第一阶段就开放
                "availability": "worldwide",
                "impact": "low"
            },
            "advanced_ai": {
                "stage": 2,  # 第二阶段开放
                "availability": "developed_countries",
                "impact": "medium"
            },
            "agi_level": {
                "stage": 5,  # 第五阶段才开放
                "availability": "restricted",
                "impact": "high",
                "requires_approval": True
            }
        }
    
    def can_use_feature(self, user, feature_name):
        """
        判断用户是否可以使用某个功能
        """
        feature = self.features[feature_name]
        
        # 检查当前阶段
        current_stage = self.get_current_stage()
        if current_stage < feature["stage"]:
            return False, "这个功能还没到开放时间，请耐心等待"
        
        # 检查地区
        if feature["availability"] == "restricted":
            if not self.has_approval(user, feature_name):
                return False, "这个功能影响太大，需要特别审批"
        
        # 检查用户是否准备好
        user_readiness = self.assess_user_readiness(user)
        if user_readiness < 0.8:
            return False, "建议先熟练掌握基础功能"
        
        return True, "欢迎使用"
