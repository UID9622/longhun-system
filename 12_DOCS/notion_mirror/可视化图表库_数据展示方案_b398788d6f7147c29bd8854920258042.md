### 1. Chart.js基础图表
```javascript
// chart-basic.js
import Chart from 'chart.js/auto';

// H武器使用趋势
const ctx = document.getElementById('h-weapon-chart');
new Chart(ctx, {
  type: 'line',
  data: {
    labels: ['1月', '2月', '3月', '4月', '5月', '6月'],
    datasets: [{
      label: 'H武器使用次数',
      data: [65, 89, 103, 145, 178, 210],
      borderColor: '#FF0000',
      tension: 0.4
    }]
  }
});
```
### 2. 易经卦象可视化
```javascript
// yijing-viz.js
function renderGua(guaNumber) {
  const gua = GUA_DATABASE[guaNumber];
  const canvas = document.getElementById('gua-canvas');
  const ctx = canvas.getContext('2d');
  
  // 绘制卦象
  const binary = gua.binary;
  for (let i = 0; i < 6; i++) {
    const y = 50 + i * 40;
    if (binary[i] === '1') {
      // 阳爻（实线）
      ctx.fillRect(50, y, 200, 10);
    } else {
      // 阴爻（断线）
      ctx.fillRect(50, y, 90, 10);
      ctx.fillRect(160, y, 90, 10);
    }
  }
}
```
