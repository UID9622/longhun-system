#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
🧭 七X 自动归类判别器 · classify_seven.py
把工作区任意「七X」页自动路由到正确家族，防再撞名。
纯标准库，跑一次自检全部样例，错一条即报错。
DNA: #龍芯⚡️2026-06-03-SEVEN-FACTOR-MAINBONE-ALIGN-v1.2
主權人: UID9622 · 龍芯北辰
"""
from __future__ import annotations
from typing import Dict

# ── 家族关键字路由表（可扩展）──
FAMILIES = {
    "套A·BC":    {"kw": ("行为密码学", "Behavioral Cryptography", "Σ(C)",
                         "conf=∏", "来源追溯", "内容认证", "BC-F"),
                 "verdict": "套A · BC-F1..F7（内容血统）→ ✅ 可证实"},
    "套B·责任轴": {"kw": ("责任塌缩", "事不关己", "老好人", "责任系数",
                         "R_coerced", "胁迫态", "责任轴"),
                 "verdict": "套B · R1..R7（人格责任）→ 🟡 待复现"},
    "七柱·TP":   {"kw": ("七柱", "时间逻辑底座", "物理时间", "创世时间",
                         "35格", "TP1"),
                 "verdict": "七柱 · TP1..TP7（时间矩阵）→ 非七因子"},
    "七维·SOUL": {"kw": ("SOUL", "七维度权重", "Σwᵢ·Eᵢ", "SOUL-D"),
                 "verdict": "七维 · SOUL-D1..D7（综合评分）→ 非七因子"},
    "七维·太极": {"kw": ("太极易经", "TJ-D", "七维人性"),
                 "verdict": "七维 · TJ-D1..D7（太极推演）→ 非七因子"},
}

def classify_seven(text: str) -> Dict[str, str]:
    """把任意页路由到正确的『七』家族；归不了标 🔖 转人工。"""
    for family, cfg in FAMILIES.items():
        if any(k in text for k in cfg["kw"]):
            return {"family": family, "verdict": cfg["verdict"]}
    return {"family": "未分类", "verdict": "🔖 未分类 · 转人工判别（特殊标志）"}

# ── 自检：跑一次，错一条就报错 ──
def selftest() -> None:
    cases = [
        ("行为密码学 七因子 Σ(C) 来源追溯", "套A·BC"),
        ("责任塌缩 事不关己 老好人 R_coerced", "套B·责任轴"),
        ("时间逻辑底座 七柱 35格 创世时间", "七柱·TP"),
        ("SOUL 七维度权重 Σwᵢ·Eᵢ", "七维·SOUL"),
        ("太极易经 七维人性推演", "七维·太极"),
        ("一段跟七毫无关系的话", "未分类"),
    ]
    for text, expect in cases:
        got = classify_seven(text)["family"]
        assert got == expect, f"判别错误：{text!r} → {got}（应为 {expect}）"
        print(f"  ✅ {expect:9s} ← {text[:18]}")
    print("🟢 判别器自检全过——新「七X」页可自动归档，归不了的标 🔖。🐉")

if __name__ == "__main__":
    selftest()
