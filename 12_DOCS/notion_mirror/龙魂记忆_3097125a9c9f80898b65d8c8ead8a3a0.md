    // 记忆库
    let memoryLibrary = [];

    // 生成DNA追溯码
    function generateDNA(content) {
        const timestamp = new Date().toISOString().split('T')[0];
        const hash = simpleHash(content).substring(0, 8);
        return `#龙芯⚡️${timestamp}-MEMORY-${hash}-UID9622`;
    }

    // 简单哈希函数
    function simpleHash(str) {
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            const char = str.charCodeAt(i);
            hash = ((hash << 5) - hash) + char;
            hash = hash & hash;
        }
        return Math.abs(hash).toString(16);
    }

    // 提取太极算法变量
    function extractTaijiVars(content) {
        const lines = content.split('\\n');
        const wordCount = content.length;
        const lineCount = lines.length;

        // 提取关键词（简化版）
        const keywords = extractKeywords(content);

        // 情感分析（简化版）
        const emotion = analyzeEmotion(content);

        // 重要程度（基于长度和关键词）
        const importance = Math.min(10, Math.ceil(wordCount / 100) + keywords.length);

        return {
            字数: wordCount,
            行数: lineCount,
            关键词: keywords,
            情感: emotion,
            重要程度: importance,
            时间戳: new Date().toISOString(),
            创建者: 'UID9622'
        };
    }

    // 提取关键词（简化版）
    function extractKeywords(text) {
        // 常见停用词
        const stopWords = ['的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一', '一个', '上', '也', '很', '到', '说', '要', '去', '你', '会', '着', '没有', '看', '好', '自己', '这'];

        // 分词（简化：按标点和空格分割）
        const words = text.split(/[，。！？；：、\\s]+/).filter(w => w.length > 1);

        // 过滤停用词
        const filtered = words.filter(w => !stopWords.includes(w));

        // 统计词频
        const freq = {};
        filtered.forEach(w => {
            freq[w] = (freq[w] || 0) + 1;
        });

        // 按词频排序，取前5个
        const sorted = Object.entries(freq).sort((a, b) => b[1] - a[1]);
        return sorted.slice(0, 5).map(([word]) => word);
    }

    // 情感分析（简化版）
    function analyzeEmotion(text) {
        const positive = ['开心', '快乐', '高兴', '幸福', '美好', '喜欢', '爱', '棒', '好', '成功'];
        const negative = ['难过', '悲伤', '痛苦', '失败', '糟糕', '讨厌', '恨', '差', '错误'];

        let posCount = 0;
        let negCount = 0;

        positive.forEach(word => {
            if (text.includes(word)) posCount++;
        });

        negative.forEach(word => {
            if (text.includes(word)) negCount++;
        });

        if (posCount > negCount) return '积极';
        if (negCount > posCount) return '消极';
        return '平静';
    }

    // 记忆压缩
    function compressMemoryContent(content, taijiVars) {
        // 提取摘要（前100字 + 关键词）
        const summary = content.substring(0, 100) + (content.length > 100 ? '...' : '');

        return {
            原始长度: content.length,
            压缩后长度: summary.length + JSON.stringify(taijiVars).length,
            压缩率: Math.round((1 - (summary.length / content.length)) * 100),
            摘要: summary,
            完整哈希: simpleHash(content),
            可还原: true
        };
    }

    // 冲突检测
    function detectConflict(newContent, taijiVars) {
        // 检查是否有相似的记忆
        for (let memory of memoryLibrary) {
            const similarity = calculateSimilarity(newContent, memory.原始内容);

            if (similarity > 0.8) {
                return {
                    hasConflict: true,
                    conflictMemory: memory,
                    similarity: Math.round(similarity * 100),
                    suggestion: similarity > 0.95 ? 'reject' : 'optimize'
                };
            }
        }

        return { hasConflict: false };
    }

    // 计算相似度（简化版）
    function calculateSimilarity(text1, text2) {
        const words1 = new Set(text1.split(/\\s+/));
        const words2 = new Set(text2.split(/\\s+/));

        let intersection = 0;
        words1.forEach(word => {
            if (words2.has(word)) intersection++;
        });

        const union = words1.size + words2.size - intersection;
        return union > 0 ? intersection / union : 0;
    }

    // 处理记忆
    function processMemory() {
        const input = document.getElementById('inputText').value.trim();

        if (!input) {
            alert('请输入记忆内容！');
            return;
        }

        // 1. 生成DNA
        const dna = generateDNA(input);

        // 2. 提取太极变量
        const taijiVars = extractTaijiVars(input);

        // 3. 检测冲突
        const conflict = detectConflict(input, taijiVars);

        // 4. 压缩记忆
        const compressed = compressMemoryContent(input, taijiVars);

        // 显示结果
        let result = `🧬 DNA追溯码：\\n${dna}\\n\\n`;

        result += `☯️ 太极算法变量：\\n`;
        result += `  字数：${taijiVars.字数}\\n`;
        result += `  行数：${taijiVars.行数}\\n`;
        result += `  关键词：${taijiVars.关键词.join(', ')}\\n`;
        result += `  情感：${taijiVars.情感}\\n`;
        result += `  重要程度：${taijiVars.重要程度}/10\\n`;
        result += `  时间戳：${taijiVars.时间戳}\\n\\n`;

        if (conflict.hasConflict) {
            result += `⚠️ 冲突检测：\\n`;
            result += `  发现相似记忆（相似度${conflict.similarity}%）\\n`;
            result += `  建议：${conflict.suggestion === 'reject' ? '🔴 拒绝（重复记忆）' : '🟡 优化（合并记忆）'}\\n`;
            result += `  冲突DNA：${conflict.conflictMemory.DNA}\\n\\n`;
        } else {
            result += `✅ 冲突检测：无冲突\\n\\n`;
        }

        result += `📦 记忆压缩：\\n`;
        result += `  原始长度：${compressed.原始长度}字\\n`;
        result += `  压缩后：${compressed.压缩后长度}字\\n`;
        result += `  压缩率：${compressed.压缩率}%\\n`;
        result += `  完整哈希：${compressed.完整哈希}\\n\\n`;

        result += `📝 记忆摘要：\\n${compressed.摘要}\\n`;

        document.getElementById('resultBox').textContent = result;

        // 如果没有冲突或建议优化，保存到记忆库
        if (!conflict.hasConflict || conflict.suggestion === 'optimize') {
            saveMemory({
                DNA: dna,
                原始内容: input,
                太极变量: taijiVars,
                压缩数据: compressed,
                创建时间: new Date().toISOString()
            });
        }

        updateStatus(compressed.压缩率);
    }

    // 压缩记忆（单独功能）
    function compressMemory() {
        const input = document.getElementById('inputText').value.trim();

        if (!input) {
            alert('请输入记忆内容！');
            return;
        }

        const taijiVars = extractTaijiVars(input);
        const compressed = compressMemoryContent(input, taijiVars);

        let result = `📦 记忆压缩结果：\\n\\n`;
        result += `原始长度：${compressed.原始长度}字\\n`;
        result += `压缩后长度：${compressed.压缩后长度}字\\n`;
        result += `压缩率：${compressed.压缩率}%\\n`;
        result += `节省空间：${compressed.原始长度 - compressed.压缩后长度}字\\n\\n`;
        result += `📝 摘要：\\n${compressed.摘要}\\n\\n`;
        result += `🔑 关键词：${taijiVars.关键词.join(', ')}\\n`;
        result += `💭 情感：${taijiVars.情感}\\n`;

        document.getElementById('resultBox').textContent = result;
        updateStatus(compressed.压缩率);
    }
