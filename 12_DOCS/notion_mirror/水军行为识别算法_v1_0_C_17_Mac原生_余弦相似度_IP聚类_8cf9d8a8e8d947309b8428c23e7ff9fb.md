```c++
// 水军行为识别算法 v1.0
// 核心：三特征联合判定
// 编译: clang++ -std=c++17 -o bot_detect bot_detect.cpp

#include <vector>
#include <numeric>
#include <cmath>
#include <string>
#include <algorithm>
#include <iostream>

using Vec = std::vector<double>;

// 余弦相似度
double cosineSim(const Vec& a, const Vec& b) {
    if (a.size() != b.size() || a.empty()) return 0.0;
    double dot = 0.0, na = 0.0, nb = 0.0;
    for (size_t i = 0; i < a.size(); ++i) {
        dot += a[i] * b[i];
        na  += a[i] * a[i];
        nb  += b[i] * b[i];
    }
    return (na * nb < 1e-9) ? 0.0 : dot / std::sqrt(na * nb);
}

struct UserBehavior {
    std::string uid;          // 用户ID
    std::string ip_prefix;    // IP前缀 (e.g. "192.168.1")
    Vec content_vec;          // 内容向量 (TF-IDF等处理后)
    double freq_per_hour;     // 小时发布频率
    int account_age_days;     // 账号年龄(天)
    double report_success_rate; // 举报成功率 [0,1]
};

struct BotScore {
    double similarity;    // 内容相似度
    double ip_cluster;    // IP聚类度
    double freq_score;    // 频率异常度
    double total;         // 综合得分
    bool   is_bot;        // 判定结果
    std::string risk_level; // 风险级别
};

BotScore detectBot(
    const UserBehavior& target,
    const std::vector<UserBehavior>& population,
    double sim_threshold  = 0.9,
    double freq_threshold = 10.0,   // 每小时超过10条算异常
    int    age_threshold  = 7       // 带号小于7天为新号
) {
    BotScore score{};

    // 1. 内容相似度（余弦）
    int similar_count = 0;
    for (const auto& u : population) {
        if (u.uid == target.uid) continue;
        if (cosineSim(target.content_vec, u.content_vec) > sim_threshold)
            ++similar_count;
    }
    int n = (int)population.size() - 1;
    score.similarity = n > 0 ? (double)similar_count / n : 0.0;

    // 2. IP聚类检测
    int ip_count = 0;
    for (const auto& u : population)
        if (u.ip_prefix == target.ip_prefix) ++ip_count;
    score.ip_cluster = (double)ip_count / std::max(1, (int)population.size());

    // 3. 频率异常度
    score.freq_score = std::min(1.0, target.freq_per_hour / freq_threshold);

    // 综合得分
    score.total = score.similarity * 0.40
                + score.ip_cluster  * 0.35
                + score.freq_score  * 0.25;

    // 新号加成
    if (target.account_age_days < age_threshold)
        score.total = std::min(1.0, score.total * 1.2);

    // 举报成功率过低→恶意举报器
    if (target.report_success_rate < 0.20)
        score.total = std::min(1.0, score.total + 0.15);

    score.is_bot = (score.total > 0.6);
    score.risk_level = score.total > 0.8 ? "🔴高危"
                     : score.total > 0.6 ? "🟡可疑"
                     : "🟢正常";
    return score;
}

int main() {
    // 模拟场景：5个用户
    std::vector<UserBehavior> pop = {
        {"u001", "10.0.0", {0.9,0.8,0.1},  15.0, 3,  0.15}, // 水军
        {"u002", "10.0.0", {0.85,0.82,0.12}, 12.0, 2, 0.10}, // 水军
        {"u003", "192.1.1",{0.3,0.1,0.8},   1.5, 365, 0.80}, // 正常
        {"u004", "10.0.0", {0.88,0.79,0.11}, 14.0, 5, 0.12}, // 水军
        {"u005", "172.1.2",{0.1,0.5,0.9},   2.0, 200, 0.90}, // 正常
    };

    for (const auto& u : pop) {
        auto s = detectBot(u, pop);
        std::cout << u.uid << " " << s.risk_level
                  << " 得分:" << s.total
                  << " 相似:" << s.similarity
                  << " IP:" << s.ip_cluster << "\n";
    }
    return 0;
}
```
