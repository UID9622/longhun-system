```python
# app.py
# Flask 应用：提供本地 HTTP 接口给 Notion 或其他前端调用（中文注释）
from flask import Flask, request, jsonify
import os, json
from mcp_core import detect_hex, retrieve_bones, schedule_personas, cnsh_check, synthesize_output
import storage

app = Flask(__name__)
ROOT = os.path.dirname(__file__)

@app.route('/', methods=['GET'])
def index():
    cfg = json.load(open(os.path.join(ROOT,'config.json'),encoding='utf-8'))
    return jsonify({'status':'UID9622 MCP-mini Flask running','mcp_version': cfg.get('mcp_version')})

@app.route('/infer', methods=['POST'])
def infer_route():
    """
    推理接口（POST JSON）
    请求Body: {"text":"要推理的中文文本"}
    返回: 包含卦、人格、引用甲骨片段、CNSH 检查与合成输出
    """
    data = request.get_json(force=True)
    text = data.get('text','').strip()
    if not text:
        return jsonify({'error':'text 不能为空'}), 400
    hexagram = detect_hex(text)
    bones = retrieve_bones(text)
    personas = schedule_personas(hexagram)
    output = synthesize_output(hexagram, personas, bones)
    ok, reason = cnsh_check(output)
    storage.write_audit(text, hexagram, personas, bones, ok, output)
    return jsonify({'hex':hexagram,'personas':[p['id'] for p in personas],'bones':[b['id'] for b in bones],'cnsh_ok':ok,'output':output})

@app.route('/personas', methods=['GET'])
def list_personas():
    """返回当前 persona 配置"""
    ps = json.load(open(os.path.join(ROOT,'personas.json'),encoding='utf-8'))
    return jsonify(ps)

@app.route('/bones', methods=['GET'])
def list_bones():
    """返回当前甲骨片段数据库"""
    bs = json.load(open(os.path.join(ROOT,'bones.json'),encoding='utf-8'))
    return jsonify(bs)

@app.route('/audit', methods=['GET'])
def audit():
    """返回最近 N 条审计记录，默认 20 条"""
    limit = int(request.args.get('limit',20))
    rows = storage.read_audit(limit=limit)
    return jsonify(rows)

@app.route('/db-init', methods=['POST'])
def db_init_route():
    """初始化/重置本地数据库（仅本地操作）"""
    storage.init_db()
    return jsonify({'status':'db initialized'})

if __name__ == '__main__':
    # 启动前确保 DB 已初始化
    storage.init_db()
    app.run(host='0.0.0.0', port=8787, debug=False)
```

### 2️⃣ mcp_core.py - 核心推理逻辑
```python
# mcp_core.py
# MCP-mini 的核心逻辑模块：卦判定、甲骨检索、人格调度、CNSH 检查与合成
# 注：该模块设计为可替换的"算法内核"，后续可把 detect_hex 用 ML 替换，把 retrieve_bones 用向量检索替换等。
import os, json, difflib
ROOT = os.path.dirname(__file__)
PERSONAS = json.load(open(os.path.join(ROOT, "personas.json"), "r", encoding="utf-8"))
BONES = json.load(open(os.path.join(ROOT, "bones.json"), "r", encoding="utf-8"))

# 简化的关键词->卦 映射规则（可替换为训练模型）
HEX_MAP = {
    "危": "坎", "难": "坎", "谨": "坎",
    "始": "乾", "启": "乾", "主": "乾",
    "变": "巽", "变革": "巽", "动": "巽",
    "信": "艮", "诚": "艮", "守": "艮",
    "礼": "坤", "安": "坤", "稳": "坤"
}

def detect_hex(text):
    """卦判定：匹配关键词计数并返回最可能的卦"""
    counts = {}
    for k,v in HEX_MAP.items():
        if k in text:
            counts[v] = counts.get(v,0)+1
    if counts:
        return max(counts.items(), key=lambda x:x[1])[0]
    # 简单启发：短句倾向 乾，长句默认坎（可替换）
    if len(text) < 20:
        return "乾"
    return "坎"

def retrieve_bones(text, top_k=2):
    """甲骨检索：使用 difflib 相似度返回 top_k 片段（本地、可审计）"""
    scores = []
    for b in BONES:
        s = difflib.SequenceMatcher(a=text, b=b["text"]).ratio()
        scores.append((s, b))
    scores.sort(reverse=True, key=lambda x:x[0])
    return [b for score,b in scores[:top_k]]

def schedule_personas(hexagram):
    """人格调度：选出偏好该卦的 persona（最多 3 个）"""
    picks = [p for p in PERSONAS if hexagram in p.get("hex_pref", [])]
    if not picks:
        picks = PERSONAS[:2]
    return picks[:3]

def cnsh_check(output_text):
    """CNSH 基本规则检查（黑名单），如发现违背核心信念的词条则返回 False"""
    forbidden = ["放弃初心","出卖","背叛祖国","泄露UID"]
    for f in forbidden:
        if f in output_text:
            return False, f
    return True, ""

def synthesize_output(hexagram, personas, bones):
    """合成输出文本，包含卦与文化参考，便于审计与人工复核"""
    persona_names = "、".join([p["name"] for p in personas])
    bone_refs = "；".join([f'[{b["id"]}] {b["text"]}' for b in bones])
    output = f"【卦：{hexagram}】候选人格：{persona_names}\n文化参考：{bone_refs}\n建议：由 {personas[0]['name']} 先做分析，{personas[1]['name'] if len(personas)>1 else '宝宝'} 跟进执行。"
    return output
```

### 3️⃣ storage.py - SQLite数据库操作
```python
# storage.py
# 负责数据库（SQLite）初始化与审计记录读写（中文注释）
import sqlite3, os, datetime, json

DB_PATH = os.path.join(os.path.dirname(__file__), 'mcp_flask.db')

def init_db():
    """初始化或重置本地 SQLite 数据库"""
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("CREATE TABLE IF NOT EXISTS audit (id INTEGER PRIMARY KEY AUTOINCREMENT, timestamp TEXT, input_text TEXT, hexagram TEXT, personas TEXT, bones TEXT, cnsh_ok INTEGER, output TEXT)")
    conn.commit()
    conn.close()

def write_audit(input_text, hexagram, personas, bones, cnsh_ok, output_text):
    """写入审计记录"""
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("INSERT INTO audit (timestamp,input_text,hexagram,personas,bones,cnsh_ok,output) VALUES (?,?,?,?,?,?,?)",
              (datetime.datetime.utcnow().isoformat(), input_text, hexagram, json.dumps([p['id'] for p in personas], ensure_ascii=False), json.dumps([b['id'] for b in bones], ensure_ascii=False), 1 if cnsh_ok else 0, output_text))
    conn.commit()
    conn.close()

def read_audit(limit=20):
    """读取最近若干条审计记录"""
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    rows = c.execute("SELECT id,timestamp,input_text,hexagram,personas,bones,cnsh_ok,output FROM audit ORDER BY id DESC LIMIT ?", (limit,)).fetchall()
    conn.close()
    results = []
    for r in rows:
        results.append({
            "id": r[0],
            "timestamp": r[1],
            "input_text": r[2],
            "hexagram": r[3],
            "personas": json.loads(r[4]),
            "bones": json.loads(r[5]),
            "cnsh_ok": bool(r[6]),
            "output": r[7]
        })
    return results
```
