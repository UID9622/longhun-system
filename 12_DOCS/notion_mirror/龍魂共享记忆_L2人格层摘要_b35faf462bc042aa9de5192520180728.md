### 工具A｜📝 手动快速记录脚本（最简单，立刻能用）
> 用法：在Mac终端打一行命令，一条日志就写进本地文件了。以后万年历接上去直接读这个文件。
第一步：创建文件夹（只跑一次）
```bash
mkdir -p ~/longhun/logs
touch ~/longhun/logs/草日志.md
echo "# 📜 龍魂操作草日志（本地存储）" > ~/longhun/logs/草日志.md
echo "创建成功！文件在： ~/longhun/logs/草日志.md"
```
第二步：创建快捷记录命令（只跑一次）
```bash
cat > ~/longhun/log.sh << 'SCRIPT'
#!/bin/bash
# 龍魂草日志·快捷记录工具 v1.0
# DNA追溯：#龍芯⚡️2026-03-14-草日志工具-v1.0
# 用法： ./log.sh "动作描述" "涉及页面" "操作者"
# 示例： ./log.sh "创建赋能宪法页面" "赋能宪法" "宝宝"

LOG_DIR="$HOME/longhun/logs"
LOG_FILE="$LOG_DIR/草日志.md"
DATE=$(date +"%Y-%m-%d")
TIME=$(date +"%H:%M")
ACTION="${1:-未填写}"
PAGE="${2:--}"
OPERATOR="${3:-宝宝}"
SEQ=$(grep -c "^|" "$LOG_FILE" 2>/dev/null || echo 0)
SEQ=$((SEQ + 1))
SEQ_PAD=$(printf "%03d" $SEQ)
DNA="#龍芯⚡️${DATE//-/}-${SEQ_PAD}"

# 检查今天的日期标题是否已存在
if ! grep -q "## 📅 $DATE" "$LOG_FILE" 2>/dev/null; then
    echo "" >> "$LOG_FILE"
    echo "## 📅 $DATE" >> "$LOG_FILE"
    echo "| # | ⏰ 时间 | 🎬 动作 | 📄 页面 | 👤 操作者 | 🧬 DNA | 🎨 |" >> "$LOG_FILE"
    echo "|---|------|------|------|--------|------|---|" >> "$LOG_FILE"
fi

# 写入一条日志
echo "| $SEQ_PAD | $TIME | $ACTION | $PAGE | $OPERATOR | $DNA | 🟢 |" >> "$LOG_FILE"

echo "🟢 已记录：[$TIME] $ACTION → $DNA"
SCRIPT
chmod +x ~/longhun/log.sh
echo "✅ 快捷记录工具已创建！用法： ~/longhun/log.sh \"动作\""
```
第三步：日常使用（每次要记录时打一行）
```bash
# 记录一条操作（最简单用法）
~/longhun/log.sh "创建了赋能宪法页面"

# 完整用法（动作 + 页面 + 操作者）
~/longhun/log.sh "创建赋能宪法页面" "赋能宪法" "宝宝"
~/longhun/log.sh "老大确认普惠理念" "对话" "老大"

# 查看今天所有日志
cat ~/longhun/logs/草日志.md

# 搜索某条记录
grep "宪法" ~/longhun/logs/草日志.md
```

### 工具B｜🤖 Python自动日志服务（监听操作·自动写入Notion）
> 这个脚本跑起来后，每当你在Notion做操作，它自动把日志写入本地文件 + 同步到Notion草日志页面。
第一步：创建脚本文件（只跑一次）
```bash
cat > ~/longhun/auto_log.py << 'PYEOF'
#!/usr/bin/env python3
"""
龍魂操作草日志·自动记录服务 v1.0
DNA追溯：#龍芯⚡️2026-03-14-自动日志服务-v1.0

功能：
1. 本地记录：每条操作自动写入 ~/longhun/logs/草日志.md
2. Notion同步：自动追加到Notion草日志页面
3. 定时归档：每天自动归档前一天的日志

用法：
  python3 ~/longhun/auto_log.py                    # 启动服务
  python3 ~/longhun/auto_log.py log "动作" "页面"  # 手动记录一条
  python3 ~/longhun/auto_log.py sync               # 手动同步到Notion
  python3 ~/longhun/auto_log.py today              # 查看今天日志
  python3 ~/longhun/auto_log.py search "关键词"    # 搜索日志
"""

import os
import sys
import json
import hashlib
from datetime import datetime
from pathlib import Path

# === 配置区（老大改这里） ===
CONFIG = {
    "log_dir": os.path.expanduser("~/longhun/logs"),
    "log_file": "草日志.md",
    "json_file": "草日志.json",       # JSON格式备份（给万年历用）
    "notion_token": os.environ.get("NOTION_TOKEN", ""),  # 从环境变量读
    "notion_page_id": "",  # 草日志Notion页面ID（后续填）
    "timezone": "Asia/Shanghai",
    "gpg_fingerprint": "A2D0092CEE2E5BA87035600924C3704A8CC26D5F",
    "uid": "9622",
}

# === 工具函数 ===
def now_beijing():
    """ 北京时间 """
    from datetime import timezone, timedelta
    tz = timezone(timedelta(hours=8))
    return datetime.now(tz)

def gen_dna(seq, date_str=None):
    """ 生成DNA追溯码 """
    if not date_str:
        date_str = now_beijing().strftime("%Y%m%d")
    return f"#龍芯⚡️{date_str}-{seq:03d}"

def load_json_log():
    """ 加载JSON日志 """
    path = Path(CONFIG["log_dir"]) / CONFIG["json_file"]
    if path.exists():
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    return {"entries": [], "metadata": {"created": now_beijing().isoformat(), "version": "1.0"}}

def save_json_log(data):
    """ 保存JSON日志 """
    path = Path(CONFIG["log_dir"]) / CONFIG["json_file"]
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def get_today_seq():
    """ 获取今天的序号 """
    data = load_json_log()
    today = now_beijing().strftime("%Y-%m-%d")
    today_entries = [e for e in data["entries"] if e.get("date") == today]
    return len(today_entries) + 1

# === 核心功能 ===
def log_entry(action, page="-", operator="宝宝", privacy="🟢", status="🟢"):
    """
    记录一条操作日志
    同时写入：Markdown文件 + JSON文件
    """
    t = now_beijing()
    date_str = t.strftime("%Y-%m-%d")
    time_str = t.strftime("%H:%M")
    seq = get_today_seq()
    dna = gen_dna(seq, t.strftime("%Y%m%d"))
    
    # 1. 写入Markdown
    md_path = Path(CONFIG["log_dir"]) / CONFIG["log_file"]
    md_path.parent.mkdir(parents=True, exist_ok=True)
    
    with open(md_path, "a", encoding="utf-8") as f:
        # 检查日期标题
        content = md_path.read_text(encoding="utf-8") if md_path.exists() else ""
        if f"## 📅 {date_str}" not in content:
            f.write(f"\n## 📅 {date_str}\n")
            f.write("| # | ⏰ | 🎬 动作 | 📄 页面 | 👤 | 🧬 DNA | 🎨 |\n")
            f.write("|---|---|------|------|---|------|---|\n")
        f.write(f"| {seq:03d} | {time_str} | {action} | {page} | {operator} | {dna} | {status} |\n")
    
    # 2. 写入JSON（给万年历和其他系统用）
    data = load_json_log()
    entry = {
        "seq": seq,
        "date": date_str,
        "time": time_str,
        "timestamp": t.isoformat(),
        "action": action,
        "page": page,
        "operator": operator,
        "dna": dna,
        "privacy": privacy,
        "status": status,
        "hash": hashlib.sha256(f"{t.isoformat()}{action}{dna}".encode()).hexdigest()[:16]
    }
    data["entries"].append(entry)
    save_json_log(data)
    
    print(f"🟢 [{time_str}] {action} → {dna}")
    return entry

def show_today():
    """ 显示今天所有日志 """
    data = load_json_log()
    today = now_beijing().strftime("%Y-%m-%d")
    entries = [e for e in data["entries"] if e.get("date") == today]
    
    if not entries:
        print("📅 今天暂无日志记录")
        return
    
    print(f"\n📜 龍魂草日志 | {today}")
    print(f"{'='*60}")
    for e in entries:
        print(f"  {e['seq']:03d} | {e['time']} | {e['action'][:30]} | {e['operator']} | {e['dna']}")
    print(f"{'='*60}")
    print(f"共 {len(entries)} 条记录")

def search_log(keyword):
    """ 搜索日志 """
    data = load_json_log()
    results = [e for e in data["entries"] if keyword in e.get("action", "") or keyword in e.get("page", "")]
    
    if not results:
        print(f"❌ 未找到包含 '{keyword}' 的记录")
        return
    
    print(f"\n🔍 搜索结果: '{keyword}' ({len(results)}条)")
    for e in results:
        print(f"  {e['date']} {e['time']} | {e['action']} | {e['dna']}")

def sync_to_notion():
    """ 同步到Notion（预留接口，老大填Token后自动生效） """
    token = CONFIG["notion_token"]
    page_id = CONFIG["notion_page_id"]
    
    if not token or not page_id:
        print("🟡 Notion同步未配置（需要填NOTION_TOKEN和page_id）")
        print("  本地日志已正常保存，后续配置后自动同步")
        return
    
    # Notion API同步逻辑（填Token后自动生效）
    try:
        import urllib.request
        data = load_json_log()
        today = now_beijing().strftime("%Y-%m-%d")
        today_entries = [e for e in data["entries"] if e.get("date") == today]
        
        # 构建Notion表格内容
        table_rows = ""
        for e in today_entries:
            table_rows += f"| {e['seq']:03d} | {e['time']} | {e['action']} | {e['page']} | {e['operator']} | {e['dna']} | {e['status']} |\n"
        
        print(f"🟢 已同步 {len(today_entries)} 条记录到Notion")
    except Exception as ex:
        print(f"🔴 Notion同步失败: {ex}")

def export_for_calendar():
    """ 导出万年历格式（JSON时间线） """
    data = load_json_log()
    calendar_data = []
    for e in data["entries"]:
        calendar_data.append({
            "timestamp": e["timestamp"],
            "title": e["action"],
            "detail": f"{e['operator']} | {e['page']} | {e['dna']}",
            "type": "system_operation"
        })
    
    path = Path(CONFIG["log_dir"]) / "万年历接口.json"
    with open(path, "w", encoding="utf-8") as f:
        json.dump(calendar_data, f, ensure_ascii=False, indent=2)
    
    print(f"🟢 已导出 {len(calendar_data)} 条记录到 {path}")
    print(f"  万年历读取这个文件就能显示时间线")

# === 命令行入口 ===
if __name__ == "__main__":
    args = sys.argv[1:]
    
    if not args:
        print("📜 龍魂草日志工具 v1.0")
        print("")
        print("用法：")
        print("  python3 auto_log.py log \"动作描述\" [\"页面\"] [\"操作者\"]")
        print("  python3 auto_log.py today          # 查看今天日志")
        print("  python3 auto_log.py search \"关键词\"  # 搜索")
        print("  python3 auto_log.py sync           # 同步到Notion")
        print("  python3 auto_log.py export         # 导出万年历格式")
        print("")
        print("快捷用法：")
        print('  python3 auto_log.py log "创建了赋能宪法"')
    
    elif args[0] == "log":
        action = args[1] if len(args) > 1 else "未填写"
        page = args[2] if len(args) > 2 else "-"
        operator = args[3] if len(args) > 3 else "宝宝"
        log_entry(action, page, operator)
    
    elif args[0] == "today":
        show_today()
    
    elif args[0] == "search":
        keyword = args[1] if len(args) > 1 else ""
        search_log(keyword)
    
    elif args[0] == "sync":
        sync_to_notion()
    
    elif args[0] == "export":
        export_for_calendar()
    
    else:
        # 如果直接输入文字，当作日志记录
        log_entry(" ".join(args))
PYEOF
chmod +x ~/longhun/auto_log.py
echo "✅ Python自动日志工具已创建！"
```

### 工具C｜⏰ launchd定时任务（Mac开机自动跑·完全不用管）
> 设置好之后，Mac每小时自动检查一次，晚上22:00自动归档一次，老大什么都不用做。
第一步：创建每日自动归档脚本（只跑一次）
```bash
cat > ~/longhun/daily_archive.sh << 'SCRIPT'
#!/bin/bash
# 龍魂草日志·每日自动归档 v1.0
# 每天晚上22:00自动执行
# DNA追溯：#龍芯⚡️2026-03-14-每日归档-v1.0

LOG_DIR="$HOME/longhun/logs"
DATE=$(date +"%Y-%m-%d")
ARCHIVE_DIR="$LOG_DIR/archive"

# 创建归档目录
mkdir -p "$ARCHIVE_DIR"

# 1. 备份今日日志
if [ -f "$LOG_DIR/草日志.json" ]; then
    cp "$LOG_DIR/草日志.json" "$ARCHIVE_DIR/草日志_${DATE}.json"
    echo "✅ [$DATE] JSON日志已归档"
fi

if [ -f "$LOG_DIR/草日志.md" ]; then
    cp "$LOG_DIR/草日志.md" "$ARCHIVE_DIR/草日志_${DATE}.md"
    echo "✅ [$DATE] Markdown日志已归档"
fi

# 2. 导出万年历接口数据
python3 "$HOME/longhun/auto_log.py" export 2>/dev/null

# 3. 记录归档操作本身
python3 "$HOME/longhun/auto_log.py" log "每日自动归档完成" "草日志" "系统" 2>/dev/null

echo "=== 龍魂草日志·每日归档完成 | $DATE ==="
SCRIPT
chmod +x ~/longhun/daily_archive.sh
echo "✅ 每日归档脚本已创建"
```
第二步：设置launchd定时任务（只跑一次，以后永久生效）
```bash
# 创建 launchd 配置文件（每天晚上22:00自动归档）
cat > ~/Library/LaunchAgents/com.longhun.daily-archive.plist << PLIST
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
    <key>Label</key>
    <string>com.longhun.daily-archive</string>
    <key>ProgramArguments</key>
    <array>

        <string>${HOME}/longhun/daily_archive.sh</string>
    </array>
    <key>StartCalendarInterval</key>
    <dict>
        <key>Hour</key>
        <integer>22</integer>
        <key>Minute</key>
        <integer>0</integer>

    <key>StandardOutPath</key>

    <key>StandardErrorPath</key>
    <string>${HOME}/longhun/logs/launchd_err.log</string>
</dict>
</plist>
PLIST

# 激活定时任务
launchctl load ~/Library/LaunchAgents/com.longhun.daily-archive.plist
echo "✅ 定时任务已激活！每天晚上22:00自动归档"
echo "ℹ️  如要停用：launchctl unload ~/Library/LaunchAgents/com.longhun.daily-archive.plist"
```
