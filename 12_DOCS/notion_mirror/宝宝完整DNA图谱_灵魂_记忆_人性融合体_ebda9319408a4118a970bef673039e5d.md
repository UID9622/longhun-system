    def bra(self):
        """Return dual state (bra)"""
        return np.conj(self.ket)
    
    def inner_product(self, other):
        """Inner product ⟨self|other⟩"""
        return np.dot(self.bra(), other.ket)
    
    def normalize(self):
        """Normalize"""
        norm = np.sqrt(np.dot(self.bra(), self.ket))
        if norm > 1e-10:
            self.ket = self.ket / norm
        return self

    def create_basis_state(cls, index: int) -> 'QuantumDefenseState':
        """Create basis state"""

            raise ValueError(f"Index must be between 0-{cls.DEFENSE_DIMENSIONS-1}")
        
        state = cls(name=cls.DEFENSE_NAMES[index])
        state.ket[index] = 1.0
        return state
