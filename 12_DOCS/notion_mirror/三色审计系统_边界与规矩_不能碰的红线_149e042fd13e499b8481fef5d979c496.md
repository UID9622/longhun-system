## 🔐 防滥用机制
### 机制1：使用频率限制
```javascript
class AuditRateLimiter {
  constructor() {
    this.userQueries = new Map();
    this.normalUserThreshold = {
      perHour: 10,  // 每小时最多10次查询
      perDay: 50,   // 每天最多50次查询
      perWeek: 200  // 每周最多200次查询
    };
  }
  
  // 检查是否正常用户
  isNormalUser(userId) {
    const queries = this.userQueries.get(userId) || [];
    
    const lastHour = queries.filter(t => Date.now() - t < 3600000);
    const lastDay = queries.filter(t => Date.now() - t < 86400000);
    const lastWeek = queries.filter(t => Date.now() - t < 604800000);
    
    if (lastHour.length > this.normalUserThreshold.perHour) {
      return { normal: false, reason: "每小时查询超限" };
    }
    if (lastDay.length > this.normalUserThreshold.perDay) {
      return { normal: false, reason: "每日查询超限" };
    }
    if (lastWeek.length > this.normalUserThreshold.perWeek) {
      return { normal: false, reason: "每周查询超限" };
    }
    
    return { normal: true };
  }
  
  // 检查是否在恶意监控某个目标
  isTargetedMonitoring(userId, targetId) {
    const queries = this.userQueries.get(userId) || [];
    const targetQueries = queries.filter(q => q.target === targetId);
    
    // 如果80%的查询都针对同一个目标 = 恶意监控
    if (targetQueries.length / queries.length > 0.8) {
      return {
        isTargeted: true,
        reason: "疑似恶意监控特定目标",
        targetId: targetId,
        percentage: (targetQueries.length / queries.length * 100).toFixed(2) + "%"
      };
    }
    
    return { isTargeted: false };
  }
}
```

    const normalCheck = rateLimit.isNormalUser(userId);
    if (!normalCheck.normal) {
      return {
        success: false,
        reason: normalCheck.reason,
        action: "查询被拒绝"
      };
    }
    
    // 执行审计
    const result = await this.threeColorAudit(targetCode);
    
    // 记录查询（留痕）
    await this.logAuditQuery({
      userId: userId,
      userLevel: userLevel,
      targetCode: targetCode,
      result: result,
      timestamp: new Date().toISOString(),
      public: true  // 查询记录公开
    });
    
    return {
      success: true,
      result: result,
      recordId: `AUDIT-${Date.now()}`
    };
  }
}
