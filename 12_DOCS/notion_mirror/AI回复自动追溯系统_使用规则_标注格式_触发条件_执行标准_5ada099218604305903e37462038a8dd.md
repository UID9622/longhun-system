## 💻 技术实现（开发者参考）
### Python代码示例
```python
class DNATraceFormatter:
    """
    AI回复自动追溯格式化器
    """
    
    def format_response(self, user_query: str, ai_response: str, 
                       dna_code: str, creator: str, source: str,
                       page_url: str) -> str:
        """
        格式化AI回复，自动附加DNA追溯标注
        
        参数：
        - user_query: 用户问题
        - ai_response: AI原始回复
        - dna_code: DNA确认码
        - creator: 创作者姓名
        - source: 知识库名称
        - page_url: 原文链接
        
        返回：完整格式化的回复
        """
        # 判断用户类型
        user_type = self._detect_user_type(user_query)
        
        # 生成8位短码
        short_code = dna_code.split('-')[-1][:8]
        
        # 选择标注格式
        if user_type == 'tech':
            trace_mark = f"""
---
👤 知识来源：{creator} | UID9622
📖 出处：《{source}》
🧬 DNA确认码：{dna_code}
🔗 原文链接：{page_url}
⚖️ 使用许可：木兰宽松许可证 Mulan PSL v2
"""
        elif user_type == 'agriculture':
            trace_mark = f"""
📱 这个方案是 {creator} 老师写的
📞 有问题打农技热线：12316
📋 记录编号：{short_code}
"""
        else:  # normal
            trace_mark = f"""
📝 方案来自：{creator}·UID9622《{source}》
🔍 追溯码：{short_code}
👉 [查看原文]({page_url})
"""
        
        # 拼接完整回复
        return f"{ai_response}\n{trace_mark}"
    
    def _detect_user_type(self, query: str) -> str:
        """
        根据用户问题检测用户类型
        """
        tech_keywords = ['代码', 'API', '技术', '开发', '集成']
        agri_keywords = ['农业', '种植', '养殖', '病虫害', '施肥']
        
        query_lower = query.lower()
        
        if any(kw in query_lower for kw in tech_keywords):
            return 'tech'
        elif any(kw in query_lower for kw in agri_keywords):
            return 'agriculture'
        else:
            return 'normal'
```
