## 🚀 一键安装UID9622三色系统
```bash
#!/bin/bash
# UID9622 三色系统一键安装脚本
# Lucky完全自主控制版本

echo "🚀 UID9622三色系统安装程序"
echo "====================================="
echo ""

# 检查Python版本
echo "📋 检查Python环境..."
python3 --version

if [ $? -ne 0 ]; then
    echo "❌ 未找到Python3，请先安装"
    exit 1
fi

echo "✅ Python环境正常"
echo ""

# 创建目录结构
echo "📁 创建目录结构..."
mkdir -p ~/UID9622/tombstone/red
mkdir -p ~/UID9622/reference/yellow
mkdir -p ~/UID9622/core/green
mkdir -p ~/UID9622/backups/notion
mkdir -p ~/UID9622/conversation_history
mkdir -p ~/UID9622/logs
mkdir -p ~/UID9622/scripts

echo "✅ 目录创建完成"
echo ""

# 安装Python依赖
echo "📦 安装Python依赖..."
pip3 install notion-client flask flask-cors

if [ $? -ne 0 ]; then
    echo "⚠️ 部分依赖安装失败，继续..."
fi

echo ""

# 下载脚本文件
echo "📥 下载核心脚本..."
cd ~/UID9622/scripts

# 这里应该从Git仓库或Notion下载脚本
# 暂时提示用户手动复制
echo "⚠️ 请将以下脚本复制到 ~/UID9622/scripts/ 目录："
echo "   - auto_classifier.py"
echo "   - notion_local_backup.py"
echo "   - tombstone_query.py"
echo "   - conversation_logger.py"
echo "   - chat_history.py"
echo ""

read -p "脚本已复制完成？(y/n): " scripts_ready

if [ "$scripts_ready" != "y" ]; then
    echo "❌ 安装已取消"
    exit 1
fi

# 设置环境变量
echo ""
echo "🔑 配置环境变量..."
echo ""
read -p "请输入Notion Token: " notion_token

echo "export NOTION_TOKEN='$notion_token'" >> ~/.zshrc
echo "export UID9622_HOME='~/UID9622'" >> ~/.zshrc

echo "✅ 环境变量已配置"
echo ""

# 创建快捷命令
echo "⚡ 创建快捷命令..."
echo "alias chat-history='python3 ~/UID9622/scripts/chat_history.py'" >> ~/.zshrc
echo "alias tombstone-query='python3 ~/UID9622/scripts/tombstone_query.py'" >> ~/.zshrc
echo "alias notion-backup='python3 ~/UID9622/scripts/notion_local_backup.py'" >> ~/.zshrc

echo "✅ 快捷命令已创建"
echo ""

# 配置定时任务
echo "⏰ 配置自动化任务..."

read -p "是否配置定时任务？(y/n): " setup_cron

if [ "$setup_cron" = "y" ]; then
    bash ~/UID9622/scripts/setup_cron.sh
    echo "✅ 定时任务已配置"
else
    echo "⏭️ 跳过定时任务配置"
fi

echo ""

# 完成
echo "🎉 安装完成！"
echo ""
echo "📋 下一步："
echo "1. 重新加载配置: source ~/.zshrc"
echo "2. 测试三色分类: cd ~/UID9622/scripts && python3 auto_classifier.py"
echo "3. 查看对话历史: chat-history today"
echo "4. 备份Notion: notion-backup"
echo ""
echo "📍 所有数据存储在: ~/UID9622/"
echo ""
```
