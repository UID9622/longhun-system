### 📄 activation-code.js（激活码生成）

/**
 * LU激活系统 - 激活码生成
 * DNA追溯码：#ZHUGEXIN⚡️2026-01-27-激活码生成-v1.0
 */

const crypto = require('crypto');
const config = require('./config');

class ActivationCodeGenerator {
  // 生成激活码
  static generate(email) {
    const emailHash = this.hashEmail(email);
    const timestamp = this.getTimestamp();
    const random = this.getRandomCode();
    
    return `LU-${emailHash}-${timestamp}-${random}`;
  }
  
  // 生成激活指纹
  static generateFingerprint(email, paymentId, deviceDNA) {
    const emailHash = this.hashEmail(email);
    const paymentHash = this.hashPayment(paymentId);
    const timestamp = this.getTimestamp();
    const deviceHash = this.hashDevice(deviceDNA);
    
    return `${config.dnaPrefix}LU-ACTIVE-${emailHash}-${paymentHash}-${timestamp}-${deviceHash}`;
  }
  
  // 邮箱哈希
  static hashEmail(email) {
    return crypto.createHash('sha256')
      .update(email)
      .digest('hex')
      .substring(0, 8)
      .toUpperCase();
  }
  
  // 支付ID哈希
  static hashPayment(paymentId) {
    return crypto.createHash('sha256')
      .update(paymentId)
      .digest('hex')
      .substring(0, 8)
      .toUpperCase();
  }
  
  // 设备DNA哈希
  static hashDevice(deviceDNA) {
    return crypto.createHash('sha256')
      .update(deviceDNA)
      .digest('hex')
      .substring(0, 6)
      .toUpperCase();
  }
  
  // 获取时间戳（YYYYMMDD格式）
  static getTimestamp() {
    return new Date().toISOString().split('T')[0].replace(/-/g, '');
  }
  
  // 获取随机码
  static getRandomCode() {
    return Math.floor(Math.random() * 10000)
      .toString()
      .padStart(4, '0');
  }
  
  // 验证激活码格式
  static validate(code) {
    const pattern = /^LU-[A-F0-9]{8}-\d{8}-\d{4}$/;
    return pattern.test(code);
  }
}

module.exports = ActivationCodeGenerator;

/**
 * LU激活系统 - 邮件发送
 * DNA追溯码：#ZHUGEXIN⚡️2026-01-27-邮件发送-v1.0
 */

const nodemailer = require('nodemailer');
const config = require('./config');

class EmailSender {
  constructor() {
    this.transporter = nodemailer.createTransport(config.email);
  }
  
  // 发送激活邮件
  async sendActivationEmail(userEmail, userName, activationCode) {
    const qrCodeUrl = `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${activationCode}`;
    
    const mailOptions = {
      from: config.email.auth.user,
      to: userEmail,
      subject: '🔐 您的LU激活码',
      html: this.generateActivationEmailHTML(userName, activationCode, qrCodeUrl)
    };
    
    try {
      await this.transporter.sendMail(mailOptions);
      console.log(`✅ 激活码已发送到 ${userEmail}`);
      return { success: true };
    } catch (error) {
      console.error(`❌ 发送失败：${error.message}`);
      return { success: false, error: error.message };
    }
  }
  
  // 发送告警邮件
  async sendAlertEmail(activationCode, reason, level) {
    const emoji = level === 'red' ? '🔴' : '🟡';
    const subject = `${emoji} LU系统${level === 'red' ? '红色熔断' : '黄色预警'}`;
    
    const mailOptions = {
      from: config.email.auth.user,
      to: config.email.auth.user, // 发给老大自己
      subject,
      html: this.generateAlertEmailHTML(activationCode, reason, level)
    };
    
    try {
      await this.transporter.sendMail(mailOptions);
      console.log(`✅ 告警邮件已发送`);
      return { success: true };
    } catch (error) {
      console.error(`❌ 发送失败：${error.message}`);
      return { success: false, error: error.message };
    }
  }
  
  // 生成激活邮件HTML
  generateActivationEmailHTML(userName, activationCode, qrCodeUrl) {
    return `
      <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto;">
        <h2>🔐 LU激活系统</h2>
        <p>尊敬的 <strong>${userName}</strong>，</p>
        <p>感谢您申请LU激活码！</p>
        
        <div style="background: #f5f5f5; padding: 20px; border-radius: 5px; margin: 20px 0;">
          <h3>您的激活码：</h3>
          <p style="font-size: 24px; font-weight: bold; color: #0066cc; letter-spacing: 2px;">
            ${activationCode}
          </p>
        </div>
        
        <h3>激活步骤：</h3>
        <ol>
          <li>扫描下方二维码</li>
          <li>绑定实名支付（微信/支付宝/华为钱包）</li>
          <li>完成激活</li>
        </ol>
        
        <div style="text-align: center; margin: 20px 0;">
          <img src="${qrCodeUrl}" alt="激活二维码" />
        </div>
        
        <hr style="margin: 30px 0;" />
        
        <p style="font-size: 12px; color: #666;">
          DNA追溯码：${config.dnaPrefix}2026-01-27-LU激活邮件<br>
          创建者：${config.system.creator}<br>
          系统：龙魂系统
        </p>
      </div>
    `;
  }
  
  // 生成告警邮件HTML
  generateAlertEmailHTML(activationCode, reason, level) {
    const emoji = level === 'red' ? '🔴' : '🟡';
    const title = level === 'red' ? '红色熔断' : '黄色预警';
    const actions = level === 'red' 
      ? '已执行：<br>✅ 封禁激活码<br>✅ 写入黑名单<br>✅ 记录DNA追溯'
      : '建议：人工审核';
    
    return `
      <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto;">
        <h2>${emoji} LU系统${title}</h2>
        <p>老大，检测到异常使用行为：</p>
        
        <div style="background: #fff3cd; padding: 20px; border-radius: 5px; margin: 20px 0;">
          <p><strong>激活码：</strong>${activationCode}</p>
          <p><strong>${level === 'red' ? '熔断' : '预警'}原因：</strong>${reason}</p>

}
