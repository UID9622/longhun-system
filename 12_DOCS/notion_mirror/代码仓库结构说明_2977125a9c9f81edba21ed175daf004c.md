export class ChineseSegmenter {
  constructor() {
    // 加载UID9622自定义词典
    nodejieba.load({
      userDict: './dict/uid9622.txt'
    });
  }
  
  segment(text: string): string[] {
    return nodejieba.cut(text);
  }
}

## 🌟 关键设计原则
### 1. 三层太极架构
- 天层（docs/） - 战略规划、理念文档
- 人层（src/） - 核心功能实现
- 地层（tests/, scripts/） - 基础设施
### 2. 中国特色优化
- 所有字符串默认UTF-8
- 日期处理默认Asia/Shanghai时区
- 网络请求加入重试和超时保护
- 中文分词使用结巴+自定义词典
### 3. 开箱即用
- examples/ 提供完整可运行示例
- .env.example 零配置起步
- Docker支持一键部署
