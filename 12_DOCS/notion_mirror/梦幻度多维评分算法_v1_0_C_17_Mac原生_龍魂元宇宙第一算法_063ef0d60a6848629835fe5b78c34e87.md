## 一句话定义
> 将任意事件的「科幻程度」通过影响范围×重要性×时间衰减计算出可比较的实数。来自 DNA L5架构原点能量场公式 E = R × I × T^(-α)。
## 公式
```c++
// 梦幻度评分算法 v1.0
// 来自 DNA-L5架构能量场公式
// 不必安装任何依赖，Mac上直接编译运行

#include <cmath>
#include <string>
#include <iostream>

// 五彩石层级对应衰减指数
struct L5Layer {
    std::string name;
    double alpha;        // 衰减指数
    std::string stone;   // 对应彩石
};

static const L5Layer LAYERS[] = {
    {"L0 永恒", 0.0,    "金石"},
    {"L1 百年", 0.01,   "銀石"},
    {"L2 十年", 0.1,    "红石"},
    {"L3 日常", 1.0,    "蓝石"},
    {"L4 瞬时", 1e9,    "绳石"},
};

struct FantasyInput {
    double R;       // 影响范围 1=个体 / 10=模块 / 100=系统 / 1000=传承 / inf=文明
    double I;       // 重要性 [0, 100]
    double T;       // 时间距离(天)
    int    layer;   // L0~L4 对应下标
};

double calcFantasy(const FantasyInput& s) {
    double alpha = LAYERS[s.layer].alpha;
    // T^(-alpha): alpha=0时T^0=1（永恒层）
    double decay = (alpha < 1e-9) ? 1.0
                 : std::pow(std::max(1.0, s.T), -alpha);
    return s.R * s.I * decay;
}

// 归一化到 [0, 100]，少于1=无看头，大于95=五星级
double normalizedScore(const FantasyInput& s) {
    double raw = calcFantasy(s);
    // log归一化
    return std::min(100.0, std::max(0.0,
        (std::log10(raw + 1.0) + 1.0) * 20.0
    ));
}

int main() {
    // 示例1：量子计算事件（30天前，全行业影响）
    FantasyInput qc = {.R=1000, .I=95, .T=30, .layer=2}; // L2十年层
    std::cout << "量子计算 梦幻度: " << normalizedScore(qc) << "/100\n";

    // 示例2：普通帖子（今天，内容一般）
    FantasyInput post = {.R=10, .I=20, .T=1, .layer=3}; // L3日常层
    std::cout << "普通帖子 梦幻度: " << normalizedScore(post) << "/100\n";

    return 0;
}
// 编译: clang++ -std=c++17 -o fantasy fantasy.cpp && ./fantasy
```
