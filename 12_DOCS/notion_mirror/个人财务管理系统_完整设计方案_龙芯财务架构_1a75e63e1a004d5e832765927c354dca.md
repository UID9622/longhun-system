## 📋 第一部分：需求分析与功能结构设计
### 🎯 核心需求分析
目标用户画像：
- 普通上班族、学生、自由职业者
- 不懂复杂财务知识
- 想知道钱花哪了、还剩多少
- 需要简单易用的记账工具
核心痛点：
1. 记账太麻烦 → 懒得记，坚持不下来
1. 看不懂报表 → 专业术语太多，看不懂
1. 不知道怎么省钱 → 缺乏数据分析和建议
1. 多账户管理混乱 → 微信、支付宝、银行卡分散

### 📊 数据库设计（核心表结构）

- account_id (主键)

- create_time (创建时间)
分类表 (categories)：
- category_id (主键)
- user_id (用户ID)
- category_name (分类名称)

- transaction_id (主键)

- budget_id (主键)

from flask import Flask, jsonify, request
from datetime import datetime, timedelta
import pandas as pd

app = Flask(__name__)

@app.route('/api/statistics/monthly', methods=['GET'])
def get_monthly_statistics():
    """获取月度收支统计"""
    user_id = request.args.get('user_id')
    year = request.args.get('year', datetime.now().year)
    month = request.args.get('month', datetime.now().month)
    
    # 查询该月所有交易记录
    transactions = db.query(
        f"""SELECT 
            transaction_type,
            category_name,
            SUM(amount) as total_amount,
            COUNT(*) as count
        FROM transactions t
        JOIN categories c ON t.category_id = c.category_id
        WHERE t.user_id = {user_id}
        AND YEAR(transaction_date) = {year}
        AND MONTH(transaction_date) = {month}
        GROUP BY transaction_type, category_name
        """
    )
    
    # 数据处理
    income_total = 0
    expense_total = 0
    category_stats = []
    
    for record in transactions:
        if record['transaction_type'] == '收入':
            income_total += record['total_amount']
        else:
            expense_total += record['total_amount']
            category_stats.append({
                'category': record['category_name'],
                'amount': record['total_amount'],
                'count': record['count']
            })
    
    # 计算分类占比
    for stat in category_stats:
        stat['percentage'] = round(
            (stat['amount'] / expense_total * 100) if expense_total > 0 else 0, 
            2
        )
    
    return jsonify({
        'income': income_total,
        'expense': expense_total,
        'balance': income_total - expense_total,
        'category_stats': category_stats
    })

@app.route('/api/statistics/trend', methods=['GET'])
def get_trend_statistics():
    """获取收支趋势（最近6个月）"""
    user_id = request.args.get('user_id')
    
    # 获取最近6个月的数据
    trend_data = []
    for i in range(5, -1, -1):
        target_date = datetime.now() - timedelta(days=30*i)
        year = target_date.year
        month = target_date.month
        
        result = db.query(
            f"""SELECT 
                SUM(CASE WHEN transaction_type='收入' THEN amount ELSE 0 END) as income,
                SUM(CASE WHEN transaction_type='支出' THEN amount ELSE 0 END) as expense
            FROM transactions
            WHERE user_id = {user_id}
            AND YEAR(transaction_date) = {year}
            AND MONTH(transaction_date) = {month}
            """
        )
        
        trend_data.append({
            'month': f'{year}-{month:02d}',
            'income': result['income'] or 0,
            'expense': result['expense'] or 0
        })
    
    return jsonify({'trend': trend_data})

class BudgetMonitor {
  // 检查预算状态
  async checkBudgetStatus(userId, categoryId, currentAmount) {
    // 获取该分类的预算设置
    const budget = await db.query(
      `SELECT budget_amount FROM budgets 
       WHERE user_id = ? AND category_id = ? 
       AND start_date <= NOW() AND end_date >= NOW()`,
      [userId, categoryId]
    )
    
    if (!budget) return { status: 'no_budget' }
    
    // 获取当前周期已使用金额
    const used = await db.query(
      `SELECT SUM(amount) as total 
       FROM transactions 
       WHERE user_id = ? AND category_id = ? 
       AND transaction_date >= (SELECT start_date FROM budgets WHERE ...)
       AND transaction_type = '支出'`,
      [userId, categoryId]
    )
    
    const usedAmount = used.total || 0
    const budgetAmount = budget.budget_amount
    const percentage = (usedAmount / budgetAmount) * 100
    
    // 判断预算状态
    if (percentage >= 100) {
      return {
        status: 'exceeded',
        message: `${categoryId}预算已超支！已使用${usedAmount}元，预算${budgetAmount}元`,
        percentage: percentage
      }
    } else if (percentage >= 80) {
      return {
        status: 'warning',
        message: `${categoryId}预算即将用完！已使用${percentage.toFixed(1)}%`,
        percentage: percentage
      }
    } else {
      return {
        status: 'normal',
        percentage: percentage
      }
    }
  }
