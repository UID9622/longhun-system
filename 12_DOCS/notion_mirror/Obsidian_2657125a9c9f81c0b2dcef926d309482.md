## 🧰 单段 AI 回复 → 一键看懂（极简版｜可选图表）
- 用法：把 AI 的回复整段粘贴到“输入文本” → 点“分析文本”。
- 默认只显示“🧠 技术解读”卡片，图表是可选项，随时开关。
---
### ▶ 复制下面这段 HTML，保存为本地 .html，用浏览器直接打开
```html

<script>
  document.addEventListener('DOMContentLoaded', () => {
    const $ = sel => document.querySelector(sel);
    const textInput = $('#text-input');
    const analyzeBtn = $('#analyze-btn');
    const results = $('#results');
    const chart = $('#chart');
    const tbody = $('#tbody');
    const insightBox = $('#insight');
    const showChart = $('#show-chart');
    const viz = $('#viz');

    analyzeBtn.addEventListener('click', () => {
      const text = (textInput.value || '').trim();
      if(!text){ alert('请输入文本'); return; }
      const counts = analyzeText(text);
      const insight = buildInsight(text, counts);
      display(counts, insight);
    });

    // 文本 → 词频（中英混合，去标点与大小写）
    function analyzeText(text){
      const clean = text.replace(/[.,，。!?！？;:；："'()（）\[\]【】\-—_\s]+/g,' ');
      let words = []; let buf='';
      for(const ch of clean){
        if(/[\u4e00-\u9fa5]/.test(ch)){ // 中文：按单字简化
          if(buf){ words.push(buf.toLowerCase()); buf=''; }
          words.push(ch);
        } else if(ch===' '){ if(buf){ words.push(buf.toLowerCase()); buf=''; } }
        else { buf += ch; }
      }
      if(buf) words.push(buf.toLowerCase());
      words = words.filter(w => w.trim()!=='');
      const map = {}; words.forEach(w=> map[w]=(map[w]||0)+1);
      return map;
    }

    function display(counts, insight){
      tbody.innerHTML=''; chart.innerHTML=''; insightBox.innerHTML='';
      results.style.display='block';

      // 技术解读卡片
      insightBox.innerHTML = `
        <h3>🧠 技术解读</h3>
        <p><strong>这是啥：</strong>${insight.oneLiner}</p>
        <p><strong>要解决的问题：</strong>${insight.problem || '—'}</p>
        <p><strong>关键要点：</strong></p>
        <ul>${insight.bullets.map(b=>`<li>${b}</li>`).join('')}</ul>
        ${insight.terms.length?`<p><strong>术语解释：</strong></p><ul>${insight.terms.map(t=>`<li>${t.term}：${t.explain}</li>`).join('')}</ul>`:''}
        <p><strong>下一步：</strong>${insight.next || '可忽略'}</p>`;

      // 图表（可选）
      const wantChart = showChart.checked;
      if(!wantChart){ viz.style.display='none'; return; }
      viz.style.display='block';

      const top = Object.entries(counts).sort((a,b)=>b[1]-a[1]).slice(0,10);
      const totalCount = Math.max(1, top.reduce((s,[_w,c])=>s+c,0));

      tbody.innerHTML = top.map(([w,c])=>{
        const pct=((c/totalCount)*100).toFixed(2)+'%';
        return `<tr><td>${w}</td><td>${c}</td><td>${pct}</td></tr>`;
      }).join('');

      const type = document.querySelector('input[name="chartType"]:checked').value;
      if(type==='pie') drawPie(top); else drawBar(top);
    }

    // 简单 SVG 条形图
    function drawBar(data){
      const margin={top:10,right:20,bottom:40,left:40};
      const width=Math.max(320, chart.clientWidth) - margin.left - margin.right;
      const height=360 - margin.top - margin.bottom;
      const svgNS='http://www.w3.org/2000/svg';
      chart.innerHTML='';
      const svg=document.createElementNS(svgNS,'svg');
      svg.setAttribute('width', width+margin.left+margin.right);
      svg.setAttribute('height', height+margin.top+margin.bottom);
      svg.style.overflow='visible';
      chart.appendChild(svg);
      const g=document.createElementNS(svgNS,'g');
      g.setAttribute('transform',`translate(${margin.left},${margin.top})`);
      svg.appendChild(g);

      const max=Math.max(1, ...data.map(d=>d[1]));
      const yScale=height/max; const barW=width/data.length;

      data.forEach(([w,c],i)=>{
        const h=c*yScale;
        const bar=document.createElementNS(svgNS,'rect');
        bar.setAttribute('class','bar');
        bar.setAttribute('x', i*barW);
        bar.setAttribute('y', height-h);
        bar.setAttribute('width', Math.max(2, barW-6));
        bar.setAttribute('height', h);
        g.appendChild(bar);
        const tx=document.createElementNS(svgNS,'text');
        tx.textContent=w; tx.setAttribute('x', i*barW + barW/2);
        tx.setAttribute('y', height+16); tx.setAttribute('fill','#9ca3af');
        tx.setAttribute('text-anchor','middle'); tx.setAttribute('transform',`rotate(45, ${i*barW + barW/2}, ${height+16})`);
        g.appendChild(tx);
        const cv=document.createElementNS(svgNS,'text');
        cv.textContent=c; cv.setAttribute('x', i*barW + barW/2);
        cv.setAttribute('y', height-h-6); cv.setAttribute('fill','#e5e7eb');
        cv.setAttribute('text-anchor','middle'); g.appendChild(cv);
      });

      const yAxis=document.createElementNS(svgNS,'line');
      yAxis.setAttribute('x1',0); yAxis.setAttribute('y1',0);
      yAxis.setAttribute('x2',0); yAxis.setAttribute('y2',height);
      yAxis.setAttribute('stroke','#6b7280'); g.appendChild(yAxis);
      const xAxis=document.createElementNS(svgNS,'line');
      xAxis.setAttribute('x1',0); xAxis.setAttribute('y1',height);
      xAxis.setAttribute('x2',width); xAxis.setAttribute('y2',height);
      xAxis.setAttribute('stroke','#6b7280'); g.appendChild(xAxis);
    }

    // 简单 SVG 饼图
    function drawPie(data){
      const svgNS='http://www.w3.org/2000/svg';
      chart.innerHTML='';
      const size=Math.max(260, Math.min(chart.clientWidth, 360));
      const r=size/2 - 6; const cx=r+6, cy=r+6;
      const svg=document.createElementNS(svgNS,'svg');
      svg.setAttribute('width', size+12); svg.setAttribute('height', size+12);
      chart.appendChild(svg);
      const total=data.reduce((s,[_w,c])=>s+c,0)||1;
      let ang=0;
      data.forEach(([w,c],i)=>{
        const slice=2*Math.PI*(c/total);
        const x1=cx + r*Math.cos(ang), y1=cy + r*Math.sin(ang);
        const x2=cx + r*Math.cos(ang+slice), y2=cy + r*Math.sin(ang+slice);
        const large=slice>Math.PI?1:0;
        const path=document.createElementNS(svgNS,'path');
        const d=`M ${cx} ${cy} L ${x1} ${y1} A ${r} ${r} 0 ${large} 1 ${x2} ${y2} Z`;
        path.setAttribute('d', d);
        path.setAttribute('fill', `hsl(${(i*360/data.length)|0} 70% 55%)`);
        path.setAttribute('opacity','0.95');
        svg.appendChild(path);
      });
    }

    // 生成“技术解读”
    function buildInsight(text, counts){
      const stop=new Set(['的','了','在','是','和','啊','嗯','呢','吧','就','也','还','我们','你们','他们','以及','一个','这个','那个','对于','如果','因为','所以','然后','the','a','an','to','of','and','or','for','in','on','with','is','are','be']);
      const topWords=Object.entries(counts).filter(([w])=>!stop.has(w)&&w.length>1).sort((a,b)=>b[1]-a[1]).slice(0,20).map(([w])=>w);
      const sents=text.split(/(?<=[。！？.!?])\s*/).map(s=>s.trim()).filter(Boolean);
      const hint=/(目标|结论|用于|实现|提供|方案|目的|定位|支持|能力|总结|综述)/;
      let one=sents.find(x=>hint.test(x))||sents[0]||'—';
      one=simplify(one);
      const scored=sents.map(s=>({s,score:topWords.reduce((a,w)=>a+(s.includes(w)?1:0),0)})).filter(x=>x.score>0).sort((a,b)=>b.score-a.score).map(x=>simplify(x.s));
      const bullets=dedup(scored).slice(0,5);
      const dict={
        'API':'应用程序接口，用于模块间调用',
        'RAG':'检索增强生成：先检索再生成',
        'Embedding':'把文本转向量做相似度检索',
        '向量数据库':'存储/检索向量，支持相似度搜索',
        'QPS':'每秒请求数，衡量吞吐',
        '延迟':'从请求到响应的时间'
      };
      const terms=Object.keys(dict).filter(k=>text.toLowerCase().includes(k.toLowerCase())).map(k=>({term:k,explain:dict[k]}));
      let next='';
      if(/(部署|上线|集成|接入|调用)/.test(text)) next='确定接入与鉴权，做一次最小可用验证';
      else if(/(性能|延迟|吞吐|QPS)/.test(text)) next='设定指标并做一次小压测';
      else if(/(需求|范围|方案|评审)/.test(text)) next='收敛范围，出一页纸方案并评审';
      else next='记录要点，必要时再跟进';
      const prob=sents.find(s=>/(问题|痛点|风险|难点|瓶颈)/.test(s))||'';
      return {oneLiner:one, problem:simplify(prob), bullets:bullets.length?bullets:[one], terms, next};

      function simplify(s){ const t=s.replace(/\s+/g,' ').trim(); return t.length>120? t.slice(0,118)+'…':t; }
      function dedup(arr){ const set=new Set(); return arr.filter(x=>{ const k=x.replace(/[，。,\.\!！?？\s]/g,''); if(set.has(k)) return false; set.add(k); return true; }); }
    }
  });
</script>
</body>
</html>
```

### 使用说明
- 默认只显示“技术解读”。想看图，就勾选“显示图表”，并在“条形图/饼图”间切换。
- 不研究代码也能用：一键粘贴、一键分析，图表只是可选的“大神观感”。
