1. 点击 "+ New integration"
1. 命名：UID9622-Integration
1. 复制 "Internal Integration Token"
### 2. 创建 .env 文件
```bash
# .env 文件（🔴 绝对不能分享！）
NOTION_API_KEY=secret_xxxxxxxxxxxxxxxxxxxx

```
### 3. 分享数据库给Integration
在Notion数据库页面右上角：
- 点击 "..." → "Connections" → 选择你的Integration

## 💻 完整代码示例
### 基础连接测试
```python
# notion_basic.py
from notion_client import Client
from dotenv import load_dotenv
import os

# 加载环境变量
load_dotenv()

# 初始化客户端
notion = Client(auth=os.getenv("NOTION_API_KEY"))

# 测试连接
try:
    me = notion.users.me()
    print(f"✅ 连接成功！当前用户：{me['name']}")
except Exception as e:
    print(f"❌ 连接失败：{e}")
```
### 查询数据库
```python
# 查询数据库所有页面
def query_database(database_id):
    try:
        results = notion.databases.query(
            database_id=database_id
        )
        
        print(f"📊 找到 {len(results['results'])} 条记录")
        
        for page in results['results']:
            # 获取标题属性（假设标题属性名为"Name"）
            title_prop = page['properties'].get('Name', {})
            if title_prop.get('title'):
                title = title_prop['title'][0]['plain_text']
                print(f"- {title}")
        
        return results
    except Exception as e:
        print(f"❌ 查询失败：{e}")
        return None

# 使用示例
database_id = os.getenv("NOTION_DATABASE_ID")
query_database(database_id)
```
### 创建页面
```python
# 在数据库中创建新页面
def create_page(database_id, title, content=""):
    try:
        new_page = notion.pages.create(
            parent={"database_id": database_id},
            properties={
                "Name": {  # 标题属性名（根据你的数据库调整）
                    "title": [
                        {
                            "text": {
                                "content": title
                            }
                        }
                    ]
                }
            },
            children=[
                {
                    "object": "block",
                    "type": "paragraph",
                    "paragraph": {
                        "rich_text": [
                            {
                                "type": "text",
                                "text": {"content": content}
                            }
                        ]
                    }
                }
            ]
        )
        
        print(f"✅ 页面创建成功！ID: {new_page['id']}")
        return new_page
    except Exception as e:
        print(f"❌ 创建失败：{e}")
        return None

# 使用示例
create_page(
    database_id=os.getenv("NOTION_DATABASE_ID"),
    title="测试页面",
    content="这是通过API创建的测试内容"
)
```
### 更新页面属性
```python
# 更新页面的属性
def update_page(page_id, new_title=None, new_status=None):
    try:
        properties = {}
        
        if new_title:
            properties["Name"] = {
                "title": [{"text": {"content": new_title}}]
            }
        
        if new_status:
            properties["Status"] = {
                "status": {"name": new_status}
            }
        
        updated_page = notion.pages.update(
            page_id=page_id,
            properties=properties
        )
        
        print(f"✅ 页面更新成功！")
        return updated_page
    except Exception as e:
        print(f"❌ 更新失败：{e}")
        return None
```
### 完整工具类（推荐使用）
```python
# notion_helper.py
from notion_client import Client
from dotenv import load_dotenv
import os
from typing import Optional, Dict, List

class NotionHelper:
    """UID9622 Notion API工具类"""
    
    def __init__(self):
        load_dotenv()
        self.client = Client(auth=os.getenv("NOTION_API_KEY"))
        self.database_id = os.getenv("NOTION_DATABASE_ID")
    
    def test_connection(self) -> bool:
        """测试API连接"""
        try:
            me = self.client.users.me()
            print(f"✅ 连接成功！用户：{me['name']}")
            return True
        except Exception as e:
            print(f"❌ 连接失败：{e}")
            return False
    
    def query_all(self, database_id: Optional[str] = None) -> List[Dict]:
        """查询数据库所有记录"""
        db_id = database_id or self.database_id
        results = []
        has_more = True
        start_cursor = None
        
        while has_more:
            response = self.client.databases.query(
                database_id=db_id,
                start_cursor=start_cursor
            )
            results.extend(response['results'])
            has_more = response['has_more']
            start_cursor = response.get('next_cursor')
        
        return results
    
    def create_simple_page(self, title: str, content: str = "") -> Optional[Dict]:
        """创建简单页面"""
        try:
            return self.client.pages.create(
                parent={"database_id": self.database_id},
                properties={
                    "Name": {
                        "title": [{"text": {"content": title}}]
                    }
                },
                children=[
                    {
                        "object": "block",
                        "type": "paragraph",
                        "paragraph": {
                            "rich_text": [{"text": {"content": content}}]
                        }
                    }
                ]
            )
        except Exception as e:
            print(f"❌ 创建失败：{e}")
            return None

# 使用示例
if __name__ == "__main__":
    helper = NotionHelper()
    helper.test_connection()
    
    # 查询所有记录
    pages = helper.query_all()
    print(f"📊 共找到 {len(pages)} 条记录")
    
    # 创建新页面
    new_page = helper.create_simple_page(
        title="API测试页面",
        content="这是通过UID9622工具类创建的"
    )
```

## 🐛 常见错误与解决
### 错误1：Unauthorized
```javascript
APIResponseError: Could not find database with ID
```
原因： Integration没有访问数据库的权限
解决：
1. 打开Notion数据库页面
1. 右上角 "..." → "Connections"
1. 添加你的Integration
### 错误2：object_not_found
```javascript
APIResponseError: object_not_found
```
原因： DATABASE_ID 或 PAGE_ID 错误
解决：
- 从URL复制正确的ID
- 格式：https://notion.so/xxxxxxxxxxxxxxxx（32位字符）
### 错误3：validation_error
```javascript
APIResponseError: body failed validation
```
原因： 属性名称不匹配
解决：
1. 检查数据库的实际属性名（区分大小写）
1. 使用 notion.databases.retrieve(database_id) 查看schema
