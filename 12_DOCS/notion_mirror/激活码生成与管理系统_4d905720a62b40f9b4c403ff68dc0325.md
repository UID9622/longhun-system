## 🔧 生成机制
### 自动生成脚本
Python脚本（可直接运行）：
```python
import hashlib
import datetime
import secrets
import string

class MetaverseActivationCodeGenerator:
    """元宇宙激活码生成器"""
    
    def __init__(self):
        self.prefix = "UID9622-METAVERSE"
        self.generated_codes = set()
    
    def generate_code(self, user_email=None):
        """生成单个激活码"""
        # 获取当前日期（北京时间）
        beijing_date = datetime.datetime.now(
            datetime.timezone(datetime.timedelta(hours=8))
        ).strftime("%Y%m%d")
        
        # 生成6位随机码
        random_code = ''.join(
            secrets.choice(string.ascii_uppercase + string.digits)
            for _ in range(6)
        )
        
        # 组合完整激活码
        activation_code = f"{self.prefix}-{beijing_date}-{random_code}"
        
        # 生成验证哈希
        if user_email:
            verify_hash = hashlib.sha256(
                f"{activation_code}-{user_email}".encode()
            ).hexdigest()[:16]
        else:
            verify_hash = hashlib.sha256(
                activation_code.encode()
            ).hexdigest()[:16]
        
        # 防止重复
        while activation_code in self.generated_codes:
            random_code = ''.join(
                secrets.choice(string.ascii_uppercase + string.digits)
                for _ in range(6)
            )
            activation_code = f"{self.prefix}-{beijing_date}-{random_code}"
        
        self.generated_codes.add(activation_code)
        
        return {
            "code": activation_code,
            "verify_hash": verify_hash,
            "generated_at": datetime.datetime.now().isoformat(),
            "user_email": user_email
        }
    
    def generate_batch(self, count, prefix_info=None):
        """批量生成激活码"""
        codes = []
        for i in range(count):
            code_info = self.generate_code()
            code_info["batch_id"] = prefix_info or f"BATCH-{datetime.datetime.now().strftime('%Y%m%d')}"
            code_info["sequence"] = i + 1
            codes.append(code_info)
        return codes

# 使用示例
if __name__ == "__main__":
    generator = MetaverseActivationCodeGenerator()
    
    # 生成单个激活码
    code = generator.generate_code(user_email="user@example.com")
    print(f"激活码: {code['code']}")
    print(f"验证哈希: {code['verify_hash']}")
    
    # 批量生成10个激活码
    batch_codes = generator.generate_batch(10, prefix_info="PUBLIC-BETA")
    for c in batch_codes:
        print(f"{c['sequence']}. {c['code']}")
```
