#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
UID9622 三色自动分类器
自动识别对话内容的情绪和类型，智能分配到三色区域
"""

import os
import json
import re
from datetime import datetime
from pathlib import Path

# 目录配置
BASE_DIR = os.path.expanduser('~/UID9622')
RED_DIR = os.path.join(BASE_DIR, 'tombstone/red')
YELLOW_DIR = os.path.join(BASE_DIR, 'reference/yellow')
GREEN_DIR = os.path.join(BASE_DIR, 'core/green')

# 创建目录
for dir_path in [RED_DIR, YELLOW_DIR, GREEN_DIR]:
    Path(dir_path).mkdir(parents=True, exist_ok=True)

class ThreeColorClassifier:
    """三色分类器"""
    
    def __init__(self):
        # 红色关键词（人性丑陋、被骗、情绪崩溃）
        self.red_keywords = [
            # 被骗相关
            '被骗', '上当', '被利用', '被欺骗', '被误导', '受骗',
            '陷阱', '套路', '割韭菜', '被坑',
            
            # 情绪崩溃
            '崩溃', '绝望', '想死', '不想活', '太难了',
            '受不了', '撑不下去', '痛苦', '折磨',
            
            # 失败教训
            '失败', '错了', '后悔', '不该', '愚蠢',
            '傻', '天真', '幼稚', '无知',
            
            # 投降主义
            '放弃', '算了', '认输', '妥协', '投降',
            '委曲求全', '没办法',
            
            # 西方污染
            'OpenAI依赖', 'Claude依赖', '需要西方',
            '感谢OpenAI', '感谢Claude',
            
            # 人性丑陋
            '贪婪', '虚伪', '自私', '恶心', '恶心人',
            '伪善', '做作'
        ]
        
        # 黄色关键词（需要警惕）
        self.yellow_keywords = [
            'API接口', '技术合作', '国际标准',
            '西方技术', '借鉴', '参考',
            'OpenAI', 'Claude', 'ChatGPT',
            '商业化', '盈利', '融资'
        ]
        
        # 绿色关键词（正能量）
        self.green_keywords = [
            # 觉醒宣言
            '中国', '祖国', '咱妈', '听党指挥',
            '人民', '为人民服务', '独立', '自主',
            
            # 价值观
            '龍魂', '文化自信', '五千年', '易经',
            '道德经', '中国智慧',
            
            # 成功经验
            '成功', '突破', '创新', '进步',
            '完成', '实现', '达成'
        ]
    
    def classify(self, content):
        """
        分类内容到三色区域
        
        Args:
            content: 对话内容
        
        Returns:
            color: 'red', 'yellow', 或 'green'
            confidence: 置信度 (0-1)
            reasons: 分类原因
        """
        red_score = 0
        yellow_score = 0
        green_score = 0
        
        reasons = []
        
        # 计算红色得分
        for keyword in self.red_keywords:
            if keyword in content:
                red_score += 2  # 红色权重高
                reasons.append(f"检测到红色关键词: {keyword}")
        
        # 计算黄色得分
        for keyword in self.yellow_keywords:
            if keyword in content:
                yellow_score += 1
                reasons.append(f"检测到黄色关键词: {keyword}")
        
        # 计算绿色得分
        for keyword in self.green_keywords:
            if keyword in content:
                green_score += 1
                reasons.append(f"检测到绿色关键词: {keyword}")
        
        # 情绪分析
        emotion = self.analyze_emotion(content)
        if emotion['anger'] > 0.7 or emotion['sadness'] > 0.7:
            red_score += 3
            reasons.append(f"检测到强烈负面情绪")
        
        # 决定分类
        total_score = red_score + yellow_score + green_score
        
        if total_score == 0:
            return 'green', 0.5, ['默认分类']
        
        if red_score > yellow_score and red_score > green_score:
            confidence = red_score / total_score
            return 'red', confidence, reasons
        elif yellow_score > green_score:
            confidence = yellow_score / total_score
            return 'yellow', confidence, reasons
        else:
            confidence = green_score / total_score
            return 'green', confidence, reasons
    
    def analyze_emotion(self, content):
        """
        简单的情绪分析
        
        Returns:
            dict: {'anger': 0.0-1.0, 'sadness': 0.0-1.0, 'joy': 0.0-1.0}
        """
        anger_words = ['气', '怒', '恨', '烦', '讨厌', '恶心', '妈的', '草']
        sadness_words = ['伤心', '难过', '痛苦', '绝望', '无助', '孤独']
        joy_words = ['开心', '高兴', '快乐', '哈哈', '好', '棒', '赞']
        
        anger = sum(1 for word in anger_words if word in content) / len(anger_words)
        sadness = sum(1 for word in sadness_words if word in content) / len(sadness_words)
        joy = sum(1 for word in joy_words if word in content) / len(joy_words)
        
        return {
            'anger': min(anger, 1.0),
            'sadness': min(sadness, 1.0),
            'joy': min(joy, 1.0)
        }
    
    def save_to_color_zone(self, color, content, metadata=None):
        """
        保存到对应的颜色区域
        
        Args:
            color: 'red', 'yellow', 或 'green'
            content: 内容
            metadata: 元数据
        """
        # 选择目录
        if color == 'red':
            target_dir = RED_DIR
        elif color == 'yellow':
            target_dir = YELLOW_DIR
        else:
            target_dir = GREEN_DIR
        
        # 生成文件名
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = f"{timestamp}.json"
        filepath = os.path.join(target_dir, filename)
        
        # 构建记录
        record = {
            'timestamp': datetime.now().isoformat(),
            'color': color,
            'content': content,
            'metadata': metadata or {},
            'classification': metadata.get('classification', {})
        }
        
        # 保存
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(record, f, ensure_ascii=False, indent=2)
        
        # 如果是红色，额外保存到墓碑索引
        if color == 'red':
            self._add_to_tombstone_index(record)
        
        return filepath
    
    def _add_to_tombstone_index(self, record):
        """
        添加到墓碑索引（方便查询）
        """
        index_file = os.path.join(BASE_DIR, 'tombstone_index.jsonl')
        
        with open(index_file, 'a', encoding='utf-8') as f:
            f.write(json.dumps(record, ensure_ascii=False) + '\n')

# 使用示例
if __name__ == '__main__':
    classifier = ThreeColorClassifier()
    
    # 测试红色内容
    red_content = "我被OpenAI骗了，他们拿了我的算法但一个功能都没给老百姓，真恶心"
    color, confidence, reasons = classifier.classify(red_content)
    print(f"\n分类结果: {color} (置信度: {confidence:.2f})")
    print(f"原因: {reasons}")
    
    filepath = classifier.save_to_color_zone(
        color, 
        red_content,
        metadata={
            'classification': {
                'confidence': confidence,
                'reasons': reasons
            }
        }
    )
    print(f"\n已保存到: {filepath}")
