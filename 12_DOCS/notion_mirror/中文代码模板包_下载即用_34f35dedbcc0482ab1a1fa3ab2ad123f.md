```sql

```
### 按条件筛选
```sql

WHERE 字段名 = "值"
ORDER BY 创建时间 DESC
```
### 统计数量
```sql

FROM "数据源URL"
```
### 分组统计
```sql

FROM "数据源URL"
GROUP BY 字段名
ORDER BY 数量 DESC
```
