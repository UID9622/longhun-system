### 1️⃣ 全球税收管辖权四要素判定
```python
# 税收管辖权判定引擎（原型）
def tax_jurisdiction(buyer_country, seller_country, goods_type, transaction_location):
    """
    四要素判定：买方国籍、卖方国籍、商品类型、交易发生地
    返回：应纳税国家列表 + 税率 + 分配比例
    """
    jurisdiction = []
    
    # 规则1：实物商品 → 交易发生地优先
    if goods_type == "physical":
        jurisdiction.append({
            "country": transaction_location,
            "tax_type": "增值税/消费税",
            "priority": 1
        })
    
    # 规则2：数字服务 → 买方所在地优先（OECD Pillar One原则）
    elif goods_type == "digital_service":
        jurisdiction.append({
            "country": buyer_country,
            "tax_type": "数字服务税",
            "priority": 1
        })
    
    # 规则3：跨境交易 → 双重课税协调
    if buyer_country != seller_country:
        jurisdiction.append({
            "country": seller_country,
            "tax_type": "所得税/预提税",
            "priority": 2,
            "note": "需检查双边税收协定"
        })
    
    return jurisdiction
```
### 2️⃣ 数字人民币母子合约自动分账
```python
# 母子合约分账引擎
def auto_split_payment(total_amount, tax_jurisdictions):
    """
    母合约：总交易额
    子合约：自动拆分为 商家收入 + 各国税款 + 系统服务费
    """
    splits = {
        "merchant": 0,      # 商家到手
        "taxes": {},         # 各国税款
        "system_fee": 0      # 系统维护（极低）
    }
    
    # 系统服务费：0.1%（仅用于服务器维护，不盈利）
    splits["system_fee"] = total_amount * 0.001
    remaining = total_amount - splits["system_fee"]
    
    # 按管辖权分配税款
    total_tax = 0
    for j in tax_jurisdictions:
        tax = remaining * j["tax_rate"]
        splits["taxes"][j["country"]] = tax
        total_tax += tax
    
    # 商家到手 = 总额 - 服务费 - 税款
    splits["merchant"] = remaining - total_tax
    
    return splits
```
