# 三色审计机制
def three_color_audit(operation):
    """
    🟢 绿色：自动通过（安全操作）
    🟡 黄色：需要审计（敏感操作）
    🔴 红色：需要批准（高风险操作）
    """
    
    # 绿色：日常操作，自动通过
    if operation.type in ["查询", "搜索", "阅读"]:
        return "🟢 自动通过"
    
    # 黄色：敏感操作，记录审计
    elif operation.type in ["创建", "更新", "删除"]:
        audit_log.record(operation)
        notify_audit_team(operation)
        return "🟡 已审计"
    
    # 红色：高风险操作，需要批准
    elif operation.type in ["支付", "授权", "导出数据"]:
        approval = request_approval(operation)
        if approval:
            audit_log.record(operation)
            return "🔴 已批准"
        else:
            return "🔴 已拒绝"
