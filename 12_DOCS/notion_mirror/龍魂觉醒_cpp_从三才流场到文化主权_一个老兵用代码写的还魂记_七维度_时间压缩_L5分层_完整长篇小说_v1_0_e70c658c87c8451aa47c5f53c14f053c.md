        if (n <= 0) return 9;
        return 1 + ((n - 1) % 9);
    }

    /*

     * 力气不浪费，方向不走偏。
     */

private:
    float perlinNoise(float x, float y, float t) {
        // 简化版Perlin噪声
        return std::sin(x * 0.1f + t) *
               std::cos(y * 0.1f + t * 0.7f);

        int gridX = static_cast<int>(x / 3) % 3;
        int gridY = static_cast<int>(y / 3) % 3;
        float luoshuVal = luoshu[gridY * 3 + gridX];
        return (luoshuVal / 9.0f) * 2.0f * 3.14159f;
    }
};
