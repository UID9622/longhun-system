## 四、迷宫求解算法 v1.3（可跑代码骨架）
```python
# 诸葛八卦村迷宫求解器 v1.3
# DNA: #龍芯⚡️2026-06-09-BAGUA-VILLAGE-MAZE-SOLVER-v1.3
# 核心：梯度下降回不动点 + 后天卦位定向 + 三感齐到收敛

LUOSHU = [[4,9,2],[3,5,7],[8,1,6]]   # 洛书九宫地场
ZHONGCHI = (1, 1)                     # 钟池=中宫5=不动点

# 后天八卦×洛书方位映射（修正版）
GUA_BEARING = {
    (0,1): ("坎", "N"),   (2,1): ("离", "S"),
    (1,0): ("震", "E"),   (1,2): ("兑", "W"),
    (0,0): ("巽", "SE"),  (2,2): ("乾", "NW"),
    (2,0): ("艮", "NE"),  (0,2): ("坤", "SW"),
}

def gradient_descent(pos):
    """第一步：往低走+跟水声 → 梯度下降回钟池"""
    path = [pos]
    x, y = pos
    while (x, y) != ZHONGCHI:
        x += (1 - x) // max(abs(1 - x), 1) if x != 1 else 0
        y += (1 - y) // max(abs(1 - y), 1) if y != 1 else 0
        path.append((x, y))
    return path

def recognize_gua(target_bearing):
    """第二步：从钟池认准目标卦位方位 → 定向"""
    for cell, (gua, bearing) in GUA_BEARING.items():
        if bearing == target_bearing:
            return cell
    raise ValueError("非本村方位")

def solve_maze(start_pos, target_bearing):
    """完整解法：迷路位置 → 目标方位"""
    step1 = gradient_descent(start_pos)         # 先回钟池
    target_cell = recognize_gua(target_bearing)  # 再认卦位
    step2 = [ZHONGCHI, target_cell]              # 中宫→目标巷
    return step1 + step2[1:]

# 示例
if __name__ == "__main__":
    路线 = solve_maze((0, 0), "E")  # 从西北角迷路，要去东方
    print("破阵路线（两步）:", 路线)
    # → [(0,0), ..., (1,1), (1,2)]·必过中宫
```
