## 公式
```javascript
🟢 置信度 > 0.90  → 通行，审计链留痕
🟡 0.50 ~ 0.90 → 待审，延迟2h判定
🔴 置信度 < 0.50  → 熔断，留证据链 + 可申诉

带操纵置信度调整：
  气话识别 → +0.15（情绪能量补偿）
  反讽识别 → +0.20（反讽不是违规）
  水军标记 → -0.30（水军和尔不山）
```
```c++
// 三色审计置信度算法 v1.0
// 龍魂审核过滤系统的核心判定引擎
// 编译: clang++ -std=c++17 -o tricolor tricolor.cpp

#include <string>
#include <vector>
#include <algorithm>
#include <iostream>
#include <cmath>

enum class AuditColor {
    GREEN,   // 置信度>0.90
    YELLOW,  // 0.50~0.90
    RED      // <0.50
};

struct AuditInput {
    std::string text;           // 原始内容
    std::string context;        // 上下文（最近10条拼接）
    bool   is_sarcasm;          // 是否识别为反讽
    bool   is_emotional;        // 是否识别为气话
    bool   is_bot;              // 是否水军标记
    double base_confidence;     // 内容模型基础置信度 [0,1]
    std::string user_history;   // 用户历史类别
};

struct AuditResult {
    AuditColor color;
    double confidence;          // 最终置信度
    double adjustment;          // 调整量
    std::string reason;         // 判定原因
    std::string action;         // 建议动作
    std::string appeal_id;      // 申诉ID（黄和红自动生成）
};

AuditResult tricolorAudit(const AuditInput& inp) {
    AuditResult r;
    double conf = inp.base_confidence;
    double adj  = 0.0;

    // 调整置信度
    if (inp.is_sarcasm)   { adj += 0.20; } // 反讽不是违规，提升置信度
    if (inp.is_emotional) { adj += 0.15; } // 气话有缓冲，提升
    if (inp.is_bot)       { adj -= 0.30; } // 水军抹黑，降低置信度

    // 历史加成
    if (inp.user_history == "high_quality") adj += 0.10;
    if (inp.user_history == "risky")        adj -= 0.15;

    conf = std::max(0.0, std::min(1.0, conf + adj));
    r.confidence = conf;
    r.adjustment = adj;

    // 三色判定
    if (conf >= 0.90) {
        r.color  = AuditColor::GREEN;
        r.reason = "置信度超过90%·内容安全";
        r.action = "通行，审计链留痕";
    } else if (conf >= 0.50) {
        r.color     = AuditColor::YELLOW;
        r.reason    = "置信度50%~90%，内容存疑";
        r.action    = "延迟2h判定 + 人工/公证团复核";
        r.appeal_id = "APP-" + inp.text.substr(0, 6) + "-" + std::to_string((int)(conf*100));
    } else {
        r.color     = AuditColor::RED;
        r.reason    = "置信度低于50%，高危内容";
        r.action    = "留证据链，不直接处罚，自动生成申诉ID";
        r.appeal_id = "APP-RED-" + std::to_string((int)(conf*100));
    }

    return r;
}

std::string colorName(AuditColor c) {
    switch(c) {
        case AuditColor::GREEN:  return "🟢 置信度>90%·通行";
        case AuditColor::YELLOW: return "🟡 置信度50%~90%·待审";
        case AuditColor::RED:    return "🔴 置信度<50%·熔断";
    }
    return "";
}

int main() {
    // 测试场景1：正常内容
    AuditResult r1 = tricolorAudit({"hello world", "", false, false, false, 0.95, "high_quality"});
    std::cout << "正常内容: " << colorName(r1.color) << " conf=" << r1.confidence << "\n";

    // 测试场景2：气话（意气山河复山古）
    AuditResult r2 = tricolorAudit({"\u6df7蛋"", "", false, true, false, 0.55, "normal"});
    std::cout << "气话内容: " << colorName(r2.color) << " conf=" << r2.confidence
              << " (adj=+" << r2.adjustment << ")\n";

    // 测试场景3：水军标记内容
    AuditResult r3 = tricolorAudit({"spam text", "", false, false, true, 0.7, "risky"});
    std::cout << "水军内容: " << colorName(r3.color) << " conf=" << r3.confidence
              << " appealID=" << r3.appeal_id << "\n";

    return 0;
}
```
## 局限与注昐
> base_confidence 通常来自 NLP 模型的输出。气话和反讽识别应由轻量识别模型运行结果提供，并非灵魔数字。
## 铁律：申诉通道
> 任何 🟡 或 🔴 判定必须自动生成申诉ID。没有申诉通道 = 数字狱警。申诉是权利，不是恩赐。
