### 2. Notion同步引擎
```python
class NotionSyncEngine:
    def __init__(self, notion_token: str):
        self.client = Client(auth=notion_token)
        self.database_configs = {
            'activation_codes': 'your_database_id_here',
            'user_requirements': 'requirements_db_id',
            'usage_analytics': 'analytics_db_id'
        }

    def sync_activation_code(self, code_data: dict):
        """同步激活码到Notion"""
        properties = {
            "激活码": {"title": [{"text": {"content": code_data['activation_code']}}]},
            "用户需求": {"rich_text": [{"text": {"content": code_data['requirement_text']}}]},
            "技术级别": {"select": {"name": code_data['technical_level']}},
            "状态": {"status": {"name": code_data['status']}},
            "创建时间": {"date": {"start": code_data['created_at'].isoformat()}},
            "功能权限": {"multi_select": [{"name": f} for f in code_data['feature_set']]},
            "使用次数": {"number": code_data['usage_count']},
            "优先级": {"select": {"name": str(code_data['priority_score'])}}
        }

        return self.client.pages.create(
            parent={"database_id": self.database_configs['activation_codes']},
            properties=properties
        )
```

```python
class RecommendationEngine:
    def __init__(self):
        self.feature_matrix = {
            'BEGINNER': ['basic_templates', 'guided_setup', 'simple_ai'],
            'INTERMEDIATE': ['custom_databases', 'automation', 'ai_personas'],
            'ADVANCED': ['api_integration', 'complex_workflows', 'custom_ai']
        }

    def recommend_features(self, requirements: dict) -> list:
        """基于需求推荐功能"""
        base_features = self.feature_matrix.get(
            requirements['technical_level'],
            self.feature_matrix['BEGINNER']
        )

        # 根据使用场景添加特殊功能
        if 'project_management' in requirements['use_cases']:
            base_features.extend(['task_tracking', 'team_collaboration'])

        if 'creative_work' in requirements['use_cases']:
            base_features.extend(['ai_writing', 'inspiration_collection'])

        return list(set(base_features))

    def calculate_upgrade_probability(self, usage_data: dict) -> float:
        """计算升级可能性"""
        factors = {
            'high_usage_frequency': usage_data.get('daily_usage', 0) > 5,
            'feature_boundary_hits': usage_data.get('limit_hits', 0) > 3,
            'advanced_features_interest': len(usage_data.get('explored_features', [])) > 10,
            'collaboration_needs': usage_data.get('share_attempts', 0) > 0
        }

        score = sum([0.25 for condition in factors.values() if condition])
        return min(score, 1.0)
```

```python
class UID9622ActivationWorkflow:
    def __init__(self):
        self.collector = ChatGPTCollector()
        self.sync_engine = NotionSyncEngine(NOTION_TOKEN)
        self.recommender = RecommendationEngine()

    async def process_user_request(self, user_message: str, session_id: str):
        """完整处理用户请求的工作流"""

        # 1. 需求收集与分析
        requirements = self.collector.collect_requirements(user_message)

        # 2. 生成定制化激活码
        activation_code = self.collector.generate_activation_code(requirements)

        # 3. 推荐功能配置
        recommended_features = self.recommender.recommend_features(requirements)

        # 4. 创建数据库记录
        code_record = {
            'activation_code': activation_code,
            'user_id': session_id,
            'requirement_text': user_message,
            'technical_level': requirements['technical_level'],
            'feature_set': recommended_features,
            'access_level': self.determine_access_level(requirements),
            'status': 'ACTIVE',
            'created_at': datetime.now()
        }

        # 5. 同步到Notion
        notion_page = await self.sync_engine.sync_activation_code(code_record)

        # 6. 生成用户响应
        response = self.generate_user_response(code_record)

        return {
            'activation_code': activation_code,
            'recommended_features': recommended_features,
            'notion_page_id': notion_page['id'],
            'user_response': response
        }

    def generate_user_response(self, code_record: dict) -> str:
        """生成给用户的回复"""
        return f"""

        ⏰ **有效期**: 永久有效
        💬 **技术支持**: 30天内可随时调整配置

        需要升级功能请联系 UID9622 系统管理员
```
