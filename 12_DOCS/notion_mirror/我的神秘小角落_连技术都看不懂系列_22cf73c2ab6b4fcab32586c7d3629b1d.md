### 公式1：活跃度计算
```javascript
// 【技术版】
function 活跃度(最后活跃日期) {
  let 天数差 = 今天 - 最后活跃日期;
  if (天数差 < 7) return "🔥 高频";
  if (天数差 < 30) return "✅ 正常";
  if (天数差 < 90) return "⚠️ 低频";
  return "❌ 休眠";
}

// 【大白话版】
// 7天内有动静 = 很活跃

// 三个月内有动静 = 有点懒
// 超过三个月没动静 = 睡着了
```
