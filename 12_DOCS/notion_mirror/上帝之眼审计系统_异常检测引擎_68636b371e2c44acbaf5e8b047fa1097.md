```python
        # 特征工程
        features = self._extract_features(event)
        
        # 训练/更新模型（每100个事件更新一次）
        if len(self.event_history) % 100 == 0:

            self.anomaly_detector.fit(historical_features)
        
        # 预测异常分数
        anomaly_score = self.anomaly_detector.score_samples([features])[0]
        # 转换为0-1范围（分数越低越异常）

        base_score = 0

            '敏感操作': 30,

            base_score += violation_weights.get(violation, 10)

    def _generate_event_id(self, event: AuditEvent) -> str:
        """生成唯一事件ID"""
        data = f"{event.user_id}{event.action}{event.timestamp.isoformat()}"
        return hashlib.sha256(data.encode()).hexdigest()[:16]
    
    def _trigger_alert(self, event: AuditEvent, level: str, details: Dict):
        """触发告警"""
        alert = {
            'timestamp': datetime.now(),
            'level': level,
            'event': event,
            'details': details,
            'message': f"🚨 {level}级别告警：用户{event.user_id}执行{event.action}，风险评分{event.risk_score}"
        }
        
        # 调用所有注册的告警回调
        for callback in self.alert_callbacks:
            callback(alert)
    
    def register_alert_callback(self, callback):
        """注册告警回调函数"""
        self.alert_callbacks.append(callback)

        """生成用户风险画像"""
        user_events = [e for e in self.event_history if e.user_id == user_id]

        violations = [v for e in user_events

            'total_events': len(user_events),

        }
    
    def _get_top_items(self, items: List, n: int) -> List:
        """获取最常见的N个项目"""
        from collections import Counter
        return [item for item, count in Counter(items).most_common(n)]
    
    def _calculate_risk_trend(self, events: List[AuditEvent]) -> str:
        """计算风险趋势"""
        if len(events) < 10:
            return '数据不足'
        
        recent_risk = np.mean([e.risk_score for e in events[-10:]])
        older_risk = np.mean([e.risk_score for e in events[-20:-10]])
        
        if recent_risk > older_risk * 1.2:
            return '⬆️ 上升'
        elif recent_risk < older_risk * 0.8:
            return '⬇️ 下降'
```

```python
            return '仅记录'
    
    def _generate_event_id(self, event: AuditEvent) -> str:
        """生成唯一事件ID"""
        data = f"{event.user_id}{event.action}{event.timestamp.isoformat()}"
        return hashlib.sha256(data.encode()).hexdigest()[:16]
    
    def _trigger_alert(self, event: AuditEvent, level: str, details: Dict):
        """触发告警"""
        alert = {
            'timestamp': datetime.now(),
            'level': level,
            'event': event,
            'details': details,
            'message': f"🚨 {level}级别告警：用户{event.user_id}执行{event.action}，风险评分{event.risk_score}"
        }
        
        for callback in self.alert_callbacks:
            callback(alert)
    
    def register_alert_callback(self, callback):
        """注册告警回调函数"""
        self.alert_callbacks.append(callback)
    
    def get_user_risk_profile(self, user_id: str) -> Dict:
        """生成用户风险画像"""
        user_events = [e for e in self.event_history if e.user_id == user_id]
        
        if not user_events:
            return {'error': '无历史数据'}
        
        total_risk = sum(e.risk_score for e in user_events)
        avg_risk = total_risk / len(user_events)
        
        violations = [v for e in user_events 
                     for v in self._check_rules(e).get('violations', [])]
        
        return {
            'user_id': user_id,
            'total_events': len(user_events),
            'average_risk_score': round(avg_risk, 2),
            'total_violations': len(violations),
            'most_common_violations': self._get_top_items(violations, 3),
            'risk_trend': self._calculate_risk_trend(user_events),
            'threat_level': self._determine_alert_level(avg_risk)
        }
    
    def _get_top_items(self, items: List, n: int) -> List:
        """获取最常见的N个项目"""
        return [item for item, count in Counter(items).most_common(n)]
    
    def _calculate_risk_trend(self, events: List[AuditEvent]) -> str:
        """计算风险趋势"""
        if len(events) < 10:
            return '数据不足'
        
        recent_risk = np.mean([e.risk_score for e in events[-10:]])
        older_risk = np.mean([e.risk_score for e in events[-20:-10]])
        
        if recent_risk > older_risk * 1.2:
            return '⬆️ 上升'
        elif recent_risk < older_risk * 0.8:
            return '⬇️ 下降'
        else:
            return '➡️ 平稳'

```
