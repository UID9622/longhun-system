## 🎯 三色审计规则

### 🟢 绿色（正常）

**判断条件：**

- 使用频率正常（<100次/天）
- 无异常行为模式
- 支付金额>=1元
- 激活状态正常

**处理动作：**

- 继续运行
- 正常记录日志
- 不触发通知

---

### 🟡 黄色（预警）

**判断条件：**

- 短时间大量请求（1分钟>100次）
- 异常使用模式（深夜高频）
- IP地址频繁变化
- 设备DNA不匹配

**处理动作：**

- 发送警告提醒
- 限制请求频率
- 记录详细日志
- 通知老大邮箱

**通知邮件：**

### 🔴 红色（熔断）

**判断条件：**

- 恶意刷量（5分钟>500次）

## 💻 熔断代码实现

```jsx
/**
 * LU激活系统 - 三色审计+熔断
 * DNA追溯码：#ZHUGEXIN⚡️2026-01-27-三色审计熔断-v1.0
 */

class ThreeColorAudit {
  constructor() {
    this.usageCache = new Map(); // 使用记录缓存
    this.blacklist = new Set();   // 黑名单
  }
  
  // 检查使用频率
  async checkUsageRate(activationCode) {
    const now = Date.now();
    const usage = this.usageCache.get(activationCode) || [];
    
    // 清理1小时前的记录
    const recentUsage = usage.filter(time => now - time < 3600000);
    
    // 统计频率
    const last1min = recentUsage.filter(time => now - time < 60000).length;
    const last5min = recentUsage.filter(time => now - time < 300000).length;
    
    // 判断等级
    if (last5min > 500) {
      return { level: 'red', reason: `5分钟内请求${last5min}次` };
    }
    if (last1min > 100) {
      return { level: 'yellow', reason: `1分钟内请求${last1min}次` };
    }
    
    return { level: 'green', reason: '使用正常' };
  }
  
  // 记录使用
  recordUsage(activationCode) {
    const usage = this.usageCache.get(activationCode) || [];
    usage.push(Date.now());
    this.usageCache.set(activationCode, usage);
  }
  
  // 熔断处理
  async triggerCircuitBreaker(activationCode, reason) {
    console.log(`🔴 触发熔断：${activationCode}`);
    console.log(`   原因：${reason}`);
    
    // 1. 加入黑名单
    this.blacklist.add(activationCode);
    
    // 2. 写入数据库
    await this.recordToDatabase({
      activationCode,
      breakTime: new Date(),
      reason,
      level: 'red',
      status: '已封禁'
    });
    
    // 3. 发送通知
    await this.sendAlert(activationCode, reason, 'red');
    
    // 4. DNA追溯码
    const dna = `#ZHUGEXIN⚡️${new Date().toISOString().split('T')[0]}-熔断-${activationCode}`;
    console.log(`   DNA追溯码：${dna}`);
    
    return { success: true, dna };
  }
  
  // 检查是否在黑名单
  isBlacklisted(activationCode) {
    return this.blacklist.has(activationCode);
  }
  
  // 发送告警
  async sendAlert(activationCode, reason, level) {
    const emoji = level === 'red' ? '🔴' : '🟡';
    const subject = `${emoji} LU系统${level === 'red' ? '红色熔断' : '黄色预警'}`;
    
    // 这里调用邮件发送函数
    console.log(`发送告警邮件：${subject}`);
  }
  
  // 写入数据库
  async recordToDatabase(data) {
    // 这里调用金山文档API
    console.log('写入熔断记录：', data);
  }
}

// 使用示例
const audit = new ThreeColorAudit();

// 每次LU指令执行前检查
async function executeLUCommand(activationCode, command) {
  // 1. 检查黑名单
  if (audit.isBlacklisted(activationCode)) {
    console.log('❌ 激活码已被封禁');
    return { success: false, error: '激活码已被封禁' };
  }
  
  // 2. 记录使用
  audit.recordUsage(activationCode);
  
  // 3. 检查频率
  const auditResult = await audit.checkUsageRate(activationCode);
  
  // 4. 根据等级处理
  if (auditResult.level === 'red') {
    await audit.triggerCircuitBreaker(activationCode, auditResult.reason);
    return { success: false, error: '触发熔断' };
  }
  
  if (auditResult.level === 'yellow') {
    await audit.sendAlert(activationCode, auditResult.reason, 'yellow');
    console.log('⚠️ 黄色预警，继续执行');
  }
  
  // 5. 执行命令
  console.log(`✅ 执行LU指令：${command}`);
  return { success: true };
}

module.exports = { ThreeColorAudit, executeLUCommand };
```
