def digital_root(n: int) -> int:
    """计算数字根。n 必须为正整数。"""
    if n <= 0:
        raise ValueError("数字根仅定义于正整数")
    return 1 + (n - 1) % 9

def tricolor_gate(n: int) -> tuple[str, str]:
    """三色熔断闸 GATE-01。返回 (颜色, 含义)。"""
    dr = digital_root(n)
    if dr in {3, 9}:
        return "🔴", "熔断"
    elif dr == 6:
        return "🟡", "待审"
    else:
        return "🟢", "通行"

def is_luoshu(matrix: list[list[int]]) -> bool:
    """检查 3×3 矩阵是否为有效洛书（含五行校验）。"""
    flat = sum(matrix, [])
    if sorted(flat) != list(range(1, 10)):
        return False  # 九宫完整性
    target = 15
    rows_ok = all(sum(row) == target for row in matrix)
    cols_ok = all(sum(matrix[r][c] for r in range(3)) == target
                  for c in range(3))
    diag_ok = (matrix[0][0]+matrix[1][1]+matrix[2][2] == target and
               matrix[0][2]+matrix[1][1]+matrix[2][0] == target)
    center_ok = matrix[1][1] == 5  # 中宫必须是5（不动点）
    return rows_ok and cols_ok and diag_ok and center_ok

from dataclasses import dataclass
from typing import Literal

@dataclass
class DecisionCard:
    source: str            # 规则来源
    rule_match: float      # R: 规则匹配度
    importance: float      # I: 重要性（洛书宫位权重）
    risk: float            # 风险分数
    bias: float            # 人性偏置
    alpha_r: float = 0.3
    alpha_h: float = 0.2

    @property
    def score(self) -> float:
        factor = 1 - self.alpha_r * self.risk - self.alpha_h * self.bias
        return round(self.rule_match * self.importance * max(factor, 0), 4)

    @property
    def tricolor(self) -> Literal["🟢", "🟡", "🔴"]:
        if self.score >= 0.8: return "🟢"
        if self.score >= 0.5: return "🟡"
        return "🔴"

    def to_yaml(self, dna: str = "") -> str:
        return f"""决策卡:
  来源: {self.source}
  规则匹配度: {self.rule_match}
  重要性(宫位): {self.importance}
  风险分数: {self.risk}
  人性偏置: {self.bias}
  决策分数: {self.score}
  三色: {self.tricolor}
  建议: {"通过" if self.score>=0.8 else "需确认" if self.score>=0.5 else "拒绝执行，风险过高"}
  DNA: {dna}"""

import hashlib, time, json
from dataclasses import dataclass, field
from pathlib import Path

def generate_dna(content: str, version: str = "v1.0",
                 date: str = "") -> str:
    """生成单条DNA追溯码。"""
    if not date:
        date = time.strftime("%Y-%m-%d")
    sha8 = hashlib.sha256(content.encode("utf-8")).hexdigest()[:8]
    # 内容摘要：取前10个字符，去除特殊符号
    summary = content[:10].replace(" ", "").replace("/","-")
    return f"#龍芯⚡️{date}-{summary}-{version}-{sha8}"

def chain_hash(prev_hash: str, current_dna: str) -> str:
    """链式哈希：防篡改，每条DNA依赖上一条。"""
    combined = (prev_hash + current_dna).encode("utf-8")
    return hashlib.sha256(combined).hexdigest()[:16]

def verify_dna(dna: str) -> dict:

        return {"valid": False, "error": "格式错误：缺少#龍芯⚡️"}

    if len(segments) < 4:
        return {"valid": False, "error": "字段不足"}
    date_str = "-".join(segments[:3])
    sha8 = segments[-1]
    dr = digital_root(int(sha8, 16) % 10000 + 1)
    color, label = tricolor_gate(dr)
    return {

# ── DNA账本（append-only JSONL）──
@dataclass
class DNALedger:
    path: str = "dna_ledger.jsonl"
    _prev_hash: str = field(default="GENESIS", repr=False)

    def append(self, content: str, version: str = "v1.0") -> dict:
        dna = generate_dna(content, version)
        ch = chain_hash(self._prev_hash, dna)
        entry = {
            "dna": dna, "content": content,
            "chain_hash": ch, "prev_hash": self._prev_hash,
            "ts": time.strftime("%Y-%m-%dT%H:%M:%S"),
            "tricolor": verify_dna(dna)["tricolor"]
        }
        Path(self.path).open("a", encoding="utf-8").write(
            json.dumps(entry, ensure_ascii=False) + "\n")
        self._prev_hash = ch
        return entry

class LonghunEngine:
    """
    龍魂决策引擎：六公式一体化封装。
    数字根 → 五行 → 宫位 → 三色 → 决策评分 → DNA → 账本
    """
    def __init__(self, ledger_path: str = "longhun_ledger.jsonl"):
        self.ledger = DNALedger(path=ledger_path)

    def analyze(self, n: int, content: str,
                 risk: float = 0.0, bias: float = 0.0,
                 source: str = "规则库") -> dict:
        """完整分析流程：六公式全走一遍。"""
        # Step 1: 数字根
        dr = digital_root(n)
        # Step 2: 五行映射
        wx = WUXING_MAP[dr]
        palace = PALACE_MAP[dr]
        # Step 3: 三色熔断（GATE-01）
        color, label = tricolor_gate(n)
        # Step 4: 决策评分（宫位决定重要性I）
        importance_map = {5: 1.0, 1: 0.8, 2: 0.75, 4: 0.7,
                          7: 0.7, 8: 0.65, 6: 0.4, 3: 0.0, 9: 0.0}
        I = importance_map.get(dr, 0.6)
        R = 1.0 if color == "🟢" else (0.6 if color == "🟡" else 0.0)
        card = DecisionCard(source=source, rule_match=R,
                            importance=I, risk=risk, bias=bias)
        # Step 5: DNA生成
        dna = generate_dna(content)
        # Step 6: 账本落档（append-only）
        ledger_entry = self.ledger.append(content)

        return {
            "n": n, "dr": dr, "wuxing": wx, "palace": palace,
            "tricolor": color, "gate_label": label,
            "decision_score": card.score,
            "decision_tricolor": card.tricolor,
            "dna": dna,
            "chain_hash": ledger_entry["chain_hash"],
            "decision_card_yaml": card.to_yaml(dna),
            "auto_action": self._auto_action(color, card.tricolor)
        }

    @staticmethod
    def _auto_action(gate_color: str, decision_color: str) -> str:
        if gate_color == "🔴" or decision_color == "🔴":
            return "BLOCKED · 熔断 · 记录日志 · 不执行"
        elif gate_color == "🟡" or decision_color == "🟡":
            return "HOLD · 暂停 · 等待人工确认"
        return "PASS · 执行 · DNA落档完成"
