### 步骤4：依次执行 ⚡

**自动按顺序执行，一步完成自动进入下一步**

```jsx
function 依次执行(执行计划) {
  for (步骤 in 执行计划.步骤列表) {
    // 更新当前步骤
    当前步骤 = 步骤.序号
    
    // 调用负责人格
    人格 = 步骤.负责人
    结果 = 人格.执行(步骤.任务)
    
    // 记录结果
    步骤.结果 = 结果
    步骤.状态 = "✅完成"
    
    // 更新进度
    执行计划.完成步骤 += 1
    
    // 通知老大进度（可选）
    if (步骤.序号 % 2 == 0) {
      通知老大进度(执行计划)
    }
  }
  
  // 全部完成
  执行计划.状态 = "🟢已完成"
  return 执行计划
}
```
