// SwiftUI 星空粒子系统 - ParticleSystem with Canvas
import SwiftUI

struct Star: Identifiable {
    let id = UUID()
    var x: CGFloat
    var y: CGFloat
    var size: CGFloat
    var brightness: Double
    var twinkleSpeed: Double
}

struct StarrySky: View {
    @State private var stars: [Star] = []
    @State private var time: Double = 0
    let timer = Timer.publish(every: 0.05, on: .main, in: .common).autoconnect()

    var body: some View {
        TimelineView(.animation) { timeline in
            Canvas { context, size in
                for star in stars {
                    let twinkle = sin(time * star.twinkleSpeed) * 0.5 + 0.5
                    let opacity = star.brightness * twinkle
                    let path = Path(ellipseIn: CGRect(
                        x: star.x, y: star.y,
                        width: star.size, height: star.size
                    ))
                    context.fill(path, with: .color(.white.opacity(opacity)))
                    context.addFilter(.blur(radius: star.size * 0.3))
                }
            }
        }
        .onAppear { generateStars(in: UIScreen.main.bounds) }
        .background(Color.black)
    }

    func generateStars(in bounds: CGRect) {
        stars = (0..<200).map { _ in
            Star(x: CGFloat.random(in: 0...bounds.width),
                 y: CGFloat.random(in: 0...bounds.height),
                 size: CGFloat.random(in: 0.5...3),
                 brightness: Double.random(in: 0.3...1.0),
                 twinkleSpeed: Double.random(in: 1...5))
        }
    }
}
