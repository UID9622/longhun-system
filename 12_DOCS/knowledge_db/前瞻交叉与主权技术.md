```python
# CompanyRootCheck 协议 v1.0
# 九宫档案系统——给公司/平台建立结构化根信息档案

class CompanyRootCheck:
    """洛书九宫骨架的公司根检查协议"""

    POSITIONS = {
        5: "任务根/主权根/UID9622锚点",  # 中宫·永恒原点
        4: "公司根（注册信息·股权·法人·资质）",
        9: "政策根（隐私政策·服务条款·定价·API限制）",
        2: "舆论表达根（公开声明·博客·社交媒体·承诺）",
        3: "产品生态根（功能列表·版本·弃用计划·兼容性）",
        7: "时间事件根（关键时间线·事故报告·变更记录）",
        8: "开源代码根（开源许可·代码仓库·贡献者·依赖）",
        1: "证据记忆根（用户反馈·第三方审计·截图存档）",
        6: "待审不确定根（未经证实·待核实·灰色地带）",
    }

    def __init__(self, company_name: str, uid9622_anchor: str = "UID9622"):
        self.company = company_name
        self.grid = {pos: {'label': label, 'data': None, 'evidence': []}
                     for pos, label in self.POSITIONS.items()}

    def set_root(self, position: int, data: dict, evidence: list = None):
        """在一个宫位落位数据"""
        if position not in self.POSITIONS:
            raise ValueError(f"无效宫位: {position}")
        self.grid[position]["data"] = data
        self.grid[position]["evidence"] = evidence or []

    def check(self, query_topic: str) -> tuple:
        """对某个话题做根检查，返回(状态, 理由)"""
        filled = sum(1 for g in self.grid.values() if g["data"] is not None)
        conflicts = self._detect_conflicts()
        if filled < 5:
            return ("⚠️", f"根不全({filled}/9宫有数据)，回答将标注边界")
        if conflicts:
            return ("❌", f"发现{len(conflicts)}处根冲突: {conflicts[0]}")
        return ("✅", "根完整·证据充足·可以回答")
```
