import math

w, m, v = 1.0, 0.0, 0.0
beta1, beta2, eps, lr = 0.9, 0.999, 1e-8, 0.1
for t in range(1, 11):
    g = 2*w  # gradient of w^2
    m = beta1*m + (1-beta1)*g
    v = beta2*v + (1-beta2)*g*g
    m_hat = m/(1-beta1**t)

# 知识点ID=97 · 强化学习

import random

Q = {}
alpha, gamma, eps = 0.5, 0.9, 0.2
actions = [0, 1]

def choose(s):
    if random.random() < eps: return random.choice(actions)
    return max(actions, key=lambda a: Q.get((s,a), 0))

for _ in range(100):
    s = random.choice([0,1])
    a = choose(s)
    r = 1 if (s==0 and a==1) or (s==1 and a==0) else 0
    ns = random.choice([0,1])
    Q[(s,a)] = Q.get((s,a),0) + alpha*(r + gamma*max(Q.get((ns,aa),0) for aa in actions) - Q.get((s,a),0))
print(Q)
```

import math

def sigmoid(x): return 1/(1+math.exp(-x))
def generator(z, w): return z*w
def discriminator(x, w): return sigmoid(x*w)

for step in range(5):

# 知识点ID=84 · XGBoost

def gradient_boosting(y, rounds=3, lr=0.3):
    pred = [0.0]*len(y)
    for _ in range(rounds):
        residuals = [yi - pi for yi, pi in zip(y, pred)]
        pred = [pi + lr*ri for pi, ri in zip(pred, residuals)]

# Toy text classifier
train = [("免费 优惠", "spam"), ("会议 明天", "ham"), ("中奖 免费", "spam")]
vocab = Counter()
cls = Counter()
for text, label in train:
    cls[label] += 1
    for w in text.split():
        vocab[(label,w)] += 1

def predict(text):
    words = text.split()
    scores = {l: cls[l]/len(train) for l in cls}
    for l in scores:
        total = sum(vocab[(l,w)] for w,_ in vocab if (l,w) in vocab)
        for w in words:
            scores[l] *= (vocab[(l,w)] + 1)/(total + len(vocab))
    return max(scores, key=scores.get)

def cov_matrix(data):

    n = len(data)

import math

def dist(a,b): return math.sqrt(sum((x-y)**2 for x,y in zip(a,b)))

def kmeans(points, k=2, iters=10):
    cents = points[:k]
    for _ in range(iters):
        clusters = [[] for _ in range(k)]
        for p in points:
            clusters[min(range(k), key=lambda i: dist(p, cents[i]))].append(p)
        cents = [[sum(coord)/len(cluster) for coord in zip(*cluster)] if cluster else cents[i]
                 for i, cluster in enumerate(clusters)]
    return cents

points = [[0,0],[0,1],[5,5],[5,6]]
print(kmeans(points))
```

from collections import Counter

def entropy(labels):
    n = len(labels); return -sum((c/n)*log2(c/n) for c in Counter(labels).values())

def info_gain(data, labels, feature_idx):
    base = entropy(labels)
    vals = set(row[feature_idx] for row in data)

                   entropy([labels[i] for i,r in enumerate(data) if r[feature_idx]==v]) for v in vals)
    return base - weighted
