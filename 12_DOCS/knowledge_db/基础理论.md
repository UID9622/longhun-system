```python
import math, cmath

def dft(x):
    N = len(x)
    return [sum(x[n]*cmath.exp(-2j*math.pi*k*n/N) for n in range(N)) for k in range(N)]
```

```python
# Tiny DFA: accepts strings ending with '01'
def dfa(s):
    state = "A"
    for ch in s:
        if state == "A": state = "B" if ch == "0" else "A"
        elif state == "B": state = "B" if ch == "0" else "C"
        elif state == "C": state = "B" if ch == "0" else "A"
    return state == "C"

print(dfa("10101"), dfa("111"))
```

```python
def quick_sort(a):
    if len(a) <= 1: return a
    p = a[0]
    return quick_sort([x for x in a[1:] if x < p]) + [p] + quick_sort([x for x in a[1:] if x >= p])

print(quick_sort([3,1,4,1,5,9,2,6]))
```
