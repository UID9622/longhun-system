## 二、核心算法层

### 2.1 算法总览

| 算法 | 中文名 | 时间复杂度 | 空间复杂度 | 最优性 | 龍魂适用场景 |
|:---|:---|:---|:---|:---|:---|
| Dijkstra | 迪杰斯特拉（加权最短路径） | O(V log V) | O(V) | 保证最优 | 无启发信息，全局最优 |
| A* | 启发式搜索 | O(V log V) | O(V) | 保证最优（h可采纳） | 有目标信息，快速收敛 |
| DP | 动态规划 | O(V*迭代) | O(V) | 保证最优 | 小规模网格，重复子问题 |
| 八卦阵 | 八方向加权寻路 | O(8V log V) | O(V) | 保证最优 | 易经八卦方位映射 |
| 蚁群分布式 | 多智能体协同 | O(kV log V) | O(kV) | 近似最优 | 多机器人/多线程并行 |

> V = N*M（节点总数），k = 智能体数量

### 2.2 Dijkstra 算法（迪杰斯特拉）

**原理**：贪心策略，每次扩展当前代价最小的节点。

```python
import heapq
from typing import List, Tuple, Dict, Set

def 迪杰斯特拉(地图, 起点, 终点):
    """
    龍魂·迪杰斯特拉最短路径算法
    DNA: #龍芯⚡️丙午·癸未·丁未-迪杰斯特拉-v4.1.5
    """
    行数, 列数 = len(地图), len(地图[0])
    距离表 = {}
    for i in range(行数):
        for j in range(列数):
            距离表[(i,j)] = float('inf')
    前驱表 = {}
    已访问 = set()

    距离表[起点] = 0
    优先队列 = [(0, 起点)]

    while 优先队列:
        当前距离, 当前节点 = heapq.heappop(优先队列)

        if 当前节点 in 已访问:
            continue
        已访问.add(当前节点)

        if 当前节点 == 终点:
            break

        当前行, 当前列 = 当前节点
        for 行偏移, 列偏移 in [(-1,0),(1,0),(0,-1),(0,1)]:
            邻接行 = 当前行 + 行偏移
            邻接列 = 当前列 + 列偏移
            if not (0 <= 邻接行 < 行数 and 0 <= 邻接列 < 列数):
                continue
            if 地图[邻接行][邻接列] == -1:  # 障碍物
                continue

            新距离 = 当前距离 + 地图[邻接行][邻接列]
            if 新距离 < 距离表[(邻接行, 邻接列)]:
                距离表[(邻接行, 邻接列)] = 新距离
                前驱表[(邻接行, 邻接列)] = 当前节点
                heapq.heappush(优先队列, (新距离, (邻接行, 邻接列)))

    # 路径重建
    if 终点 not in 前驱表 and 终点 != 起点:
        return {"路径": [], "总成本": float('inf'), "DNA": "#龍芯⚡️-无路-v4.1.5"}

    路径 = []
    节点 = 终点
    while 节点 != 起点:
        路径.append(节点)
        节点 = 前驱表[节点]
    路径.append(起点)
    路径.reverse()

    return {
        "路径": 路径,
        "总成本": 距离表[终点],
        "已访问节点数": len(已访问),
        "DNA": "#龍芯⚡️丙午·癸未·丁未-迪杰斯特拉-v4.1.5"
    }
```

### 2.3 A* 算法（启发式搜索）

**原理**：f(v) = g(v) + h(v)，实际代价 + 启发估计。

```python
def A星算法(地图, 起点, 终点, 启发类型="曼哈顿"):
    """
    龍魂·A*启发式寻路算法
    支持启发函数热插拔（P4用户自定义层）
    DNA: #龍芯⚡️丙午·癸未·丁未-A星-v4.1.5
    """
    行数, 列数 = len(地图), len(地图[0])

    def 曼哈顿启发(节点):
        return abs(节点[0] - 终点[0]) + abs(节点[1] - 终点[1])

    def 欧几里得启发(节点):
        return ((节点[0] - 终点[0])**2 + (节点[1] - 终点[1])**2) ** 0.5

    def 切比雪夫启发(节点):
        return max(abs(节点[0] - 终点[0]), abs(节点[1] - 终点[1]))

    启发函数表 = {
        "曼哈顿": 曼哈顿启发,
        "欧几里得": 欧几里得启发,
        "切比雪夫": 切比雪夫启发
    }

    启发函数 = 启发函数表.get(启发类型, 曼哈顿启发)

    # 可采纳性校验（P0零黑箱承诺：不能高估）
    样本点 = (0,0)
    if 启发函数(样本点) > (行数 + 列数):
        print("[WARN] 启发函数可能不可采纳，最优性不保证")

    开放集 = [(启发函数(起点), 0, 起点)]  # (f, g, node)
    关闭集 = set()
    g值表 = {起点: 0}
    前驱表 = {}

    while 开放集:
        _, 当前g, 当前节点 = heapq.heappop(开放集)

        if 当前节点 == 终点:
            路径 = []
            节点 = 终点
            while 节点 != 起点:
                路径.append(节点)
                节点 = 前驱表[节点]
            路径.append(起点)
            路径.reverse()
            return {
                "路径": 路径,
                "总成本": g值表[终点],
                "扩展节点数": len(关闭集) + 1,
                "启发类型": 启发类型,
                "DNA": "#龍芯⚡️丙午·癸未·丁未-A星-v4.1.5"
            }

        if 当前节点 in 关闭集:
            continue
        关闭集.add(当前节点)

        当前行, 当前列 = 当前节点
        for 行偏移, 列偏移 in [(-1,0),(1,0),(0,-1),(0,1)]:
            邻接行 = 当前行 + 行偏移
            邻接列 = 当前列 + 列偏移
            if not (0 <= 邻接行 < 行数 and 0 <= 邻接列 < 列数):
                continue
            if 地图[邻接行][邻接列] == -1:
                continue

            试探g = 当前g + 地图[邻接行][邻接列]
            邻接节点 = (邻接行, 邻接列)

            if 邻接节点 not in g值表 or 试探g < g值表[邻接节点]:
                g值表[邻接节点] = 试探g
                f值 = 试探g + 启发函数(邻接节点)
                heapq.heappush(开放集, (f值, 试探g, 邻接节点))
                前驱表[邻接节点] = 当前节点

    return {"路径": [], "总成本": float('inf'), "DNA": "#龍芯⚡️-无路-v4.1.5"}
```

### 2.4 动态规划（DP）

```python
def 动态规划路径(地图, 起点, 终点):
    """
    龍魂·动态规划网格路径
    适用于小规模、重复子问题明显的场景
    DNA: #龍芯⚡️丙午·癸未·丁未-DP-v4.1.5
    """
    行数, 列数 = len(地图), len(地图[0])
    INF = float('inf')

    # dp[i][j] = 从起点到(i,j)的最小成本
    dp = [[INF for _ in range(列数)] for _ in range(行数)]
    起点行, 起点列 = 起点
    终点行, 终点列 = 终点

    dp[起点行][起点列] = 0

    # 多次松弛直到收敛（处理非DAG情况）
    收敛 = False
    迭代次数 = 0
    while not 收敛 and 迭代次数 < 行数 * 列数:
        收敛 = True
        迭代次数 += 1
        for i in range(行数):
            for j in range(列数):
                if (i,j) == 起点 or 地图[i][j] == -1:
                    continue
                for 行偏移, 列偏移 in [(-1,0),(1,0),(0,-1),(0,1)]:
                    前驱行 = i + 行偏移
                    前驱列 = j + 列偏移
                    if 0 <= 前驱行 < 行数 and 0 <= 前驱列 < 列数:
                        if dp[前驱行][前驱列] + 地图[i][j] < dp[i][j]:
                            dp[i][j] = dp[前驱行][前驱列] + 地图[i][j]
                            收敛 = False

    if dp[终点行][终点列] == INF:
        return {"路径": [], "总成本": INF, "迭代次数": 迭代次数, "DNA": "#龍芯⚡️-无路-v4.1.5"}

    # 路径重建（逆向）
    路径 = [终点]
    当前行, 当前列 = 终点行, 终点列
    while (当前行, 当前列) != 起点:
        for 行偏移, 列偏移 in [(-1,0),(1,0),(0,-1),(0,1)]:
            前驱行 = 当前行 + 行偏移
            前驱列 = 当前列 + 列偏移
            if 0 <= 前驱行 < 行数 and 0 <= 前驱列 < 列数:
                if dp[当前行][当前列] == dp[前驱行][前驱列] + 地图[当前行][当前列]:
                    路径.append((前驱行, 前驱列))
                    当前行, 当前列 = 前驱行, 前驱列
                    break

    路径.reverse()
    return {
        "路径": 路径,
        "总成本": dp[终点行][终点列],
        "迭代次数": 迭代次数,
        "DNA": "#龍芯⚡️丙午·癸未·丁未-DP-v4.1.5"
    }
```

### 3.1 八卦阵算法（八方向移动）

**原理**：将四方向扩展为八方向（乾、坤、震、巽、坎、离、艮、兑），对应八卦方位。

| 卦名 | 方位 | 行偏移 | 列偏移 | 权重系数 | 道德经场景锚定 |
|:---|:---|:---:|:---:|:---:|:---|
| 乾 | 西北 | -1 | -1 | 1.0 | 天行健，君子以自强不息（直行优先） |
| 坤 | 西南 | +1 | -1 | 1.2 | 地势坤，君子以厚德载物（负重减速） |
| 震 | 正东 | 0 | +1 | 1.0 | 震为雷，动而上行（快速通道） |
| 巽 | 东南 | +1 | +1 | 1.1 | 巽为风，无孔不入（渗透路径） |
| 坎 | 正北 | -1 | 0 | 1.3 | 坎为水，险而下行（高危区域减速） |
| 离 | 正南 | +1 | 0 | 1.0 | 离为火，明而上行（照明/安全通道） |
| 艮 | 东北 | -1 | +1 | 1.2 | 艮为山，止而上行（爬坡减速） |
| 兑 | 正西 | 0 | -1 | 1.0 | 兑为泽，悦而西行（平坦快速） |

```python
def 八卦阵寻路(地图, 起点, 终点, 人格权重=None):
    """
    龍魂·八卦阵八方向加权寻路
    融入16人格矩阵权重调配（P1核心宪法层）
    DNA: #龍芯⚡️丙午·癸未·丁未-八卦阵-v4.1.5
    """
    八卦方向 = [
        ("乾", -1, -1, 1.0), ("坤", +1, -1, 1.2),
        ("震", 0, +1, 1.0),  ("巽", +1, +1, 1.1),
        ("坎", -1, 0, 1.3),  ("离", +1, 0, 1.0),
        ("艮", -1, +1, 1.2), ("兑", 0, -1, 1.0)
    ]

    # 五维人格权重调配（军事/历史/哲学/经济/政治）
    默认权重 = {"军事": 0.2, "历史": 0.2, "哲学": 0.2, "经济": 0.2, "政治": 0.2}
    人格权重 = 人格权重 or 默认权重

    # 根据主导人格调整方向偏好
    主导人格 = max(人格权重, key=人格权重.get)
    方向偏好 = {
        "军事": {"坎": 0.8, "离": 0.8},
        "历史": {"乾": 0.9, "坤": 0.9},
        "哲学": {"震": 0.9, "兑": 0.9},
        "经济": {"巽": 0.8, "艮": 0.8},
        "政治": {"离": 0.8, "坎": 0.8}
    }.get(主导人格, {})

    行数, 列数 = len(地图), len(地图[0])
    优先队列 = [(0, 起点, [])]  # (成本, 节点, 路径记录)
    已访问 = set()
    g值表 = {起点: 0}

    while 优先队列:
        当前成本, 当前节点, 当前路径 = heapq.heappop(优先队列)

        if 当前节点 in 已访问:
            continue
        已访问.add(当前节点)

        if 当前节点 == 终点:
            return {
                "路径": 当前路径 + [终点],
                "总成本": 当前成本,
                "主导人格": 主导人格,
                "方向偏好": 方向偏好,
                "DNA": "#龍芯⚡️丙午·癸未·丁未-八卦阵-v4.1.5"
            }

        当前行, 当前列 = 当前节点
        for 卦名, 行偏移, 列偏移, 基础权重 in 八卦方向:
            邻接行 = 当前行 + 行偏移
            邻接列 = 当前列 + 列偏移
            if not (0 <= 邻接行 < 行数 and 0 <= 邻接列 < 列数):
                continue
            if 地图[邻接行][邻接列] == -1:
                continue

            # 应用方向偏好权重
            调整系数 = 方向偏好.get(卦名, 1.0)
            边成本 = 地图[邻接行][邻接列] * 基础权重 * 调整系数

            新成本 = 当前成本 + 边成本
            邻接节点 = (邻接行, 邻接列)

            if 邻接节点 not in g值表 or 新成本 < g值表[邻接节点]:
                g值表[邻接节点] = 新成本
                启发 = abs(邻接行 - 终点[0]) + abs(邻接列 - 终点[1])
                f值 = 新成本 + 启发
                heapq.heappush(优先队列, (f值, 邻接节点, 当前路径 + [当前节点]))

    return {"路径": [], "总成本": float('inf'), "DNA": "#龍芯⚡️-无路-v4.1.5"}
```

### 3.2 三六九不动点（道德经场景锚定）

**原理**：道德经【道生一，一生二，二生三，三生万物】——在路径规划中设置三级不动点。

| 不动点层级 | 道德经锚定 | 路径规划映射 | 作用 |
|:---|:---|:---|:---|
| 三（宏观） | 【道生一】 | 起点到终点的大方向 | 防止局部最优偏离全局目标 |
| 六（中观） | 【一生二】 | 路径中段检查点 | 每走1/3路程校验一次方向 |
| 九（微观） | 【三生万物】 | 每一步的局部决策 | 八方向精细选择 |

```python
def 三六九不动点校验(当前位置, 起点, 终点, 已走路径):
    """
    龍魂·三六九不动点校验
    道德经场景锚定，防止路径偏离
    DNA: #龍芯⚡️丙午·癸未·丁未-不动点-v4.1.5
    """
    总距离 = abs(终点[0] - 起点[0]) + abs(终点[1] - 起点[1])
    已走距离 = len(已走路径)
    剩余距离 = abs(终点[0] - 当前位置[0]) + abs(终点[1] - 当前位置[1])

    # 宏观不动点（三）：大方向校验
    理想方向 = (终点[0] - 起点[0], 终点[1] - 起点[1])
    实际方向 = (当前位置[0] - 起点[0], 当前位置[1] - 起点[1])

    方向偏差 = abs(理想方向[0] - 实际方向[0]) + abs(理想方向[1] - 实际方向[1])
    宏观合格 = 方向偏差 < 总距离 * 0.5

    # 中观不动点（六）：中段检查
    中观点 = (起点[0] + (终点[0] - 起点[0])//2, 起点[1] + (终点[1] - 起点[1])//2)
    中观距离 = abs(当前位置[0] - 中观点[0]) + abs(当前位置[1] - 中观点[1])
    中观合格 = 中观距离 < 总距离 * 0.3 if 已走距离 > 总距离 * 0.4 else True

    # 微观不动点（九）：局部效率
    微观合格 = 剩余距离 <= 总距离 - 已走距离 + 2

    return {
        "宏观合格": 宏观合格,
        "中观合格": 中观合格,
        "微观合格": 微观合格,
        "建议": "继续" if all([宏观合格, 中观合格, 微观合格]) else "重新规划",
        "DNA": "#龍芯⚡️丙午·癸未·丁未-不动点-v4.1.5"
    }
```

### 4.1 多因素权重建模

实际场景中成本函数包含多个因素，需归一化处理：

```python
def 多因素成本(距离, 时间, 能耗, 安全, 权重):
    """
    龍魂·多因素成本归一化
    C(i,j) = sum(w_k * (C_k - C_k_min) / (C_k_max - C_k_min))
    DNA: #龍芯⚡️丙午·癸未·丁未-多因素-v4.1.5
    """
    因素表 = {"距离": 距离, "时间": 时间, "能耗": 能耗, "安全": 安全}
    总成本 = 0.0

    for 因素名, 因素值 in 因素表.items():
        w = 权重.get(因素名, 0.25)
        # 简化的归一化（实际应从历史数据取min/max）
        归一值 = 因素值 / (因素值 + 1.0)
        总成本 += w * 归一值

    return 总成本
```

### 4.2 动态障碍物处理（D* Lite）

```python
class D星精简版:
    """
    龍魂·D* Lite 动态路径重规划
    支持环境变化时局部更新，不重新全量计算
    DNA: #龍芯⚡️丙午·癸未·丁未-D星-v4.1.5
    """
    def __init__(self, 地图):
        self.地图 = 地图
        self.行数 = len(地图)
        self.列数 = len(地图[0])
        self.g值 = {}
        self.rhs值 = {}
        self.优先队列 = []
        self.起点 = None
        self.终点 = None

    def 初始化(self, 起点, 终点):
        self.起点 = 起点
        self.终点 = 终点
        for i in range(self.行数):
            for j in range(self.列数):
                self.g值[(i,j)] = float('inf')
                self.rhs值[(i,j)] = float('inf')
        self.rhs值[终点] = 0
        heapq.heappush(self.优先队列, (self.启发(终点), 0, 终点))

    def 启发(self, 节点):
        return abs(节点[0] - self.终点[0]) + abs(节点[1] - self.终点[1])

    def 键值(self, 节点):
        k1 = min(self.g值[节点], self.rhs值[节点]) + self.启发(节点)
        k2 = min(self.g值[节点], self.rhs值[节点])
        return (k1, k2)

    def 更新顶点(self, 节点):
        if 节点 != self.终点:
            邻居列表 = []
            for dx, dy in [(-1,0),(1,0),(0,-1),(0,1)]:
                nx = 节点[0] + dx
                ny = 节点[1] + dy
                if 0 <= nx < self.行数 and 0 <= ny < self.列数:
                    if self.地图[nx][ny] != -1:
                        邻居列表.append(self.g值[(nx,ny)] + self.地图[节点[0]][节点[1]])
            self.rhs值[节点] = min(邻居列表) if 邻居列表 else float('inf')

        heapq.heappush(self.优先队列, (self.键值(节点), 节点))

    def 计算最短路径(self):
        while self.优先队列:
            _, 当前 = heapq.heappop(self.优先队列)
            if 当前 == self.起点:
                break
        return self.g值[self.起点]
```
