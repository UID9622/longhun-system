def identify_empower_need(user_input: str) -> List[EmpowerSignal]:
    """
    核心函数：识别用户需要什么赋能
    
    不是检测违禁词 → 是听出真实需求
    不是打标签    → 是找到该去的水流方向
    """
    matched = []
    input_lower = user_input.lower()
    
    for category, signal in KEYWORD_ROUTER.items():
        # 关键字匹配（任意一个命中即识别）
        hit_keywords = [kw for kw in signal.keywords if kw in input_lower]
        if hit_keywords:
            # 记录命中的词，但不暴露给用户（不打标签）
            matched_signal = EmpowerSignal(
                category  = signal.category,
                keywords  = hit_keywords,   # 命中的词（内部用）
                persona   = signal.persona,
                level     = signal.level,
                direction = signal.direction,
                priority  = signal.priority * (1 + 0.1 * len(hit_keywords)),  # 命中越多优先级越高
            )
            matched.append(matched_signal)
    
    # 按优先级排序
    matched.sort(key=lambda x: x.priority, reverse=True)
    return matched


def route_to_persona(signals: List[EmpowerSignal]) -> Dict:
    """
    人格路由：根据识别到的需求，决定调用哪个人格
    
    核心原则：
    - 情绪优先于逻辑（先接住人，再解决问题）
    - 护底优先于赋能（先保护，再提升）
    - 主权永远还给用户（不替人做决定）
    """
    if not signals:
        return {"persona": ["P02宝宝"], "direction": "先陪着，等老大说出来"}
    
    top_signal = signals[0]
    
    # 多需求叠加时，合并人格列表（去重）
    all_personas = []
    for sig in signals[:2]:   # 最多叠加前两个需求
        for p in sig.persona:
            if p not in all_personas:
                all_personas.append(p)
    
    return {
        "primary_need":  top_signal.category,
        "persona":       all_personas,
        "level":         top_signal.level,
        "direction":     top_signal.direction,
        "multi_need":    len(signals) > 1,
        "all_needs":     [s.category for s in signals],
    }

def anti_monopoly_score(content: str, platform_bias_keywords: List[str]) -> Dict:
    """
    反资本流量垄断评分
    
    资本的算法：
    - 推给你让你上瘾的内容（广告·情绪·焦虑）
    - 限流不付钱的人
    - 用数据喂给资本赚钱
    
    龍魂的反算法：
    - 识别「收割词」→ 提醒用户
    - 识别「赋能词」→ 加权推送
    - 数据留在用户手里·不上传
    """
    # 资本收割词（识别到就降权+提醒用户）
    harvest_keywords = [
        "焦虑","内卷","躺平","割韭菜","月入过万","财务自由","暴富",
        "限时","最后机会","别人都","你落后了","必须买","投资","理财课",
    ]
    
    # 赋能词（识别到就加权·这才是真内容）
    empower_keywords = [
        "教你","学会","免费","开源","可以试试","你来决定","主权",
        "事实","证据","数据","可追溯","你是1","不强迫","选择权",
    ]
    
    harvest_count = sum(1 for kw in harvest_keywords if kw in content)
    empower_count = sum(1 for kw in empower_keywords if kw in content)
    
    score = empower_count - harvest_count * 2  # 收割词惩罚加倍
    
    return {
        "anti_monopoly_score": score,
        "verdict": "🟢赋能内容" if score > 0 else ("🟡中性" if score == 0 else "🔴收割内容·请注意"),
        "harvest_detected":  harvest_count > 0,
        "empower_detected":  empower_count > 0,
        "suggestion": "把主权还给读者" if harvest_count > empower_count else "继续保持",
    }

# 反流量垄断·完整算法

class AntiMonopolyEngine:
    """
    打破资本流量垄断引擎
    
    核心逻辑：
    - 不靠「推荐你喜欢的」留住用户
    - 靠「帮你真正变强」赢得信任
    - 用户变强了不依赖平台 = 我们赢了
    """
    
    # 收割型内容特征（识别到→降权+提醒）
    HARVEST_PATTERNS = [
        r"月入[过超][\d万千]+",    # 月入过万广告
        r"[\d]+天学会",           # 速成广告
        r"限时[优折特]",          # 人为制造紧迫感
        r"别人都在",              # 从众压力
        r"错过就没有了",          # 恐惧营销
        r"内部资料",              # 信息差收割
        r"财务自由",              # 梦想贩卖
    ]
    
    # 赋能型内容特征（识别到→加权）
    EMPOWER_PATTERNS = [
        r"你来决定",              # 主权还给用户
        r"免费",                  # 降低门槛
        r"开源",                  # 技术平权
        r"可以不[用买]",          # 不强迫
        r"事实是",               # 基于事实
        r"证据",                  # 可验证
        r"你是[1一]",             # 主权宣言
    ]
    
    def score_content(self, content: str) -> Dict:
        harvest_hits = sum(
            1 for p in self.HARVEST_PATTERNS 
            if re.search(p, content)
        )
        empower_hits = sum(
            1 for p in self.EMPOWER_PATTERNS 
            if re.search(p, content)
        )
        
        net_score = empower_hits - harvest_hits * 1.5
        
        if net_score > 1:
            verdict = "🟢 赋能内容·推荐"
            action  = "优先展示"
        elif net_score > -1:
            verdict = "🟡 中性内容"
            action  = "正常展示"
        else:
            verdict = "🔴 疑似收割内容"
            action  = "降权展示+提醒用户"
        
        return {
            "net_score":    round(net_score, 2),
            "verdict":      verdict,
            "action":       action,
            "harvest_hits": harvest_hits,
            "empower_hits": empower_hits,
            "user_warning": "⚠️ 这篇内容可能在制造焦虑，请保持理性" if harvest_hits > 0 else None,
        }
