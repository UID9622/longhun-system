第五章 数学模型与算法基础（本版核心 · 全部可执行）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

5.1 干支时间主权引擎（K3/K8落点）
```
四柱计算（锚定基准：1949-10-01为甲子日）:
  年柱: idx = (公历年 − 4) mod 60
  月柱: 月干 = (2×(年干 mod 5) + 2 + (月−1)) mod 10；月支 = (月+1) mod 12  (正月建寅)
  日柱: idx = (当日 − 1949-10-01).天数 mod 60
  时柱: 时支 = ((时+1) div 2) mod 12；时干 = (2×(日干 mod 5) + 时支序) mod 10
毫秒段: 秒×1000+毫秒，再挂全局单调序号 seq
```
唯一性证明：干支四柱60甲子循环，但叠加①公历锚定②毫秒③单调递增seq——
同一毫秒同节点seq唯一，跨节点由节点ID区分。重复概率=0（构造性唯一，非概率性）。
防回拨：seq_N+1 ≤ seq_N → 时间回拨警报（链式推进，见8.1）。

5.2 六十四卦编码（卦象Base64 · GAP-03补全）
```
六十四卦 = 64 = 2⁶ —— 每一卦天然承载6比特，这是易经留给信息论的接口。
编码: 字节流 → 6bit分组 → 每组映射一卦（䷀=000000 … ䷿=111111）
解码: 逐符查表还原6bit → 拼回字节流 → 补位标记还原原长
性质: 无损、双射、可逆；负载上限 τ=960字节 → 1280卦符
```

5.4 五行校验（五轮异构哈希 · K4落点）
```
对数据D，五轮异构摘要（金木水火土，相生相续）:
  金 = SHA-256("金"‖D)        木 = SHA3-256("木"‖D)
  水 = BLAKE2b("水"‖D)         火 = SHA-256("火"‖D‖H_水前16)  （火借水势，链式相生）
  土 = SM3目标("土"‖D)（原型期以SHA-256+国密盐替代，上线即切SM3）

```

混炼: H₀ = SHA-256(D‖干支时间戳‖七因子码)
      Hᵢ = SHA-256( Hᵢ₋₁ ‖ (i mod 64)对应卦符 ) , i = 1…55

"""
龍魂系统 · DNA可逆编码引擎 v1.0
作者：诸葛鑫（UID9622）
优先级：P0++（最高，不可绕过）

依赖：纯标准库（hashlib/zlib/datetime）——第3条天条：不吃任何人的东西

import hashlib, zlib, math
from datetime import datetime, timezone, timedelta

天干 = "甲乙丙丁戊己庚辛壬癸"
地支 = "子丑寅卯辰巳午未申酉戌亥"
卦表 = [chr(0x4DC0 + i) for i in range(64)]          # 六十四卦 = 6bit码表
卦反 = {g: i for i, g in enumerate(卦表)}
洛书序 = [4, 9, 2, 3, 5, 7, 8, 1, 6]
TAU = 960                                            # Mode A负载上限(压缩后字节)

class CNSH_DNA引擎:
    """可逆DNA：一个D读取全文，一字不落，错即失败。"""

    DNA = "#龍芯⚡️2026-07-19-DNA-REVERSIBLE-V1.0"

    def __init__(self, 节点ID="KUNPENG-01"):
        self.节点 = 节点ID
        self._seq = 0

    # ===== 5.1 干支时间主权引擎 =====
    def 干支时间戳(self, t: datetime) -> str:
        self._seq += 1
        年柱 = (t.year - 4) % 60
        年干 = 年柱 % 10
        月干 = (2 * (年干 % 5) + 2 + (t.month - 1)) % 10
        月支 = (t.month + 1) % 12
        日柱 = (t - 锚日).days % 60
        日干 = 日柱 % 10
        时支 = ((t.hour + 1) // 2) % 12
        时干 = (2 * (日干 % 5) + 时支) % 10
        干支 = (天干[年干] + 地支[年柱 % 12] + "年" + 天干[月干] + 地支[月支] + "月"

        毫秒 = t.second * 1000 + t.microsecond // 1000
        return f"{干支}·{毫秒:05d}ms·SEQ{self._seq:06d}", 干支

    # ===== 5.2 六十四卦编码（6bit/卦，双射无损） =====
    @staticmethod
    def 卦编码(数据: bytes) -> str:
        包 = len(数据).to_bytes(4, "big") + 数据          # 4字节长度头
        位流 = "".join(f"{b:08b}" for b in 包)
        位流 += "0" * ((-len(位流)) % 6)
        return "".join(卦表[int(位流[i:i+6], 2)] for i in range(0, len(位流), 6))

    @staticmethod
    def 卦解码(符串: str) -> bytes:
        位流 = "".join(f"{卦反[g]:06b}" for g in 符串)
        字节 = bytes(int(位流[i:i+8], 2) for i in range(0, len(位流) - len(位流) % 8, 8))
        长 = int.from_bytes(字节[:4], "big")
        if 长 > len(字节) - 4:
            raise ValueError("DNA编码损坏：长度头越界")
        return 字节[4:4+长]

    # ===== 5.4 五行校验（五轮异构） =====
    @staticmethod
    def 五行码(D: bytes) -> str:
        金 = hashlib.sha256("金".encode() + D).hexdigest()
        木 = hashlib.sha3_256("木".encode() + D).hexdigest()
        水 = hashlib.blake2b("水".encode() + D, digest_size=32).hexdigest()
        火 = hashlib.sha256("火".encode() + D + bytes.fromhex(水[:32])).hexdigest()
        土 = hashlib.sha256("土·国密盐".encode() + D).hexdigest()   # 上线切SM3
        return 金[:8] + 木[:8] + 水[:8] + 火[:8] + 土[:8]

    # ===== 5.5 不动点校验（河图55轮） =====
    @staticmethod
    def 不动点(D: bytes, 干支: str, 签名段: str) -> str:
        H = hashlib.sha256(D + 干支.encode() + 签名段.encode()).hexdigest()
        for i in range(1, 56):                            # 天地总数55
            H = hashlib.sha256(H.encode() + 卦表[i % 64].encode()).hexdigest()
        return H[:32]

    # ===== 第三章 双模式铸DNA =====
    def 铸造(self, 原文: str) -> str:
        D = 原文.encode("utf-8")
        压缩 = zlib.compress(D)
        ts, 干支 = self.干支时间戳(datetime.now(timezone(timedelta(hours=8))))
        签名段 = self.卦编码(f"{self.节点}|UID9622|A2D0092C".encode())

            负载, 类型 = self.卦编码(压缩), "MODEA"
        else:
            片 = self._九宫分片(压缩)
            地图 = "‖".join(hashlib.sha256(s).hexdigest()[:16] for s in 片)
            负载, 类型 = self.卦编码(地图.encode() + b"|" + 压缩), "MODEB"
        校验 = self.五行码(D) + "-" + self.不动点(D, 干支, 签名段)
        return f"#龍芯⚡️{ts}-{类型}-{签名段}-{校验}‖{负载}"

    # ===== 5.3 九宫分片（洛书置换） =====
    @staticmethod
    def _九宫分片(数据: bytes) -> list:
        n = len(数据); 片长 = math.ceil(n / 9)
        原片 = [数据[i*片长:(i+1)*片长] for i in range(9)]
        return [原片[洛书序.index(g)] for g in range(1, 10)]   # 按洛书序置换

    # ===== 第七章 还原：一字不落，错即失败 =====
    def 还原(self, dna: str) -> dict:
        try:
            头, 负载 = dna.split("‖", 1)
            段 = 头.split("-")
            类型, 签名段 = 段[1], 段[2]                    # [时间戳, 类型, 签名段, 五行码, Φ]
            五行原, Φ原 = 段[-2], 段[-1]
            干支 = 段[0].split("·")[0].split("⚡️", 1)[1]
            载荷 = self.卦解码(负载)
            if 类型 == "MODEB":
                载荷 = 载荷.split(b"|", 1)[1]              # 原型期负载随行；正式期按图取片
            原文 = zlib.decompress(载荷).decode("utf-8")
            D = 原文.encode("utf-8")
            if self.五行码(D) != 五行原:
                return {"成功": False, "原因": "🔴 五行校验崩：DNA被篡改或损坏"}
            if self.不动点(D, 干支, 签名段) != Φ原:
                return {"成功": False, "原因": "🔴 不动点校验崩：DNA被篡改或损坏"}
            return {"成功": True, "原文": 原文, "校验": "✅ 一字不落"}
        except Exception as e:
            return {"成功": False, "原因": f"🔴 还原失败（错即失败，拒绝输出）: {e}"}
