## 8.3 一键训练脚本（v4.1）

```bash
#!/bin/bash
# 龍魂v4.1训练脚本 · 审计协议注入 + 数据扩量
# DNA: #龍芯⚡️丙午·乙申·辛亥·亥时·乾-MODEL-LORA-TRAINER-v4.1

echo "[龍魂] v4.1训练启动 · 审计协议v2.0融合 + 数据扩量"
echo "[龍魂] 底座: Llama-3.1-8B-Instruct (MLX)"
echo "[龍魂] 目标: Val loss < 0.5 (第一步), 最终 < 0.1"
echo "[龍魂] 数据: 1145(现有) + 审计指令 + 扩量 → 6000+"
echo ""

# 1. 扩量数据（扫描全项目文件 → 生成Q&A）
echo "[Step 1/5] 数据扩量..."
python3 bin/lh_data_expand_v41.py

# 2. 合并审计指令
echo "[Step 2/5] 合并审计协议v2.0训练数据..."
python3 bin/lh_audit_protocol_inject.py

# 3. 启动训练（LoRA rank=64）
echo "[Step 3/5] LoRA训练中..."
python3 bin/lh_lora_trainer_v4.py train \
    --rank 64 --alpha 128 --layers 24 \
    --iters 1500 --lr 5e-5 \
    --data models/longhun-v1.0/lora_output_v4/data_v41/train.jsonl \
    --adapter models/longhun-v1.0/lora_output_v4/adapter_v4.1

# 4. Fuse + Export
echo "[Step 4/5] Fuse + Export..."
python3 bin/lh_lora_trainer_v4.py fuse --adapter adapter_v4.1 --name v4.1
python3 bin/lh_lora_trainer_v4.py export --name v4.1

# 5. Ollama部署
echo "[Step 5/5] Ollama部署..."
ollama create longhun-v4.1 -f models/longhun-v1.0/lora_output_v4/gguf_v4.1/Modelfile.v4.1

echo ""
echo "[龍魂] v4.1训练完成！"
echo "[龍魂] 测试: ollama run longhun-v4.1"
```
