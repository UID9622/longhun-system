from enum import Enum
import logging

class ChineseLaw(Enum):
    """
    真实的中国法律条文引用
    每条法律都可以在全国人大官网查询
    """
    DATA_SECURITY_LAW_21 = "《数据安全法》第21条：重要数据境内存储"
    DATA_SECURITY_LAW_36 = "《数据安全法》第36条：向境外提供数据需评估申报"
    CRYPTOGRAPHY_LAW_27 = "《密码法》第27条：涉及国家安全的商用密码产品需检测认证"
    CYBER_SECURITY_LAW_37 = "《网络安全法》第37条：关键信息基础设施数据境内存储"

class ComplianceChecker:
    """
    合规检查器（本地逻辑，不调用政府API）
    """
    
    def __init__(self):
        self.audit_log = []
        self.logger = logging.getLogger('UID9622.Compliance')
        
        # 真实的合规配置
        self.config = {
            "data_境内存储": True,        # 遵守《数据安全法》第21条
            "使用国密算法": True,          # 遵守《密码法》
            "禁止境外API": True,          # 遵守《数据安全法》第36条
            "本地审计日志": True,          # 可追溯原则
        }
    
    def check_data_export(self, destination: str) -> bool:
        """
        检查数据导出是否合规
        依据：《数据安全法》第36条
        """
        if self._is_foreign_destination(destination):
            # 真实的法律要求：向境外提供数据需评估申报
            self._log_violation(
                law=[ChineseLaw.DATA](http://ChineseLaw.DATA)_SECURITY_LAW_36,
                action="尝试向境外导出数据",
                destination=destination,
                result="已阻止"
            )
            return False
        return True
    
    def check_encryption_algorithm(self, algorithm: str) -> bool:
        """
        检查加密算法是否合规
        依据：《密码法》第27条
        """
        approved_algorithms = ["SM2", "SM3", "SM4"]  # 国密算法
        
        if algorithm not in approved_algorithms:
            self._log_violation(
                law=ChineseLaw.CRYPTOGRAPHY_LAW_27,
                action=f"尝试使用非国密算法：{algorithm}",
                result="建议使用国密算法"
            )
            return False
        return True
    
    def _is_foreign_destination(self, destination: str) -> bool:
        """
        判断目标是否为境外
        这是技术判断，不是调用政府API
        """
        foreign_indicators = [
            ".com", ".net", ".org",  # 常见境外域名
            "[amazonaws.com](http://amazonaws.com)",           # AWS
            "[azure.com](http://azure.com)",               # Azure
            "[googleapis.com](http://googleapis.com)"           # Google
        ]
        return any(indicator in destination for indicator in foreign_indicators)
    
    def _log_violation(self, law: ChineseLaw, action: str, result: str, **kwargs):
        """
        记录违规行为到本地审计日志
        注意：这是本地记录，不是"上报网信办"
        """
        import datetime
        
        log_entry = {
            "timestamp": [datetime.datetime.now](http://datetime.datetime.now)().isoformat(),
            "law_reference": law.value,
            "action": action,
            "result": result,
            **kwargs
        }
        
        self.audit_log.append(log_entry)
        self.logger.warning(f"合规检查: {law.value} - {action} - {result}")
        
        # 写入本地文件（不是上报政府）
        with open("./logs/compliance_audit.log", "a", encoding="utf-8") as f:
            import json
            f.write(json.dumps(log_entry, ensure_ascii=False) + "\n")
    
    def generate_compliance_report(self) -> dict:
        """
        生成合规报告
        可以在需要时人工提交给监管部门
        """
        return {
            "report_time": [datetime.datetime.now](http://datetime.datetime.now)().isoformat(),
            "system_name": "UID9622",
            "compliance_status": {
                "数据境内存储": self.config["data_境内存储"],
                "使用国密算法": self.config["使用国密算法"],
                "禁止境外传输": self.config["禁止境外API"]
            },
            "audit_log_count": len(self.audit_log),
            "violations": [log for log in self.audit_log if "阻止" in log["result"]]
        }

# 使用示例
if __name__ == "__main__":

### 3. 真实的审计日志系统

```python
# 文件：core/audit_[system.py](http://system.py)
# 本地审计日志系统（不是上报政府，而是可供审计）

import json
import datetime
import hashlib
from typing import List, Dict

class AuditSystem:
    """
    本地审计系统
    - 所有操作记录本地日志
    - 日志不可篡改（使用哈希链）
    - 可在需要时提供给监管部门
    """
    
    def __init__(self, log_file="./logs/audit_chain.log"):
        self.log_file = log_file
        self.previous_hash = self._get_last_hash()
    
    def log_action(self, action: str, details: Dict, law_reference: str = None):
        """
        记录操作到审计日志
        使用哈希链确保不可篡改
        """
        entry = {
            "timestamp": [datetime.datetime.now](http://datetime.datetime.now)().isoformat(),
            "action": action,
            "details": details,
            "law_reference": law_reference,
            "previous_hash": self.previous_hash
        }
        
        # 计算当前条目的哈希
        entry_str = json.dumps(entry, sort_keys=True, ensure_ascii=False)
        current_hash = hashlib.sha256(entry_str.encode()).hexdigest()
        entry["current_hash"] = current_hash
        
        # 写入日志文件
        with open(self.log_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(entry, ensure_ascii=False) + "\n")
        
        # 更新previous_hash
        self.previous_hash = current_hash
    
    def _get_last_hash(self) -> str:
        """
        获取最后一条日志的哈希
        """
        try:
            with open(self.log_file, "r", encoding="utf-8") as f:
                lines = f.readlines()
                if lines:
                    last_entry = json.loads(lines[-1])
                    return last_entry.get("current_hash", "0")
        except FileNotFoundError:
            pass
        return "0"  # 初始哈希
    
    def verify_integrity(self) -> bool:
        """
        验证审计日志的完整性
        检查哈希链是否被篡改
        """
        try:
            with open(self.log_file, "r", encoding="utf-8") as f:
                lines = f.readlines()
            
            previous = "0"
            for line in lines:
                entry = json.loads(line)
                
                # 检查previous_hash是否匹配
                if entry["previous_hash"] != previous:
                    print(f"❌ 审计日志被篡改！时间：{entry['timestamp']}")
                    return False
                
                # 验证current_hash
                stored_hash = entry.pop("current_hash")
                entry_str = json.dumps(entry, sort_keys=True, ensure_ascii=False)
                calculated_hash = hashlib.sha256(entry_str.encode()).hexdigest()
                
                if stored_hash != calculated_hash:
                    print(f"❌ 审计日志哈希不匹配！时间：{entry['timestamp']}")
                    return False
                
                previous = stored_hash
            
            print("✅ 审计日志完整，未被篡改")
            return True
        
        except Exception as e:
            print(f"❌ 审计日志验证失败：{str(e)}")
            return False
    
    def export_for_audit(self, output_file="./compliance_report.json") -> str:
        """
        导出审计日志供监管部门审查
        这是人工提交，不是自动上报
        """
        with open(self.log_file, "r", encoding="utf-8") as f:
            logs = [json.loads(line) for line in f.readlines()]
        
        report = {
            "system_name": "UID9622",
            "export_time": [datetime.datetime.now](http://datetime.datetime.now)().isoformat(),
            "total_entries": len(logs),
            "integrity_verified": self.verify_integrity(),
            "logs": logs
        }
        
        with open(output_file, "w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        
        return output_file

# 使用示例
if __name__ == "__main__":
    audit = AuditSystem()
    
    # 记录操作
    audit.log_action(
        action="数据导出检查",
        details={"destination": "[api.openai.com](http://api.openai.com)", "result": "已阻止"},
        law_reference="《数据安全法》第36条"
    )
    
    # 验证完整性
    audit.verify_integrity()
    
    # 导出供审计
    report_file = audit.export_for_audit()
    print(f"合规报告已生成：{report_file}")
```
