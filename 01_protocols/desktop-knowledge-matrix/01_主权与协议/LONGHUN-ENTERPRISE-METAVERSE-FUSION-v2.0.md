### 2.2 蚁群节点模型

```python
class LonghunAntNode:
    """
    龍魂蚁群节点 · 每只蚂蚁都是自治体
    
    四种角色:
    - 蚁后(queen): 高存储+高在线 → 数据持久化
    - 兵蚁(soldier): 高计算 → 任务执行
    - 侦察蚁(scout): 高网络 → 节点发现
    - 工蚁(worker): 通用 → 消息传递
    """

    def __init__(self, dna: str, platform: str, chip: str):
        self.dna = dna                    # 身份DNA
        self.platform = platform          # ios/harmony/windows/line
        self.chip = chip                  # 麒麟/鲲鹏/龙芯/昇腾
        self.role = self.determine_role()
        self.capabilities = self.probe()
        self.neighbors = []               # 邻居节点
        self.pheromones = {}              # 信息素表
        self.state = "active"
        self.lu_commands = {}             # 本地LU指令缓存

    def determine_role(self) -> str:
        caps = self.capabilities
        if caps["storage_gb"] > 100 and caps["uptime_h"] > 720:
            return "queen"
        elif caps["compute_score"] > 80:
            return "soldier"
        elif caps["network_score"] > 80:
            return "scout"
        return "worker"

    def emit_pheromone(self, ptype: str, data: dict, ttl: int = 3600):
        """
        释放信息素 — 蚁群通信核心
        
        信息素类型:
        - message: 点对点消息（加密·不留存）
        - task: 任务分发（计算/存储/审计）
        - lu_cmd: LU指令传播（人格调度/跨平台联动）
        - heartbeat: 心跳存活检测
        - product: 产品同步（1:1真实产品信息）
        - audit: 审计信号（三色审计联动）
        """
        pheromone = {
            "id": f"PHE-{uuid.uuid4().hex[:8]}",
            "type": ptype,
            "source": self.dna,
            "timestamp": time.time(),
            "ttl": ttl,
            "data": data,
            "hops": 0,
            "max_hops": 10,
        }
        for neighbor in self.neighbors:
            neighbor.receive_pheromone(pheromone)
```

### 2.3 P2P通信原则

```
点对点通讯铁律：

1. 消息内容加密传输（SM4国密）
2. 不经过中心服务器
3. 服务器只做"地址发现"（告诉A去哪里找B）
4. 服务器不存消息内容
5. 消息阅后即焚（按L5时间轴自动清理）
6. 群聊=多路P2P叠加
7. 语音/视频=WebRTC直连（不经过中转）
```

class InboxToLayerMapper:
    """四类收纳 → 六层架构 自动映射引擎"""

    MAP_RULES = {
        # (bucket, keywords) → (layer, confidence)
        ("系统规则", ["协议", "宪法", "铁律", "治理", "芯片", "架构"]): ("L1", 0.95),
        ("系统规则", ["审计", "熔断", "合规", "访问控制"]): ("L6", 0.90),
        ("系统规则", ["备份", "同步", "API", "数据库"]): ("L5", 0.85),
        ("操作日志", ["人格", "签章", "红蓝", "对抗"]): ("L2", 0.90),
        ("操作日志", ["部署", "健康检查", "监控", "告警"]): ("L5", 0.85),
        ("操作日志", ["认证", "登录", "密钥", "签名"]): ("L6", 0.85),
        ("灵感碎片", ["推演", "沙盒", "实验", "假设", "新想法"]): ("L3", 0.85),
        ("灵感碎片", ["知识", "文章", "教程", "论文"]): ("L4", 0.75),
        ("知识派系", ["论文", "研究", "报告", "技术文档"]): ("L4", 0.95),
        ("知识派系", ["推演", "哲学", "文化", "易经"]): ("L3", 0.80),
    }

    def suggest_layer(self, bucket: str, content: str) -> dict:
        """对每条inbox条目建议目标层"""
        best = {"layer": None, "confidence": 0}
        for (bk, keywords), (layer, conf) in self.MAP_RULES.items():
            if bk == bucket:
                score = sum(1 for kw in keywords if kw in content) / len(keywords)
                adjusted_conf = conf * (0.5 + 0.5 * score)
                if adjusted_conf > best["confidence"]:
                    best = {"layer": layer, "confidence": round(adjusted_conf, 2)}
        return best

    def auto_map(self, bucket: str, content: str) -> dict:
        """自动映射 + 三色路径"""
        suggestion = self.suggest_layer(bucket, content)
        if suggestion["confidence"] >= 0.85:
            return {"action": "🟢 auto_apply", "layer": suggestion["layer"], "confidence": suggestion["confidence"]}
        elif suggestion["confidence"] >= 0.70:
            return {"action": "🟡 suggest_review", "layer": suggestion["layer"], "confidence": suggestion["confidence"]}
        else:
            return {"action": "🔴 manual_required", "layer": None, "confidence": suggestion["confidence"]}

    """

    五维循环:
    1. 节点加入 → 蚁群能力增强
    2. 算法吸收 → 万国仓库排排坐

    4. 协议进化 → 社区共识→版本迭代
    5. 反馈回灌 → 闭环优化
    """

    def growth_cycle(self):
        # 维1: 节点自组织
        new_nodes = self.discover_ants()
        for node in new_nodes:
            self.integrate_node(node)

        # 维2: 算法道引吸收
        new_algorithms = self.daoyin_scan_github()
        for algo in new_algorithms:
            # 排排站（待审核）→ 排排坐（已吸收）

        new_terms = self.registry_auto_sync.scan_and_apply()

        # 维4: 协议进化
        evolved = self.evolve_protocols_from_usage()
        for protocol in evolved:
            self.protocol_registry.upgrade(protocol)
