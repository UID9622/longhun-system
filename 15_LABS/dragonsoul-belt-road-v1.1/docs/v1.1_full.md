## 十、快速开始指南·Quick Start

> **目标：5分钟内完成部署，10分钟内跑通第一个场景。**

### 10.1 前置检查（自动）

运行一键检查脚本：

```bash
# 自动检测硬件、操作系统、网络、依赖
curl -fsSL https://dragonsoul.dev/install/check.sh | bash
# 或本地运行
./scripts/preflight-check.sh
```

检查项：
- [ ] CPU 架构（x86_64 / ARM64 / RISC-V）
- [ ] 内存 ≥ 8GB（微型部署）
- [ ] 磁盘空间 ≥ 50GB

## 十二、一键部署脚本·CB-003

### 12.1 主部署脚本：`install.sh`

```bash
#!/bin/bash
# 龍魂 · 一带一路入门包 · 一键部署脚本 v1.1
# DNA: #龍芯⚡️丙午·丙申·戊午·戊午·䷱鼎-BELT-ROAD-PACK-UID9622
# 确认码: #CONFIRM🌌9622-ONLY-ONCE🧬LK9X-772Z
# 许可: MulanPSL v2

set -euo pipefail

DRAGONSOUL_VERSION="1.1"
INSTALL_DIR="${DRAGONSOUL_HOME:-$HOME/dragonsoul-belt-road}"
LOG_FILE="/tmp/dragonsoul-install.log"
MODEL_SIZE="${MODEL_SIZE:-7b}"
LANG="${DRAGONSOUL_LANG:-en}"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log() { echo -e "${GREEN}[龍魂]${NC} $1" | tee -a "$LOG_FILE"; }
warn() { echo -e "${YELLOW}[警告]${NC} $1" | tee -a "$LOG_FILE"; }
error() { echo -e "${RED}[错误]${NC} $1" | tee -a "$LOG_FILE"; exit 1; }

# ========== 前置检查 ==========
check_system() {
    log "=== 系统前置检查 ==="

    # CPU 架构
    ARCH=$(uname -m)
    log "CPU架构: $ARCH"
    case "$ARCH" in
        x86_64|amd64) ARCH_TAG="amd64" ;;
        aarch64|arm64) ARCH_TAG="arm64" ;;
        *) error "不支持的架构: $ARCH" ;;
    esac

    # 内存检查
    MEM_KB=$(grep MemTotal /proc/meminfo 2>/dev/null | awk '{print $2}' || echo 0)
    MEM_GB=$((MEM_KB / 1024 / 1024))
    log "内存: ${MEM_GB}GB"
    if [ "$MEM_GB" -lt 4 ]; then
        warn "内存不足4GB，仅支持微型模型(1B-3B)"
        MODEL_SIZE="3b"
    elif [ "$MEM_GB" -lt 8 ]; then
        warn "内存不足8GB，建议使用小型模型(3B-7B)"
        MODEL_SIZE="7b"
    fi

    # 磁盘检查
    DISK_AVAIL=$(df -BG "$HOME" | tail -1 | awk '{print $4}' | sed 's/G//')
    log "可用磁盘: ${DISK_AVAIL}GB"
    if [ "${DISK_AVAIL:-0}" -lt 20 ]; then
        error "磁盘空间不足20GB，无法安装"
    fi

    # 操作系统
    OS=$(uname -s)
    log "操作系统: $OS"

    # 依赖检查
    if command -v docker &> /dev/null; then
        log "Docker 已安装 ✓"
        HAS_DOCKER=1
    else
        warn "Docker 未安装，将使用裸机模式"
        HAS_DOCKER=0
    fi

    if command -v python3 &> /dev/null; then
        PY_VER=$(python3 --version | awk '{print $2}')
        log "Python: $PY_VER ✓"
    else
        warn "Python3 未安装，部分功能受限"
    fi

    log "前置检查完成 ✓"
}

# ========== 安装模式选择 ==========
select_mode() {
    if [ "$HAS_DOCKER" -eq 1 ]; then
        MODE="docker"
        log "自动选择 Docker Compose 模式"
    else
        MODE="bare"
        log "自动选择裸机模式（Ollama）"
    fi

    # 离线模式检测
    if [ "${OFFLINE:-0}" -eq 1 ] || ! ping -c 1 github.com &>/dev/null; then
        OFFLINE_MODE=1
        warn "离线模式检测：将使用本地缓存/预下载包"
    else
        OFFLINE_MODE=0
    fi
}

# ========== Docker 模式安装 ==========
install_docker() {
    log "=== Docker Compose 安装 ==="
    mkdir -p "$INSTALL_DIR"
    cd "$INSTALL_DIR"

    # 下载编排文件
    if [ "$OFFLINE_MODE" -eq 0 ]; then
        curl -fsSL "https://dragonsoul.dev/releases/v${DRAGONSOUL_VERSION}/docker-compose.yml"             -o docker-compose.yml
        curl -fsSL "https://dragonsoul.dev/releases/v${DRAGONSOUL_VERSION}/.env.example"             -o .env.example
    else
        cp "${OFFLINE_PACKAGE:-/tmp/dragonsoul-offline}"/docker-compose.yml . 2>/dev/null ||             error "离线包中缺少 docker-compose.yml"
    fi

    # 生成环境配置
    cat > .env <<EOF
# 龍魂 · 一带一路入门包 · 环境配置
DRAGONSOUL_VERSION=${DRAGONSOUL_VERSION}
MODEL_SIZE=${MODEL_SIZE}
DEFAULT_LANG=${LANG}
DATA_DIR=./data
MODELS_DIR=./models
LOGS_DIR=./logs
PORT=8080
OFFLINE_MODE=${OFFLINE_MODE}
EOF

    # 创建目录结构
    mkdir -p data models logs configs scenarios

    # 启动服务
    docker-compose up -d

    # 等待健康检查
    log "等待服务启动..."
    for i in {1..30}; do
        if curl -fsSL http://localhost:8080/health &>/dev/null; then
            log "服务启动成功 ✓"
            break
        fi
        sleep 2
    done

    # 下载模型（如未离线）
    if [ "$OFFLINE_MODE" -eq 0 ]; then
        log "下载多语言模型（约7GB，请等待）..."
        docker-compose exec dragonsoul ollama pull dragonsoul/belt-road-${MODEL_SIZE}-multilingual
    fi

    log "Docker 模式安装完成 ✓"
}

# ========== 裸机模式安装 ==========
install_bare() {
    log "=== 裸机模式安装（Ollama） ==="
    mkdir -p "$INSTALL_DIR"
    cd "$INSTALL_DIR"

    # 安装 Ollama（如未安装）
    if ! command -v ollama &> /dev/null; then
        log "安装 Ollama..."
        curl -fsSL https://ollama.com/install.sh | sh
    fi

    # 下载模型
    if [ "$OFFLINE_MODE" -eq 0 ]; then
        log "下载模型 dragonsoul/belt-road-${MODEL_SIZE}-multilingual..."
        ollama pull dragonsoul/belt-road-${MODEL_SIZE}-multilingual
    else
        log "离线模式：请手动将 .gguf 模型文件放入 $INSTALL_DIR/models/"
    fi

    # 安装 Python 依赖
    if command -v python3 &> /dev/null; then
        log "安装 Python 依赖..."
        pip3 install -q dragonsoul-belt-road==${DRAGONSOUL_VERSION} ||             pip3 install -q -e .
    fi

    # 创建启动脚本
    cat > start.sh <<'EOF'
#!/bin/bash
# 龍魂启动脚本
MODEL="${MODEL:-dragonsoul/belt-road-7b-multilingual}"
LANG="${DRAGONSOUL_LANG:-en}"
PORT="${PORT:-8080}"

echo "[龍魂] 启动服务..."
echo "模型: $MODEL"
echo "语言: $LANG"
echo "端口: $PORT"

# 启动 Ollama 服务（后台）
ollama serve &
sleep 3

# 启动龍魂网关
python3 -m dragonsoul.gateway --model "$MODEL" --lang "$LANG" --port "$PORT"
EOF
    chmod +x start.sh

    log "裸机模式安装完成 ✓"
    log "运行 ./start.sh 启动服务"
}

# ========== 验证安装 ==========
verify_install() {
    log "=== 安装验证 ==="

    # 健康检查
    if curl -fsSL http://localhost:8080/health &>/dev/null; then
        log "API 服务: 运行中 ✓"
    else
        warn "API 服务未响应，请检查日志"
    fi

    # 模型检查
    if command -v ollama &> /dev/null; then
        if ollama list | grep -q "belt-road"; then
            log "模型: 已加载 ✓"
        else
            warn "模型: 未加载"
        fi
    fi

    # 场景测试
    log "运行场景测试 AGRI-001..."
    if [ -f "scenarios/AGRI-001/test.json" ]; then
        python3 -m dragonsoul.test --scenario AGRI-001 --quiet &&             log "场景测试: 通过 ✓" || warn "场景测试: 失败"
    fi

    log "=== 安装验证完成 ==="
}

# ========== 主流程 ==========
main() {
    log "🌍 龍魂 · 一带一路老铁入门包 · 一键部署 v${DRAGONSOUL_VERSION}"
    log "DNA: #龍芯⚡️丙午·丙申·戊午·戊午·䷱鼎-BELT-ROAD-PACK-UID9622"

    check_system
    select_mode

    case "$MODE" in
        docker) install_docker ;;
        bare) install_bare ;;
    esac

    verify_install

    log ""
    log "═══════════════════════════════════════════════════════"
    log "  🎉 部署完成！"
    log "═══════════════════════════════════════════════════════"
    log "  访问地址: http://localhost:8080"
    log "  文档地址: http://localhost:8080/docs"
    log "  日志文件: $LOG_FILE"
    log "  安装目录: $INSTALL_DIR"
    log "  切换语言: export DRAGONSOUL_LANG=ar && ./start.sh"
    log "═══════════════════════════════════════════════════════"
}

main "$@"
```

## 十四、故障排查·FAQ

### 14.1 常见问题速查

| 症状 | 可能原因 | 解决方案 |
|------|----------|----------|
| 服务启动失败 | 端口冲突 | `export PORT=8081` 更换端口 |
| 模型加载超时 | 内存不足 | 换用更小模型或增加内存 |
| 阿拉伯语乱码 | 字体缺失 | 安装 `fonts-noto-arabic` |
| 离线模式无法推理 | 模型未缓存 | 首次需联网下载，或手动放置 `.gguf` |
| 场景测试失败 | 数据格式错误 | 检查 JSON Schema：`python -m dragonsoul.validate --input file.json` |
| Docker 容器退出 | 资源限制 | 调整 `docker-compose.yml` 中 `mem_limit` |
| 多语言切换无效 | 语言包未加载 | 确认 `lang-adapter-XX` 已安装 |
