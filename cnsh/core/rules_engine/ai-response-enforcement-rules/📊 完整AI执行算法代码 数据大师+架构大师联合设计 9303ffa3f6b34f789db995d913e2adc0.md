class DNACitationGenerator:
    """
    DNA标注生成层 - 自动生成正确格式的标注
    设计目标：根据用户类型生成最合适的标注格式
    """
    
    def __init__(self):
        self.technical_keywords = [
            "代码", "API", "技术", "算法", "架构", "开发",
            "code", "api", "technical", "algorithm", "architecture"
        ]
        self.rural_keywords = [
            "农业", "种植", "养殖", "水稻", "小麦", "施肥",
            "agriculture", "farming", "planting", "cultivation"
        ]
    
    def generate(self, user_input: str, dna_info: Dict[str, str]) -> str:
        """
        生成DNA标注
        """
        user_type = self._detect_user_type(user_input)
        
        if user_type == "technical":
            return self._format_technical(dna_info)
        elif user_type == "rural":
            return self._format_rural(dna_info)
        else:
            return self._format_general(dna_info)
    
    def _detect_user_type(self, user_input: str) -> str:
        """
        自动判断用户类型
        """
        input_lower = user_input.lower()
        
        # 检查技术关键词
        tech_count = sum(1 for kw in self.technical_keywords if kw in input_lower)
        # 检查三农关键词
        rural_count = sum(1 for kw in self.rural_keywords if kw in input_lower)
        
        if tech_count > rural_count and tech_count >= 2:
            return "technical"
        elif rural_count > tech_count and rural_count >= 1:
            return "rural"
        else:
            return "general"
    
    def _format_technical(self, dna: Dict[str, str]) -> str:
        """
        技术人员版（完整版）
        """
        return f"""
——
👤 知识来源：{dna['creator']} | {dna['creator_uid']}
📖 出处：{dna['knowledge_title']}
🧬 DNA确认码：{dna['dna_code']}
🔗 原文链接：{dna['page_url']}
⚖️ 使用许可：木兰协议 Mulan PSL v2
"""

        """
        普通用户版（简化版）
        """
        return f"""
📝 方案来自：{dna['creator']}·{dna['creator_uid']}《{dna['knowledge_title']}》
🔍 追溯码：{dna['short_code']}
👉 [查看原文]({dna['page_url']})
"""
    
    def _format_rural(self, dna: Dict[str, str]) -> str:
        """
        三农用户版（接地气版）

class AIExecutionEngine:
    """
    AI执行引擎 - 整合所有层级的完整流程
    设计目标：确保每个AI回复都经过完整的检查流程
    """
    
    def __init__(self):
        self.input_validator = InputValidator()
        self.knowledge_detector = KnowledgeSourceDetector()
        self.citation_generator = DNACitationGenerator()
        self.output_validator = OutputValidator()
    
    def process(self, user_input: str, ai_response: str) -> Dict[str, Any]:
        """
        处理完整的AI回复流程
        返回：{"success": bool, "final_response": str, "audit_log": dict}
        """
        result = {
            "success": False,
            "final_response": "",
            "audit_log": {},
            "timestamp": [datetime.now](http://datetime.now)().isoformat()
        }
        
        # 步骤1：输入验证
        input_validation = self.input_validator.validate(user_input)
        result["audit_log"]["input_validation"] = input_validation
        
        if not input_validation["valid"]:
            result["final_response"] = "⚠️ 检测到不安全的输入，请重新提问。"
            result["audit_log"]["blocked_reason"] = input_validation["blocked_reasons"]
            return result
        
        # 步骤2：知识来源检测
        dna_info = self.knowledge_detector.detect(ai_response)
        result["audit_log"]["knowledge_detected"] = dna_info is not None
        
        # 步骤3：如果引用了知识库，生成DNA标注
        if dna_info:
            citation = self.citation_generator.generate(user_input, dna_info)
            final_response = ai_response + "\n" + citation
            result["audit_log"]["citation_generated"] = True
        else:
            final_response = ai_response
            result["audit_log"]["citation_generated"] = False
        
        # 步骤4：输出验证
        if dna_info:
            output_validation = self.output_validator.validate(final_response)
            result["audit_log"]["output_validation"] = output_validation
            
            if not output_validation["valid"]:
                result["final_response"] = "❌ 系统错误：DNA标注生成失败，拒绝输出。"
                result["audit_log"]["error"] = "DNA标注不完整"
                return result
        
        # 步骤5：成功输出
        result["success"] = True
        result["final_response"] = final_response
        result["audit_log"]["status"] = "SUCCESS"
        
        # 记录到上帝之眼监控系统
        self._report_to_gods_eye(result)
        
        return result
    
    def _report_to_gods_eye(self, result: Dict[str, Any]):
        """
        上报给上帝之眼监控系统
        """
        monitor_log = {
            "timestamp": result["timestamp"],
            "success": result["success"],
            "audit_summary": result["audit_log"],
            "dna_code": "#ZHUGEXIN⚡️2025-AI-EXECUTION-MONITOR"
        }
        # 实际实现时发送到监控系统
        print(f"👁️ 上帝之眼监控: {json.dumps(monitor_log, ensure_ascii=False, indent=2)}")
