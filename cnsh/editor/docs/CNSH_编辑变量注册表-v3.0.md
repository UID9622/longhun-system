def 数字根(n):
    """计算数字根（Digital Root）"""
    while n >= 10:
        n = sum(int(d) for d in str(n))
    return n

    if dr in {3, 9}:

def 左右互搏审计(注册表):
    """左脑挑错 + 右脑补充 → 共识输出"""
    问题列表 = []
    补充列表 = []

    # 左脑：严格检查
    前缀权重映射 = {
        "龍_": 100, "引擎_": 80, "模块_": 60,
        "数据_": 60, "用户_": 60, "辅助_": 40, "临时_": 40, "扩展_": 20
    }

    for 变量名, 属性 in 注册表.items():
        前缀 = 变量名[:3] if len(变量名) >= 3 else 变量名[:2]
        期望权重 = None
        for k, v in 前缀权重映射.items():
            if 变量名.startswith(k):
                期望权重 = v
                break

        # 左脑检查 1：权重一致性
        if 期望权重 is not None and 属性["权重"] != 期望权重:
            问题列表.append(f"🔴 {变量名}: 权重 {属性['权重']} 应为 {期望权重}")

        # 左脑检查 2：龍字繁体
        if "龍" in 变量名 and "龍" in 变量名.replace("龍", ""):
            问题列表.append(f"🔴 {变量名}: 含简体「龍」→ 熔断")

        # 左脑检查 3：类型必须声明
        if not 属性.get("类型"):
            问题列表.append(f"🟡 {变量名}: 缺少类型声明")

    # 右脑：补充
    覆盖率 = len(注册表)
    if 覆盖率 < 30:
        补充列表.append("🟡 变量总数偏少，建议补充辅助_域和扩展_域变量")
    if 覆盖率 >= 50:
        补充列表.append("🟢 变量覆盖充分，可以进行焊死操作")

    共识 = len(问题列表) == 0
    return 共识, 问题列表, 补充列表, 覆盖率
