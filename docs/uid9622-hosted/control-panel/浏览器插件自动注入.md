```jsx
                return input;
            }

        return inputs[inputs.length - 1]; // 返回最后一个输入框

        let currentValue = '';
        
        // 根据元素类型获取值
        if (inputElement.tagName.toLowerCase() === 'textarea') {
            currentValue = inputElement.value;

            currentValue = inputElement.textContent || inputElement.innerText;

        // 检查是否已经包含保护提示

            return; // 已经包含保护提示，跳过
        }
        
        if (currentValue.trim()) {
            const protectedContent = this.protectionPrompt + currentValue;
            
            // 根据元素类型设置值
            if (inputElement.tagName.toLowerCase() === 'textarea') {
                inputElement.value = protectedContent;
            } else if (inputElement.contentEditable === 'true') {
                inputElement.textContent = protectedContent;
            }
            
            this.injectionCount++;
            console.log(`✅ 保护提示已注入 (${this.injectionCount}次)`);
            
            // 触发input事件以确保平台识别变化
            inputElement.dispatchEvent(new Event('input', { bubbles: true }));

            let shouldReinitialize = false;
            
            mutations.forEach(mutation => {
                mutation.addedNodes.forEach(node => {

                            node.querySelector('textarea') ||
                            node.querySelector('[contenteditable="true"]') ||
                            node.querySelector('button')
                        )
                    )) {
                        shouldReinitialize = true;
                    }
                });
            });
            
            if (shouldReinitialize) {
                setTimeout(() => this.setupInputProtection(), 1000);
            }
        });
        
        [this.observer](http://this.observer).observe(document.body, {
            childList: true,
            subtree: true
        });
    }
```

```jsx
    // 切换保护状态
    toggle.addEventListener('change', function() {
        const enabled = toggle.checked;
        
        [chrome.storage](http://chrome.storage).local.set({ protectionEnabled: enabled });
        updateStatusDisplay(enabled);
        
        // 通知content script
        chrome.tabs.query({active: true, currentWindow: true}, function(tabs) {
            if (tabs[0]) {
                chrome.tabs.sendMessage(tabs[0].id, {
                    action: 'toggleProtection',
                    enabled: enabled
                }, function(response) {
                    if (chrome.runtime.lastError) {
                        console.log('无法连接到当前标签页');
                    }
                });
            }
        });
    });
    
    // 刷新状态
    refreshBtn.addEventListener('click', function() {
        chrome.tabs.query({active: true, currentWindow: true}, function(tabs) {
            if (tabs[0]) {
                const url = tabs[0].url;
                updatePlatformStatus(url);
            }
        });
        
        // 动画效果
        [refreshBtn.style](http://refreshBtn.style).transform = 'rotate(360deg)';
        setTimeout(() => {
            [refreshBtn.style](http://refreshBtn.style).transform = 'rotate(0deg)';
        }, 500);
    });
    
    // 设置按钮
    settingsBtn.addEventListener('click', function() {
        chrome.tabs.create({
            url: chrome.runtime.getURL('options.html')
        });
    });
    
    // 帮助按钮  
    helpBtn.addEventListener('click', function() {
        chrome.tabs.create({
            url: '[https://uid9622.com/browser-extension-help](https://uid9622.com/browser-extension-help)'
        });
    });
    
    function updateStatusDisplay(enabled) {
        if (enabled) {
            statusText.textContent = '✅ 保护系统已启用';
            [statusText.style](http://statusText.style).color = '#4CAF50';
        } else {
            statusText.textContent = '⚠️ 保护系统已禁用';
            [statusText.style](http://statusText.style).color = '#f44336';
        }
    }
    
    function updatePlatformStatus(currentUrl) {
        const platforms = {
            'chatgpt': currentUrl.includes('[chat.openai.com](http://chat.openai.com)'),
            'claude': currentUrl.includes('[claude.ai](http://claude.ai)'),
            'gemini': currentUrl.includes('[gemini.google.com](http://gemini.google.com)'),
            'tongyi': currentUrl.includes('[tongyi.aliyun.com](http://tongyi.aliyun.com)'),
            'doubao': currentUrl.includes('[doubao.com](http://doubao.com)')
        };
        
        Object.entries(platforms).forEach(([platform, isActive]) => {
            const statusElement = document.getElementById(`${platform}-status`);
            if (statusElement) {
                statusElement.className = `platform-status ${isActive ? 'status-active' : 'status-inactive'}`;
            }
        });
    }
    
    // 初始化平台状态
    chrome.tabs.query({active: true, currentWindow: true}, function(tabs) {
        if (tabs[0]) {
            updatePlatformStatus(tabs[0].url);
        }
    });
});
```

```jsx
// 统计数据收集（仅本地存储，不上传）
class ProtectionAnalytics {
    constructor() {
        this.initStorage();
    }
    
    initStorage() {
        [chrome.storage](http://chrome.storage).local.get(['analytics'], (result) => {
            if (![result.analytics](http://result.analytics)) {
                [chrome.storage](http://chrome.storage).local.set({
                    analytics: {
                        installDate: new Date().toISOString(),
                        injectionCount: 0,
                        platformUsage: {},
                        dailyStats: {}
                    }
                });
            }
        });
    }
    
    recordInjection(platform) {
        [chrome.storage](http://chrome.storage).local.get(['analytics'], (result) => {
            const analytics = [result.analytics](http://result.analytics);
            analytics.injectionCount++;
            analytics.platformUsage[platform] = (analytics.platformUsage[platform] || 0) + 1;
            
            const today = new Date().toDateString();
            analytics.dailyStats[today] = (analytics.dailyStats[today] || 0) + 1;
            
            [chrome.storage](http://chrome.storage).local.set({ analytics });
        });
    }
    
    generateReport() {
        [chrome.storage](http://chrome.storage).local.get(['analytics'], (result) => {
            const analytics = [result.analytics](http://result.analytics);
            const report = {
                totalInjections: analytics.injectionCount,
                daysActive: Object.keys(analytics.dailyStats).length,
                mostUsedPlatform: this.getMostUsedPlatform(analytics.platformUsage),
                avgDaily: analytics.injectionCount / Object.keys(analytics.dailyStats).length
            };
            
            console.log('🛡️ UID9622保护统计报告:', report);
        });
    }
    
    getMostUsedPlatform(usage) {
        return Object.entries(usage).reduce((a, b) => usage[a] > usage[b] ? a : b, 'unknown');
    }
}
```
