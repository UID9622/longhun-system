#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""📚 中央藏经阁存储系统 v3.0"""

import chromadb
import sqlite3
from pathlib import Path
from datetime import datetime

class 中央藏经阁:
    def __init__(self, 路径="~/龍魂系统/藏经阁"):
        路径 = Path(路径).expanduser()
        路径.mkdir(parents=True, exist_ok=True)
        
        # 初始化Chroma向量库
        self.向量库 = chromadb.PersistentClient(path=str(路径))
        self.集合 = self.向量库.get_or_create_collection("气韵转化")
        
        # 初始化SQLite关系库
        self.关系库 = sqlite3.connect(str(路径 / "术语映射.db"))
        self.初始化表结构()
        
        print(f"✅ 中央藏经阁初始化完成：{路径}")
    
    def 初始化表结构(self):
        """创建数据表"""
        cursor = self.关系库.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS 术语映射表 (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                外部术语 TEXT UNIQUE,
                龍魂术语 TEXT,
                来源 TEXT,
                更新时间 DATETIME DEFAULT CURRENT_TIMESTAMP
            )
        """)
        self.关系库.commit()
    
    def 存储(self, 外部术语, 龍魂术语, 来源="系统"):
        """存储术语映射"""
        cursor = self.关系库.cursor()
        cursor.execute("""
            INSERT OR REPLACE INTO 术语映射表
            (外部术语, 龍魂术语, 来源, 更新时间)
            VALUES (?, ?, ?, datetime('now'))
        """, (外部术语, 龍魂术语, 来源))
        self.关系库.commit()
        print(f"✅ 存储成功：{外部术语} → {龍魂术语}")
    
    def 查询(self, 外部术语):
        """查询术语映射"""
        cursor = self.关系库.cursor()
        cursor.execute(
            "SELECT 龍魂术语, 来源, 更新时间 FROM 术语映射表 WHERE 外部术语 = ?",
            (外部术语,)
        )
        result = cursor.fetchone()
        if result:
            return {
                "龍魂术语": result[0],
                "来源": result[1],
                "更新时间": result[2]
            }
        return None
    
    def 列出所有术语(self):
        """列出所有存储的术语"""
        cursor = self.关系库.cursor()
        cursor.execute(
            "SELECT 外部术语, 龍魂术语, 来源 FROM 术语映射表 ORDER BY id"
        )
        results = cursor.fetchall()
        
        print(f"\n📚 藏经阁术语清单（共{len(results)}条）")
        print("=" * 70)
        for 外, 龍, 源 in results:
            print(f"  {外:25s} → {龍:15s} (来源: {源})")
        print("=" * 70)
    
    def 批量存储(self, 术语列表):
        """批量存储术语映射"""
        for 外部术语, 龍魂术语 in 术语列表.items():
            self.存储(外部术语, 龍魂术语)
        print(f"\n✅ 批量存储完成，共 {len(术语列表)} 个术语")
    
    def __del__(self):
        """析构函数，关闭数据库连接"""
        if hasattr(self, '关系库'):
            self.关系库.close()

if __name__ == "__main__":
    藏经阁 = 中央藏经阁()
    
    print("\n🐉 中央藏经阁测试")
    print("=" * 70)
    
    print("\n测试1：单个存储")
    藏经阁.存储("Prompt", "道令", "系统")
    藏经阁.存储("Agent", "灵使", "系统")
    藏经阁.存储("RAG", "博古通今", "系统")
    
    print("\n测试2：查询术语")
    result = 藏经阁.查询("Prompt")

    print("\n测试3：批量存储")
