# 日志分析与清理工具

代码内容: #!/bin/bash
# UID9622 日志分析与清理工具
# 智能分析系统日志，自动清理过期文件

# 颜色设置
RED='033[0;31m'
GREEN='033[0;32m'
YELLOW='033[1;33m'
BLUE='033[0;34m'
PURPLE='033[0;35m'
NC='033[0m'

# 配置变量
LOG_DIRS=(
    "/var/log"
    "/usr/local/var/log"
    "$HOME/Library/Logs"  # macOS
    "~/.local/share/logs"
    "./logs"
)

MAX_AGE_DAYS=30
MAX_SIZE_MB=100
BACKUP_DIR="$HOME/log_backups"
REPORT_FILE="log_analysis_$(date +%Y%m%d_%H%M%S).txt"

# 显示帮助信息
show_help() {
    echo -e "${BLUE}UID9622 日志分析与清理工具${NC}"
    echo "===================================="
    echo "用法: $0 [options]"
    echo ""
    echo "参数:"
    echo "  -a, --analyze      分析日志文件"
    echo "  -c, --clean        清理过期日志"
    echo "  -s, --size-check   检查大文件"
    echo "  -e, --errors       提取错误信息"
    echo "  -d, --directory    指定日志目录"
    echo "  -t, --days         保留天数 (默认: $MAX_AGE_DAYS)"
    echo "  --dry-run          预览模式不实际删除"
    echo "  -h, --help         显示帮助信息"
    echo ""
    echo "示例:"
    echo "  $0 --analyze                    # 分析所有日志"
    echo "  $0 --clean --days 7             # 清理 7 天前的日志"
    echo "  $0 -d /var/log -e               # 分析指定目录错误"
    echo "  $0 --clean --dry-run            # 预览清理操作"
}

# 记录日志
log_message() {
    local level=$1
    local message=$2
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    
    case $level in
        "INFO")
            echo -e "${BLUE}[信息]${NC} $message" | tee -a "$REPORT_FILE"
            ;;
        "SUCCESS")
            echo -e "${GREEN}[成功]${NC} $message" | tee -a "$REPORT_FILE"
            ;;
        "WARNING")
            echo -e "${YELLOW}[警告]${NC} $message" | tee -a "$REPORT_FILE"
            ;;
        "ERROR")
            echo -e "${RED}[错误]${NC} $message" | tee -a "$REPORT_FILE"
            ;;
        "STAT")
            echo -e "${PURPLE}[统计]${NC} $message" | tee -a "$REPORT_FILE"
            ;;
    esac
    
    echo "$timestamp [$level] $message" >> "$REPORT_FILE"
}

# 获取文件大小（MB）
get_file_size_mb() {
    local file="$1"
    if [[ "$OSTYPE" == "darwin" ]]; then
        # macOS
        stat -f%z "$file" 2>/dev/null | awk '{print int($1/1024/1024)}'
    else
        # Linux
        stat -c%s "$file" 2>/dev/null | awk '{print int($1/1024/1024)}'
    fi
}

# 检查目录是否存在且可访问
check_directory() {
    local dir="$1"
    
    # 展开波浪号
    dir=$(eval echo "$dir")
    
    if [[ ! -d "$dir" ]]; then
        return 1
    fi
    
    if [[ ! -r "$dir" ]]; then
        log_message "WARNING" "目录不可读: $dir"
        return 1
    fi
    
    return 0
}

# 分析日志文件
analyze_logs() {
    local target_dir="$1"
    
    log_message "INFO" "=== 开始分析日志文件 ==="
    
    local total_files=0
    local total_size_mb=0
    local large_files=0
    local old_files=0
    
    # 如果指定目录，只分析指定目录
    local dirs_to_check
    if [[ -n "$target_dir" ]]; then
        dirs_to_check=("$target_dir")
    else
        dirs_to_check=("${LOG_DIRS[@]}")
    fi
    
    for dir in "${dirs_to_check[@]}"; do
        if ! check_directory "$dir"; then
            continue
        fi
        
        dir=$(eval echo "$dir")  # 展开变量
        log_message "INFO" "分析目录: $dir"
        
        # 查找日志文件
        while IFS= read -r -d '' file; do
            if [[ -f "$file" ]]; then
                local file_size_mb=$(get_file_size_mb "$file")
                local file_age_days
                
                if [[ "$OSTYPE" == "darwin" ]]; then
                    # macOS
                    file_age_days=$(( ($(date +%s) - $(stat -f%m "$file")) / 86400 ))
                else
                    # Linux
                    file_age_days=$(( ($(date +%s) - $(stat -c%Y "$file")) / 86400 ))
                fi
                
                total_files=$((total_files + 1))
                total_size_mb=$((total_size_mb + file_size_mb))
                
                # 检查大文件
                if [[ $file_size_mb -gt $MAX_SIZE_MB ]]; then
                    large_files=$((large_files + 1))
                    log_message "WARNING" "大文件: $file (${file_size_mb}MB)"
                fi
                
                # 检查老文件
                if [[ $file_age_days -gt $MAX_AGE_DAYS ]]; then
                    old_files=$((old_files + 1))
                    log_message "INFO" "过期文件: $file (${file_age_days}天前)"
                fi
            fi
        done < <(find "$dir" -type f ( -name ".log" -o -name ".log." -o -name ".out" -o -name ".err" ) -print0 2>/dev/null)
    done
    
    # 显示统计信息
    log_message "STAT" "总计: $total_files 个日志文件"
    log_message "STAT" "总大小: ${total_size_mb}MB"
    log_message "STAT" "大文件: $large_files 个 (>${MAX_SIZE_MB}MB)"
    log_message "STAT" "过期文件: $old_files 个 (>${MAX_AGE_DAYS}天)"
}

# 提取错误信息
extract_errors() {
    local target_dir="$1"
    
    log_message "INFO" "=== 提取错误信息 ==="
    
    local dirs_to_check
    if [[ -n "$target_dir" ]]; then
        dirs_to_check=("$target_dir")
    else
        dirs_to_check=("${LOG_DIRS[@]}")
    fi
    
    local error_patterns=(
        "ERROR"
        "FATAL"
        "CRITICAL"
        "EXCEPTION"
        "FAILED"
        "PANIC"
        "SEGFAULT"
    )
    
    for dir in "${dirs_to_check[@]}"; do
        if ! check_directory "$dir"; then
            continue
        fi
        
        dir=$(eval echo "$dir")
        
        # 查找近24小时内的日志文件
        while IFS= read -r -d '' file; do
            if [[ -f "$file" ]]; then
                # 检查文件是否在近24小时内修改
                local file_age_hours
                if [[ "$OSTYPE" == "darwin" ]]; then
                    file_age_hours=$(( ($(date +%s) - $(stat -f%m "$file")) / 3600 ))
                else
                    file_age_hours=$(( ($(date +%s) - $(stat -c%Y "$file")) / 3600 ))
                fi
                
                if [[ $file_age_hours -le 24 ]]; then
                    for pattern in "${error_patterns[@]}"; do
                        local error_count=$(grep -c "$pattern" "$file" 2>/dev/null || echo 0)
                        if [[ $error_count -gt 0 ]]; then
                            log_message "ERROR" "$file: $error_count 个 $pattern 错误"
                            # 显示最近3条错误
                            log_message "INFO" "最新错误示例:"
                            grep "$pattern" "$file" | tail -3 | while read -r line; do
                                log_message "INFO" "  $line"
                            done
                        fi
                    done
                fi
            fi
        done < <(find "$dir" -type f ( -name ".log" -o -name ".log." ) -print0 2>/dev/null)
    done
}

# 清理过期日志
clean_old_logs() {
    local target_dir="$1"
    local is_dry_run="$2"
    
    log_message "INFO" "=== 清理过期日志 (${MAX_AGE_DAYS}天前) ==="
    
    if [[ "$is_dry_run" == "true" ]]; then
        log_message "INFO" "[预览模式] 不会实际删除文件"
    fi
    
    # 创建备份目录
    if [[ "$is_dry_run" != "true" ]] && [[ ! -d "$BACKUP_DIR" ]]; then
        mkdir -p "$BACKUP_DIR"
        log_message "INFO" "创建备份目录: $BACKUP_DIR"
    fi
    
    local dirs_to_check
    if [[ -n "$target_dir" ]]; then
        dirs_to_check=("$target_dir")
    else
        dirs_to_check=("${LOG_DIRS[@]}")
    fi
    
    local deleted_count=0
    local backed_up_count=0
    local freed_space_mb=0
    
    for dir in "${dirs_to_check[@]}"; do
        if ! check_directory "$dir"; then
            continue
        fi
        
        dir=$(eval echo "$dir")
        
        # 查找过期文件
        while IFS= read -r -d '' file; do
            if [[ -f "$file" ]]; then
                local file_age_days
                if [[ "$OSTYPE" == "darwin" ]]; then
                    file_age_days=$(( ($(date +%s) - $(stat -f%m "$file")) / 86400 ))
                else
                    file_age_days=$(( ($(date +%s) - $(stat -c%Y "$file")) / 86400 ))
                fi
                
                if [[ $file_age_days -gt $MAX_AGE_DAYS ]]; then
                    local file_size_mb=$(get_file_size_mb "$file")
                    freed_space_mb=$((freed_space_mb + file_size_mb))
                    
                    if [[ "$is_dry_run" == "true" ]]; then
                        log_message "INFO" "[预览] 将删除: $file (${file_size_mb}MB, ${file_age_days}天前)"
                        deleted_count=$((deleted_count + 1))
                    else
                        # 备份重要文件
                        if [[ $file_size_mb -gt 50 ]] || [[ "$file" == "error" ]] || [[ "$file" == "critical" ]]; then
                            local backup_file="$BACKUP_DIR/$(basename "$file")$(date +%Y%m%d)"
                            if cp "$file" "$backup_file" 2>/dev/null; then
                                log_message "SUCCESS" "备份成功: $file -> $backup_file"
                                backed_up_count=$((backed_up_count + 1))
                            fi
                        fi
                        
                        # 删除文件
                        if rm "$file" 2>/dev/null; then
                            log_message "SUCCESS" "删除成功: $file (${file_size_mb}MB)"
                            deleted_count=$((deleted_count + 1))
                        else
                            log_message "ERROR" "删除失败: $file"
                        fi
                    fi
                fi
            fi
        done < <(find "$dir" -type f ( -name ".log" -o -name ".log." -o -name ".out" ) -print0 2>/dev/null)
    done
    
    log_message "STAT" "处理统计: $deleted_count 个文件"
    log_message "STAT" "备份文件: $backed_up_count 个"
    log_message "STAT" "释放空间: ${freed_space_mb}MB"
}

# 检查大文件
check_large_files() {
    local target_dir="$1"
    
    log_message "INFO" "=== 检查大文件 (>${MAX_SIZE_MB}MB) ==="
    
    local dirs_to_check
    if [[ -n "$target_dir" ]]; then
        dirs_to_check=("$target_dir")
    else
        dirs_to_check=("${LOG_DIRS[@]}")
    fi
    
    for dir in "${dirs_to_check[@]}"; do
        if ! check_directory "$dir"; then
            continue
        fi
        
        dir=$(eval echo "$dir")
        
        # 使用 find 查找大文件
        find "$dir" -type f ( -name ".log" -o -name ".log." -o -name ".out" ) -size +${MAX_SIZE_MB}M 2>/dev/null | while read -r file; do
            local file_size_mb=$(get_file_size_mb "$file")
            log_message "WARNING" "发现大文件: $file (${file_size_mb}MB)"
            
            # 提供处理建议
            log_message "INFO" "处理建议:"
            log_message "INFO" "  1. 压缩: gzip '$file'"
            log_message "INFO" "  2. 截断: tail -n 1000 '$file' > '${file}.recent' && rm '$file'"
            log_message "INFO" "  3. 分割: split -l 10000 '$file' '${file}.part'"
        done
    done
}

# 主函数
main() {
    local do_analyze=false
    local do_clean=false
    local do_size_check=false
    local do_extract_errors=false
    local target_directory=""
    local is_dry_run=false
    
    # 初始化报告文件
    echo "UID9622 日志分析报告 - $(date)" > "$REPORT_FILE"
    echo "========================================" >> "$REPORT_FILE"
    
    # 解析命令行参数
    while [[ $# -gt 0 ]]; do
        case $1 in
            -a|--analyze)
                do_analyze=true
                shift
                ;;
            -c|--clean)
                do_clean=true
                shift
                ;;
            -s|--size-check)
                do_size_check=true
                shift
                ;;
            -e|--errors)
                do_extract_errors=true
                shift
                ;;
            -d|--directory)
                target_directory="$2"
                shift 2
                ;;
            -t|--days)
                MAX_AGE_DAYS="$2"
                shift 2
                ;;
            --dry-run)
                is_dry_run=true
                shift
                ;;
            -h|--help)
                show_help
                exit 0
                ;;
            *)
                echo "未知参数: $1"
                show_help
                exit 1
                ;;
        esac
    done
    
    # 如果没有参数，显示帮助
    if [[ $do_analyze == false && $do_clean == false && $do_size_check == false && $do_extract_errors == false ]]; then
        show_help
        exit 1
    fi
    
    log_message "INFO" "UID9622 日志分析工具启动"
    log_message "INFO" "报告文件: $REPORT_FILE"
    
    # 执行相应操作
    [[ $do_analyze == true ]] && analyze_logs "$target_directory"
    [[ $do_size_check == true ]] && check_large_files "$target_directory"
    [[ $do_extract_errors == true ]] && extract_errors "$target_directory"
    [[ $do_clean == true ]] && clean_old_logs "$target_directory" "$is_dry_run"
    
    log_message "SUCCESS" "日志分析完成，详细结果请查看: $REPORT_FILE"
}

# 运行主函数
main "$@"
