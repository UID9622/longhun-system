"""

安全管理和配置环境变量，支持加密存储
"""

import os
import sys
import json
import base64
import argparse
from pathlib import Path
from datetime import datetime
import subprocess

try:
    from cryptography.fernet import Fernet
    CRYPTO_AVAILABLE = True
except ImportError:
    CRYPTO_AVAILABLE = False
    print("⚠️  加密功能不可用，请安装: pip install cryptography")

class EnvManager:
    def init(self, config_dir=None):
        self.config_dir = Path(config_dir or Path.home() / ".uid9622_env")
        self.config_dir.mkdir(exist_ok=True)
        
        self.env_file = self.config_dir / "environments.json"
        self.key_file = self.config_dir / "env.key"
        self.templates_file = self.config_dir / "templates.json"
        
        self.environments = self.load_environments()
        self.templates = self.load_templates()
        
        # 初始化加密密钥
        if CRYPTO_AVAILABLE:
            self.cipher = self.init_encryption()
    
    def init_encryption(self):
        """初始化加密系统"""
        if self.key_file.exists():
            with open(self.key_file, 'rb') as f:
                key = http://f.read()
        else:
            key = Fernet.generate_key()
            with open(self.key_file, 'wb') as f:
                f.write(key)
            # 设置严格的文件权限
            os.chmod(self.key_file, 0o600)
        
        return Fernet(key)
    
    def encrypt_value(self, value):
        """加密值"""
        if not CRYPTO_AVAILABLE:
            return value
        
        if isinstance(value, str):
            value = value.encode()
        return self.cipher.encrypt(value).decode()
    
    def decrypt_value(self, encrypted_value):
        """解密值"""
        if not CRYPTO_AVAILABLE:
            return encrypted_value
        
        try:
            if isinstance(encrypted_value, str):
                encrypted_value = encrypted_value.encode()
            return self.cipher.decrypt(encrypted_value).decode()
        except:
            # 如果解密失败，可能是明文
            return encrypted_value.decode() if isinstance(encrypted_value, bytes) else encrypted_value
    
    def load_environments(self):
        """加载环境配置"""
        if self.env_file.exists():
            with open(self.env_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        return {}
    
    def save_environments(self):
        """保存环境配置"""
        with open(self.env_file, 'w', encoding='utf-8') as f:
            json.dump(self.environments, f, indent=2, ensure_ascii=False)
        os.chmod(self.env_file, 0o600)
    
    def load_templates(self):
        """加载模板配置"""
        if self.templates_file.exists():
            with open(self.templates_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        return self.get_default_templates()
    
    def save_templates(self):
        """保存模板配置"""
        with open(self.templates_file, 'w', encoding='utf-8') as f:
            json.dump(self.templates, f, indent=2, ensure_ascii=False)
    
    def get_default_templates(self):
        """获取默认模板"""
        return {
            "python-dev": {
                "description": "Python 开发环境",
                "variables": {
                    "PYTHONPATH": "./src:./lib",
                    "PYTHON_ENV": "development",
                    "DEBUG": "true",
                    "LOG_LEVEL": "INFO"
                }
            },
            "web-dev": {
                "description": "Web 开发环境",
                "variables": {
                    "NODE_ENV": "development",
                    "PORT": "3000",
                    "API_URL": "http://localhost:8000",
                    "DEBUG_MODE": "true"
                }
            },
            "database": {
                "description": "数据库连接配置",
                "variables": {
                    "DB_HOST": "http://localhost",
                    "DB_PORT": "5432",
                    "DB_NAME": "myapp",
                    "DB_USER": "username",
                    "DB_PASSWORD": "加密密码"
                },
                "encrypted_vars": ["DB_PASSWORD"]
            }
        }
    
    def create_environment(self, name, description="", template=None):
        """创建新环境"""
        if name in self.environments:
            print(f"❌ 环境 '{name}' 已存在")
            return False
        
        env_config = {
            "description": description,
            "created_at": http://datetime.now().isoformat(),
            "variables": {},
            "encrypted_vars": []
        }
        
        # 使用模板
        if template and template in self.templates:
            template_config = self.templates[template]
            env_config["description"] = description or template_config["description"]
            env_config["variables"] = template_config["variables"].copy()
            env_config["encrypted_vars"] = template_config.get("encrypted_vars", [])
            
            # 加密指定变量
            for var in env_config["encrypted_vars"]:
                if var in env_config["variables"]:
                    env_config["variables"][var] = self.encrypt_value(env_config["variables"][var])
        
        self.environments[name] = env_config
        http://self.save_environments()
        print(f"✅ 创建环境 '{name}' 成功")
        return True
    
    def delete_environment(self, name):
        """删除环境"""
        if name not in self.environments:
            print(f"❌ 环境 '{name}' 不存在")
            return False
        
        del self.environments[name]
        http://self.save_environments()
        print(f"✅ 删除环境 '{name}' 成功")
        return True
    
    def list_environments(self):
        """列出所有环境"""
        if not self.environments:
            print("📁 没有配置环境")
            return
        
        print("🌍 环境列表:")
        print("="  50)
        
        for name, config in self.environments.items():
            var_count = len(config.get("variables", {}))
            encrypted_count = len(config.get("encrypted_vars", []))
            created = config.get("created_at", "Unknown")
            
            print(f"📝 {name}")
            print(f"   描述: {config.get('description', 'No description')}")
            print(f"   变量: {var_count} 个 (加密: {encrypted_count} 个)")
            print(f"   创建: {created[:19] if created != 'Unknown' else created}")
            print()
    
    def set_variable(self, env_name, var_name, var_value, encrypt=False):
        """设置环境变量"""
        if env_name not in self.environments:
            print(f"❌ 环境 '{env_name}' 不存在")
            return False
        
        env_config = self.environments[env_name]
        
        # 加密存储
        if encrypt:
            var_value = self.encrypt_value(var_value)
            if var_name not in env_config.get("encrypted_vars", []):
                env_config.setdefault("encrypted_vars", []).append(var_name)
        
        env_config.setdefault("variables", {})[var_name] = var_value
        http://self.save_environments()
        
        status = "🔒 [加密]" if encrypt else "🔓 [明文]"
        print(f"✅ {status} 设置变量 {env_name}.{var_name}")
        return True
    
    def get_variable(self, env_name, var_name, decrypt=True):
        """获取环境变量"""
        if env_name not in self.environments:
            print(f"❌ 环境 '{env_name}' 不存在")
            return None
        
        env_config = self.environments[env_name]
        variables = env_config.get("variables", {})
        
        if var_name not in variables:
            print(f"❌ 变量 '{var_name}' 不存在")
            return None
        
        value = variables[var_name]
        
        # 解密
        if decrypt and var_name in env_config.get("encrypted_vars", []):
            value = self.decrypt_value(value)
        
        return value
    
    def show_environment(self, env_name, show_encrypted=False):
        """显示环境详情"""
        if env_name not in self.environments:
            print(f"❌ 环境 '{env_name}' 不存在")
            return
        
        env_config = self.environments[env_name]
        encrypted_vars = env_config.get("encrypted_vars", [])
        
        print(f"🌍 环境: {env_name}")
        print(f"📄 描述: {env_config.get('description', 'No description')}")
        print(f"📅 创建: {env_config.get('created_at', 'Unknown')}")
        print("n🔧 环境变量:")
        print("="  30)
        
        variables = env_config.get("variables", {})
        if not variables:
            print("没有配置变量")
            return
        
        for var_name, var_value in variables.items():
            if var_name in encrypted_vars:
                if show_encrypted:
                    decrypted_value = self.decrypt_value(var_value)
                    print(f"🔒 {var_name} = {decrypted_value}")
                else:
                    print(f"🔒 {var_name} = * [加密] 使用 --show-encrypted 查看")
            else:
                print(f"🔓 {var_name} = {var_value}")
    
    def export_environment(self, env_name, format_type="bash", output_file=None):
        """导出环境变量"""
        if env_name not in self.environments:
            print(f"❌ 环境 '{env_name}' 不存在")
            return False
        
        env_config = self.environments[env_name]
        variables = env_config.get("variables", {})
        encrypted_vars = env_config.get("encrypted_vars", [])
        
        if not variables:
            print("没有变量可导出")
            return False
        
        # 生成导出内容
        export_lines = []
        export_lines.append(f"# UID9622 环境变量: {env_name}")
        export_lines.append(f"# 生成时间: {http://datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        export_lines.append("")
        
        for var_name, var_value in variables.items():
            # 解密变量
            if var_name in encrypted_vars:
                var_value = self.decrypt_value(var_value)
            
            if format_type == "bash":
                export_lines.append(f"export {var_name}='{var_value}'")
            elif format_type == "fish":
                export_lines.append(f"set -x {var_name} '{var_value}'")
            elif format_type == "powershell":
                export_lines.append(f"$env:{var_name} = '{var_value}'")
            elif format_type == "docker":
                export_lines.append(f"{var_name}={var_value}")
            elif format_type == "json":
                # JSON 格式单独处理
                pass
        
        # JSON 格式特殊处理
        if format_type == "json":
            json_data = {}
            for var_name, var_value in variables.items():
                if var_name in encrypted_vars:
                    var_value = self.decrypt_value(var_value)
                json_data[var_name] = var_value
            export_content = json.dumps(json_data, indent=2, ensure_ascii=False)
        else:
            export_content = "n".join(export_lines)
        
        # 输出到文件或控制台
        if output_file:
            with open(output_file, 'w', encoding='utf-8') as f:
                f.write(export_content)
            print(f"✅ 导出到文件: {output_file}")
        else:
            print(export_content)
        
        return True
    
    def import_from_file(self, env_name, file_path, format_type="auto"):
        """从文件导入环境变量"""
        if not Path(file_path).exists():
            print(f"❌ 文件不存在: {file_path}")
            return False
        
        with open(file_path, 'r', encoding='utf-8') as f:
            content = http://f.read()
        
        # 自动检测格式
        if format_type == "auto":
            if file_path.endswith('.json'):
                format_type = "json"
            elif file_path.endswith('.env'):
                format_type = "env"
            else:
                format_type = "bash"
        
        variables = {}
        
        try:
            if format_type == "json":
                data = json.loads(content)
                if isinstance(data, dict):
                    variables = data
            else:
                # 解析 bash/env 格式
                for line in content.split('n'):
                    line = line.strip()
                    if line and not line.startswith('#'):
                        if '=' in line:
                            key, value = line.split('=', 1)
                            key = key.replace('export ', '').strip()
                            value = value.strip().strip('"').strip("'")
                            variables[key] = value
        except Exception as e:
            print(f"❌ 解析文件失败: {e}")
            return False
        
        if not variables:
            print("❌ 没有解析到变量")
            return False
        
        # 创建或更新环境
        if env_name not in self.environments:
            self.create_environment(env_name, f"从 {file_path} 导入")
        
        env_config = self.environments[env_name]
        env_config.setdefault("variables", {}).update(variables)
        http://self.save_environments()
        
        print(f"✅ 导入 {len(variables)} 个变量到环境 '{env_name}'")
        return True
    
    def activate_environment(self, env_name, shell="bash"):
        """生成激活环境的脚本"""
        if env_name not in self.environments:
            print(f"❌ 环境 '{env_name}' 不存在")
            return None
        
        script_file = self.config_dir / f"activate_{env_name}.{shell}"
        success = self.export_environment(env_name, shell, str(script_file))
        
        if success:
            print(f"🎆 环境激洿脚本: {script_file}")
            print(f"📄 使用方法: source {script_file}")
            return str(script_file)
        
        return None

    set_parser.add_argument('--encrypt', action='store_true', help='加密存储')

    get_parser = subparsers.add_parser('get', help='获取环境变量')
    get_parser.add_argument('env_name', help='环境名称')

    # 导出环境

    export_parser.add_argument('name', help='环境名称')

    # 激活环境

    activate_parser.add_argument('--shell', choices=['bash', 'fish', 'powershell'], 
                                default='bash', help='Shell 类型')

    args = parser.parse_args()
    
    if not args.command:

        return
    
    # 初始化管理器
    manager = EnvManager(args.config_dir)
    
    # 执行命令
    if args.command == 'create':
        manager.create_environment(http://args.name, args.description, args.template)
    elif args.command == 'list':
        manager.list_environments()
    elif args.command == 'show':
        http://manager.show_environment(http://args.name, http://args.show_encrypted)
    elif args.command == 'set':
        manager.set_variable(args.env_name, args.var_name, args.var_value, args.encrypt)
    elif args.command == 'get':
        value = manager.get_variable(args.env_name, args.var_name)

            print(value)

    elif args.command == 'activate':
        manager.activate_environment(http://args.name, http://args.shell)
    elif args.command == 'delete':
        manager.delete_environment(http://args.name)
