# 1. 快速创建项目结构
create_project() {
    echo -e "${BLUE}选择项目类型：${NC}"
    echo "1. Python 项目"
    echo "2. Node.js 项目"
    echo "3. 通用项目"
    read -p "输入选择: " project_type
    
    read -p "输入项目名称: " project_name
    
    if [[ -z "$project_name" ]]; then
        echo -e "${RED}项目名称不能为空${NC}"
        return
    fi
    
    mkdir -p "$project_name"
    cd "$project_name"
    
    case $project_type in
        1)
            # Python 项目
            mkdir -p {src,tests,docs,scripts}
            echo "# $project_name" > http://README.md
            echo ".pycn__pycache__/n.envnvenv/" > .gitignore
            echo "requests>=2.28.0" > requirements.txt
            touch src/init.py
            touch src/http://main.py
            echo -e "${GREEN}Python 项目创建成功！${NC}"
            ;;
        2)
            # Node.js 项目  
            mkdir -p {src,tests,docs}
            echo "# $project_name" > http://README.md
            echo "node_modules/n.envnnpm-debug.log" > .gitignore
            echo '{"name":"'$project_name'","version":"1.0.0"}' > package.json
            touch src/index.js
            echo -e "${GREEN}Node.js 项目创建成功！${NC}"
            ;;
        3)
            # 通用项目
            mkdir -p {docs,scripts,config}
            echo "# $project_name" > http://README.md
            touch .gitignore
            echo -e "${GREEN}通用项目创建成功！${NC}"
            ;;
    esac
    cd ..
}

# 2. 查找大文件

    size_mb=${size_mb:-100}
    
    echo -e "${BLUE}查找大于 ${size_mb}MB 的文件：${NC}"
    find . -type f -size +${size_mb}M -exec ls -lh {} ; 2>/dev/null | sort -k5 -hr
}

# 3. 压缩/解压缩工具
archive_tools() {
    echo -e "${BLUE}选择操作：${NC}"
    echo "1. 压缩文件夹"
    echo "2. 解压缩文件"
    read -p "输入选择: " choice
    
    case $choice in
        1)
            read -p "输入要压缩的文件夹: " folder
            if [[ -d "$folder" ]]; then
                tar -czf "${folder}.tar.gz" "$folder"
                echo -e "${GREEN}压缩成功: ${folder}.tar.gz${NC}"
                echo "文件大小: $(ls -lh ${folder}.tar.gz | awk '{print $5}')"
            else
                echo -e "${RED}文件夹不存在${NC}"
            fi
            ;;
        2)
            read -p "输入要解压缩的文件: " archive
            if [[ -f "$archive" ]]; then
                case "$archive" in
                    .tar.gz) tar -xzf "$archive" ;;
                    .zip) unzip "$archive" ;;
                    .tar) tar -xf "$archive" ;;
                    ) echo -e "${RED}不支持的文件格式${NC}" ;;
                esac
                echo -e "${GREEN}解压缩成功${NC}"
            else
                echo -e "${RED}文件不存在${NC}"
            fi
            ;;
    esac
}

# 4. 系统资源监控
system_monitor() {
    echo -e "${BLUE}系统资源状态：${NC}"
    echo "CPU使用率:"
    if [[ "$OSTYPE" == "darwin" ]]; then
        top -l 1 | grep "CPU usage"
    else
        grep 'cpu ' /proc/stat | awk '{usage=($2+$4)100/($2+$3+$4+$5)} END {print usage "%"}'
    fi
    
    echo "n内存使用率:"
    if [[ "$OSTYPE" == "darwin" ]]; then
        vm_stat | grep -E "(free|active|inactive|wired)"
    else
        free -h
    fi
    
    echo "n磁盘使用率:"
    df -h | grep -E "(Filesystem|/dev/)"
}

# 5. 网络工具
network_tools() {
    echo -e "${BLUE}选择网络工具：${NC}"
    echo "1. 查看本机IP"
    echo "2. 网络连通性测试"

        1)

            else
                hostname -I
            fi
            echo "n外网IP:"
            curl -s http://ifconfig.me
            echo
            ;;
        2)
            read -p "输入要测试的域名/IP: " target

                ping -c 4 "$target"
            fi
            ;;
        3)
            read -p "输入IP地址: " ip
            read -p "输入端口范围(eg: 80-443): " ports
            if command -v nmap >/dev/null 2>&1; then
                nmap -p "$ports" "$ip"
            else
                echo -e "${RED}nmap 未安装${NC}"
            fi
            ;;
    esac
}
