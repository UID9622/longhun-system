```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""🔐 龍魂自动备份系统 v3.0
完全本地化、零依赖的智能备份系统
Lucky｜UID9622 专属
"""

import os
import shutil
import sqlite3
import hashlib
from pathlib import Path
from datetime import datetime, timedelta
import tarfile
import json
import sys

class 龍魂备份系统:
    def __init__(self):
        self.龍魂目录 = Path.home() / "龍魂系统"
        self.备份目录 = self.龍魂目录 / "备份"
        self.备份目录.mkdir(parents=True, exist_ok=True)
        
        # 需要备份的目录
        self.备份目标 = [
            self.龍魂目录 / "核心层",
            self.龍魂目录 / "藏经阁",
            self.龍魂目录 / "应用层",
            self.龍魂目录 / "兼容层",
        ]
        
        # 备份保留天数
        self.保留天数 = 30
        
        print(f"🐉 龍魂备份系统初始化完成")
        print(f"📂 备份目录：{self.备份目录}")
    
    def 计算文件哈希(self, 文件路径):
        """计算文件MD5哈希值"""
        if not Path(文件路径).exists():
            return None
        
        md5 = [hashlib.md](http://hashlib.md)5()
        with open(文件路径, 'rb') as f:
            for chunk in iter(lambda: [f.read](http://f.read)(4096), b""):
                md5.update(chunk)
        return md5.hexdigest()
    
    def 创建备份(self, 备份类型="增量"):
        """创建备份"""
        时间戳 = [datetime.now](http://datetime.now)().strftime("%Y%m%d_%H%M%S")
        备份名 = f"龍魂备份_{备份类型}_{时间戳}.tar.gz"
        备份文件 = self.备份目录 / 备份名
        
        print(f"\n📦 开始创建 [{'增量' if 备份类型 == '增量' else '全量'}] 备份...")
        print(f"⏳ 时间：{[datetime.now](http://datetime.now)().strftime('%Y-%m-%d %H:%M:%S')}")
        
        # 创建备份元数据
        元数据 = {
            "备份时间": 时间戳,
            "备份类型": 备份类型,
            "版本": "v3.0",
            "文件列表": []
        }
        
        # 创建压缩包
        总大小 = 0
        文件数量 = 0
        
        with [tarfile.open](http://tarfile.open)(备份文件, "w:gz") as tar:
            for 目标 in self.备份目标:
                if 目标.exists():
                    print(f"  ✅ 备份：{[目标.name](http://目标.name)}")
                    tar.add(目标, arcname=[目标.name](http://目标.name))
                    
                    # 统计文件
                    for 文件 in 目标.rglob("*"):
                        if [文件.is](http://文件.is)_file():
                            文件大小 = 文件.stat().st_size
                            总大小 += 文件大小
                            文件数量 += 1
                            元数据["文件列表"].append({
                                "路径": str(文件.relative_to(self.龍魂目录)),
                                "大小": 文件大小,
                                "修改时间": datetime.fromtimestamp(文件.stat().st_mtime).isoformat()
                            })
            
            # 添加元数据文件
            元数据_path = self.备份目录 / f"metadata_{时间戳}.json"
            with open(元数据_path, 'w', encoding='utf-8') as f:
                json.dump(元数据, f, ensure_ascii=False, indent=2)
            tar.add(元数据_path, arcname="metadata.json")
            元数据_path.unlink()
        
        # 计算备份文件哈希
        文件哈希 = self.计算文件哈希(备份文件)
        文件大小 = 备份文件.stat().st_size / 1024 / 1024  # MB
        
        print(f"\n✅ 备份完成！")
        print(f"💾 文件名：{备份名}")
        print(f"📊 大小：{文件大小:.2f} MB")
        print(f"📁 文件数：{文件数量}")
        print(f"🔐 哈希：MD5: {文件哈希[:16]}...")
        print(f"📂 位置：{备份文件}")
        
        # 记录备份历史
        self.记录备份历史({
            "时间": 时间戳,
            "文件": 备份名,
            "类型": 备份类型,
            "大小_MB": round(文件大小, 2),
            "文件数": 文件数量,
            "哈希": 文件哈希
        })
        
        return 备份文件
    
    def 记录备份历史(self, 备份信息):
        """记录备份历史到SQLite"""
        db_path = self.备份目录 / "备份历史.db"
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS 备份记录 (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                时间 TEXT,
                文件 TEXT,
                类型 TEXT,
                大小_MB REAL,
                文件数 INTEGER,
                哈希 TEXT,
                创建时间 DATETIME DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        cursor.execute("""
            INSERT INTO 备份记录 (时间, 文件, 类型, 大小_MB, 文件数, 哈希)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (
            备份信息["时间"],
            备份信息["文件"],
            备份信息["类型"],
            备份信息["大小_MB"],
            备份信息["文件数"],
            备份信息["哈希"]
        ))
        
        conn.commit()
        conn.close()
    
    def 列出备份(self):
        """列出所有备份"""
        db_path = self.备份目录 / "备份历史.db"
        if not db_path.exists():
            print("⚠️  暂无备份记录")
            return
        
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()
        cursor.execute("""
            SELECT 时间, 文件, 类型, 大小_MB, 文件数, 哈希, 创建时间
            FROM 备份记录
            ORDER BY 创建时间 DESC
            LIMIT 20
        """)
        
        results = cursor.fetchall()
        conn.close()
        
        if not results:
            print("⚠️  暂无备份记录")
            return
        
        print(f"\n📚 龍魂备份历史（最近20条）")
        print("=" * 110)
        print(f"{'时间戳':<17} | {'类型':<6} | {'大小(MB)':<10} | {'文件数':<8} | {'文件名'}")
        print("-" * 110)
        
        for 时间, 文件, 类型, 大小, 文件数, 哈希, 创建 in results:
            print(f"{时间:<17} | {类型:<6} | {大小:>9.2f} | {文件数:>8} | {文件}")
        
        print("=" * 110)
        print(f"\n共 {len(results)} 条记录\n")
    
    def 恢复备份(self, 备份文件名):
        """从备份恢复数据"""
        备份文件 = self.备份目录 / 备份文件名
        
        if not 备份文件.exists():
            print(f"❌ 备份文件不存在：{备份文件名}")
            return False
        
        print(f"\n⚠️  警告：即将从备份恢复数据，当前数据将被覆盖！")
        print(f"📦 备份文件：{备份文件名}")
        确认 = input("是否继续？(输入 'YES' 确认): ")
        
        if 确认 != "YES":
            print("❌ 恢复已取消")
            return False
        
        print(f"\n🔄 开始恢复...")
        
        try:
            # 解压备份
            with [tarfile.open](http://tarfile.open)(备份文件, "r:gz") as tar:
                tar.extractall(self.龍魂目录)
            
            print(f"✅ 恢复完成！")
            print(f"📂 位置：{self.龍魂目录}")
            return True
        except Exception as e:
            print(f"❌ 恢复失败：{e}")
            return False
    
    def 清理过期备份(self):
        """清理过期备份文件"""
        截止日期 = [datetime.now](http://datetime.now)() - timedelta(days=self.保留天数)
        删除计数 = 0
        释放空间 = 0
        
        print(f"\n🧹 开始清理过期备份（保留{self.保留天数}天）...")
        
        for 备份文件 in self.备份目录.glob("龍魂备份_*.tar.gz"):
            文件时间 = datetime.fromtimestamp(备份文件.stat().st_mtime)
            if 文件时间 < 截止日期:
                文件大小 = 备份文件.stat().st_size / 1024 / 1024
                print(f"  🗑️  删除：{[备份文件.name](http://备份文件.name)} ({文件大小:.2f} MB)")
                释放空间 += 文件大小
                备份文件.unlink()
                删除计数 += 1
        
        if 删除计数 > 0:
            print(f"\n✅ 清理完成，共删除 {删除计数} 个过期备份")
            print(f"💾 释放空间：{释放空间:.2f} MB")
        else:
            print(f"\n✅ 没有过期备份需要清理")
    
    def 验证备份(self, 备份文件名):
        """验证备份文件完整性"""
        备份文件 = self.备份目录 / 备份文件名
        
        if not 备份文件.exists():
            print(f"❌ 备份文件不存在：{备份文件名}")
            return False
        
        print(f"\n🔍 验证备份文件：{备份文件名}")
        
        try:
            # 尝试打开并读取tar文件
            with [tarfile.open](http://tarfile.open)(备份文件, "r:gz") as tar:
                成员列表 = tar.getmembers()
                print(f"✅ 压缩包完整，包含 {len(成员列表)} 个文件")
            
            # 计算哈希
            文件哈希 = self.计算文件哈希(备份文件)
            print(f"🔐 文件哈希：{文件哈希}")
            
            return True
        except Exception as e:
            print(f"❌ 验证失败：{e}")
            return False

def main():
    """主程序"""
    备份系统 = 龍魂备份系统()
    
    if len(sys.argv) < 2:
        print("\n🐉 龍魂自动备份系统 v3.0")
        print("=" * 60)
        print("\n使用方法：")
        print("  python3 [龍魂备份引擎.py](http://龍魂备份引擎.py) backup        # 创建增量备份")
        print("  python3 [龍魂备份引擎.py](http://龍魂备份引擎.py) backup-full   # 创建全量备份")
        print("  python3 [龍魂备份引擎.py](http://龍魂备份引擎.py) list          # 列出所有备份")
        print("  python3 [龍魂备份引擎.py](http://龍魂备份引擎.py) restore <文件> # 恢复备份")
        print("  python3 [龍魂备份引擎.py](http://龍魂备份引擎.py) verify <文件>  # 验证备份")
        print("  python3 [龍魂备份引擎.py](http://龍魂备份引擎.py) cleanup       # 清理过期备份")
        print("")
        return
    
    命令 = sys.argv[1]
    
    if 命令 == "backup":
        备份系统.创建备份("增量")
        备份系统.清理过期备份()
    
    elif 命令 == "backup-full":
        备份系统.创建备份("全量")
        备份系统.清理过期备份()
    
    elif 命令 == "list":
        备份系统.列出备份()
    
    elif 命令 == "restore":
        if len(sys.argv) < 3:
            print("❌ 请指定要恢复的备份文件名")
            备份系统.列出备份()
            return
        备份系统.恢复备份(sys.argv[2])
    
    elif 命令 == "verify":
        if len(sys.argv) < 3:
            print("❌ 请指定要验证的备份文件名")
            return
        备份系统.验证备份(sys.argv[2])
    
    elif 命令 == "cleanup":
        备份系统.清理过期备份()
    
    else:
        print(f"❌ 未知命令：{命令}")

if __name__ == "__main__":
    main()
```

```bash
# 龍魂自动备份调度脚本 v3.0
# Lucky｜UID9622 专属

# 配置
LONGHUN_DIR="$HOME/龍魂系统"
BACKUP_SCRIPT="$LONGHUN_DIR/核心层/[龍魂备份引擎.py](http://龍魂备份引擎.py)"

    echo -e "${RED}❌ 备份脚本未找到：$BACKUP_SCRIPT${NC}" | tee -a "$LOG_FILE"
    exit 1
fi

# 执行备份
echo -e "${YELLOW}⏳ 正在执行备份...${NC}" | tee -a "$LOG_FILE"
python3 "$BACKUP_SCRIPT" backup 2>&1 | tee -a "$LOG_FILE"

if [ ${PIPESTATUS[0]} -eq 0 ]; then
    echo -e "${GREEN}✅ 备份成功！${NC}" | tee -a "$LOG_FILE"
else
    echo -e "${RED}❌ 备份失败！${NC}" | tee -a "$LOG_FILE"
    exit 1
fi

# 如果外部磁盘存在，同步备份
if [ -d "$EXTERNAL_DISK" ]; then
    echo -e "${CYAN}💾 检测到外部磁盘，开始同步...${NC}" | tee -a "$LOG_FILE"
    mkdir -p "$EXTERNAL_DISK/龍魂备份"
    cp "$LONGHUN_DIR/备份"/龍魂备份_*.tar.gz "$EXTERNAL_DISK/龍魂备份/" 2>&1 | tee -a "$LOG_FILE"
    echo -e "${GREEN}✅ 已同步到外部磁盘${NC}" | tee -a "$LOG_FILE"
fi

echo "" | tee -a "$LOG_FILE"
```
