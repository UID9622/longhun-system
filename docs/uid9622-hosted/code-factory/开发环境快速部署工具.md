# UID9622 开发环境快速部署工具
# 一键安装和配置常用开发工具

set -e  # 出错即退出

# 颜色定义
RED='033[0;31m'
GREEN='033[0;32m'
YELLOW='033[1;33m'
BLUE='033[0;34m'
PURPLE='033[0;35m'
NC='033[0m'

# 日志函数
log() {
    echo -e "${BLUE}[$(date '+%H:%M:%S')]${NC} $1"
}

success() {
    echo -e "${GREEN}[u2713]${NC} $1"
}

warn() {
    echo -e "${YELLOW}[!]${NC} $1"
}

error() {
    echo -e "${RED}[u2717]${NC} $1"
}

# 检测操作系统
detect_os() {
    if [[ "$OSTYPE" == "darwin" ]]; then
        echo "macos"
    elif [[ "$OSTYPE" == "linux-gnu" ]]; then
        echo "linux"
    else
        echo "unknown"
    fi
}

# 检测命令是否存在
command_exists() {
    command -v "$1" >/dev/null 2>&1
}

# 安装 Homebrew (macOS)
install_homebrew() {
    if ! command_exists brew; then
        log "Installing Homebrew..."
        /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
        success "Homebrew installed"
    else
        success "Homebrew already installed"
    fi
}

# 安装基础工具
install_basic_tools() {
    local os=$(detect_os)
    
    log "Installing basic development tools..."
    
    if [[ "$os" == "macos" ]]; then
        # macOS 使用 Homebrew
        install_homebrew
        brew install git curl wget tree htop neofetch jq
        
    elif [[ "$os" == "linux" ]]; then
        # Linux 使用包管理器
        if command_exists apt; then
            sudo apt update
            sudo apt install -y git curl wget tree htop neofetch jq build-essential
        elif command_exists yum; then
            sudo yum install -y git curl wget tree htop neofetch jq gcc gcc-c++ make
        elif command_exists pacman; then
            sudo pacman -S --noconfirm git curl wget tree htop neofetch jq base-devel
        fi
    fi
    
    success "Basic tools installed"
}

# 安装 Node.js
install_nodejs() {
    if ! command_exists node; then
        log "Installing Node.js..."
        
        # 安装 nvm
        if ! command_exists nvm; then
            curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.0/install.sh | bash
            export NVM_DIR="$HOME/.nvm"
            [ -s "$NVM_DIR/http://nvm.sh" ] && . "$NVM_DIR/http://nvm.sh"
        fi
        
        # 安装最新 LTS 版本
        nvm install --lts
        nvm use --lts
        
        # 安装全局包
        npm install -g yarn pnpm typescript nodemon pm2
        
        success "Node.js and tools installed"
    else
        success "Node.js already installed ($(node --version))"
    fi
}

# 安装 Python
install_python() {
    if ! command_exists python3; then
        log "Installing Python 3..."
        
        local os=$(detect_os)
        if [[ "$os" == "macos" ]]; then
            brew install python@3.11
        elif [[ "$os" == "linux" ]]; then
            if command_exists apt; then
                sudo apt install -y python3 python3-pip python3-venv
            elif command_exists yum; then
                sudo yum install -y python3 python3-pip
            fi
        fi
        
        success "Python 3 installed"
    else
        success "Python 3 already installed ($(python3 --version))"
    fi
    
    # 安装常用 Python 包
    if command_exists pip3; then
        log "Installing Python packages..."
        pip3 install --user virtualenv pipenv poetry requests flask django jupyter
        success "Python packages installed"
    fi
}

# 安装 Docker
install_docker() {
    if ! command_exists docker; then
        log "Installing Docker..."
        
        local os=$(detect_os)
        if [[ "$os" == "macos" ]]; then
            warn "Please install Docker Desktop manually from https://www.docker.com/products/docker-desktop"
        elif [[ "$os" == "linux" ]]; then
            curl -fsSL https://get.docker.com -o http://get-docker.sh
            sh http://get-docker.sh
            sudo usermod -aG docker $USER
            rm http://get-docker.sh
            success "Docker installed (please re-login to use docker without sudo)"
        fi
    else
        success "Docker already installed ($(docker --version))"
    fi
}

# 安装 VS Code
install_vscode() {
    if ! command_exists code; then
        log "Installing Visual Studio Code..."
        
        local os=$(detect_os)
        if [[ "$os" == "macos" ]]; then
            brew install --cask visual-studio-code
        elif [[ "$os" == "linux" ]]; then
            wget -qO- https://packages.microsoft.com/keys/microsoft.asc | gpg --dearmor > http://packages.microsoft.gpg
            sudo install -o root -g root -m 644 http://packages.microsoft.gpg /etc/apt/trusted.gpg.d/
            sudo sh -c 'echo "deb [arch=amd64,arm64,armhf signed-by=/etc/apt/trusted.gpg.d/http://packages.microsoft.gpg] https://packages.microsoft.com/repos/code stable main" > /etc/apt/sources.list.d/vscode.list'
            sudo apt update
            sudo apt install code
        fi
        
        # 安装常用插件
        if command_exists code; then
            log "Installing VS Code extensions..."
            code --install-extension ms-python.python
            code --install-extension ms-vscode.vscode-typescript-next
            code --install-extension ms-vscode.vscode-json
            code --install-extension bradlc.vscode-tailwindcss
            code --install-extension esbenp.prettier-vscode
            code --install-extension ms-vscode.vscode-eslint
            success "VS Code extensions installed"
        fi
        
        success "VS Code installed"
    else
        success "VS Code already installed"
    fi
}

# 配置 Git
configure_git() {
    if command_exists git; then
        log "Configuring Git..."
        
        read -p "Enter your Git username: " git_username
        read -p "Enter your Git email: " git_email
        
        git config --global http://user.name "$git_username"
        git config --global http://user.email "$git_email"
        git config --global init.defaultBranch main
        git config --global core.editor "code --wait"
        
        # 生成 SSH 密钥
        if [ ! -f ~/.ssh/id_rsa ]; then
            log "Generating SSH key..."
            ssh-keygen -t rsa -b 4096 -C "$git_email" -f ~/.ssh/id_rsa -N ""
            success "SSH key generated at ~/.ssh/id_http://rsa.pub"
            echo "Please add the following SSH key to your Git provider:"
            cat ~/.ssh/id_http://rsa.pub
        fi
        
        success "Git configured"
    fi
}

# 创建开发目录结构
create_dev_structure() {
    log "Creating development directory structure..."
    
    local dev_dir="$HOME/Development"
    mkdir -p "$dev_dir"/{Projects,Playground,Scripts,Tools}
    
    # 创建快捷脚本
    cat > "$dev_dir/Scripts/quick_http://start.sh" << 'EOF'
#!/bin/bash
# Quick development environment starter

echo "Starting development environment..."

# 启动常用服务
if command -v docker >/dev/null 2>&1; then
    echo "Starting Docker services..."
    # docker-compose up -d  # 根据需要可取消注释
fi

# 打开常用应用
if command -v code >/dev/null 2>&1; then
    echo "Opening VS Code..."
    code ~/Development/Projects
fi

echo "Development environment ready!"
EOF
    
    chmod +x "$dev_dir/Scripts/quick_http://start.sh"
    
    success "Development directory structure created at $dev_dir"
}

# 安装其他工具
install_additional_tools() {
    log "Installing additional development tools..."
    
    local os=$(detect_os)
    
    if [[ "$os" == "macos" ]]; then
        brew install --cask iterm2 postman insomnia figma
        brew install fzf ripgrep fd bat exa zsh-autosuggestions zsh-syntax-highlighting
        
    elif [[ "$os" == "linux" ]]; then
        if command_exists apt; then
            sudo apt install -y fzf ripgrep fd-find bat zsh
        fi
        
        # 安装 Snap 包
        if command_exists snap; then
            sudo snap install postman insomnia code --classic
        fi
    fi
    
    success "Additional tools installed"
}

# 配置 Shell
configure_shell() {
    log "Configuring shell environment..."
    
    # 安装 Oh My Zsh
    if [ ! -d "$HOME/.oh-my-zsh" ]; then
        sh -c "$(curl -fsSL https://raw.github.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" "" --unattended
        success "Oh My Zsh installed"
    fi
    
    # 配置 .zshrc
    cat >> ~/.zshrc << 'EOF'

# UID9622 Development Environment
export PATH="$HOME/Development/Scripts:$PATH"
export EDITOR="code --wait"

# Aliases
alias ll="ls -la"
alias la="ls -la"
alias ..="cd .."
alias ...="cd ../.."
alias dev="cd ~/Development"
alias projects="cd ~/Development/Projects"
alias scripts="cd ~/Development/Scripts"

# Git aliases
alias gs="git status"
alias ga="git add"
alias gc="git commit"
alias gp="git push"
alias gl="git pull"
alias gd="git diff"
alias gb="git branch"
alias gco="git checkout"

# Quick commands
alias serve="python3 -m http.server 8000"
alias myip="curl http://ifconfig.me"
alias ports="lsof -i -P -n | grep LISTEN"
EOF
    
    success "Shell configured"
}

# 显示安装总结
show_summary() {
    echo
    echo -e "${PURPLE}================================${NC}"
    echo -e "${PURPLE}  UID9622 Dev Environment Setup  ${NC}"
    echo -e "${PURPLE}================================${NC}"
    echo
    echo -e "${GREEN}✓ Development environment setup completed!${NC}"
    echo
    echo "Installed tools:"
    command_exists git && echo "  ✓ Git $(git --version | cut -d' ' -f3)"
    command_exists node && echo "  ✓ Node.js $(node --version)"
    command_exists python3 && echo "  ✓ Python $(python3 --version | cut -d' ' -f2)"
    command_exists docker && echo "  ✓ Docker $(docker --version | cut -d' ' -f3 | tr -d ',')"
    command_exists code && echo "  ✓ VS Code"
    
    echo
    echo "Development directories:"
    echo "  ~/Development/Projects   - Your main projects"
    echo "  ~/Development/Playground - Experimental code"
    echo "  ~/Development/Scripts    - Utility scripts"
    echo "  ~/Development/Tools      - Development tools"
    echo
    echo "Quick commands (after restart shell):"
    echo "  dev      - Go to development directory"
    echo "  projects - Go to projects directory"
    echo "  serve    - Start local HTTP server"
    echo
    echo -e "${YELLOW}Please restart your terminal or run 'source ~/.zshrc' to apply changes${NC}"
    echo
}

# 主函数
main() {
    echo -e "${BLUE}"
    echo " _    _ __ __      __      "
    echo "| |  | |_   |  _  | / _  / / |_ |__  "
    echo "| |  | | | | | | | |/ / / /_   ) |  ) |"
    echo "| |  | | | | | | | ||  _  | '_  / / / / "
    echo "| |__| || || |/ / | | | | | | / // / "
    echo " \/ \/ |/  \| ||| |||_|"
    echo
    echo "Development Environment Setup Tool"
    echo -e "${NC}"
    
    read -p "This will install development tools. Continue? (y/N): " -n 1 -r
    echo
    if [[ ! $REPLY =~ ^[Yy]$ ]]; then
        warn "Installation cancelled"
        exit 0
    fi
    
    log "Starting development environment setup..."
    
    install_basic_tools
    install_nodejs
    install_python
    install_docker
    install_vscode
    configure_git
    create_dev_structure
    install_additional_tools
    configure_shell
    
    show_summary
}

# 运行主函数
if [[ "${BASH_SOURCE[0]}" == "${0}" ]]; then
    main "$@"
fi
