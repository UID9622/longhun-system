        """创建新用户"""
        try:
            # 生成盐和密码哈希

    def encrypt_data_v2(self, data: str, level: int = 1) -> bytes:
        """多层加密数据"""
        if level == 1:
            return self.cipher_suite.encrypt(data.encode('utf-8'))
        elif level == 2:
            # 双层加密
            first_encrypted = self.cipher_suite.encrypt(data.encode('utf-8'))
            return self.session_cipher.encrypt(first_encrypted)
        else:
            raise ValueError("不支持的加密级别")
    
    def decrypt_data_v2(self, encrypted_data: bytes, level: int = 1) -> str:
        """多层解密数据"""
        if level == 1:
            return self.cipher_suite.decrypt(encrypted_data).decode('utf-8')
        elif level == 2:
            # 双层解密
            first_decrypted = self.session_cipher.decrypt(encrypted_data)
            return self.cipher_suite.decrypt(first_decrypted).decode('utf-8')
        else:
            raise ValueError("不支持的解密级别")
