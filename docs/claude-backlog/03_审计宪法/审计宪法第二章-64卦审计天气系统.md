class HexagramTimeEngine:
    """卦象时辰自动引擎"""
    
    def get_current_hexagram_by_time(self, current_time):
        """
        根据当前时辰，自动获取对应卦象
        """
        hour = current_time.hour
        
        time_hexagram_map = {
            (23, 1): "坎卦",   # 子时
            (1, 3): "艮卦",    # 丑时
            (3, 5): "震卦",    # 寅时
            (5, 7): "震卦",    # 卯时

            (15, 17): "兑卦",  # 申时
            (17, 19): "兑卦",  # 酉时
            (19, 21): "乾卦",  # 戌时
            (21, 23): "乾卦",  # 亥时
        }
        
        for (start, end), hexagram in time_hexagram_map.items():
            if start <= hour < end:
                return hexagram
        
        # 特殊处理子时跨日
        if 23 <= hour or hour < 1:
            return "坎卦"
    
    def combine_hexagrams(self, state_hexagram, time_hexagram):
        """
        结合系统状态卦象和时辰卦象
        """
        # 状态卦象权重70%，时辰卦象权重30%
        return {
            "primary": state_hexagram,
            "secondary": time_hexagram,
            "strategy": self.merge_strategy(state_hexagram, time_hexagram)
        }
