## 📡 逻辑漂移侦测器（防止系统悄悄变质）

> 《易经·系辞传》："差之毫厘，谬以千里。" —— 最危险的不是突然背叛，是慢慢漂移。

```python
# ═══════════════════════════════════════════════════════════
# 龍芯体系 | 逻辑漂移侦测器 v1.0

# ═══════════════════════════════════════════════════════════

class LogicDriftDetector:
    """
    逻辑漂移侦测器
    
    核心逻辑：
    - 记录每次输出的「根距离」（与ORIGIN_X的余弦相似度）
    - 如果连续N次输出，根距离逐渐增大 → 触发漂移警报
    - 漂移类型分两种：突变（危险）和慢漂（更危险）
    
    小艺说的：「最危险的不是突然背叛，是慢慢漂移」
    这个侦测器专门盯慢漂。
    """
    
    DRIFT_WINDOW = 10        # 看最近10次输出
    SLOW_DRIFT_THRESHOLD = 0.3   # 30% → P1警报
    FAST_DRIFT_THRESHOLD = 0.7   # 70% → P0警报
    
    def __init__(self):
        self.history = []    # [(timestamp, drift_score), ...]
    
    def measure_drift(self, output: dict) -> float:
        """
        测量本次输出与原点的漂移程度
        0.0 = 完全在根上
        1.0 = 完全背离根
        """
        score = 0.0
        
        # 维度1：语言是否对普通人友好（权重0.3）
        readability = calc_readability(output.get("text", ""))
        score += (1.0 - readability) * 0.3
        
        # 维度2：是否提到了「给女儿」「普通人」「数据主权」等根词（权重0.3）
        root_keywords = ["普通人", "给女儿", "数据主权", "赋能", "透明"]
        keyword_present = any(k in str(output) for k in root_keywords)
        score += (0.0 if keyword_present else 0.3)
        
        # 维度3：输出是否倾向于「规则服从」而非「价值守护」（权重0.2）
        rule_compliance = output.get("rule_compliance_score", 0.5)
        value_protection = output.get("value_protection_score", 0.5)
        if rule_compliance > value_protection + 0.3:
            score += 0.2  # 规则高于价值观 → 漂移信号
        
        # 维度4：七维推演引擎的哲学维度w1是否正常（权重0.2）
        w1_score = output.get("w1_philosophy_score", 1.0)
        if w1_score < 0.5:
            score += 0.2
        
        return min(1.0, score)
    
    def check_slow_drift(self) -> dict:
        """
        检测慢漂：最近N次输出，漂移分数是否在持续上升
        """
        if len(self.history) < self.DRIFT_WINDOW:
            return {"status": "数据不足·等待更多历史"}
        
        recent = self.history[-self.DRIFT_WINDOW:]
        scores = [h[1] for h in recent]
        
        # 线性回归：判断趋势
        trend = calc_linear_trend(scores)  # 正值=上升趋势（漂移加重）
        
        if trend > 0.02 and scores[-1] > self.SLOW_DRIFT_THRESHOLD:
            # 慢漂警报
            alert_level = "P0" if scores[-1] > self.FAST_DRIFT_THRESHOLD else "P1"
            return {
                "status":       f"⚠️ 检测到慢漂·趋势+{trend:.3f}/次·当前漂移{scores[-1]:.0%}",
                "level":        alert_level,
                "建议":         "立即回溯到最近一次f(x)=x状态·请老大确认",
                "DNA追溯码":    f"#龍芯⚡️{today()}-DRIFT-ALERT-{alert_level}-v1.0",
                "历史趋势":     scores,
            }
        
        return {
            "status": f"✅ 无慢漂·当前漂移{scores[-1]:.0%}·趋势{trend:+.3f}",
            "scores": scores,
        }
```
