        const canvas = document.getElementById('deepSea');

        class Bubble {
            constructor(x, y, fromMouse) {
                this.x = x || Math.random() * W;
                this.y = y || H + Math.random() * 50;
                this.r = Math.random() * 4 + 1;
                this.speed = Math.random() * 0.8 + 0.2;
                this.wobble = Math.random() * Math.PI * 2;
                this.wobbleSpeed = Math.random() * 0.02 + 0.01;
                this.opacity = fromMouse ? 0.8 : Math.random() * 0.4 + 0.1;
                this.life = fromMouse ? 100 : Infinity;
            }
            update() {
                this.y -= this.speed;
                this.wobble += this.wobbleSpeed;
                this.x += Math.sin(this.wobble) * 0.5;
                if (this.life !== Infinity) {
                    this.life--;
                    this.opacity *= 0.99;
                }
                return this.y > -10 && this.opacity > 0.01 && this.life > 0;
            }
            draw() {
                const style = getComputedStyle(document.documentElement);
                const accent = style.getPropertyValue('--accent').trim() || '#00d4ff';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.r, 0, Math.PI * 2);
                ctx.fillStyle = accent.replace(')', `,${this.opacity})`.replace('rgb', 'rgba');
                if (!accent.startsWith('rgba')) {
                    ctx.fillStyle = `rgba(0,212,255,${this.opacity})`;
                }
                ctx.fill();
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.r + 3, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(0,212,255,${this.opacity * 0.2})`;
                ctx.fill();
            }
        }
