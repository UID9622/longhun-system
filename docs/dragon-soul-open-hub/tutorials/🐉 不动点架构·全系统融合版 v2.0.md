class FixpointGapFiller:

    def __init__(self):
        self.personas = PERSONA_REGISTRY
        self.digital_humans = DIGITAL_HUMAN_REGISTRY
        self.gate_engine = ThreeGateEngine()
        self.templates = {}  # 排版模板缓存
        self.fill_stats = {pid: 0 for pid in PERSONA_REGISTRY}  # 使用统计

    def identify_gaps(self, user_input: str) -> List[dict]:
        """识别用户输入中的'坑'，路由到不动点"""
        gaps = []
        for pid, persona in self.personas.items():
            match_score = 0
            matched_triggers = []
            for trigger in persona["triggers"]:
                if trigger in user_input:
                    match_score += 1
                    matched_triggers.append(trigger)

            if match_score > 0:
                gaps.append({
                    "persona_id": pid,
                    "persona_name": persona["name"],
                    "layer": persona["layer"],
                    "skill": persona["skills"][0],
                    "output": persona["output"],
                    "weight": persona["weight"],
                    "match_score": match_score,
                    "triggers_hit": matched_triggers,
                    "quarantine": persona.get("quarantine", False)
                })

        # 按权重×匹配分数排序
        gaps.sort(key=lambda g: g["weight"] * g["match_score"], reverse=True)
        return gaps

        """填坑主流程: 识别→闸门→路由→填充→合成"""

            "template_key": template_key
        }

    def _build_template(self, components: list) -> dict:
        """构建排版模板（一次生成，永久缓存）"""
        return {
            "sections": len(components),
            "persona_order": [c["persona_id"] for c in components],
            "layers": list(set(c["layer"] for c in components)),
            "created": time.time()
        }

    def stats(self) -> dict:
        """不动点使用统计"""
        total = sum(self.fill_stats.values())
        return {
            "total_calls": total,
            "templates_cached": len(self.templates),
            "top_personas": sorted(self.fill_stats.items(), key=lambda x: x[1], reverse=True)[:5]
        }
