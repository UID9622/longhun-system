# 權重：任意選擇，僅為實驗目的

def weight_zeta(s):
    """天軸：|ζ(s)|"""
    return np.abs(zeta(s))

def weight_symmetric(s):
    """地軸：|ζ(1-s)|"""
    return np.abs(zeta(1 - s))

def weight_factor(s):
    """人軸：|χ(s)| 的近似"""
    # χ(s) = 2^s π^(s-1) sin(πs/2) Γ(1-s)
    # 這裡用 |Γ(1-s)| 作為簡化近似
