## 4️⃣ 报表 [report.py](http://report.py)（把 audit_log.jsonl 统计成 report.json）

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import json
import os
from collections import Counter

def read_jsonl(path):
    if not os.path.exists(path):
        return []
    rows = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rows.append(json.loads(line))
    return rows

def main():
    rows = read_jsonl("audit_log.jsonl")

    colors = Counter()
    scenes = Counter()
    roles = Counter()
    outcomes = Counter()

    for r in rows:
        res = (r.get("result") or {})
        colors[res.get("result_color", "unknown")] += 1
        scenes[r.get("scene", "unknown")] += 1
        roles[r.get("role", "unknown")] += 1
        outcomes[r.get("outcome", "unknown")] += 1

    report = {
        "total": len(rows),
        "by_color": dict(colors),
        "by_scene": dict(scenes),
        "by_role": dict(roles),
        "by_outcome": dict(outcomes)
    }

    with open("report.json", "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)

    print(json.dumps(report, ensure_ascii=False, indent=2))

if __name__ == "__main__":
    main()
```
