# ═══════════════════════════════════════════════════════════
# 🧬 DNA自动生成器 v1.0
# DNA追溯码：#龍芯⚡️2026-03-31-DNA-v1.0
# ═══════════════════════════════════════════════════════════

def 生成DNA(操作名称: str, 层级="L4") -> str:
    """
    L0: 永恒锚点（手动创建）
    L1: 重大决策
    L2: 系统升级
    L3: 日常操作 ← 最常用
    L4: 实时执行 ← 最常用
    """
    from datetime import date
    import hashlib
    
    日期 = date.today().strftime("%Y-%m-%d")
    哈希 = hashlib.md5(操作名称.encode()).hexdigest()[:8].upper()
    
    DNA格式 = {
        "L3": f"#龍芯⚡️{日期}-{操作名称[:10].replace(' ','-')}-{哈希}",
        "L4": f"#龍芯⚡️{日期}-{操作名称[:8].replace(' ','-')}-EXEC-{哈希[:4]}",
    }
    
    return DNA格式.get(层级, DNA格式["L4"])
