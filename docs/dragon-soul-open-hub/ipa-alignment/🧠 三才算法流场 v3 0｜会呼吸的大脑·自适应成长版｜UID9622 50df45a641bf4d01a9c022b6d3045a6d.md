## 🔬 P06 公式层（数学大师提取）

### 公式一：呼吸节律（三才权重动态调制）

$$
\text{breathPhase} = \sin(\text{frameCount} \times 0.008)
$$

$$
wH = 0.35 + \text{breathPhase} \times 0.12 \quad \text{(天场吸气扩张)}
$$

$$
wE = 0.15 - \text{breathPhase} \times 0.06 \quad \text{(地场呼气收紧)}
$$

$$
wR = \max(0.38,\ 1.0 - wH - wE) \quad \text{(人场自动补足，保证 \Sigma=1)}
$$

**呼吸周期：** $T = \dfrac{2\pi}{0.008} \approx 785$ 帧·约每13秓一次完整呼吸

---

### 公式二：赫布学习（突触权重动态更新）

$$
w_{\text{synaptic}}(t+1) = \min\left(1.0,\ w_{\text{synaptic}}(t) \times 0.94 + n_{\text{near}} \times 0.018\right)
$$

- $n_{\text{near}}$ = 65px半径内的激活邻居数
- $0.94$ = 自然衰减（遗忘曲线）
- $0.018$ = 使用增强（「用进废退」）

---

### 公式三：年轮引力场（甜甜圈）

$$
d = \sqrt{(x_p - x_r)^2 + (y_p - y_r)^2}
$$

$$
\text{diff} = d - r_{\text{ring}}, \quad f_{\text{mag}} = -\frac{\text{diff}}{|\text{diff}| + 60} \times S_{\text{ring}} \times 0.12
$$

- 粒子在环内即被向外推，在环外则被向内拉 → 最终沿环边运动
- $S_{text{ring}} = max(0.08, 1.0 - text{age}/4000)$（年轮老化定律）

---

### 公式四：神经可塑性（种子偏置自适应漂移）

$$
\theta_{\text{seed}}(t+1) = \theta_{\text{seed}}(t) + \sin(\text{age} \times 0.018) \times P_{\text{plasticity}} \times 0.0008
$$

- $P_{text{plasticity}} in [0.2, 1.0]$：每个粒子独立分配
- 高可塑性：学习快，逐渐对齐流场主方向
- 低可塑性：稳定性强，提供系统骨架

---

### 公式五：合力积分（粒子运动方程）

$$
\vec{F}_{\text{total}} = \vec{F}_{\text{Heaven}} + \vec{F}_{\text{Earth}} + \vec{F}_{\text{Human}} + \vec{F}_{\text{Rings}}
$$

$$
\vec{v}(t+1) = \vec{v}(t) \times 0.87 + \vec{F}_{\text{total}} \times 1.3, \quad |\vec{v}| \leq 3.8
$$

    const SEED=9622,N=1200,TRAIL_LEN=18,BREATH_RATE=0.008,BREATH_DEPTH=0.12;
    const GROWTH_INTERVAL=300,MAX_RINGS=12;
    let wH=0.35,wE=0.15,wR=0.50,breathPhase=0,noiseOffset=0;
    let growthTimer=0,growthRings=[],particles=[];
    const LUOSHU=[
      {rx:0.50,ry:0.50,v:5,attract:true},{rx:0.50,ry:0.10,v:9,attract:true},
      {rx:0.50,ry:0.90,v:1,attract:true},{rx:0.10,ry:0.50,v:3,attract:true},
      {rx:0.90,ry:0.50,v:7,attract:true},{rx:0.10,ry:0.10,v:4,attract:false},

    ];
    class GrowthRing{
      constructor(x,y){this.x=x;this.y=y;this.radius=5;this.maxR=random(80,220);
        this.strength=1.0;this.age=0;this.hue=random(180,260);}
      update(){this.age++;if(this.radius<this.maxR)this.radius+=0.08;
        this.strength=max(0.08,1.0-this.age/4000);}
      forceOn(px,py){let dx=this.x-px,dy=this.y-py,d=sqrt(dx*dx+dy*dy)+0.001;
        let diff=d-this.radius,mag=-diff/(abs(diff)+60)*this.strength*0.12;
        return{fx:(dx/d)*mag,fy:(dy/d)*mag};}
      draw(){let a=this.strength*25;noFill();stroke(`hsla(${this.hue},70%,65%,${a/255})`);
        strokeWeight(0.6);circle(this.x,this.y,this.radius*2);
        if(this.age<60){fill(`hsla(${this.hue},90%,80%,${(60-this.age)/60*0.8})`);noStroke();circle(this.x,this.y,6);}}
    }
    class Particle{
      constructor(id){this.id=id;this.trail=[];this.synaptic=0;
        randomSeed(SEED+id*7919);this.plasticity=random(0.2,1.0);this.baseHue=random(190,250);this._init();}
      _init(){randomSeed(SEED+this.id*137);this.x=random(width);this.y=random(height);
        this.vx=0;this.vy=0;this.seedBias=random(TWO_PI);
        this.life=random(300,900);this.maxLife=this.life;this.sz=random(1.5,3.5);this.age=0;}
      update(){this.age++;this.life--;this.trail.push({x:this.x,y:this.y});
        if(this.trail.length>TRAIL_LEN)this.trail.shift();
        let ang=noise(this.x*0.003,this.y*0.003,noiseOffset)*TWO_PI*2;
        let fxH=cos(ang)*wH,fyH=sin(ang)*wH,fxE=0,fyE=0;
        for(let a of LUOSHU){let ax=a.rx*width,ay=a.ry*height,dx=ax-this.x,dy=ay-this.y;
          let d=sqrt(dx*dx+dy*dy)+0.001,bMod=a.attract?(1+breathPhase*0.35):(1-breathPhase*0.2);
          let force=(a.v/(d+70))*wE*bMod;if(!a.attract)force*=-0.5;
          fxE+=(dx/d)*force;fyE+=(dy/d)*force;}
        this.seedBias+=sin(this.age*0.018)*this.plasticity*0.0008;
        let fxR=cos(this.seedBias)*wR,fyR=sin(this.seedBias)*wR,fxG=0,fyG=0;
        for(let ring of growthRings){let f=ring.forceOn(this.x,this.y);fxG+=f.fx;fyG+=f.fy;}
        let ax=fxH+fxE+fxR+fxG,ay=fyH+fyE+fyR+fyG;
        this.vx=this.vx*0.87+ax*1.3;this.vy=this.vy*0.87+ay*1.3;
        let spd=sqrt(this.vx*this.vx+this.vy*this.vy);
        if(spd>3.8){this.vx*=3.8/spd;this.vy*=3.8/spd;}
        this.x+=this.vx;this.y+=this.vy;
        if(this.x<0)this.x+=width;if(this.x>width)this.x-=width;
        if(this.y<0)this.y+=height;if(this.y>height)this.y-=height;
        if(this.life<=0)this._init();}
      draw(){let lifeRatio=this.life/this.maxLife;
        for(let i=1;i<this.trail.length;i++){let prog=i/this.trail.length,alpha=prog*40*lifeRatio;
          let hue=this.baseHue+this.synaptic*50;
          stroke(`hsla(${hue},75%,65%,${alpha/255})`);strokeWeight(0.4+prog*this.sz*0.25);
          let t=this.trail[i-1],t2=this.trail[i];line(t.x,t.y,t2.x,t2.y);}
        noStroke();
        if(this.synaptic>0.35){let glowA=lifeRatio*0.12*this.synaptic;
          fill(`rgba(180,100,255,${glowA})`);circle(this.x,this.y,this.sz*6*(1+breathPhase*0.2));}
        let coreA=lifeRatio*200,coreR=this.sz*(1+breathPhase*0.25);
        fill(`rgba(140,200,255,${coreA/255})`);circle(this.x,this.y,coreR);}}
    function setup(){createCanvas(windowWidth,windowHeight);colorMode(RGB);
      for(let i=0;i<N;i++)particles.push(new Particle(i));
      growthRings.push(new GrowthRing(width/2,height/2));}
    function draw(){background(8,8,20,18);
      breathPhase=sin(frameCount*BREATH_RATE);
      wH=0.35+breathPhase*BREATH_DEPTH;wE=0.15-breathPhase*(BREATH_DEPTH*0.5);
      wR=max(0.38,1.0-wH-wE);growthTimer++;
      if(growthTimer>=GROWTH_INTERVAL){growthTimer=0;
        let cx=width/2+random(-width*0.3,width*0.3),cy=height/2+random(-height*0.25,height*0.25);

      for(let r of growthRings){r.update();r.draw();}
      if(frameCount%30===0){let sample=min(particles.length,300);
        for(let i=0;i<sample;i++){let p=particles[i],near=0;
          for(let j=0;j<sample;j++){if(i===j)continue;
            let q=particles[j];if(dist(p.x,p.y,q.x,q.y)<65)near++;}
          p.synaptic=min(1.0,p.synaptic*0.94+near*0.018);}}
      noiseOffset+=0.003;
      for(let p of particles){p.update();p.draw();}}
    function mousePressed(){growthRings.push(new GrowthRing(mouseX,mouseY));
      if(growthRings.length>MAX_RINGS)growthRings.shift();}
    function windowResized(){resizeCanvas(windowWidth,windowHeight);}
    </script></body></html>
