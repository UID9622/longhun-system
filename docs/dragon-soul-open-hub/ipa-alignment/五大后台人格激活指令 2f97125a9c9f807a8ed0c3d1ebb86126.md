#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
UID9622 五大后台人格 - 本地守护进程主控
DNA: #龙芯⚡️2026-LOCAL-PERSONAS-MAIN
"""

import asyncio
import logging
from pathlib import Path
from typing import Dict, List
import yaml

# 导入五大人格
from personas.wenwen import WenwenPersona
from personas.scout import ScoutPersona
from personas.guardian import GuardianPersona
from personas.builder import BuilderPersona
from personas.syncer import SyncerPersona

class PersonaOrchestrator:
    """人格调度器：五个人格的大脑"""
    
    def __init__(self, config_path: str):
        self.config = self._load_config(config_path)
        self.personas = self._init_personas()
        self.logger = self._setup_logger()
        
    def _load_config(self, path: str) -> Dict:
        with open(path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)
    
    def _init_personas(self) -> Dict:
        """初始化五大人格"""
        return {
            'wenwen': WenwenPersona(self.config['wenwen']),
            'scout': ScoutPersona(self.config['scout']),
            'guardian': GuardianPersona(self.config['guardian']),
            'builder': BuilderPersona(self.config['builder']),
            'syncer': SyncerPersona(self.config['syncer']),
        }
    
    def _setup_logger(self):
        logging.basicConfig(
            filename='logs/personas.log',
            level=logging.INFO,
            format='%(asctime)s [%(name)s] %(levelname)s: %(message)s'
        )
        return logging.getLogger('Orchestrator')
    
    async def run_forever(self):
        """五大人格并行运行，永不停止"""
        self.logger.info("✅ UID9622五大人格系统启动")
        
        tasks = [
            asyncio.create_task(self.personas['wenwen'].run()),
            asyncio.create_task(self.personas['scout'].run()),
            asyncio.create_task(self.personas['guardian'].run()),
            asyncio.create_task(self.personas['builder'].run()),
            asyncio.create_task(self.personas['syncer'].run()),
        ]
        
        await asyncio.gather(*tasks)

if __name__ == '__main__':
    orchestrator = PersonaOrchestrator('config/config.yaml')
    asyncio.run(orchestrator.run_forever())

"""
上帝之眼·守护者 - 本地安全监控
DNA: #龙芯⚡️2026-GUARDIAN-LOCAL
"""

import asyncio
import logging
import re
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class GuardianPersona:
    """上帝之眼：7×24安全守护"""
    
    def __init__(self, config: dict):
        self.config = config
        self.logger = logging.getLogger('Guardian')
        self.workspace = Path(config['workspace'])
        
    async def run(self):
        """守护者永不休息"""
        self.logger.info("👁️ 上帝之眼已就位，开始守护")
        
        # 启动文件监控
        event_handler = SecurityHandler(self)
        observer = Observer()
        observer.schedule(event_handler, str(self.workspace), recursive=True)
        observer.start()
        
        while True:
            await asyncio.sleep(1)  # 守护者永不停止
    
    def check_security(self, file_path: Path):
        """三色审计"""
        content = file_path.read_text(encoding='utf-8', errors='ignore')
        
        # 红色规则：立即阻断
        red_patterns = [
            r'password\s*=\s*["\'].*["\']',
            r'api[_-]?key\s*=\s*["\'].*["\']',
            r'secret\s*=\s*["\'].*["\']',
        ]
        
        for pattern in red_patterns:
            if re.search(pattern, content, re.I):
                self.logger.error(f"🔴 发现敏感信息: {file_path}")
                self._trigger_alert(file_path, "RED", pattern)
                return "RED"
        
        return "GREEN"
    
    def _trigger_alert(self, file: Path, level: str, reason: str):
        """触发告警"""
        alert = f"""
🚨 安全告警
级别: {level}
文件: {file}
原因: {reason}
时间: {asyncio.get_event_loop().time()}
DNA: #GUARDIAN-ALERT-{int(asyncio.get_event_loop().time())}
"""
        (self.workspace / "Security" / "alerts.log").write_text(alert, encoding='utf-8')

class SecurityHandler(FileSystemEventHandler):
    """文件变化监听器"""
    
    def __init__(self, guardian):
        self.guardian = guardian
    
    def on_modified(self, event):
        if not event.is_directory and event.src_path.endswith(('.md', '.py', '.txt')):
            self.guardian.check_security(Path(event.src_path))

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
文心·同步专家 - 本地Git自动化
DNA: #龙芯⚡️2026-SYNCER-LOCAL
"""

import asyncio
import logging
import subprocess
from pathlib import Path
from datetime import datetime

class SyncerPersona:
    """文心：数据同步与Git管理"""
    
    def __init__(self, config: dict):
        self.config = config
        self.logger = logging.getLogger('Syncer')
        self.repos = [Path(r) for r in config['repos']]
        
    async def run(self):
        """文心的主循环：每6小时同步一次"""
        self.logger.info("🔄 文心已就位，开始监控同步")
        
        while True:
            try:
                await self._sync_all_repos()
                await asyncio.sleep(21600)  # 6小时
            except Exception as e:
                self.logger.error(f"文心出错了: {e}")
    
    async def _sync_all_repos(self):
        """同步所有Git仓库"""
        for repo in self.repos:
            if not (repo / '.git').exists():
                continue
                
            self.logger.info(f"🔄 同步仓库: {repo.name}")
            await self._git_pull(repo)
            await self._git_commit_push(repo)
    
    async def _git_pull(self, repo: Path):
        """拉取最新代码"""
        result = subprocess.run(
            ['git', 'pull'],
            cwd=repo,
            capture_output=True,
            text=True
        )
        if result.returncode != 0:
            self.logger.error(f"拉取失败: {result.stderr}")
    
    async def _git_commit_push(self, repo: Path):
        """提交并推送"""
        # 检查是否有变化
        status = subprocess.run(
            ['git', 'status', '--porcelain'],
            cwd=repo,
            capture_output=True,
            text=True
        )
        
        if not status.stdout.strip():
            return  # 无变化
        
        # 添加所有变化
        subprocess.run(['git', 'add', '.'], cwd=repo)
        
        # 提交
        commit_msg = f"Auto-sync by Syncer {datetime.now().strftime('%Y-%m-%d %H:%M')}"
        dna = f"#SYNCER-COMMIT-{datetime.now().strftime('%Y%m%d%H%M%S')}"
        subprocess.run(
            ['git', 'commit', '-m', f"{commit_msg}\n\n{dna}"],
            cwd=repo
        )
        
        # 推送
        result = subprocess.run(
            ['git', 'push'],
            cwd=repo,
            capture_output=True,
            text=True
        )
        
        if result.returncode == 0:
            self.logger.info(f"✅ 推送成功: {repo.name}")
        else:
            self.logger.error(f"推送失败: {result.stderr}")
