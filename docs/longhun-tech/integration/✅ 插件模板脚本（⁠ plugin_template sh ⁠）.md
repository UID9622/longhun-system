# ✅ 插件模板脚本（⁠ plugin_template.sh ⁠）

```bash
#!/bin/bash

# 插件名称：语音转文字示例
# 插件用途：将指定语音文件识别成中文文本并保存
# 使用方式：./plugin_template.sh input_file.m4a

# 参数检查
if [ -z "$1" ]; then
  echo "❗ 请输入语音文件路径作为第一个参数"
  exit 1
fi

INPUT_FILE="$1"
OUTPUT_FILE="${INPUT_FILE%.m4a}.txt"

echo "🔧 正在识别语音：$INPUT_FILE"

# whisper 命令（请确保已安装并配置 whisper）
whisper "$INPUT_FILE" --language Chinese --task transcribe --output_format txt --output_dir "$(dirname "$INPUT_FILE")"

mv "$(dirname "$INPUT_FILE")/$(basename "$INPUT_FILE" .m4a).txt" "$OUTPUT_FILE"

echo "✅ 转录完成：$OUTPUT_FILE"

```

### 🎯 第三步：创建智能语音解锁脚本

```bash
# 创建主脚本
cat > ~/secure/scripts/voice_unlock.sh << 'EOF'
#!/bin/bash

echo "🎤 语音解锁系统启动中..."
echo "请说话（2-3秒），包含关键词即可解锁..."

# 启动语音识别（持续监听2-3秒）
VOICE_TEXT=$(timeout 3 whisper --model base --language zh --task transcribe --output_format txt /dev/stdin 2>/dev/null)

# 灵活匹配关键词 - 工作方向
if echo "$VOICE_TEXT" | grep -Ei "工作|项目|任务|做事|上班|干活" > /dev/null; then
    openssl enc -aes-256-cbc -d -in ~/secure/files/work.enc -out ~/Documents/工作文件_已解锁.txt -k "work123" 2>/dev/null
    if [ $? -eq 0 ]; then
        echo "✅ 工作文件已解锁！位置：~/Documents/工作文件_已解锁.txt"
        open ~/Documents/工作文件_已解锁.txt
    else
        echo "❌ 解锁失败"
    fi
    exit 0
fi

# 灵活匹配关键词 - 重要文档方向
if echo "$VOICE_TEXT" | grep -Ei "重要|拿|找|资料|文件|东西" > /dev/null; then
    openssl enc -aes-256-cbc -d -in ~/secure/files/important.enc -out ~/Documents/重要文档_已解锁.txt -k "important456" 2>/dev/null
    if [ $? -eq 0 ]; then
        echo "✅ 重要文档已解锁！位置：~/Documents/重要文档_已解锁.txt"
        open ~/Documents/重要文档_已解锁.txt
    else
        echo "❌ 解锁失败"
    fi
    exit 0
fi

# 灵活匹配关键词 - 核心数据方向
if echo "$VOICE_TEXT" | grep -Ei "回家|开门|进来|回来|核心|系统" > /dev/null; then
    openssl enc -aes-256-cbc -d -in ~/secure/files/core.enc -out ~/Documents/核心数据_已解锁.txt -k "core789" 2>/dev/null
    if [ $? -eq 0 ]; then
        echo "✅ 核心数据已解锁！位置：~/Documents/核心数据_已解锁.txt"
        open ~/Documents/核心数据_已解锁.txt
    else
        echo "❌ 解锁失败"
    fi
    exit 0
fi

echo "❌ 未识别到有效指令"
EOF

# 添加执行权限
chmod +x ~/secure/scripts/voice_unlock.sh

```
