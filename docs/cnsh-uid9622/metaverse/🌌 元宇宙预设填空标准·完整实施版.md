```python
            'id': f"obj_{name.lower().replace(' ', '_')}",
```

## 📊 引擎映射转换器

**支持4个游戏引擎：**

- ✅ Unity（C#代码）
- ✅ Three.js（JavaScript代码）
- ✅ Unreal（Blueprint JSON）
- ✅ Godot（GDScript代码）

```python
"""
元宇宙引擎映射转换器
DNA: #龙芯⚡️2026-02-01-引擎映射器-v1.0
"""

class EngineMapper:
    """引擎映射转换器"""
    
    def __init__(self):
        self.mappings = {
            'unity': self.map_to_unity,
            'unreal': self.map_to_unreal,
            'threejs': self.map_to_threejs,
            'godot': self.map_to_godot
        }
    
    def convert(self, scene_yaml, target_engine):
        """转换场景到目标引擎"""
        if target_engine not in self.mappings:
            raise ValueError(f"不支持的引擎: {target_engine}")
        
        with open(scene_yaml, 'r', encoding='utf-8') as f:
            scene_data = yaml.safe_load(f)
        
        engine_data = self.mappings[target_engine](scene_data)
        return engine_data

# 使用示例
if __name__ == "__main__":
    mapper = EngineMapper()
    
    # 转换到Unity
    unity_code = mapper.convert("forest_entrance.yaml", "unity")
    print("✅ Unity代码已生成")
    
    # 转换到Three.js
    threejs_code = mapper.convert("forest_entrance.yaml", "threejs")
    print("✅ Three.js代码已生成")
```
