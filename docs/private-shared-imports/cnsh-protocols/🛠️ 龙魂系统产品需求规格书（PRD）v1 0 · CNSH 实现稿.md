            raise Exception("DNA冲突，系统拒绝注册")

        self.hook_system.execute_hook('after_redline_check', dna)

        """人格向量更新，投影到 [-1,1]^28（对接 §7.2 人格演化系统）"""

            self.personality_vectors[h] = [0.0] * 28

            'hash': new_hash,
            'prev_hash': prev_hash,
            'action': action,
            'timestamp': datetime.now().isoformat()
