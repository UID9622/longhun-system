### 4｜统计检验与建模


```python
# ── 正态性检验
stat, p = stats.shapiro(df_clean['目标变量'])
print(f"Shapiro-Wilk: stat={stat:.4f}, p={p:.4f}")
print("正态分布" if p > 0.05 else "非正态分布")

# ── 两组比较（t检验 或 Mann-Whitney）
group_a = df_clean[df_clean['分组'] == 'A']['目标变量']
group_b = df_clean[df_clean['分组'] == 'B']['目标变量']

if p > 0.05:  # 正态
    t_stat, p_val = stats.ttest_ind(group_a, group_b)
    print(f"独立样本t检验: t={t_stat:.4f}, p={p_val:.4f}")
else:  # 非正态
    u_stat, p_val = stats.mannwhitneyu(group_a, group_b)
    print(f"Mann-Whitney U: U={u_stat:.4f}, p={p_val:.4f}")

# ── 效应量 Cohen's d
cohen_d = (group_a.mean() - group_b.mean()) / \
           np.sqrt((group_a.std()**2 + group_b.std()**2) / 2)
print(f"Cohen's d 效应量: {cohen_d:.4f}")
print("小效应" if abs(cohen_d)<0.5 else ("中效应" if abs(cohen_d)<0.8 else "大效应"))
```
