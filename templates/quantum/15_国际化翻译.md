## 💻 通义翻译 Python 工具包


```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
国际化技术文档翻译工具 v1.0
使用通义千问(Qwen) API 实现多语言翻译
DNA: #龍芯⚡️2026-08-31-国际化翻译模板-v1.0-UID9622
"""

import os
import re
from openai import OpenAI  # 通义兼容OpenAI SDK

# 通义API配置
client = OpenAI(
    api_key=os.getenv("DASHSCOPE_API_KEY"),  # 从阿里云控制台获取
    base_url="https://dashscope.aliyuncs.com/compatible-mode/v1"
)

# 技术术语保护列表（这些词不翻译）
TECH_TERMS = [
    "API", "SDK", "JSON", "URL", "HTTP", "REST", "GraphQL",
    "Docker", "Kubernetes", "Git", "GitHub", "CI/CD",
    "Python", "JavaScript", "TypeScript", "React", "Node.js",
    "DNA", "UID", "UUID", "Base64", "SHA256", "GPG"
]

# 翻译风格提示
STYLE_PROMPTS = {
    "技术": "使用准确的技术术语，保持专业简洁，代码块不翻译",
    "学术": "使用学术规范语言，被动语态，符合期刊投稿风格",
    "普通": "使用通俗易懂的语言，适合普通读者，必要时添加解释",
    "开发者": "面向开发者，保留英文技术名词，简洁直接"
}

def 保护代码块(text: str) -> tuple[str, dict]:
    """提取并保护代码块，防止被翻译"""
    placeholders = {}
    code_pattern = re.compile(r'```[\s\S]*?```', re.MULTILINE)
    inline_pattern = re.compile(r'`[^`]+`')
    
    counter = 0
    def replace_code(match):
        nonlocal counter
        key = f"__CODE_BLOCK_{counter}__"
        placeholders[key] = match.group(0)
        counter += 1
        return key
    
    protected = code_pattern.sub(replace_code, text)
    protected = inline_pattern.sub(replace_code, protected)
    return protected, placeholders

def 还原代码块(text: str, placeholders: dict) -> str:
    """将占位符还原为原始代码块"""
    for key, value in placeholders.items():
        text = text.replace(key, value)
    return text

def 翻译文档(
    content: str,
    source_lang: str = "zh-CN",
    target_lang: str = "en",
    style: str = "技术",
    model: str = "qwen-max"
) -> dict:
    """主翻译函数"""
    
    # 保护代码块
    protected_content, placeholders = 保护代码块(content)
    
    # 语言映射
    lang_names = {
        "zh-CN": "简体中文", "en": "英文", "ja": "日文",
        "ko": "韩文", "es": "西班牙文", "fr": "法文",
        "ar": "阿拉伯文", "hi": "印地文"
    }
    
    style_note = STYLE_PROMPTS.get(style, STYLE_PROMPTS["技术"])
    
    system_prompt = f"""你是专业的技术文档翻译专家。
翻译规则：
1. 将{lang_names.get(source_lang, source_lang)}翻译为{lang_names.get(target_lang, target_lang)}
2. {style_note}
3. 保持Markdown格式（标题、列表、表格、链接）
4. 以下技术术语保留英文原文：{', '.join(TECH_TERMS)}
5. 代码块占位符（__CODE_BLOCK_X__）直接原样保留，不要翻译
6. 只输出翻译结果，不要添加任何解释"""
    
    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": protected_content}
        ],
        temperature=0.3,  # 低温度=更准确的技术翻译
        max_tokens=8192
    )
    
    translated_protected = response.choices[0].message.content
    
    # 还原代码块
    final_translation = 还原代码块(translated_protected, placeholders)
    
    return {
        "source_lang": source_lang,
        "target_lang": target_lang,
        "style": style,
        "model": model,
        "original_length": len(content),
        "translated_length": len(final_translation),
        "translation": final_translation,
        "tokens_used": response.usage.total_tokens
    }

# ── 批量多语言翻译
def 批量翻译(content: str, target_langs: list, **kwargs) -> dict:
    """一键翻译为多种语言"""
    results = {}
    for lang in target_langs:
        print(f"正在翻译为 {lang}...")
        results[lang] = 翻译文档(content, target_lang=lang, **kwargs)
    return results


# ── 使用示例
if __name__ == "__main__":
    # 单语言翻译
    chinese_doc = """
    ## 安装
```

pip install your-package


```javascript
## 快速开始
导入模块并初始化：
```

from your_package import Client

client = Client(api_key="your_key")


```javascript
"""

result = 翻译文档(chinese_doc, source_lang="zh-CN", target_lang="en", style="开发者")
print(result["translation"])
print(f"消耗tokens: {result['tokens_used']}")

# 批量翻译
multi_results = 批量翻译(
    chinese_doc,
    target_langs=["en", "ja", "es"],
    style="开发者"
)
```
