## 🎯 设计目标

构建一个健壮、可扩展的论文元数据提取与评估系统，支持多种来源（学术数据库、预印本平台、通用网页），输出统一的评估报告。


---


## 🧠 核心思路

1. 多源适配：优先使用结构化 API（Crossref、Semantic Scholar、OpenAlex），失败后回退到 HTML 解析。
1. 统一数据模型：所有来源的元数据映射到标准字段。
1. 评估算法：基于引用次数、发表时间、作者影响力、期刊/会议等级等指标计算综合评分。
1. 可配置性：允许用户自定义权重和阈值。

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
论文评估提取模板 v1.0
支持多源提取和综合评估
DNA: #龍芯⚡️2026-08-31-论文评估提取模板-v1.0-UID9622
"""

import requests
import re
import math
from typing import Dict, List, Optional, Any
from urllib.parse import urlparse
from datetime import datetime
from bs4 import BeautifulSoup


class PaperEvaluator:
    """论文评估器"""

    def __init__(self, config: Optional[Dict] = None):
        self.config = {
            "weights": {
                "citations": 0.4,
                "recency": 0.2,
                "author_score": 0.2,
                "venue_score": 0.2
            },
            "citation_scale": 100,
            "max_age_years": 10,
        }
        if config:
            self.config.update(config)
        self.headers = {
            "User-Agent": "PaperEvaluator/1.0 (your-email@example.com)"
        }

    # ─── 提取方法 ───────────────────────────────────────────────

    def extract_from_url(self, url: str) -> Dict[str, Any]:
        """从 URL 提取论文元数据（自动选择最佳来源）"""
        doi = self._extract_doi_from_url(url)
        if doi:
            for fetcher, source in [
                (self._fetch_from_crossref, "Crossref"),
                (self._fetch_from_semantic_scholar, "Semantic Scholar"),
                (self._fetch_from_openalex, "OpenAlex"),
            ]:
                metadata = fetcher(doi)
                if metadata:
                    metadata["source"] = source
                    return metadata
        metadata = self._parse_html(url)
        if metadata:
            metadata["source"] = "HTML"
            return metadata
        raise ValueError(f"无法从 {url} 提取论文元数据")

    def extract_from_doi(self, doi: str) -> Dict[str, Any]:
        """从 DOI 提取论文元数据"""
        for fetcher in [self._fetch_from_crossref,
                        self._fetch_from_semantic_scholar,
                        self._fetch_from_openalex]:
            try:
                metadata = fetcher(doi)
                if metadata:
                    metadata["doi"] = doi
                    return metadata
            except Exception:
                continue
        raise ValueError(f"无法从 DOI {doi} 提取元数据")

    # ─── 辅助提取函数 ──────────────────────────────────────────

    def _extract_doi_from_url(self, url: str) -> Optional[str]:
        doi_match = re.search(r'doi\.org/(10\.\d{4,9}/[^\s]+)', url)
        if doi_match:
            return doi_match.group(1)
        doi_match = re.search(r'(10\.\d{4,9}/[^\s?#]+)', url)
        if doi_match:
            return doi_match.group(1)
        parsed = urlparse(url)
        if 'doi.org' in parsed.netloc:
            return parsed.path.lstrip('/')
        return None

    def _fetch_from_crossref(self, doi: str) -> Optional[Dict[str, Any]]:
        try:
            resp = requests.get(
                f"https://api.crossref.org/works/{doi}",
                headers=self.headers, timeout=10
            )
            if resp.status_code != 200:
                return None
            data = resp.json()["message"]
            title = data.get("title", [""])[0] if data.get("title") else ""
            abstract = re.sub(r'<[^>]+>', '', data.get("abstract", ""))
            authors = [
                f"{a.get('given', '')} {a.get('family', '')}".strip()
                for a in data.get("author", [])
            ]
            pub_date = None
            for key in ["published-print", "published-online", "issued"]:
                if key in data:
                    parts = data[key].get("date-parts", [[None]])[0]
                    pub_date = self._format_date_parts(parts)
                    break
            venue = (data.get("container-title") or [""])[0] or \
                    data.get("event", {}).get("name", "")
            return {
                "title": title, "abstract": abstract,
                "keywords": data.get("subject", []),
                "authors": authors, "publish_date": pub_date,
                "citations": None, "venue": venue, "doi": doi
            }
        except Exception:
            return None

    def _fetch_from_semantic_scholar(
        self, doi: str = None, paper_id: str = None
    ) -> Optional[Dict[str, Any]]:
        try:
            if doi:
                url = f"https://api.semanticscholar.org/graph/v1/paper/DOI:{doi}"
            elif paper_id:
                url = f"https://api.semanticscholar.org/graph/v1/paper/{paper_id}"
            else:
                return None
            resp = requests.get(url, params={
                "fields": "title,abstract,authors,year,publicationDate,citationCount,venue,externalIds"
            }, headers=self.headers, timeout=10)
            if resp.status_code != 200:
                return None
            data = resp.json()
            return {
                "title": data.get("title", ""),
                "abstract": data.get("abstract", ""),
                "keywords": [],
                "authors": [a["name"] for a in data.get("authors", [])],
                "publish_date": data.get("publicationDate") or str(data.get("year", "")),
                "citations": data.get("citationCount", 0),
                "venue": data.get("venue", ""),
                "doi": doi
            }
        except Exception:
            return None

    def _fetch_from_openalex(self, doi: str) -> Optional[Dict[str, Any]]:
        try:
            resp = requests.get(
                f"https://api.openalex.org/works/doi:{doi}",
                headers=self.headers, timeout=10
            )
            if resp.status_code != 200:
                return None
            data = resp.json()
            abstract = self._reconstruct_abstract(
                data.get("abstract_inverted_index")
            )
            return {
                "title": data.get("title", ""),
                "abstract": abstract,
                "keywords": [c["display_name"] for c in data.get("concepts", [])],
                "authors": [
                    a["author"]["display_name"]
                    for a in data.get("authorships", [])
                ],
                "publish_date": data.get("publication_date", ""),
                "citations": data.get("cited_by_count", 0),
                "venue": data.get("primary_location", {}).get("source", {}).get("display_name", ""),
                "doi": doi
            }
        except Exception:
            return None

    def _parse_html(self, url: str) -> Optional[Dict[str, Any]]:
        try:
            resp = requests.get(url, headers=self.headers, timeout=10)
            if resp.status_code != 200:
                return None
            soup = BeautifulSoup(resp.text, 'html.parser')
            def get_meta(name):
                tag = soup.find('meta', attrs={'name': name}) or \
                      soup.find('meta', attrs={'property': name})
                return tag.get('content') if tag else None
            doi = get_meta('citation_doi')
            if doi:
                return self.extract_from_doi(doi)
            title = get_meta('citation_title') or \
                    (soup.find('h1').text.strip() if soup.find('h1') else None)
            authors = [m.get('content') for m in soup.find_all('meta', attrs={'name': 'citation_author'})]
            abstract = get_meta('description') or \
                       (soup.find('div', class_=re.compile('abstract', re.I)) or {}).get_text(strip=True)
            pub_date = get_meta('citation_publication_date') or \
                       (soup.find('time') or {}).get('datetime')
            keywords = [m.get('content') for m in soup.find_all('meta', attrs={'name': 'citation_keywords'})]
            if title or abstract:
                return {
                    "title": title or "", "abstract": abstract or "",
                    "keywords": keywords, "authors": authors,
                    "publish_date": pub_date or "",
                    "citations": None, "venue": None, "doi": None
                }
            return None
        except Exception:
            return None

    # ─── 工具函数 ──────────────────────────────────────────────

    def _format_date_parts(self, parts: List) -> Optional[str]:
        parts = [p for p in (parts or []) if p is not None]
        if not parts:
            return None
        if len(parts) == 1:
            return str(parts[0])
        if len(parts) == 2:
            return f"{parts[0]}-{parts[1]:02d}"
        return f"{parts[0]}-{parts[1]:02d}-{parts[2]:02d}"

    def _reconstruct_abstract(self, inverted_index: Optional[Dict]) -> str:
        if not inverted_index:
            return ""
        pairs = [(pos, word) for word, positions in inverted_index.items() for pos in positions]
        return " ".join(w for _, w in sorted(pairs))

    # ─── 评估逻辑 ──────────────────────────────────────────────

    def evaluate(self, metadata: Dict[str, Any]) -> Dict[str, Any]:
        """根据元数据计算综合评分"""
        scores = {}
        citations = metadata.get("citations")
        if citations is not None:
            scores["citations"] = min(1.0, math.log10(citations + 1) / math.log10(self.config["citation_scale"] + 1))
        else:
            scores["citations"] = 0.0
        pub_date = metadata.get("publish_date")
        if pub_date:
            try:
                date_obj = datetime.fromisoformat(pub_date[:10])
                years_old = (datetime.now() - date_obj).days / 365.25
                scores["recency"] = max(0.0, 1.0 - max(0, years_old) / self.config["max_age_years"])
            except Exception:
                scores["recency"] = 0.5
        else:
            scores["recency"] = 0.0
        authors = metadata.get("authors", [])
        scores["author_score"] = min(1.0, len(authors) / 5) * 0.5 if authors else 0.0
        scores["venue_score"] = 0.5 if metadata.get("venue") else 0.0
        total = sum(scores.get(k, 0.0) * w for k, w in self.config["weights"].items())
        return {
            "rating": round(total * 100, 2),
            "scores": {k: round(v, 4) for k, v in scores.items()},
            "weights": self.config["weights"],
            "metadata": metadata
        }

    def evaluate_from_url(self, url: str) -> Dict[str, Any]:
        return self.evaluate(self.extract_from_url(url))

    def evaluate_from_doi(self, doi: str) -> Dict[str, Any]:
        return self.evaluate(self.extract_from_doi(doi))


# ─── 使用示例 ──────────────────────────────────────────────────

if __name__ == "__main__":
    evaluator = PaperEvaluator()

    # 示例1：通过 DOI 评估（AlphaFold）
    doi = "10.1038/s41586-020-2649-2"
    result = evaluator.evaluate_from_doi(doi)
    print(f"标题: {result['metadata']['title']}")
    print(f"引用次数: {result['metadata']['citations']}")
    print(f"综合评分: {result['rating']}/100")
    print(f"各项得分: {result['scores']}")

    # 示例2：通过 arXiv URL 评估（Attention is All You Need）
    arxiv_url = "https://arxiv.org/abs/1706.03762"
    result2 = evaluator.evaluate_from_url(arxiv_url)
    print(f"标题: {result2['metadata']['title']}")
    print(f"综合评分: {result2['rating']}/100")
```
